\section{Edges in a vertex set}
\label{sec:edgesIn}

Laman graphs are defined by counting edges: a graph on $n \ge 2$
vertices is a Laman graph if it has $2n - 3$ edges and every set of
$m \ge 2$ of its vertices spans at most $2m - 3$ of them. This section
and the next develop this counting with general constants $k$ and
$\ell$ in place of $2$ and $3$, and \cref{sec:laman} returns to Laman
graphs. Throughout both sections, $G$ is a simple graph on a vertex
set $V$, which we assume to be finite only where a statement says so.

This section studies the set of edges of $G$ with both endpoints in a
given vertex set. We show that this set grows with the vertex set and
compute it at a few small sets. We then count the edges that remain
when a vertex is deleted, and finally compare the counts at two vertex
sets with the counts at their union and intersection.
\Cref{sec:sparsity} defines
$(k,\ell)$-sparse and $(k,\ell)$-tight graphs and develops their basic
theory.

\subsection{Definition and basic properties}

\begin{definition}[Edges induced on a vertex set]
  \label{def:edgesIn}
  \lean{SimpleGraph.edgesIn}
  \leanok
  For $S \subseteq V$, let
  \[
    \edgesIn{S} \;:=\; \{\,e \in E(G) : e \subseteq S\,\}
  \]
  be the set of edges of $G$ with both endpoints in $S$, that is, the
  edge set of the subgraph of $G$ induced on $S$. We say that $S$
  \emph{spans} these edges.
\end{definition}

\begin{lemma}[Monotonicity]
  \label{lem:edgesIn-mono}
  \lean{SimpleGraph.edgesIn_mono}
  \leanok
  \uses{def:edgesIn}
  Let $S \subseteq T \subseteq V$. Then
  $\edgesIn{S} \subseteq \edgesIn{T}$.
\end{lemma}
\begin{proof}
  \leanok
  An edge $e \subseteq S$ also satisfies $e \subseteq T$.
\end{proof}

\begin{lemma}[Corner cases]
  \label{lem:edgesIn-corners}
  \lean{SimpleGraph.edgesIn_univ, SimpleGraph.edgesIn_empty, SimpleGraph.edgesIn_singleton, SimpleGraph.edgesIn_bot}
  \leanok
  \uses{def:edgesIn}
  Let $v \in V$ and $S \subseteq V$. Then $\edgesIn{V} = E(G)$,
  $\edgesIn{\emptyset} = \emptyset$ and $\edgesIn{\{v\}} = \emptyset$.
  If $G$ has no edges, then $\edgesIn{S} = \emptyset$.
\end{lemma}
\begin{proof}
  \leanok
  All but the third are immediate from the definition. A singleton
  $\{v\}$ contains no edge, since $G$ has no loops and so the two
  endpoints of an edge are distinct.
\end{proof}

\begin{fmlnote}
  The four pins are the four claims, in the order stated.
\end{fmlnote}

\subsection{Vertex deletion}

Let $v \in V$, and write $\{v\}^c = V \setminus \{v\}$. The set
$\edgesIn{\{v\}^c}$ consists of the edges of $G$ that remain when $v$
is deleted. Let $I_G(v) := \{\,e \in E(G) : v \in e\,\}$ be the set of
edges of $G$ incident to $v$. When $V$ is finite, $|I_G(v)|$ is the
degree of $v$. The next two lemmas show that deleting $v$ removes
exactly the edges of $I_G(v)$, and count what remains. The count is
used in \cref{lem:isLaman-two-le-degree} to show that every vertex of a
Laman graph on at least three vertices has degree at least two.

\begin{lemma}[Edges avoiding a vertex]
  \label{lem:mem-edgesIn-compl-singleton}
  \lean{SimpleGraph.mem_edgesIn_compl_singleton, SimpleGraph.edgesIn_compl_singleton}
  \leanok
  \uses{def:edgesIn}
  Let $v \in V$. Then
  \[
    \edgesIn{\{v\}^{c}} \;=\; E(G) \,\setminus\, I_G(v);
  \]
  that is, a pair $e$ of vertices lies in $\edgesIn{\{v\}^c}$ if and
  only if $e \in E(G)$ and $v \notin e$.
\end{lemma}
\begin{proof}
  \leanok
  An edge has both endpoints in $\{v\}^c$ exactly when neither
  endpoint is $v$, that is, when $v \notin e$. So
  $\edgesIn{\{v\}^c}$ consists of the edges of $G$ not incident to $v$,
  which is $E(G) \setminus I_G(v)$.
\end{proof}

\begin{lemma}[Vertex-deletion partition of edges]
  \label{lem:ncard-edgesIn-compl-singleton}
  \lean{SimpleGraph.ncard_edgesIn_compl_singleton_add_ncard_incidenceSet}
  \leanok
  \uses{lem:mem-edgesIn-compl-singleton}
  Let $V$ be finite and let $v \in V$. Then
  \[
    \bigl|\edgesIn{\{v\}^{c}}\bigr| \;+\; |I_G(v)| \;=\; |E(G)|.
  \]
\end{lemma}
\begin{proof}
  \leanok
  By \cref{lem:mem-edgesIn-compl-singleton}, $E(G)$ is the disjoint
  union of $\edgesIn{\{v\}^c}$ and $I_G(v)$, so the cardinalities add.
\end{proof}

\subsection{Modularity: intersection, union, and supermodular counts}

The map $S \mapsto \edgesIn{S}$ commutes with intersection but not with
union. An edge with one endpoint in $S \setminus T$ and the other in
$T \setminus S$ lies in $\edgesIn{S \cup T}$ but in neither
$\edgesIn{S}$ nor $\edgesIn{T}$. Counting edges therefore gives an
inequality (\cref{lem:edgesIn-ncard-add-le}). It becomes an identity,
Jord\'an's Lemma~2.1.1~\cite{jordan2016}, when the number of these
crossing edges is added to its left side. We need only the inequality,
which is the counting step in the proof that tight sets are closed
under union and intersection (\cref{thm:isTightOn-union-inter}).

\begin{lemma}[Distribution over intersection]
  \label{lem:edgesIn-inter}
  \lean{SimpleGraph.edgesIn_inter}
  \leanok
  \uses{def:edgesIn}
  Let $S, T \subseteq V$. Then
  $\edgesIn{S \cap T} = \edgesIn{S} \cap \edgesIn{T}$.
\end{lemma}
\begin{proof}
  \leanok
  An edge has both endpoints in $S \cap T$ if and only if both
  endpoints lie in $S$ and both lie in $T$.
\end{proof}

\begin{lemma}[Supermodularity of edge counts]
  \label{lem:edgesIn-ncard-add-le}
  \lean{SimpleGraph.edgesIn_ncard_add_le}
  \leanok
  \uses{lem:edgesIn-mono, lem:edgesIn-inter}
  Let $V$ be finite and let $S, T \subseteq V$. Then
  \[
    \bigl|\edgesIn{S}\bigr| + \bigl|\edgesIn{T}\bigr|
    \;\le\;
    \bigl|\edgesIn{S \cup T}\bigr| + \bigl|\edgesIn{S \cap T}\bigr|.
  \]
\end{lemma}
\begin{proof}
  \leanok
  Let $A = \edgesIn{S}$ and $B = \edgesIn{T}$, so that
  $|A| + |B| = |A \cup B| + |A \cap B|$. By \cref{lem:edgesIn-inter},
  $A \cap B = \edgesIn{S \cap T}$. By monotonicity
  (\cref{lem:edgesIn-mono}), $A \cup B \subseteq \edgesIn{S \cup T}$, so
  $|A \cup B| \le |\edgesIn{S \cup T}|$. Together these give the
  inequality.
\end{proof}

\section{\texorpdfstring{$(k,\ell)$}{(k,l)}-sparsity and \texorpdfstring{$(k,\ell)$}{(k,l)}-tightness}
\label{sec:sparsity}

Throughout this section $k$ and $\ell$ are non-negative integers. We
define $(k,\ell)$-sparse and $(k,\ell)$-tight graphs, as Lee and
Streinu~\cite{leeStreinu2008} do for multigraphs, and $(k,\ell)$-tight
sets of vertices. The Laman graphs of \cref{sec:laman} are the case
$(k,\ell) = (2,3)$, which Jord\'an's survey~\cite{jordan2016} treats in
\S 2.1.

After the definitions we show that sparsity passes to subgraphs and
bounds the number of edges. Two facts about $(2,3)$-sparse graphs
follow: such a graph has a vertex of degree at most three, and no
vertex has three pairwise adjacent neighbours. Sparsity and tightness
are invariant under isomorphism. The main result is that two tight
sets in a sparse graph have a tight union and a tight intersection if
their intersection is large enough (\cref{thm:isTightOn-union-inter}).
Two variants enlarge a tight set by counting edges that it does not
span. Next comes a counting lemma for reversing the Type~II Henneberg
move (\cref{def:typeII}), that is, for deleting a vertex of degree
three and joining two of its neighbours by an edge. The section ends
with the reverse decomposition of a $(2,3)$-sparse graph with an edge:
it has a vertex whose deletion leaves a $(2,3)$-sparse graph, after
such an edge is added if the vertex has degree three
(\cref{thm:isSparse-exists-typeI-or-typeII-reverse}).

\subsection{Definitions}

\begin{definition}[$(k,\ell)$-sparse]
  \label{def:isSparse}
  \lean{SimpleGraph.IsSparse}
  \leanok
  \uses{def:edgesIn}
  $G$ is \emph{$(k,\ell)$-sparse} if every finite set $s \subseteq V$
  with $\ell \le k\,|s|$ spans at most $k\,|s| - \ell$ edges, that is,
  \[
    \bigl|\edgesIn{s}\bigr| + \ell \;\le\; k\,|s|.
  \]
\end{definition}
\begin{fmlnote}
  The formal definition states the bound in the additive form displayed
  above. Edge counts there are natural numbers, in which
  $k\,|s| - \ell$ would be truncated at $0$. Under the hypothesis
  $\ell \le k\,|s|$ the two forms agree.
\end{fmlnote}

\begin{definition}[$(k,\ell)$-tight]
  \label{def:isTight}
  \lean{SimpleGraph.IsTight}
  \leanok
  \uses{def:isSparse}
  Let $V$ be finite. The graph $G$ is \emph{$(k,\ell)$-tight} if it is
  $(k,\ell)$-sparse and
  \[
    |E(G)| + \ell \;=\; k\,|V|.
  \]
\end{definition}

\begin{definition}[Tight subset]
  \label{def:isTightOn}
  \lean{SimpleGraph.IsTightOn}
  \leanok
  \uses{def:edgesIn}
  A finite set $s \subseteq V$ is \emph{$(k,\ell)$-tight in $G$} if it
  attains the sparsity bound:
  \[
    \bigl|\edgesIn{s}\bigr| + \ell \;=\; k\,|s|.
  \]
\end{definition}

Tightness of a set is the local form of tightness of the graph. Since
$\edgesIn{V} = E(G)$ (\cref{lem:edgesIn-corners}), a $(k,\ell)$-sparse
graph $G$ on a finite set $V$ is $(k,\ell)$-tight exactly when $V$ is a
$(k,\ell)$-tight set in $G$. At $(k,\ell) = (2,3)$ the hypothesis
$\ell \le k\,|s|$ of \cref{def:isSparse} says that $|s| \ge 2$. So a
finite simple graph is $(2,3)$-sparse exactly when it is sparse in the
sense of Jord\'an~\cite{jordan2016}~\S 2.1, and its $(2,3)$-tight sets
are what he calls \emph{critical sets}. We use ``tight'' throughout.

\subsection{Basic properties and the global edge bound}

Sparsity passes to subgraphs. Applied to the whole vertex set, it
bounds the number of edges of $G$, so that a tight graph has as many
edges as a sparse graph on $V$ can have.

\begin{lemma}[Subgraphs of sparse graphs are sparse]
  \label{lem:isSparse-mono-left}
  \lean{SimpleGraph.IsSparse.mono_left}
  \leanok
  \uses{def:isSparse}
  Let $G'$ be a graph on $V$ with $E(G) \subseteq E(G')$. If $G'$ is
  $(k,\ell)$-sparse, then so is $G$.
\end{lemma}
\begin{proof}
  \leanok
  For a finite set $s \subseteq V$, every edge of $G$ with both
  endpoints in $s$ is an edge of $G'$ with both endpoints in $s$, so
  $|\edgesIn[G]{s}| \le |\edgesIn[G']{s}|$. The sparsity bound for
  $G'$ at $s$ therefore bounds $|\edgesIn[G]{s}|$.
\end{proof}

\begin{lemma}[Global edge bound]
  \label{lem:isSparse-edgeSet-bound}
  \lean{SimpleGraph.IsSparse.edgeSet_ncard_add_le}
  \leanok
  \uses{def:isSparse}
  Let $V$ be finite, and let $G$ be $(k,\ell)$-sparse with
  $\ell \le k\,|V|$. Then
  \[
    |E(G)| + \ell \;\le\; k\,|V|.
  \]
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:edgesIn-corners}
  Apply the sparsity bound at $s = V$ and use $\edgesIn{V} = E(G)$
  (\cref{lem:edgesIn-corners}).
\end{proof}

\begin{lemma}[A proper supergraph of a tight graph is not sparse]
  \label{lem:isTight-not-isSparse-of-lt}
  \lean{SimpleGraph.IsTight.not_isSparse_of_lt}
  \leanok
  \uses{def:isTight, lem:isSparse-edgeSet-bound}
  Let $V$ be finite, let $G$ be $(k,\ell)$-tight, and let $H$ be a
  graph on $V$ with $E(G) \subsetneq E(H)$. Then $H$ is not
  $(k,\ell)$-sparse.
\end{lemma}
\begin{proof}
  \leanok
  By tightness $|E(G)| + \ell = k\,|V|$; in particular
  $\ell \le k\,|V|$. Since $E(G)$ is a proper subset of $E(H)$, we have
  $|E(G)| < |E(H)|$, and so $|E(H)| + \ell > k\,|V|$. By
  \cref{lem:isSparse-edgeSet-bound}, $H$ is not $(k,\ell)$-sparse.
\end{proof}

\subsection{Low-degree vertex and non-adjacency consequences}

Both lemmas below concern $(2,3)$-sparse graphs. The first is applied
to Laman graphs in \cref{lem:isLaman-exists-two-le-degree-le-three}.
The second is used in the reverse decomposition of a $(2,3)$-sparse
graph, at a vertex of degree three
(\cref{thm:isSparse-exists-typeI-or-typeII-reverse}).

\begin{lemma}[A vertex of degree at most 3]
  \label{lem:isSparse-exists-degree-le-three}
  \lean{SimpleGraph.IsSparse.exists_degree_le_three}
  \leanok
  \uses{def:isSparse, lem:isSparse-edgeSet-bound}
  Let $V$ be finite with $|V| \ge 2$, and let $G$ be $(2,3)$-sparse.
  Then some vertex of $G$ has degree at most $3$.
\end{lemma}
\begin{proof}
  \leanok
  Suppose that every vertex has degree at least $4$. By the handshake
  identity, $2|E(G)| = \sum_v \deg_G(v) \ge 4|V|$. The global edge
  bound (\cref{lem:isSparse-edgeSet-bound}) applies at
  $(k, \ell) = (2, 3)$, since $3 \le 2|V|$ when $|V| \ge 2$. It gives
  $|E(G)| + 3 \le 2|V|$, so $2|E(G)| \le 4|V| - 6 < 4|V|$, a
  contradiction.
\end{proof}

\begin{lemma}[Non-adjacent pair among three neighbors]
  \label{lem:isSparse-exists-nonadj-among-three-neighbors}
  \lean{SimpleGraph.IsSparse.exists_nonadj_among_three_neighbors}
  \leanok
  \uses{def:isSparse}
  Let $V$ be finite and let $G$ be $(2,3)$-sparse. Suppose that
  $v \in V$ has three pairwise distinct neighbors $a, b, c$. Then at
  least one of the pairs $\{a, b\}$, $\{a, c\}$, $\{b, c\}$ is
  non-adjacent in $G$.
\end{lemma}
\begin{proof}
  \leanok
  Suppose that all three pairs are adjacent, and let
  $X = \{v, a, b, c\}$. The vertices $a$, $b$ and $c$ are distinct, and
  each differs from $v$ because $G$ has no loops, so $|X| = 4$. The set
  $X$ spans the six distinct edges $va$, $vb$, $vc$, $ab$, $ac$ and
  $bc$, so $|\edgesIn{X}| \ge 6$. But $(2,3)$-sparsity applies at $X$,
  since $3 \le 2 \cdot 4$, and gives
  $|\edgesIn{X}| \le 2 \cdot 4 - 3 = 5$, a contradiction.
\end{proof}

\subsection{Invariance under isomorphism}

Invariance under isomorphism is evident for properties defined by
counting edges. We record it because a Henneberg move
(\cref{sec:henneberg}) produces a graph on a new vertex set, which is
identified with a given graph only up to isomorphism.

\begin{lemma}[Isomorphisms preserve spanned edges]
  \label{lem:iso-image-edgesIn}
  \lean{SimpleGraph.Iso.image_edgesIn}
  \leanok
  \uses{def:edgesIn}
  Let $\varphi$ be an isomorphism from $G$ to a graph $H$, and let
  $s \subseteq V$. Then
  \[
    \varphi\bigl[\edgesIn[G]{s}\bigr] = \edgesIn[H]{\varphi(s)},
  \]
  where $\varphi$ maps a pair $\{u, v\}$ of vertices to
  $\{\varphi(u), \varphi(v)\}$.
\end{lemma}
\begin{proof}
  \leanok
  Since $\varphi$ preserves adjacency in both directions, a pair
  $\{p, q\}$ is an edge of $G$ inside $s$ if and only if
  $\{\varphi(p), \varphi(q)\}$ is an edge of $H$ inside $\varphi(s)$.
  Every pair of vertices of $H$ inside $\varphi(s)$ is of this form.
\end{proof}

\begin{lemma}[Sparsity is preserved by isomorphism]
  \label{lem:isSparse-iso}
  \lean{SimpleGraph.IsSparse.iso}
  \leanok
  \uses{def:isSparse, lem:iso-image-edgesIn}
  Let $\varphi$ be an isomorphism from $G$ to a graph $H$. If $G$ is
  $(k,\ell)$-sparse, then so is $H$.
\end{lemma}
\begin{proof}
  \leanok
  Let $s$ be a finite set of vertices of $H$ with $\ell \le k\,|s|$, and
  let $s' = \varphi^{-1}(s)$, so that $|s'| = |s|$. The map on pairs
  induced by $\varphi$ is injective, and it maps $\edgesIn[G]{s'}$ onto
  $\edgesIn[H]{s}$ (\cref{lem:iso-image-edgesIn}). So
  $|\edgesIn[H]{s}| = |\edgesIn[G]{s'}|$, and the sparsity bound for
  $G$ at $s'$ is the bound for $H$ at $s$.
\end{proof}

\begin{lemma}[Tightness is preserved by isomorphism]
  \label{lem:isTight-iso}
  \lean{SimpleGraph.IsTight.iso}
  \leanok
  \uses{def:isTight, lem:isSparse-iso}
  Let $\varphi$ be an isomorphism from $G$ to a graph $H$. If $G$ is
  $(k,\ell)$-tight, then so is $H$.
\end{lemma}
\begin{proof}
  \leanok
  The graph $H$ is $(k,\ell)$-sparse by \cref{lem:isSparse-iso}. The
  isomorphism is a bijection between the vertex sets of $G$ and $H$, and
  it induces a bijection from $E(G)$ to $E(H)$. So $G$ and $H$ have the
  same numbers of vertices and of edges, and the equation
  $|E(G)| + \ell = k\,|V|$ gives the same equation for $H$.
\end{proof}

\subsection{Unions and intersections of tight sets}

Each proof in this subsection shows that a set spans at least as many
edges as tightness requires. The sparsity bound gives the reverse
inequality, so the bound is attained. The first lemma records this
step.

\begin{lemma}[Tightness from a lower bound]
  \label{lem:isSparse-isTightOn-of-le}
  \lean{SimpleGraph.IsSparse.isTightOn_of_le}
  \leanok
  \uses{def:isSparse, def:isTightOn}
  Let $G$ be $(k,\ell)$-sparse, and let $s \subseteq V$ be a finite set
  with $\ell \le k\,|s|$ and
  \[
    k\,|s| \;\le\; \bigl|\edgesIn{s}\bigr| + \ell.
  \]
  Then $s$ is $(k,\ell)$-tight in $G$.
\end{lemma}
\begin{proof}
  \leanok
  Sparsity at $s$ gives $|\edgesIn{s}| + \ell \le k\,|s|$, so equality
  holds.
\end{proof}

At $(k,\ell) = (2,3)$ the size hypothesis of the next theorem says
that $|s \cap t| \ge 2$, and the theorem is the part of Jord\'an's
Lemma~2.1.2~\cite{jordan2016} that concerns critical sets. For general
$k$ and $\ell < 2k$, see Lee and
Streinu~\cite[Theorem~5(1)]{leeStreinu2008}, who call the subgraph
induced on a tight set a \emph{block}.

\begin{theorem}[Tight subsets are closed under union and intersection]
  \label{thm:isTightOn-union-inter}
  \lean{SimpleGraph.IsTightOn.union_inter}
  \leanok
  \uses{def:isSparse, def:isTightOn, lem:edgesIn-ncard-add-le}
  Let $V$ be finite and let $G$ be $(k,\ell)$-sparse. Let
  $s, t \subseteq V$ be $(k,\ell)$-tight in $G$, and suppose that
  $\ell \le k\,|s \cap t|$. Then $s \cup t$ and $s \cap t$ are
  $(k,\ell)$-tight in $G$.
\end{theorem}
\begin{proof}
  \leanok
  Since $|s \cup t| \ge |s \cap t|$, the hypothesis
  $\ell \le k\,|s \cap t|$ lets us apply sparsity at $s \cap t$ and at
  $s \cup t$:
  \begin{align*}
    \bigl|\edgesIn{s \cap t}\bigr| + \ell &\le k\,|s \cap t|, \\
    \bigl|\edgesIn{s \cup t}\bigr| + \ell &\le k\,|s \cup t|.
  \end{align*}
  Adding these and using $|s \cup t| + |s \cap t| = |s| + |t|$ gives
  \[
    \bigl|\edgesIn{s \cup t}\bigr|
      + \bigl|\edgesIn{s \cap t}\bigr|
      + 2\ell
    \;\le\; k\,|s| + k\,|t|.
  \]

  On the other hand, $s$ and $t$ are tight, and supermodularity
  (\cref{lem:edgesIn-ncard-add-le}) gives
  \[
    k\,|s| + k\,|t|
    \;=\; \bigl|\edgesIn{s}\bigr| + \bigl|\edgesIn{t}\bigr| + 2\ell
    \;\le\;
    \bigl|\edgesIn{s \cup t}\bigr|
      + \bigl|\edgesIn{s \cap t}\bigr| + 2\ell.
  \]
  So the sum of the two sparsity inequalities above is an equality.
  Hence each of them is an equality, that is, $s \cup t$ and $s \cap t$
  are tight.
\end{proof}

The next two results need no size hypothesis. The first combines two
tight sets $s_1$ and $s_2$ and counts edges that join
$s_1 \setminus s_2$ to $s_2 \setminus s_1$, which are the edges that
the inequality of \cref{lem:edgesIn-ncard-add-le} leaves out. Without
such edges, and under the size hypothesis of
\cref{thm:isTightOn-union-inter}, it gives the union half of that
theorem. The second adds a vertex to a tight set together with $k$
edges joining it to the set. The reverse decomposition of a $(2,3)$-sparse graph at a
vertex of degree three uses both
(\cref{thm:isSparse-exists-typeI-or-typeII-reverse}).

\begin{theorem}[Union of tight sets joined by edges]
  \label{thm:isTightOn-union-with-bonus}
  \lean{SimpleGraph.IsTightOn.union_with_bonus}
  \leanok
  \uses{def:isSparse, def:isTightOn}
  Let $V$ be finite, let $G$ be $(k,\ell)$-sparse, and let
  $s_1, s_2 \subseteq V$ be $(k,\ell)$-tight in $G$. Let $F$ be a set of
  edges of $G$ inside $s_1 \cup s_2$ that lie in neither
  $\edgesIn{s_1}$ nor $\edgesIn{s_2}$, and suppose that
  \[
    \bigl|\edgesIn{s_1 \cap s_2}\bigr| + \ell
    \;\le\; k\,|s_1 \cap s_2| + |F|.
  \]
  Then $s_1 \cup s_2$ is $(k,\ell)$-tight in $G$.
\end{theorem}
\begin{proof}
  \leanok
  \uses{lem:isSparse-isTightOn-of-le, lem:edgesIn-mono,
    lem:edgesIn-inter}
  The sets $\edgesIn{s_1}$, $\edgesIn{s_2}$ and $F$ lie in
  $\edgesIn{s_1 \cup s_2}$ (\cref{lem:edgesIn-mono}), and $F$ is
  disjoint from the other two. Their intersection is
  $\edgesIn{s_1} \cap \edgesIn{s_2} = \edgesIn{s_1 \cap s_2}$
  (\cref{lem:edgesIn-inter}). Counting their union as in the proof of
  \cref{lem:edgesIn-ncard-add-le} gives
  \[
    \bigl|\edgesIn{s_1}\bigr| + \bigl|\edgesIn{s_2}\bigr| + |F|
    \;\le\;
    \bigl|\edgesIn{s_1 \cup s_2}\bigr|
      + \bigl|\edgesIn{s_1 \cap s_2}\bigr|.
  \]

  Since $s_1$ and $s_2$ are tight, the left side equals
  $k\,|s_1| + k\,|s_2| - 2\ell + |F|$. We substitute
  $|s_1| + |s_2| = |s_1 \cup s_2| + |s_1 \cap s_2|$ and bound
  $|\edgesIn{s_1 \cap s_2}|$ by the hypothesis on $F$. This gives
  \[
    k\,|s_1 \cup s_2| \;\le\; \bigl|\edgesIn{s_1 \cup s_2}\bigr| + \ell.
  \]
  Tightness of $s_1$ also gives $\ell \le k\,|s_1| \le k\,|s_1 \cup s_2|$.
  So $s_1 \cup s_2$ is tight by \cref{lem:isSparse-isTightOn-of-le}.
\end{proof}

\begin{theorem}[Adding a vertex with $k$ edges to a tight set]
  \label{thm:isTightOn-insert-vertex-with-edges}
  \lean{SimpleGraph.IsTightOn.insert_vertex_with_edges}
  \leanok
  \uses{def:isSparse, def:isTightOn}
  Let $V$ be finite, let $G$ be $(k,\ell)$-sparse, and let
  $s \subseteq V$ be $(k,\ell)$-tight in $G$. Let $w \in V \setminus s$,
  and let $F$ be a set of at least $k$ edges of $G$ inside
  $s \cup \{w\}$ that are not in $\edgesIn{s}$. Then $s \cup \{w\}$ is
  $(k,\ell)$-tight in $G$.
\end{theorem}
\begin{proof}
  \leanok
  \uses{lem:edgesIn-mono, lem:isSparse-isTightOn-of-le}
  The sets $\edgesIn{s}$ and $F$ are disjoint, and both lie in
  $\edgesIn{s \cup \{w\}}$ (\cref{lem:edgesIn-mono}). Since $s$ is tight
  and $w \notin s$,
  \[
    \bigl|\edgesIn{s \cup \{w\}}\bigr| + \ell
    \;\ge\; \bigl|\edgesIn{s}\bigr| + \ell + |F|
    \;\ge\; k\,|s| + k \;=\; k\,|s \cup \{w\}|.
  \]
  Tightness of $s$ also gives $\ell \le k\,|s| \le k\,|s \cup \{w\}|$.
  So $s \cup \{w\}$ is tight by \cref{lem:isSparse-isTightOn-of-le}.
\end{proof}

\subsection{Splittings and tight sets}
\label{sec:isSparse-typeII-reverse-blocker}

Let $v$ be a vertex of degree three with neighbours $x$, $y$ and $z$,
where $x$ and $y$ are not adjacent. Deleting $v$ and joining $x$ to $y$
by an edge reverses the Type~II move on $xy$ and $z$
(\cref{def:typeII}). Jord\'an~\cite{jordan2016} calls this operation
\emph{splitting} $v$ on the pair $vx$, $vy$, and calls the splitting
\emph{suitable} if the resulting graph is sparse (\S 2.1, p.~44). His
proof of Lemma~2.1.4(b) begins with an observation about a sparse
graph: if a splitting is not suitable, then some critical set, that
is, some $(2,3)$-tight set, contains $x$ and $y$ but not $v$. We prove
this for any vertex $v$ and any vertices $x, y \ne v$, since the count
does not use the degree of $v$.

\begin{lemma}[Tight sets block unsuitable splittings]
  \label{lem:isSparse-typeII-reverse-blocker}
  \lean{SimpleGraph.IsSparse.typeII_reverse_blocker}
  \leanok
  \uses{def:isSparse, def:isTightOn,
    lem:isSparse-isTightOn-of-le}
  Let $V$ be finite, let $G$ be $(2,3)$-sparse, and let $v$, $x$ and
  $y$ be vertices of $G$ with $x, y \ne v$. Let $G - v + xy$ be the
  graph on $V \setminus \{v\}$ obtained from $G$ by deleting $v$ and
  adding the edge $xy$. If $G - v + xy$ is not $(2,3)$-sparse, then
  some set $S \subseteq V \setminus \{v\}$ that contains $x$ and $y$ is
  $(2,3)$-tight in $G$.
\end{lemma}
\begin{proof}
  \leanok
  Since $G - v + xy$ is not $(2,3)$-sparse, some finite set
  $S \subseteq V \setminus \{v\}$ with $|S| \ge 2$ spans more than
  $2|S| - 3$ edges of $G - v + xy$. As $v \notin S$, these edges are the
  edges of $G$ inside $S$, together with $xy$ if both $x$ and $y$ lie
  in $S$.

  If $x \notin S$ or $y \notin S$, then $S$ spans the same edges in
  $G - v + xy$ as in $G$, and this contradicts the sparsity of $G$. So
  $x, y \in S$, and $S$ spans at most one more edge in $G - v + xy$
  than in $G$. Hence $|\edgesIn{S}| + 1 > 2|S| - 3$, that is,
  $2|S| \le |\edgesIn{S}| + 3$. By \cref{lem:isSparse-isTightOn-of-le},
  $S$ is $(2,3)$-tight in $G$.
\end{proof}
\begin{fmlnote}
  \label{fmlnote:graph-minus-vertex}
  The formal statement does not build $G - v + xy$ on the subset
  $V \setminus \{v\}$ of $V$. Its vertex set is a separate copy of
  $V \setminus \{v\}$, which the inclusion map sends into $V$. Two of
  its vertices are adjacent if their images are adjacent in $G$ or are
  $x$ and $y$. The proof therefore first maps the violating set into
  $V$. The inclusion is injective, so the image has as many vertices as
  the set, and the edges of $G$ inside the image correspond one to one
  to the edges of $G - v$ inside the set. The reverse decompositions
  \cref{thm:isSparse-exists-typeI-or-typeII-reverse} and
  \cref{thm:isLaman-exists-typeI-or-typeII-reverse} build $G - v$ and
  $G - v + xy$ in the same way.
\end{fmlnote}

\subsection{Sparse-graph reverse decomposition}
\label{sec:isSparse-reverse-decomposition}

Below is the reverse decomposition of Laman graphs
(\cref{sec:henneberg-decomp}) in its form for $(2,3)$-sparse graphs,
where the vertex may also have degree one. The induction of
\cref{sec:rigidity-matroid-hard-direction} removes the vertex that the
theorem gives at each step. We write $G - v$ for the graph obtained
from $G$ by deleting a vertex $v$, and $G - v + ab$ for the graph
obtained from $G - v$ by adding the edge $ab$, as in
\cref{lem:isSparse-typeII-reverse-blocker}.

Jord\'an~\cite{jordan2016} finds a vertex of degree one, two or three
in his proof of Theorem~2.2.1 (p.~45). He proves the cases of degree
two and three as his Lemma~2.1.4 (p.~44): if $v$ has degree two, then
$G - v$ is sparse, and if $v$ has degree three, then $v$ has a
suitable splitting (\cref{sec:isSparse-typeII-reverse-blocker}). His
proof of the second part uses two facts about critical sets, which are
our $(2,3)$-tight sets. The first is his Lemma~2.1.2, and the part of
it that we use is \cref{thm:isTightOn-union-inter} at
$(k,\ell) = (2,3)$. The second is his Lemma~2.1.3, which says in part
that if three tight sets meet pairwise in exactly one vertex and have
no vertex in common, then their union is tight. We prove the case of
Lemma~2.1.3 that we need within the proof of the theorem.

\begin{theorem}[Sparse-graph reverse decomposition]
  \label{thm:isSparse-exists-typeI-or-typeII-reverse}
  \lean{SimpleGraph.IsSparse.exists_typeI_or_typeII_reverse}
  \leanok
  \uses{def:isSparse}
  Let $V$ be finite, and let $G$ be a $(2,3)$-sparse graph on $V$ with
  at least one edge. Then $G$ has a vertex $v$ for which one of the
  following holds.
  \begin{enumerate}
    \item The vertex $v$ has degree one, with neighbour $a$, and
      $G - v$ is $(2,3)$-sparse.
    \item The vertex $v$ has degree two, with neighbours $a$ and $b$,
      and $G - v$ is $(2,3)$-sparse.
    \item The vertex $v$ has degree three, with neighbours $a$, $b$ and
      $c$, where $a$ and $b$ are not adjacent, and $G - v + ab$ is
      $(2,3)$-sparse.
  \end{enumerate}
  Here $G - v$ is the graph on $V \setminus \{v\}$ obtained from $G$
  by deleting $v$, and $G - v + ab$ is obtained from $G - v$ by adding
  the edge $ab$.
\end{theorem}
\begin{proof}
  \leanok
  \uses{def:isSparse, thm:isTightOn-union-inter,
    lem:isSparse-typeII-reverse-blocker,
    lem:isSparse-exists-nonadj-among-three-neighbors,
    lem:edgesIn-corners, thm:isTightOn-insert-vertex-with-edges,
    thm:isTightOn-union-with-bonus}
  We find $v$ by the argument of
  \cref{lem:isSparse-exists-degree-le-three}, applied to the vertices
  of positive degree. Let $U$ be the set of these vertices. Every edge
  of $G$ has both ends in $U$. So $U$ spans every edge of $G$, and
  $|U| \ge 2$, since $G$ has an edge. Sparsity at $U$ gives
  $|E(G)| + 3 \le 2|U|$. If every vertex in $U$ had degree at least
  four, the handshake identity would give
  $2|E(G)| = \sum_{u \in U} \deg_G(u) \ge 4|U|$, a contradiction. So
  some vertex $v$ has degree one, two or three
  (\texttt{IsSparse.\allowbreak{}exists\_\allowbreak{}one\_\allowbreak{}le\_\allowbreak{}degree\_\allowbreak{}le\_\allowbreak{}three}).

  If $v$ has degree one or two, then a set of vertices of $G - v$
  spans the same edges in $G - v$ as in $G$, so $G - v$ is
  $(2,3)$-sparse. For degree two this is Jord\'an's Lemma~2.1.4(a).

  Let $v$ have degree three, with neighbours $a$, $b$ and $c$. We must
  show that $v$ has a suitable splitting, that is, two neighbours $x$
  and $y$ that are not adjacent and for which $G - v + xy$ is
  $(2,3)$-sparse. Suppose that it has none. We use an observation from
  Jord\'an's proof: no $(2,3)$-tight set $T$ with $v \notin T$ contains
  all three neighbours of $v$
  (\texttt{IsSparse.\allowbreak{}no\_\allowbreak{}isTightOn\_\allowbreak{}excluding\_\allowbreak{}three\_\allowbreak{}neighbors}).
  Otherwise
  $T \cup \{v\}$ spans the edges inside $T$ and the edges $va$, $vb$
  and $vc$, so that
  \[
    \bigl|\edgesIn{T \cup \{v\}}\bigr|
    \;\ge\; \bigl|\edgesIn{T}\bigr| + 3
    \;=\; 2|T| \;=\; 2\,|T \cup \{v\}| - 2,
  \]
  against sparsity. We derive a contradiction by constructing such a
  set $T$.

  If two neighbours of $v$ are not adjacent, the splitting of $v$ on
  them is not suitable, so some tight set that does not contain $v$
  contains both of them (\cref{lem:isSparse-typeII-reverse-blocker}).
  At least one pair of neighbours of $v$ is not adjacent
  (\cref{lem:isSparse-exists-nonadj-among-three-neighbors}). Jord\'an's
  proof takes such a tight set for each of the three pairs; for an
  adjacent pair, the pair itself is such a set. We follow the formal
  proof instead, which distinguishes how many pairs are not adjacent
  (\texttt{IsSparse.\allowbreak{}False\_\allowbreak{}of\_\allowbreak{}pairwise\_\allowbreak{}blocker\_\allowbreak{}or\_\allowbreak{}edge}).

  Suppose first that exactly one pair is not adjacent, say $a$ and $b$
  (\texttt{IsSparse.\allowbreak{}contradiction\_\allowbreak{}one\_\allowbreak{}pair}).
  Let $S$ be a tight set with $a, b \in S$ and $v \notin S$. If
  $c \in S$, let $T = S$. Otherwise the edges $ca$ and $cb$ join $c$ to
  $S$, and $T = S \cup \{c\}$ is tight by
  \cref{thm:isTightOn-insert-vertex-with-edges} with $k = 2$.

  Suppose next that exactly two pairs are not adjacent, say $a, c$ and
  $b, c$, so that $ab$ is an edge
  (\texttt{IsSparse.\allowbreak{}contradiction\_\allowbreak{}two\_\allowbreak{}pair}).
  Let $S_1$ and $S_2$ be tight sets
  that do not contain $v$, with $a, c \in S_1$ and $b, c \in S_2$, and
  let $T = S_1 \cup S_2$. If $|S_1 \cap S_2| \ge 2$, then $T$ is tight
  by \cref{thm:isTightOn-union-inter}. Otherwise
  $S_1 \cap S_2 = \{c\}$, so $a \notin S_2$ and $b \notin S_1$. Then
  the edge $ab$ lies inside $T$ but in neither $\edgesIn{S_1}$ nor
  $\edgesIn{S_2}$. So \cref{thm:isTightOn-union-with-bonus} applies with
  $F = \{ab\}$, where its hypothesis reads $0 + 3 \le 2 \cdot 1 + 1$,
  and $T$ is tight.

  Suppose finally that no two of $a$, $b$ and $c$ are adjacent
  (\texttt{IsSparse.\allowbreak{}contradiction\_\allowbreak{}three\_\allowbreak{}pair}).
  Let
  $S_{ab}$, $S_{ac}$ and $S_{bc}$ be tight sets that do not contain
  $v$, each containing the two vertices in its index. If two of them
  meet in at least two vertices, then their union contains $a$, $b$
  and $c$, and it is tight by \cref{thm:isTightOn-union-inter}; let $T$
  be this union. Otherwise
  \[
    S_{ab} \cap S_{ac} = \{a\}, \qquad
    S_{ab} \cap S_{bc} = \{b\}, \qquad
    S_{ac} \cap S_{bc} = \{c\}.
  \]
  This is the configuration of Jord\'an's Lemma~2.1.3. We show that
  $T = S_{ab} \cup S_{ac} \cup S_{bc}$ is tight by applying
  supermodularity (\cref{lem:edgesIn-ncard-add-le}) twice. The set
  $S_{ab} \cap S_{ac} = \{a\}$ spans no edge, so
  \[
    \bigl|\edgesIn{S_{ab}}\bigr| + \bigl|\edgesIn{S_{ac}}\bigr|
    \;\le\; \bigl|\edgesIn{S_{ab} \cup S_{ac}}\bigr|.
  \]
  The set $(S_{ab} \cup S_{ac}) \cap S_{bc} = \{b, c\}$ spans no edge
  either, since $b$ and $c$ are not adjacent, so
  \[
    \bigl|\edgesIn{S_{ab} \cup S_{ac}}\bigr| + \bigl|\edgesIn{S_{bc}}\bigr|
    \;\le\; \bigl|\edgesIn{T}\bigr|.
  \]
  The three sets are tight, and
  $|T| = |S_{ab}| + |S_{ac}| + |S_{bc}| - 3$. Hence
  \[
    \bigl|\edgesIn{T}\bigr|
    \;\ge\; (2|S_{ab}| - 3) + (2|S_{ac}| - 3) + (2|S_{bc}| - 3)
    \;=\; 2|T| - 3,
  \]
  and $T$ is tight by \cref{lem:isSparse-isTightOn-of-le}.

  In each case $T$ is a tight set that contains $a$, $b$ and $c$ but
  not $v$, which is impossible. So $v$ has a suitable splitting.
\end{proof}
