\section{The combinatorial induction: graph operations and Theorem 4.9}
\label{sec:molecular-induction}

This chapter proves Katoh--Tanigawa's Theorem~4.9
\cite[\S3--\S4]{katohTanigawa2011}: every minimal $0$-dof-graph on at
least two vertices reduces to the two-vertex double edge by a sequence
of two graph operations, splitting-off at a degree-2 vertex and
contraction of a proper rigid subgraph. This is the \emph{combinatorial}
half of the molecular conjecture's proof; the algebraic induction of
\cref{sec:molecular-algebraic-induction} realizes each reduction step at
the rigidity-matrix rank.

Throughout $D = \binom{d+1}{2}$ is the body-bar dimension and
$\tilde G = (D-1) \cdot G$ is the multiplied graph of
\cref{sec:body-hinge}.

\medskip
\noindent\textbf{The argument.}
\Cref{sec:molecular-induction-inherited} sharpens two structural facts
from \cref{sec:molecular-deficiency} (KT Lemmas~3.4 and~3.5) to the form
the reduction needs: a circuit of $M(\tilde G)$ induces a rigid
subgraph, and contracting a rigid subgraph conserves the deficiency.
\Cref{sec:molecular-induction-operations} defines the graph operations
themselves. \Cref{sec:molecular-induction-forest} works out the
degree-of-freedom bookkeeping a splitting-off or a vertex removal
induces: Katoh--Tanigawa's own argument is a forest surgery (Lemma~4.1)
whose base case is over-quantified as stated, and whose proof of the
intended case glosses an unstated balanced-packing assumption
(\cref{rem:kt-lemma-41} works out both the disproof and the repair) ---
but the bookkeeping Theorem~4.9 actually needs is reached instead by a
direct deficiency-count argument (\cref{sec:molecular-induction-dof}),
so the splitting-off surgery, though fully formalized as the repair, and
its edge-splitting inverse (KT Lemma~4.2), never enter the proof of
Theorem~4.9. \Cref{sec:molecular-induction-thm49} assembles Theorem~4.9
itself, by strong induction on the vertex count, dispatching on whether
$G$ has a proper rigid subgraph (contraction) or not (a reducible
degree-2 vertex, via KT Lemma~4.6). The closing sections extract the
finer degree-2 structure the algebraic induction consumes from a graph
with no proper rigid subgraph: an adjacent degree-2 pair at $d = 3$
(\cref{sec:simplicity-and-adjacent-pair}), and, at general $d$, the
length-$d$ chain-or-cycle dichotomy of KT Lemma~4.6.

\subsection{Circuits and the contraction of a rigid subgraph}
\label{sec:molecular-induction-inherited}

This subsection proves Katoh and Tanigawa's Lemmas~3.4
and~3.5~\cite[\S3.2]{katohTanigawa2011}. By KT Lemma~3.4, the ends of
the edges of a circuit of $M(\tilde G)$ induce a rigid subgraph.
\Cref{lem:circuit-rigid} is its first step, and we complete it with KT
eq.~(2.4). By KT Lemma~3.5, contracting a rigid subgraph of a minimal
$k$-dof-graph gives a minimal $k$-dof-graph. We define the contraction,
and prove the deficiency and minimality statements of KT Lemma~3.5 for
the corresponding contraction of the matroid $M(\tilde G)$. We prove the
form for the graph in the algebraic induction, where it is used.

\begin{definition}[The subgraph induced by the ends of a set of edges]
  \label{def:induced-span}
  Let $G$ be a multigraph, let $X$ be a set of edges of $\tilde G$, and
  let $V(X)$ be the set of vertices incident with an edge of $X$. Then
  $G[V(X)]$ is the subgraph of $G$ induced by $V(X)$. Its vertices are
  those of $V(X)$, and its edges are the edges of $G$ with both ends in
  $V(X)$.
\end{definition}

\begin{lemma}[A circuit induces a rigid subgraph; KT Lemma~3.4]
  \label{lem:circuit-induces-rigid}
  \lean{Graph.circuit_induces_isTight, Graph.circuit_induces_isRigidSubgraph}
  \leanok
  \uses{lem:circuit-rigid, def:induced-span, def:rigid-subgraph, thm:def-eq-corank}
  Let $D \ge 1$, let $G$ be a multigraph, let $X$ be a circuit of
  $M(\tilde G)$, and let $e \in X$. Then $|X - e| = D(|V(X)| - 1)$, and
  $X - e$ is the union of $D$ edge-disjoint spanning trees on $V(X)$.
  Moreover, $G[V(X)]$ is a rigid subgraph of $G$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{def:matroid-MG, lem:rank-matroidMG-le, lem:matroid-restrict-subgraph, thm:def-eq-corank}
  For the count we follow Katoh and
  Tanigawa~\cite[p.~658]{katohTanigawa2011}. As $X$ is a circuit, $X - e$
  is independent, so $\tilde G \restriction (X - e)$ is $(D, D)$-sparse
  (\cref{lem:circuit-rigid}). If $X - e$ is not empty, sparsity applied
  to $X - e$ itself gives
  \[
    |X - e| \;\le\; D\,(|V(X - e)| - 1) \;\le\; D\,(|V(X)| - 1).
  \]
  If $X - e$ is empty, the bound holds as $V(X)$ is not empty.
  Conversely, $X$ is dependent, so some nonempty $F \subseteq X$ has
  $|F| > D(|V(F)| - 1)$ (\cref{def:matroid-MG}). If $F$ were a proper
  subset of $X$, it would miss some $f \in X$ and lie in the independent
  set $X - f$, whose sparsity gives $|F| \le D(|V(F)| - 1)$. Hence
  $F = X$, and $|X| > D(|V(X)| - 1)$, that is,
  $|X - e| \ge D(|V(X)| - 1)$. So $|X - e| = D(|V(X)| - 1)$.

  If $X - e$ is empty, then $|V(X)| = 1$, and the $D$ spanning trees are
  empty. Otherwise both inequalities above are equalities, so
  $V(X - e) = V(X)$, and the multigraph on $V(X)$ with the edges $X - e$
  is $(D, D)$-tight. It is connected, by the count in the proof of
  \cref{lem:deficiency-zero-iff-tree-packing}. By
  \cref{cor:k-spanning-trees}, $X - e$ is the union of $D$ edge-disjoint
  spanning trees on $V(X)$.

  Let $H = G[V(X)]$. The ends of each edge of $X$ lie in $V(X)$, so
  $X \subseteq E(\tilde H)$. Hence $X - e$ is independent in
  $M(\tilde G)\,|\,E(\tilde H)$, which is $M(\tilde H)$
  (\cref{lem:matroid-restrict-subgraph}). So
  $\operatorname{rank} M(\tilde H) \ge |X - e| = D(|V(H)| - 1)$, and the
  reverse inequality is \cref{lem:rank-matroidMG-le}. By KT eq.~(2.4)
  (\cref{thm:def-eq-corank}), $\operatorname{def}(\tilde H) = 0$. As $H$
  is a subgraph of $G$, it is a rigid subgraph of $G$.
\end{proof}
\begin{fmlnote}
  The first pin is the count, and the second says that $G[V(X)]$ is a
  rigid subgraph. The partition of $X - e$ into spanning trees is not
  formalized.
\end{fmlnote}

If $G$ has no proper rigid subgraph (\cref{def:rigid-subgraph}), the
lemma shows that every circuit $X$ of $M(\tilde G)$ with
$|V(X)| \ge 2$ has $V(X) = V(G)$. Katoh and Tanigawa use it in this
form in the proofs of their Lemmas~4.5 and~4.8~\cite[pp.~663
and~666]{katohTanigawa2011}.

Katoh and Tanigawa's induction also contracts proper rigid
subgraphs~\cite[p.~659]{katohTanigawa2011}. We define the contraction
for any subgraph, which we contract to one of its vertices.

\begin{definition}[Contracting a subgraph]
  \label{def:rigid-contraction}
  \lean{Graph.rigidContract}
  \leanok
  \uses{def:rigid-subgraph}
  Let $H$ be a subgraph of a multigraph $G$, and let $r$ be a vertex of
  $H$. The \emph{contraction} $G/E(H)$ of $H$ to $r$ is obtained from
  $G$ by deleting the edges of $H$ and identifying every vertex of $H$
  with $r$. Its vertices are $r$ and the vertices of $G$ not in $H$. Its
  edges are the edges of $G$ not in $H$, and each end of such an edge
  that lies in $V(H)$ is replaced by $r$.
\end{definition}

So $G/E(H)$ has $|V(G)| - |V(H)| + 1$ vertices, and
\[
  D\,\bigl(|V(G/E(H))| - 1\bigr) \;=\; D\,\bigl(|V(G)| - |V(H)|\bigr).
\]

Katoh and Tanigawa~\cite[pp.~658--659]{katohTanigawa2011} prove their
Lemma~3.5 with partitions of bases into forests. We work instead with
the contraction of $M(\tilde G)$ by $E(\tilde H)$. For a matroid $M$ on
a set $E$ and a set $T \subseteq E$, the contraction $M/T$ is the
matroid on $E \setminus T$ whose rank function is
$X \mapsto r_M(X \cup T) - r_M(T)$~\cite[\S3.1 and
Proposition~3.1.6]{oxley2011}. The ground set of
$M(\tilde G)/E(\tilde H)$ consists of the copies of the edges of $G$
not in $H$, which are the edges of $\widetilde{G/E(H)}$. We first
compute its rank.

\begin{lemma}[The rank of a rigid subgraph; KT Lemma~3.5, first step]
  \label{lem:rigid-full-rank}
  \lean{Graph.rank_matroidMG_of_isKDof_zero}
  \leanok
  \uses{def:rigid-subgraph, thm:def-eq-corank, def:matroid-MG}
  Let $D \ge 1$, and let $H$ be a rigid subgraph of a multigraph $G$,
  with at least one vertex. Then
  $\operatorname{rank} M(\tilde H) = D(|V(H)| - 1)$.
\end{lemma}
\begin{proof}
  \leanok
  By KT eq.~(2.4) (\cref{thm:def-eq-corank}),
  $\operatorname{rank} M(\tilde H) + \operatorname{def}(\tilde H) =
  D(|V(H)| - 1)$, and $\operatorname{def}(\tilde H) = 0$ as $H$ is
  rigid.
\end{proof}

\begin{lemma}[The rank of $M(\tilde G)/E(\tilde H)$]
  \label{lem:contract-rank-bridge}
  \lean{Graph.contract_matroidMG_rank}
  \leanok
  \uses{def:rigid-contraction, def:matroid-MG, lem:matroid-restrict-subgraph}
  Let $H$ be a subgraph of a multigraph $G$. Then
  \[
    \operatorname{rank}\bigl(M(\tilde G) / E(\tilde H)\bigr)
      + \operatorname{rank} M(\tilde H)
      = \operatorname{rank} M(\tilde G).
  \]
\end{lemma}
\begin{proof}
  \leanok
  A matroid $M$ on $E$ and a set $T \subseteq E$ satisfy
  $\operatorname{rank}(M/T) = \operatorname{rank} M - r_M(T)$, by
  Proposition~3.1.6 of Oxley~\cite{oxley2011} at $X = E \setminus T$.
  Take $M = M(\tilde G)$ and $T = E(\tilde H)$. Then $r_M(T)$ is the
  rank of $M(\tilde G)\,|\,E(\tilde H)$, which is $M(\tilde H)$
  (\cref{lem:matroid-restrict-subgraph}).
\end{proof}

\begin{lemma}[Contracting a rigid subgraph keeps the deficiency; KT Lemma~3.5]
  \label{lem:contract-deficiency-conservation}
  \lean{Graph.contract_matroidMG_deficiency_eq}
  \leanok
  \uses{lem:rigid-full-rank, lem:contract-rank-bridge, thm:def-eq-corank}
  Let $D \ge 1$, and let $H$ be a rigid subgraph of a multigraph $G$,
  with at least one vertex. Then
  \[
    D\bigl(|V(G)| - |V(H)|\bigr) -
    \operatorname{rank}\bigl(M(\tilde G) / E(\tilde H)\bigr)
      = \operatorname{def}(\tilde G).
  \]
\end{lemma}
\begin{proof}
  \leanok
  By \cref{lem:contract-rank-bridge} and \cref{lem:rigid-full-rank},
  \[
    \operatorname{rank}\bigl(M(\tilde G) / E(\tilde H)\bigr) \;=\;
    \operatorname{rank} M(\tilde G) - D\,(|V(H)| - 1).
  \]
  The vertex set of $G$ contains that of $H$, so it is not empty, and
  KT eq.~(2.4) (\cref{thm:def-eq-corank}) gives
  $\operatorname{rank} M(\tilde G) = D(|V(G)| - 1) -
  \operatorname{def}(\tilde G)$. Substituting this into the first
  identity gives the lemma. This is Katoh and Tanigawa's
  count~\cite[p.~658]{katohTanigawa2011}.
\end{proof}

The next lemma is the analogue of \cref{lem:subgraph-minimality} for
the contraction.

\begin{lemma}[Minimality passes to the contraction; KT Lemma~3.5]
  \label{lem:contract-minimality-transport}
  \lean{Graph.contract_minimality_transport}
  \leanok
  \uses{lem:subgraph-minimality, def:k-dof, def:rigid-contraction}
  Let $G$ be a minimal $k$-dof-graph, and let $H$ be a subgraph of $G$.
  Then every base of $M(\tilde G)/E(\tilde H)$ meets $\tilde e$ for every
  edge $e$ of $G$ that is not an edge of $H$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{def:matroid-MG}
  Let $B'$ be a base of $M(\tilde G)/E(\tilde H)$, and let $J$ be a
  maximal subset of $E(\tilde H)$ that is independent in $M(\tilde G)$.
  By Corollary~3.1.8 of Oxley~\cite{oxley2011}, $B' \cup J$ is a base of
  $M(\tilde G)$. Let $e$ be an edge of $G$ that is not an edge of $H$.
  As $G$ is minimal, $B' \cup J$ meets $\tilde e$. The copies of $e$ are
  not in $E(\tilde H)$, which contains $J$, so $B'$ meets $\tilde e$.
  This is Katoh and Tanigawa's argument~\cite[p.~659]{katohTanigawa2011}.
\end{proof}

\begin{lemma}[The contraction of a proper rigid subgraph; KT Lemma~3.5]
  \label{lem:contraction-minimality}
  \lean{Graph.contraction_isMinimalKDof}
  \leanok
  \uses{def:rigid-subgraph, lem:subgraph-minimality, def:k-dof, lem:rigid-full-rank, lem:contract-rank-bridge, lem:contract-deficiency-conservation, lem:contract-minimality-transport}
  Let $D \ge 1$, let $G$ be a minimal $k$-dof-graph, and let $H$ be a
  proper rigid subgraph of $G$. Then
  \[
    D\bigl(|V(G)| - |V(H)|\bigr) -
    \operatorname{rank}\bigl(M(\tilde G) / E(\tilde H)\bigr) = k,
  \]
  and every base of $M(\tilde G)/E(\tilde H)$ meets $\tilde e$ for every
  edge $e$ of $G$ that is not an edge of $H$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:contract-deficiency-conservation, lem:contract-minimality-transport}
  As $H$ is proper, it has at least two vertices, and
  $\operatorname{def}(\tilde G) = k$ as $G$ is a $k$-dof-graph. So the
  identity is \cref{lem:contract-deficiency-conservation}, and the
  second claim is \cref{lem:contract-minimality-transport}.
\end{proof}

Katoh and Tanigawa state their Lemma~3.5 for the graph $G/E(H)$. It
follows from \cref{lem:contraction-minimality} and KT eq.~(2.4) for
$G/E(H)$ if the matroid $M(\widetilde{G/E(H)})$ is
$M(\tilde G)/E(\tilde H)$. The two matroids have the same ground set,
and they are equal because $H$ is rigid. The reason is Katoh and
Tanigawa's (3.1)~\cite[p.~658]{katohTanigawa2011}. Let a base of
$M(\tilde G)$ contain a base of $M(\tilde H)$, and let it be the union
of $D$ disjoint forests. Then each of these forests meets $E(\tilde H)$
in a spanning tree on $V(H)$. The algebraic induction applies its
induction hypothesis to $G/E(H)$, and we prove the identity there, for
$D \ge 2$ (\cref{lem:rigidContract-isMinimalKDof}).

\subsection{Removal, splitting off and edge-splitting}
\label{sec:molecular-induction-operations}

The other operation of the induction acts at a vertex of degree two.
Katoh and Tanigawa define splitting off at a vertex $v$ as the removal
of $v$ followed by the insertion of new edges between neighbours of
$v$, and they use it only when $v$ has degree two~\cite[\S4.1,
p.~660]{katohTanigawa2011}. They also use its inverse, edge-splitting
(p.~660), and the removal of $v$ alone (p.~662).

\begin{definition}[Removal, splitting off and edge-splitting]
  \label{def:graph-operations}
  \lean{Graph.removeVertex, Graph.splitOff, Graph.edgeSplit, Graph.endsOf}
  \leanok
  \uses{def:k-dof}
  Let $G$ be a multigraph, and let $v$ be a vertex of $G$ of degree two,
  whose two edges join $v$ to vertices $a$ and $b$.
  \begin{itemize}
    \item The \emph{removal} of $v$ gives $G_v = G - v$, obtained from
      $G$ by deleting $v$ and its two edges.
    \item The \emph{splitting off} at $v$ gives $G_v^{ab}$, obtained from
      $G$ by deleting $v$ and its two edges and adding a new edge $ab$.
  \end{itemize}
  Let $H$ be a multigraph with an edge $ab$. The \emph{edge-splitting}
  along $ab$ gives $H_{ab}^v$, obtained from $H$ by deleting $ab$ and
  adding a new vertex $v$ and two new edges $va$ and $vb$.
\end{definition}
\begin{fmlnote}
  \label{fmlnote:graph-operations}
  In the formalization the edges of a multigraph are elements of a fixed
  set of labels. An operation that adds an edge takes the label of the
  new edge as an argument: the label $e_0$ of the edge $ab$ of
  $G_v^{ab}$, and the labels of $va$ and $vb$ in $H_{ab}^v$. The formal
  splitting off deletes every edge with the label $e_0$ before it adds
  $ab$. So it gives $G_v^{ab}$ when no edge of $G - v$ has the label
  $e_0$, for instance when $e_0 \notin E(G)$. A label outside $E(G)$
  exists when there are enough labels (\cref{rem:fresh-edge-supply}).
  The fourth pin fixes, for each edge of $G$, an ordered pair of its
  ends, the \emph{endpoint choice} of $G$. The realizations of Cases~I
  and~II of the algebraic induction take the ends of the edges of $G$
  from it (\cref{fmlnote:panel-hinge-nondegenerate}).
\end{fmlnote}

Splitting off the new vertex $v$ of $H_{ab}^v$ gives back $H$.

\subsection{Forest surgery: KT Lemmas~4.1 and~4.2}
\label{sec:molecular-induction-forest}

Let $v$ be a vertex of $G$ of degree two, whose edges $e_a$ and $e_b$
join it to vertices $a, b \neq v$. For an edge $e$ of $G$, let
$\tilde e$ be the set of its $D - 1$ copies in $\tilde G$
(\cref{def:k-dof}). The edges of $\tilde G$ incident with $v$ are the
$2(D - 1)$ edges of $\tilde e_a \cup \tilde e_b$. We write
$\widetilde{ab}$ for the set of the $D - 1$ copies of the new edge $ab$
of $G_v^{ab}$. Katoh and Tanigawa~\cite[p.~660]{katohTanigawa2011}
compare the independent sets of $M(\tilde G)$ and of
$M(\widetilde{G_v^{ab}})$ in their Lemma~4.1. They prove it by a
\emph{forest surgery}: they partition an independent set of
$M(\tilde G)$ into $D$ forests of $\tilde G$, and change each forest at
$v$. Their proof assumes that every forest of the partition has an edge
incident with $v$.

We discuss the lemma and this assumption in \cref{rem:kt-lemma-41}, and
then prove the lemma for bases in three steps. The independent sets of
$M(\tilde G)$ are the unions of $D$ forests
(\cref{lem:forest-packing-decomp}). A base has a partition into $D$
forests each of which has an edge incident with $v$
(\cref{lem:balanced-forest-packing}). When $a \neq b$, the surgery on
such a partition gives the two conclusions of the lemma
(\cref{lem:forest-surgery-count}). The second step uses KT Lemma~4.4
(\cref{lem:removal-deficiency}). We prove that lemma in
\cref{sec:molecular-induction-dof} by comparing the values of partitions
of $V(G)$, without forests. KT Lemma~4.2 (\cref{lem:edge-splitting}),
the converse for edge-splitting, ends the subsection.

\begin{remark}[On KT Lemma~4.1]
  \label{rem:kt-lemma-41}
  KT Lemma~4.1~\cite[p.~660]{katohTanigawa2011} states that for any
  independent set $I$ of $M(\tilde G)$ there is an independent set $I'$
  of $M(\widetilde{G_v^{ab}})$ with $|I'| = |I| - D$ and
  $|\widetilde{ab} \cap I'| < D - 1$. The arXiv version of the paper
  (arXiv:0902.0236, 2009, Lemma~5.1, p.~11) has essentially the same
  statement and proof. We separate three points.
  \begin{enumerate}
    \item \emph{The statement holds only for bases.} If $|I| < D$, for
      instance if $I = \emptyset$, then no set $I'$ has
      $|I'| = |I| - D$ (\cref{ex:kt-41-overquantified}). Katoh and
      Tanigawa apply the lemma to a base, in the proof of their
      Lemma~4.3, and we read it for bases.
    \item \emph{The proof assumes a balanced partition.} Let
      $F_1, \dots, F_D$ be a partition of $I$ into forests of $\tilde G$
      (\cref{lem:forest-packing-decomp}). A forest has at most two edges
      incident with $v$, at most one in $\tilde e_a$ and one in
      $\tilde e_b$, since two copies of an edge form a cycle. Let
      $h = |(\tilde e_a \cup \tilde e_b) \cap I|$, and let $h'$ be the
      number of forests with two edges incident with $v$. Katoh and
      Tanigawa note that $2h' + (D - h') = h$, and they turn each $F_i$
      into a forest $F'_i$ of $\widetilde{G_v^{ab}}$ with
      $|F'_i| = |F_i| - 1$. Both steps assume that every $F_i$ has an
      edge incident with $v$. We call such a partition \emph{balanced at
      $v$}. The proof does not show that $I$ has a balanced partition,
      and if $h < D$, it has none.

      We show that a base $B$ has one. First, $B$ contains at least $D$
      edges incident with $v$ (\cref{lem:base-vfiber-count}). Now let
      $F_j$ be a forest of a partition of $B$ with no edge incident with
      $v$. The other $D - 1$ forests contain at least $D$ such edges, so
      one of them contains two. Moving one of these two to $F_j$ gives a
      new partition of $B$ into forests, as the moved edge is a
      pendant edge at $v$ of the new $F_j$
      (\cref{lem:acyclic-insert-vfiber}).
      Each move lowers the number of forests with no edge incident with
      $v$, so repeated moves give a balanced partition
      (\cref{lem:balanced-forest-packing}). So, for bases, this step is a
      gap in the proof and not an error.

      On a balanced partition the surgery goes as Katoh and Tanigawa
      describe it. A forest with two edges $p_a \in \tilde e_a$ and
      $p_b \in \tilde e_b$ replaces them by one copy of $ab$
      (\cref{lem:reroute-cycle-lift}). A forest with one edge incident
      with $v$ loses it and gains nothing
      (\cref{lem:reroute-vfree-transport}). Each forest loses one edge,
      so $|I'| = |I| - D$: the $h$ edges incident with $v$ leave, and
      $h'$ copies of $ab$ enter. This count is the equation
      $2h' + (D - h') = h$. As $h \le 2(D - 1)$, it gives
      $h' \le D - 2$. So the $D - 1$ copies of $ab$ suffice for distinct
      forests to take distinct copies, and
      $|\widetilde{ab} \cap I'| = h' < D - 1$
      (\cref{lem:forest-surgery-count}). The count and KT eq.~(2.4)
      (\cref{thm:def-eq-corank}) give
      $\operatorname{def}(\widetilde{G_v^{ab}}) \le
      \operatorname{def}(\tilde G)$, which is
      \cref{lem:forest-surgery-split}.
    \item \emph{What the induction uses.} The proof of KT Theorem~4.9
      needs KT Lemma~4.1 only through its consequence
      $\operatorname{def}(\widetilde{G_v^{ab}}) \le
      \operatorname{def}(\tilde G)$, the upper bound of KT
      Lemma~4.3(i). We prove the deficiency bounds of KT Lemmas~4.3(i)
      and~4.4 in \cref{sec:molecular-induction-dof}, by comparing the
      values of partitions of $V(G)$, without forests. These bounds
      (\cref{lem:dof-tracking}) are what our proof of KT Theorem~4.9
      uses. The surgery also gives the second conclusion of KT Lemma~4.1,
      $|\widetilde{ab} \cap I'| < D - 1$. From it we derive one half of
      KT Lemma~4.3(ii) (\cref{lem:splitoff-kdof-criterion}), which the
      algebraic induction uses in Cases~II and~III. The other half uses
      KT Lemma~4.2, the converse for edge-splitting, which adds edges at
      the new vertex and needs no balanced partition
      (\cref{lem:edge-splitting}).
  \end{enumerate}
\end{remark}

\begin{example}[KT Lemma~4.1 fails for $I = \emptyset$]
  \label{ex:kt-41-overquantified}
  \lean{Graph.kt_lemma_41_overquantified}
  \leanok
  \uses{def:matroid-MG, rem:kt-lemma-41}
  Let $D \ge 1$. The empty set is independent in $M(\tilde G)$, and no
  set $I'$ has $|I'| + D = |\emptyset| = 0$. So no independent set $I'$
  of $M(\widetilde{G_v^{ab}})$ has $|I'| = |\emptyset| - D$, and KT
  Lemma~4.1, stated for every independent set, fails at
  $I = \emptyset$.
\end{example}

\begin{lemma}[The independent sets of $M(\tilde G)$ are the unions of $D$ forests]
  \label{lem:forest-packing-decomp}
  \lean{Graph.matroidMG_indep_iff_exists_forest_packing}
  \leanok
  \uses{def:matroid-MG}
  Let $G$ be a multigraph. A set $I$ of edges of $\tilde G$ is
  independent in $M(\tilde G)$ if and only if
  $I = F_1 \cup \dots \cup F_D$ for some forests $F_1, \dots, F_D$ of
  $\tilde G$.
\end{lemma}
\begin{proof}
  \leanok
  By \cref{def:matroid-MG}, $M(\tilde G)$ is the union of $D$ copies of
  the cycle matroid of $\tilde G$. The independent sets of the cycle
  matroid are the forests (\cref{sec:body-bar-matroid-union}). The
  independent sets of a union of matroids are the unions of independent
  sets of the matroids (\cref{lem:union-indep-iff}).
\end{proof}

A subset of a forest is a forest. So removing from each $F_i$ the edges
of $F_1, \dots, F_{i-1}$ gives pairwise disjoint forests with the same
union. Hence every independent set of $M(\tilde G)$ has a partition
into $D$ forests. Katoh and Tanigawa begin their proofs of Lemmas~4.1
and~4.2 with such a partition.

\begin{lemma}[A base contains at least $D$ edges at a vertex of degree two]
  \label{lem:base-vfiber-count}
  \lean{Graph.isBase_vfiber_ncard_ge}
  \leanok
  \uses{def:graph-operations, def:matroid-MG, thm:def-eq-corank, lem:removal-deficiency, rem:kt-lemma-41}
  Let $D \ge 2$, and let $G$ be a multigraph with a vertex $v$ of degree
  two, whose edges $e_a$ and $e_b$ join $v$ to vertices $a, b \neq v$.
  Then every base $B$ of $M(\tilde G)$ contains at least $D$ of the
  $2(D - 1)$ edges of $\tilde G$ incident with $v$:
  \[
    \bigl|\, B \cap (\tilde e_a \cup \tilde e_b)\,\bigr| \;\ge\; D,
  \]
  where $\tilde e$ is the set of the $D - 1$ copies of $e$ in
  $\tilde G$.
\end{lemma}
\begin{proof}
  \leanok
  Let $G_v = G - v$ (\cref{def:graph-operations}), and let
  $B_0 = B \setminus (\tilde e_a \cup \tilde e_b)$. An edge of $B_0$ is
  a copy of an edge of $G$ other than $e_a$ and $e_b$, so it is not
  incident with $v$. Hence $B_0$ is a set of edges of
  $\widetilde{G_v}$. It is independent in $M(\tilde G)$, and hence in
  the restriction $M(\widetilde{G_v})$ of $M(\tilde G)$
  (\cref{lem:matroid-restrict-subgraph}). So
  $|B_0| \le \operatorname{rank} M(\widetilde{G_v})$.

  The multigraph $G_v$ has the vertex $a$, so KT eq.~(2.4)
  (\cref{thm:def-eq-corank}) applies to $G$ and to $G_v$:
  \[
    |B| = D\,(|V(G)| - 1) - \operatorname{def}(\tilde G), \qquad
    \operatorname{rank} M(\widetilde{G_v}) = D\,(|V(G)| - 2) -
    \operatorname{def}(\widetilde{G_v}).
  \]
  Hence
  \[
    \bigl|\, B \cap (\tilde e_a \cup \tilde e_b)\,\bigr| \;=\;
    |B| - |B_0| \;\ge\;
    D + \operatorname{def}(\widetilde{G_v}) - \operatorname{def}(\tilde G)
    \;\ge\; D.
  \]
  The last step is KT Lemma~4.4 (\cref{lem:removal-deficiency}), which
  uses $D \ge 2$: it gives
  $\operatorname{def}(\tilde G) \le \operatorname{def}(\widetilde{G_v})$.
\end{proof}

\begin{lemma}[A forest with no edge at $v$ accepts an edge at $v$]
  \label{lem:acyclic-insert-vfiber}
  \lean{Graph.acyclicSet_insert_vfiber_of_not_inc}
  \leanok
  \uses{def:matroid-MG, rem:kt-lemma-41, lem:base-vfiber-count}
  Let $G$ be a multigraph with a vertex $v$ of degree two. Let $F$ be a
  forest of $\tilde G$ none of whose edges is incident with $v$, and let
  $x$ be an edge of $\tilde G$ that joins $v$ to a vertex $w \neq v$.
  Then $F \cup \{x\}$ is a forest of $\tilde G$.
\end{lemma}
\begin{proof}
  \leanok
  In $\tilde G \restriction (F \cup \{x\})$ the only edge incident with
  $v$ is $x$, and $x$ is not a loop. So deleting $x$ leaves $v$
  isolated, and in particular separates $v$ from $w$: the edge $x$ is a
  bridge. A bridge lies on no cycle, and $F$ contains no cycle, so
  $F \cup \{x\}$ contains none. The argument uses only that no edge of
  $F$ is incident with $v$, not the degree of $v$.
\end{proof}

\begin{lemma}[Moving an edge at $v$ into a forest with none]
  \label{lem:packing-move}
  \lean{Graph.exists_packing_move_of_not_inc}
  \leanok
  \uses{def:matroid-MG, lem:acyclic-insert-vfiber, lem:base-vfiber-count}
  Let $G$ be a multigraph with a vertex $v$ of degree two, and let
  $F_1, \dots, F_D$ be forests of $\tilde G$ with union $I$. Suppose
  that no edge of $F_j$ is incident with $v$, and let $x \in I$ be an
  edge of $\tilde G$ that joins $v$ to a vertex $w \neq v$. Then there
  are forests $F'_1, \dots, F'_D$ of $\tilde G$ with union $I$ such that
  $x \in F'_j$.
\end{lemma}
\begin{proof}
  \leanok
  Let $F'_j = F_j \cup \{x\}$, and let $F'_i = F_i \setminus \{x\}$ for
  $i \neq j$. Then $F'_j$ is a forest by
  \cref{lem:acyclic-insert-vfiber}, and each $F'_i$ with $i \neq j$ is
  a forest as a subset of $F_i$. The union is still $I$, since $x \in I$
  moves from the other forests to $F'_j$.
\end{proof}

\begin{lemma}[A base has a balanced partition into forests]
  \label{lem:balanced-forest-packing}
  \lean{Graph.exists_balanced_forest_packing}
  \leanok
  \uses{def:matroid-MG, lem:forest-packing-decomp, lem:base-vfiber-count, lem:acyclic-insert-vfiber, lem:packing-move, rem:kt-lemma-41}
  Let $D \ge 2$, and let $G$ be a multigraph with a vertex $v$ of degree
  two, whose edges join $v$ to vertices $a, b \neq v$. Then every base
  $B$ of $M(\tilde G)$ is the union of $D$ pairwise disjoint forests of
  $\tilde G$, each of which has an edge incident with $v$.
\end{lemma}
\begin{proof}
  \leanok
  By \cref{lem:forest-packing-decomp}, $B$ is the union of $D$ forests,
  which we make pairwise disjoint as above. We induct on the number of
  these forests with no edge incident with $v$. If there is none, we are
  done. Otherwise let $F_j$ be one. By \cref{lem:base-vfiber-count}, $B$
  contains at least $D$ edges incident with $v$, and they lie in the
  $D - 1$ forests other than $F_j$. So one of these, $F_i$, contains two
  of them. Let $x$ be one of the two. It joins $v$ to $a$ or to $b$.

  Move $x$ from $F_i$ to $F_j$, as in the proof of
  \cref{lem:packing-move}. The new $F_j$ is a forest
  (\cref{lem:acyclic-insert-vfiber}), and the forests are still pairwise
  disjoint with union $B$, since $x$ lay in $F_i$ only. Now $F_j$ has an
  edge incident with $v$, and $F_i$ keeps its other one. So the number
  of forests with no edge incident with $v$ drops by one, and the
  induction hypothesis gives the partition.
\end{proof}

We now perform the surgery on a balanced partition of a base.

\begin{lemma}[Removing the edges at $v$ from a forest]
  \label{lem:reroute-vfree-transport}
  \lean{Graph.isAcyclicSet_splitOff_of_diff_fiberAtVertex}
  \leanok
  \uses{def:graph-operations, def:matroid-MG}
  Let $G$ be a multigraph with a vertex $v$ of degree two, whose edges
  join $v$ to vertices $a, b \neq v$, and let $F$ be a forest of
  $\tilde G$. Then the edges of $F$ that are not incident with $v$ form a
  forest of $\widetilde{G_v^{ab}}$.
\end{lemma}
\begin{proof}
  \leanok
  The edges of $\tilde G$ not incident with $v$ are the edges of
  $\widetilde{G_v}$, where $G_v = G - v$ (\cref{def:graph-operations}).
  As $G_v$ is a subgraph of $G_v^{ab}$, they are edges of
  $\widetilde{G_v^{ab}}$ with the same ends. Those of them in $F$
  contain no cycle, since $F$ contains none.
\end{proof}

\begin{lemma}[Replacing the two edges at $v$ of a forest by a copy of $ab$]
  \label{lem:reroute-cycle-lift}
  \lean{Graph.isAcyclicSet_splitOff_reroute}
  \leanok
  \uses{def:graph-operations, def:matroid-MG, lem:reroute-vfree-transport}
  Let $G$ be a multigraph with a vertex $v$ of degree two, whose edges
  $e_a$ and $e_b$ join $v$ to vertices $a, b \neq v$. Let $F$ be a forest
  of $\tilde G$ whose edges incident with $v$ are $p_a \in \tilde e_a$ and
  $p_b \in \tilde e_b$, and let $r \in \widetilde{ab}$. Then
  $(F \setminus \{p_a, p_b\}) \cup \{r\}$ is a forest of
  $\widetilde{G_v^{ab}}$.
\end{lemma}
\begin{proof}
  \leanok
  Let $F_0 = F \setminus \{p_a, p_b\}$, the set of edges of $F$ not
  incident with $v$. By \cref{lem:reroute-vfree-transport}, $F_0$ is a
  forest of $\widetilde{G_v^{ab}}$, so a cycle $C$ of $F_0 \cup \{r\}$
  passes through $r$. Then $C - r$ is a path from $a$ to $b$ in $F_0$. Its
  edges have the same ends in $\tilde G$, and it does not pass through
  $v$. So together with $p_a$ and $p_b$ it forms a cycle of $\tilde G$ in
  $F$, which is impossible. Hence $F_0 \cup \{r\}$ contains no cycle.
\end{proof}

\begin{lemma}[The surgery on a balanced partition]
  \label{lem:forest-surgery-count}
  \lean{Graph.forest_surgery_count}
  \leanok
  \uses{def:graph-operations, def:matroid-MG, lem:balanced-forest-packing, lem:reroute-cycle-lift, lem:reroute-vfree-transport}
  Let $D \ge 2$, and let $G$ be a multigraph with a vertex $v$ of degree
  two, whose edges $e_a$ and $e_b$ join $v$ to distinct vertices
  $a, b \neq v$. Let $F_1, \dots, F_D$ be pairwise disjoint forests of
  $\tilde G$, each of which has an edge incident with $v$, and let $I$ be
  their union. Then there are pairwise disjoint forests
  $F'_1, \dots, F'_D$ of $\widetilde{G_v^{ab}}$ whose union $I'$ is
  independent in $M(\widetilde{G_v^{ab}})$ and satisfies
  \[
    |I'| = |I| - D \qquad\text{and}\qquad
    |\widetilde{ab} \cap I'| < D - 1.
  \]
\end{lemma}
\begin{proof}
  \leanok
  Two copies of an edge form a cycle, so each $F_i$ has at most one edge
  in $\tilde e_a$ and at most one in $\tilde e_b$. These are its only
  edges incident with $v$, and it has at least one, so it has one or two.
  Let $h'$ be the number of forests with two, and let
  $h = |(\tilde e_a \cup \tilde e_b) \cap I|$. As the $F_i$ are disjoint,
  $h = 2h' + (D - h') = D + h'$. As
  $|\tilde e_a \cup \tilde e_b| = 2(D - 1)$, also $h \le 2(D - 1)$, so
  $h' \le D - 2$.

  Since $\widetilde{ab}$ has $D - 1 > h'$ elements, we can choose a
  different copy $r_i \in \widetilde{ab}$ for each forest $F_i$ with two
  edges incident with $v$. For such an $i$, let $F'_i$ be obtained from
  $F_i$ by replacing these two edges by $r_i$. It is a forest of
  $\widetilde{G_v^{ab}}$ by \cref{lem:reroute-cycle-lift}. For every
  other $i$, let $F'_i$ be $F_i$ without its edge incident with $v$, a
  forest of $\widetilde{G_v^{ab}}$ by \cref{lem:reroute-vfree-transport}.
  In both cases $|F'_i| = |F_i| - 1$.

  The edges of the $F'_i$ other than the $r_i$ lie in the disjoint sets
  $F_i$, and none of them is a copy of $ab$. So the $F'_i$ are pairwise
  disjoint, and their union $I'$ has
  $|I'| = \sum_i |F_i| - D = |I| - D$. Its copies of $ab$ are the $h'$
  copies $r_i$, so $|\widetilde{ab} \cap I'| = h' < D - 1$. By
  \cref{lem:forest-packing-decomp}, $I'$ is independent in
  $M(\widetilde{G_v^{ab}})$, as the union of $D$ forests.
\end{proof}

\begin{lemma}[Splitting off does not increase the deficiency; KT Lemma~4.1]
  \label{lem:forest-surgery-split}
  \lean{Graph.forest_surgery_split}
  \leanok
  \uses{def:graph-operations, def:matroid-MG}
  Let $D \ge 2$, and let $G$ be a multigraph with a vertex $v$ of degree
  two, whose edges join $v$ to distinct vertices $a, b \neq v$. Then
  \[
    \operatorname{def}(\widetilde{G_v^{ab}}) \;\le\;
    \operatorname{def}(\tilde G).
  \]
\end{lemma}
\begin{proof}
  \leanok
  \uses{thm:def-eq-corank, lem:balanced-forest-packing, lem:reroute-cycle-lift, lem:reroute-vfree-transport, lem:forest-surgery-count}
  Let $B$ be a base of $M(\tilde G)$. By
  \cref{lem:balanced-forest-packing}, $B$ is the union of $D$ pairwise
  disjoint forests of $\tilde G$, each of which has an edge incident with
  $v$. By \cref{lem:forest-surgery-count}, $M(\widetilde{G_v^{ab}})$ has
  an independent set $I'$ with $|I'| = |B| - D$. Hence
  \[
    \operatorname{rank} M(\widetilde{G_v^{ab}}) \;\ge\; |B| - D.
  \]

  The vertices of $G_v^{ab}$ are the $|V(G)| - 1$ vertices of $G$ other
  than $v$, among them $a$. So KT eq.~(2.4) (\cref{thm:def-eq-corank})
  applies to $G_v^{ab}$ and to $G$, and gives
  \[
    \operatorname{def}(\widetilde{G_v^{ab}}) \;=\;
    D\,(|V(G)| - 2) - \operatorname{rank} M(\widetilde{G_v^{ab}})
    \;\le\; D\,(|V(G)| - 1) - |B| \;=\; \operatorname{def}(\tilde G).
  \]
  Katoh and Tanigawa~\cite[p.~661]{katohTanigawa2011} derive their
  eq.~(4.1), $\operatorname{def}(\widetilde{G_v^{ab}}) \le k$ for a
  minimal $k$-dof-graph $G$, from their Lemma~4.1 in the same way.
\end{proof}

The bound of \cref{lem:forest-surgery-split} is the upper bound of KT
Lemma~4.3(i). We prove the same bound by comparing the values of
partitions (\cref{lem:splitoff-deficiency}), for every $D \ge 1$ and
without assuming $a \neq b$. Here we restate it as a consequence of the
forest surgery.

\begin{corollary}[The upper bound of KT Lemma~4.3(i) by forest surgery]
  \label{cor:forest-surgery-deficiency}
  \lean{Graph.splitOff_deficiency_le_of_forest_surgery}
  \leanok
  \uses{lem:forest-surgery-split, lem:splitoff-deficiency}
  Let $D \ge 2$, and let $G$ be a multigraph with a vertex $v$ of degree
  two, whose edges join $v$ to distinct vertices $a, b \neq v$. Then
  \[
    \operatorname{def}(\widetilde{G_v^{ab}}) \;\le\;
    \operatorname{def}(\tilde G).
  \]
\end{corollary}
\begin{proof}
  \leanok
  This is \cref{lem:forest-surgery-split}.
\end{proof}

We turn to KT Lemma~4.2. Let $H$ be a multigraph with an edge $ab$, and
let $H_{ab}^v$ be obtained from $H$ by edge-splitting along $ab$
(\cref{def:graph-operations}). Katoh and
Tanigawa~\cite[pp.~660--661]{katohTanigawa2011} partition an independent
set $I'$ of $M(\tilde H)$ into $D$ forests and change each forest at the
new vertex $v$. When no forest contains a copy of $ab$, each forest
gains one edge at $v$.

\begin{lemma}[Edge-splitting and forests avoiding $ab$; KT Lemma~4.2(i)]
  \label{lem:forest-surgery-unsplit}
  \uses{def:graph-operations, lem:forest-packing-decomp}
  Let $H$ be a multigraph with an edge $ab$, and let $H_{ab}^v$ be
  obtained from $H$ by edge-splitting along $ab$. Let
  $F'_1, \dots, F'_D$ be pairwise disjoint forests of $\tilde H$, none of
  which contains a copy of $ab$, and let $I'$ be their union. Then there
  are pairwise disjoint forests $F_1, \dots, F_D$ of
  $\widetilde{H_{ab}^v}$ with $F'_i \subseteq F_i$ and
  $|F_i| = |F'_i| + 1$ for each $i$. In particular, their union $I$ has
  $|I| = |I'| + D$.
\end{lemma}
\begin{proof}
  The forests $F'_i$ contain no copy of $ab$, so they are sets of edges of
  $\widetilde{H_{ab}^v}$ with the same ends as in $\tilde H$, and none of
  their edges is incident with $v$. Let $F_i = F'_i \cup \{(va, i)\}$
  for $i \le D - 1$, and $F_D = F'_D \cup \{(vb, 1)\}$, where $(va, i)$
  is the $i$th copy of the new edge $va$. Each added edge joins $v$ to a
  vertex of $H$ and is the only edge of $F_i$ at $v$, so it lies on no
  cycle of $F_i$. Hence $F_i$ is a forest. The added edges are distinct
  and are not edges of $\tilde H$, so the $F_i$ are pairwise disjoint.
  This is Katoh and Tanigawa's construction when no forest contains a
  copy of $ab$.
\end{proof}

Katoh and Tanigawa~\cite[p.~662]{katohTanigawa2011} use this case to
prove their Lemma~4.4, which we prove without forests
(\cref{lem:removal-deficiency}). In the form of independent sets, the
lemma above is the case $h' = 0$ of KT Lemma~4.2(i) below
(\cref{lem:edge-splitting}), with $G = H_{ab}^v$, for $D \ge 2$ and
$a \neq b$. So we do not formalize it separately.

We state KT Lemma~4.2 for a vertex $v$ of $G$ of degree two, whose edges
$e_a$ and $e_b$ join $v$ to distinct vertices $a, b \neq v$. Then $G$ is
obtained from $G_v^{ab}$ by edge-splitting along $ab$, so the lemma
applies to $H = G_v^{ab}$. In its proof, a forest of
$\widetilde{G_v^{ab}}$ that contains a copy of $ab$ loses it and gains a
copy of $e_a$ and a copy of $e_b$, and some of the other forests gain a
copy of $e_a$ or of $e_b$. The next two lemmas show that each change
gives a forest of $\tilde G$.

\begin{lemma}[Replacing a copy of $ab$ in a forest by the two edges at $v$]
  \label{lem:reverse-reroute-cycle-lift}
  \lean{Graph.isAcyclicSet_mulTilde_of_splitOff_reroute}
  \leanok
  \uses{def:graph-operations, def:matroid-MG, lem:reroute-cycle-lift}
  Let $G$ be a multigraph with a vertex $v$ of degree two, whose edges
  $e_a$ and $e_b$ join $v$ to distinct vertices $a, b \neq v$. Let $F'$
  be a forest of $\widetilde{G_v^{ab}}$ that contains a copy $r$ of $ab$,
  and let $p_a \in \tilde e_a$ and $p_b \in \tilde e_b$. Then
  $(F' \setminus \{r\}) \cup \{p_a, p_b\}$ is a forest of $\tilde G$.
\end{lemma}
\begin{proof}
  \leanok
  We reverse the argument of \cref{lem:reroute-cycle-lift}. Two copies
  of $ab$ form a cycle, so $r$ is the only copy of $ab$ in $F'$. Hence the
  edges of $F_0 = F' \setminus \{r\}$ are edges of $\widetilde{G_v}$.
  They have the same ends in $\tilde G$ as in $\widetilde{G_v^{ab}}$,
  and none of them is incident with $v$. Let $C$ be a cycle of $\tilde G$
  in $F_0 \cup \{p_a, p_b\}$. If $C$ does not pass through $v$, then $C$
  lies in $F_0$, and it is a cycle of $\widetilde{G_v^{ab}}$ in $F'$.
  Otherwise $C$ uses both $p_a$ and $p_b$, the only edges of the set at
  $v$. Then the rest of $C$ is a path from $b$ to $a$ in $F_0$, and with
  $r$ it forms a cycle of $\widetilde{G_v^{ab}}$ in $F'$. Both cases are
  impossible, as $F'$ is a forest.
\end{proof}

\begin{lemma}[Adding an edge at $v$ to a forest with no copy of $ab$]
  \label{lem:reverse-pendant-insert}
  \lean{Graph.isAcyclicSet_mulTilde_insert_vfiber_of_splitOff}
  \leanok
  \uses{def:graph-operations, def:matroid-MG, lem:reroute-vfree-transport, lem:acyclic-insert-vfiber}
  Let $G$ be a multigraph with a vertex $v$ of degree two, whose edges
  join $v$ to vertices $a, b \neq v$. Let $F$ be a forest of
  $\widetilde{G_v^{ab}}$ that contains no copy of $ab$, and let $x$ be an
  edge of $\tilde G$ that joins $v$ to a vertex $w \neq v$. Then
  $F \cup \{x\}$ is a forest of $\tilde G$.
\end{lemma}
\begin{proof}
  \leanok
  As $F$ contains no copy of $ab$, its edges are edges of
  $\widetilde{G_v}$. As in the proof of
  \cref{lem:reroute-vfree-transport}, they have the same ends in
  $\tilde G$ as in $\widetilde{G_v^{ab}}$. So $F$ is a forest of
  $\tilde G$, and none of its edges is incident with $v$. By
  \cref{lem:acyclic-insert-vfiber}, $F \cup \{x\}$ is a forest of
  $\tilde G$.
\end{proof}

We add to Katoh and Tanigawa's statement the equation
$I \setminus (\tilde e_a \cup \tilde e_b) = I' \setminus \widetilde{ab}$,
which their proof gives. Their proof of Lemma~4.3 relies on it: a
base of $M(\widetilde{G_v^{ab}})$ with no copy of an edge $e \neq ab$
extends to a base of $M(\tilde G)$ with no copy of
$e$~\cite[p.~661]{katohTanigawa2011}. As $|\widetilde{ab}| = D - 1$, the
two cases of the lemma cover every $I'$.

\begin{lemma}[Independent sets under edge-splitting; KT Lemma~4.2]
  \label{lem:edge-splitting}
  \lean{Graph.splitOff_indep_extend_of_fiber_subset,
    Graph.splitOff_indep_extend_of_fiber_lt}
  \leanok
  \uses{def:graph-operations, def:matroid-MG, lem:forest-packing-decomp,
    lem:reverse-reroute-cycle-lift, lem:reverse-pendant-insert}
  Let $D \ge 2$, and let $G$ be a multigraph with a vertex $v$ of degree
  two, whose edges $e_a$ and $e_b$ join $v$ to distinct vertices
  $a, b \neq v$. Let $I'$ be an independent set of
  $M(\widetilde{G_v^{ab}})$, and let $h' = |\widetilde{ab} \cap I'|$.
  Then $M(\tilde G)$ has an independent set $I$ with
  \[
    I \setminus (\tilde e_a \cup \tilde e_b) \;=\;
    I' \setminus \widetilde{ab}
  \]
  such that (i) if $h' < D - 1$, then $|I| = |I'| + D$ and
  $|\tilde e_b \cap I| = h' + 1$, and (ii) if $h' = D - 1$, then
  $|I| = |I'| + D - 1$.
\end{lemma}
\begin{proof}
  \leanok
  We follow Katoh and Tanigawa's proof. By
  \cref{lem:forest-packing-decomp}, $I'$ is the union of $D$ forests of
  $\widetilde{G_v^{ab}}$, which we make pairwise disjoint as after that
  lemma. Two copies of $ab$ form a cycle, so exactly $h'$ of the forests
  contain a copy of $ab$, one each. Renumbering the forests and the
  copies of $ab$, we call them $F'_1, \dots, F'_D$, where $F'_i$ contains
  $(ab, i)$ for $i \le h'$ and no copy of $ab$ for $i > h'$.

  For $i \le h'$, let $F_i$ be $F'_i$ with $(ab, i)$ replaced by
  $(e_a, i)$ and $(e_b, i)$, a forest of $\tilde G$ by
  \cref{lem:reverse-reroute-cycle-lift}. In case~(i), let
  $F_i = F'_i \cup \{(e_a, i)\}$ for $h' < i \le D - 1$, and
  $F_D = F'_D \cup \{(e_b, h' + 1)\}$. These are forests of $\tilde G$ by
  \cref{lem:reverse-pendant-insert}. In case~(ii), let $F_D = F'_D$. It
  contains no copy of $ab$, so it is a forest of $\tilde G$, as in the
  proof of \cref{lem:reverse-pendant-insert}.

  The added edges are different copies of $e_a$ and $e_b$, and they are
  not edges of $\widetilde{G_v^{ab}}$. So $F_1, \dots, F_D$ are pairwise
  disjoint forests of $\tilde G$, and their union $I$ is independent in
  $M(\tilde G)$ (\cref{lem:forest-packing-decomp}). Deleting the copies
  of $e_a$ and $e_b$ from $I$ leaves the union of the $F'_i$ without
  their copies of $ab$, which is $I' \setminus \widetilde{ab}$. In
  case~(i) every forest gains one edge, so $|I| = |I'| + D$, and the
  copies of $e_b$ in $I$ are $(e_b, 1), \dots, (e_b, h' + 1)$. In
  case~(ii) the forests $F_1, \dots, F_{D-1}$ gain one edge each and
  $F_D$ none, so $|I| = |I'| + D - 1$.
\end{proof}
\begin{fmlnote}
  The first pin is case~(ii), with $h' = D - 1$ stated as
  $\widetilde{ab} \subseteq I'$, and the second is case~(i). The formal
  proof does not renumber the copies of $ab$. A forest that contains the
  copy $(ab, j)$ receives $(e_a, j)$ and $(e_b, j)$, and the copies added
  to the other forests take the indices $j$ for which $(ab, j)$ is not in
  $I'$.
\end{fmlnote}

\subsection{Splitting off and removal: KT Lemmas~4.3--4.8}
\label{sec:molecular-induction-dof}

This subsection proves KT Lemmas~4.3 to~4.8~\cite[pp.~661--666]{katohTanigawa2011}.
Let $v$ be a vertex of $G$ of degree two, whose edges $e_a$ and $e_b$
join it to vertices $a, b \neq v$. KT Lemmas~4.3(i) and~4.4 compare the
deficiency of $G$ with the deficiencies of the splitting off $G_v^{ab}$
and of the removal $G_v$ (\cref{def:graph-operations}). Splitting off
does not raise the deficiency and lowers it by at most one. Removal does
not lower it when $D \ge 2$. Katoh and Tanigawa prove these bounds with
forests and KT eq.~(2.4). We compare the values of partitions of $V(G)$
instead (\cref{def:D-deficiency}), and we compare the two proofs of KT
Lemma~4.4 in \cref{rem:kt-lemma-44}. Two variants of the upper bound for
splitting off follow the first lemma; the pencil chapter uses them.

When $G$ is a $k$-dof-graph, KT Lemma~4.3(ii) tells whether $G_v^{ab}$
is a $k$-dof-graph or a $(k-1)$-dof-graph, in terms of the bases of
$M(\widetilde{G_v^{ab}})$. Its proof uses the forest surgery and
edge-splitting (\cref{sec:molecular-induction-forest}). KT Lemmas~4.5
to~4.7 concern a minimal $k$-dof-graph with no proper rigid subgraph.
They bound its number of edges, find a vertex of degree two in it when
it is $2$-edge-connected, and show that removing such a vertex raises
its deficiency. KT Lemma~4.8 ends the subsection: splitting off at such
a vertex gives a minimal $0$-dof-graph if $k = 0$, and a minimal
$(k-1)$-dof-graph if $k > 0$. This is the step of the induction in KT
Theorem~4.9, together with the contraction of a proper rigid subgraph
(\cref{lem:contraction-minimality}).

\begin{lemma}[Splitting off lowers the deficiency by zero or one; KT Lemma~4.3(i)]
  \label{lem:splitoff-deficiency}
  \lean{Graph.splitOff_deficiency_le, Graph.splitOff_deficiency_ge}
  \leanok
  \uses{def:graph-operations, def:matroid-MG, thm:def-eq-corank}
  Let $D \ge 1$, and let $G$ be a multigraph with a vertex $v$ of degree
  two, whose edges $e_a$ and $e_b$ join $v$ to vertices $a, b \neq v$.
  Then
  \[
    \operatorname{def}(\tilde G) - 1
      \;\le\; \operatorname{def}(\widetilde{G_v^{ab}})
      \;\le\; \operatorname{def}(\tilde G).
  \]
  In particular, if $G$ is a $k$-dof-graph, then $G_v^{ab}$ is a
  $k$-dof-graph or a $(k-1)$-dof-graph.
\end{lemma}
\begin{proof}
  \leanok
  We compare the values $\operatorname{def}_{\tilde G}(P)$ of partitions
  (\cref{def:D-deficiency}). The vertices of $G_v^{ab}$ are those of $G$
  other than $v$. As $D \ge 1$, the value
  $D\,(|P| - 1) - (D - 1)\,d_G(P)$ does not increase when $d_G(P)$
  increases.

  For the upper bound, let $P'$ be a partition of $V(G) - v$, and let
  $P$ be the partition of $V(G)$ obtained by adding $v$ to the part of
  $P'$ that contains $a$. Then $|P| = |P'|$. The edge $e_a$ joins two
  vertices of one part of $P$. The edge $e_b$ joins different parts of
  $P$ exactly when $a$ and $b$ lie in different parts of $P'$, that is,
  when the new edge $ab$ of $G_v^{ab}$ joins different parts of $P'$.
  Every other edge has the same ends in $G$ and in $G_v^{ab}$. Hence
  $d_G(P) = d_{G_v^{ab}}(P')$, and the value of $P'$ for
  $\widetilde{G_v^{ab}}$ is the value of $P$ for $\tilde G$, which is at
  most $\operatorname{def}(\tilde G)$.

  For the lower bound, let $P$ be a partition of $V(G)$ of value
  $\operatorname{def}(\tilde G)$, and let $P'$ be the partition of
  $V(G) - v$ into the nonempty sets $X - v$ with $X \in P$. The edges of
  $G_v^{ab}$ are the edges of $G$ other than $e_a$ and $e_b$, with the
  same ends, and the new edge $ab$.

  Suppose first that $v$ is not alone in its part of $P$. Then
  $|P'| = |P|$. If $ab$ joins different parts of $P'$, then $v$ lies in a
  different part from $a$ or from $b$, so $e_a$ or $e_b$ joins different
  parts of $P$. Hence $d_{G_v^{ab}}(P') \le d_G(P)$, and the value of
  $P'$ is at least $\operatorname{def}(\tilde G)$.

  Suppose now that $\{v\}$ is a part of $P$. Then $|P'| = |P| - 1$, and
  both $e_a$ and $e_b$ join different parts of $P$, so
  $d_{G_v^{ab}}(P') \le d_G(P) - 1$. The value of $P'$ is then at least
  \[
    D\,(|P| - 2) - (D - 1)\,(d_G(P) - 1) \;=\;
    \operatorname{def}(\tilde G) - 1.
  \]
  In both cases
  $\operatorname{def}(\widetilde{G_v^{ab}}) \ge \operatorname{def}(\tilde G) - 1$.
\end{proof}

KT Lemma~4.3 assumes that $G$ is a minimal $k$-dof-graph, and its
part~(i) adds that $G_v^{ab}$ is minimal when it is a
$(k-1)$-dof-graph. Katoh and Tanigawa use this clause only in the proof
of their Lemma~4.8(ii). We prove the minimality there
(\cref{lem:reduction-step-pos}).

The ear steps of the pencil chapter (\cref{sec:main-component-short})
suppress a vertex of degree two, and keep one of its two edges as the
new edge that joins its neighbours.

\begin{lemma}[Suppressing a vertex of degree two]
  \label{lem:splitoff-deficiency-reuse}
  \lean{Graph.splitOff_deficiency_le_of_eq_left}
  \leanok
  \uses{def:graph-operations, def:D-deficiency}
  Let $D \ge 1$, and let $G$ be a multigraph with a vertex $v$ of degree
  two, whose edges $e_a$ and $e_b$ join $v$ to vertices $a, b \neq v$.
  Let $G''$ be obtained from $G$ by replacing the end $v$ of $e_a$ by
  $b$, and deleting $v$ and $e_b$. Then
  \[
    \operatorname{def}(\tilde G'') \;\le\; \operatorname{def}(\tilde G).
  \]
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:splitoff-deficiency}
  The graph $G''$ is $G_v^{ab}$ with $e_a$ as its new edge $ab$, and we
  argue as for the upper bound of \cref{lem:splitoff-deficiency}. Let
  $P'$ be a partition of $V(G) - v$, and let $P$ be the partition of
  $V(G)$ obtained by adding $v$ to the part of $a$. Then $|P| = |P'|$. In
  $G$ the edge $e_a$ joins two vertices of one part of $P$, and $e_b$
  joins different parts of $P$ exactly when $e_a$ joins different parts
  of $P'$ in $G''$. Every other edge has the same ends in $G$ and in
  $G''$. So $P'$ has the same value for $\tilde G''$ as $P$ for
  $\tilde G$.
\end{proof}
\begin{fmlnote}
  In the formalization $G''$ is the splitting off at $v$ with $e_0 = e_a$
  as the label of the new edge (\cref{fmlnote:graph-operations}), while
  the upper bound of \cref{lem:splitoff-deficiency} assumes
  $e_0 \notin E(G)$. The two bounds have one proof, which uses only that
  $e_0$ does not join different parts of $P$ in $G$.
\end{fmlnote}

The ear steps of the pencil chapter whose ends are not adjacent
(\cref{sec:main-component-orbit}) compare a graph with the graph
obtained from it by adding an edge.

\begin{lemma}[Splitting off with a gap in the merged deficiency]
  \label{lem:splitoff-deficiency-merged}
  \lean{Graph.splitOff_deficiency_add_le_of_deficiencyMerged}
  \leanok
  \uses{def:graph-operations, def:D-deficiency, def:deficiency-merged}
  Let $D \ge 1$. Let $v$, $a$ and $b$ be vertices of a multigraph $G$
  with $a, b \neq v$. Let $G'$ be obtained from $G$ by deleting $v$, and
  let $H$ be obtained from $G'$ by adding a new edge $ab$. Suppose that
  \[
    \operatorname{def}(\tilde G'; a, b) \;\le\;
    \operatorname{def}(\tilde G') - (D - 1)
  \]
  (\cref{def:deficiency-merged}). Then
  $\operatorname{def}(\tilde H) \le \operatorname{def}(\tilde G') - (D - 1)$.
\end{lemma}
\begin{proof}
  \leanok
  The graphs $H$ and $G'$ have the same vertices, and $H$ has one more
  edge, $ab$. Let $P$ be a partition of $V(G')$. If $P$ separates $a$
  and $b$, then $d_H(P) = d_{G'}(P) + 1$. So the value of $P$ for
  $\tilde H$ is its value for $\tilde G'$ minus $D - 1$, which is at most
  $\operatorname{def}(\tilde G') - (D - 1)$. If $a$ and $b$ lie in one
  part of $P$, then $d_H(P) = d_{G'}(P)$. So the value of $P$ for
  $\tilde H$ is at most $\operatorname{def}(\tilde G'; a, b)$, which the
  hypothesis bounds.
\end{proof}
\begin{fmlnote}
  The pin builds $H$ with the formal splitting off at $v$
  (\cref{fmlnote:graph-operations}), which is defined at a vertex of any
  degree, and a label $e_0$ for the new edge. It asks only that every
  edge of $G$ with the label $e_0$ have $v$ as an end. So $e_0$ is not
  the label of an edge of $G'$, and it may be the label of an edge at
  $v$, as in \cref{lem:splitoff-deficiency-reuse}.
\end{fmlnote}

\begin{lemma}[Removal does not lower the deficiency; KT Lemma~4.4]
  \label{lem:removal-deficiency}
  \lean{Graph.removeVertex_deficiency_ge}
  \leanok
  \uses{def:graph-operations, lem:splitoff-deficiency}
  Let $D \ge 2$, and let $G$ be a multigraph with a vertex $v$ of degree
  two, whose edges $e_a$ and $e_b$ join $v$ to vertices $a, b \neq v$.
  Then
  \[
    \operatorname{def}(\tilde G) \;\le\;
    \operatorname{def}(\widetilde{G_v}).
  \]
  In particular, if $G$ is a $k$-dof-graph, then
  $\operatorname{def}(\widetilde{G_v}) \ge k$.
\end{lemma}
\begin{proof}
  \leanok
  We compare the values of partitions, as for the lower bound of
  \cref{lem:splitoff-deficiency}. Let $P$ be a partition of $V(G)$ of
  value $\operatorname{def}(\tilde G)$, and let $P'$ be the partition of
  $V(G) - v$ into the nonempty sets $X - v$ with $X \in P$. The edges of
  $G_v$ are the edges of $G$ other than $e_a$ and $e_b$, the only edges
  at $v$, with the same ends. So
  \[
    d_{G_v}(P') \;=\; d_G(P) - c,
  \]
  where $c \in \{0, 1, 2\}$ is the number of the edges $e_a$ and $e_b$
  that join different parts of $P$.

  Suppose first that $v$ is not alone in its part of $P$. Then
  $|P'| = |P|$, and the value of $P'$ for $\widetilde{G_v}$ is
  $\operatorname{def}(\tilde G) + (D - 1)\,c \ge \operatorname{def}(\tilde G)$.

  Suppose now that $\{v\}$ is a part of $P$. Then $|P'| = |P| - 1$. The
  neighbours $a$ and $b$ of $v$ lie in other parts, so both $e_a$ and
  $e_b$ join different parts of $P$, and $c = 2$. The value of $P'$ is
  \[
    D\,(|P| - 2) - (D - 1)\,(d_G(P) - 2) \;=\;
    \operatorname{def}(\tilde G) + D - 2,
  \]
  which is at least $\operatorname{def}(\tilde G)$ as $D \ge 2$. In both
  cases $\operatorname{def}(\widetilde{G_v}) \ge \operatorname{def}(\tilde G)$.
\end{proof}

\begin{remark}[On KT Lemma~4.4]
  \label{rem:kt-lemma-44}
  \uses{lem:splitoff-deficiency, lem:removal-deficiency}
  Katoh and Tanigawa~\cite[p.~662]{katohTanigawa2011} prove KT
  Lemma~4.4 with forests. Let $B'$ be a base of $M(\widetilde{G_v})$. As
  $G_v$ is a subgraph of $G_v^{ab}$, the set $B'$ is independent in
  $M(\widetilde{G_v^{ab}})$, and it contains no copy of $ab$. The graph
  $G$ is obtained from $G_v^{ab}$ by edge-splitting along $ab$, so KT
  Lemma~4.2(i) with $h' = 0$ gives an independent set $I$ of
  $M(\tilde G)$ with $|I| = |B'| + D$ and $|\tilde e_b \cap I| = 1$. In
  the form of forests, each of the $D$ forests whose union is $B'$ gains
  one edge at $v$ (\cref{lem:forest-surgery-unsplit}).

  By KT eq.~(2.4) (\cref{thm:def-eq-corank}),
  $|B'| = D\,(|V(G)| - 2) - \operatorname{def}(\widetilde{G_v})$ and
  $|I| \le D\,(|V(G)| - 1) - \operatorname{def}(\tilde G)$. Hence
  $\operatorname{def}(\tilde G) \le \operatorname{def}(\widetilde{G_v})$.
  If equality holds, then $I$ is a base of $M(\tilde G)$ with
  $|\tilde e_b \cap I| = 1$. This is the second assertion of KT
  Lemma~4.4, and we prove it in this way
  (\cref{lem:removal-deficiency-strict}).

  Katoh and Tanigawa's argument is correct. It also needs $D \ge 2$,
  since KT Lemma~4.2(i) at $h' = 0$ asks that $0 < D - 1$. Our proof of
  \cref{lem:removal-deficiency} compares the values of partitions
  instead, and needs neither forests nor KT eq.~(2.4). The comparison
  works because of the case where $\{v\}$ is a part of $P$. There $P'$
  has one part fewer than $P$, which lowers the value by $D$. But the
  neighbours $a$ and $b$ of $v$ lie in other parts, so both $e_a$ and
  $e_b$ join different parts of $P$. Hence
  $d_{G_v}(P') = d_G(P) - 2$, which raises the value by $2(D - 1)$. The
  value changes by $2(D - 1) - D = D - 2 \ge 0$. When $v$ is not alone in
  its part, the number of parts stays the same, and fewer edges joining
  different parts can only raise the value.

  Without the forced crossing of $e_a$ and $e_b$ the count fails. If we
  used only that the number of edges joining different parts does not
  increase, as in the other case, we would get only
  $\operatorname{def}(\widetilde{G_v}) \ge \operatorname{def}(\tilde G) - D$.
  Rank monotonicity gives the same bound. As $M(\widetilde{G_v})$ is a
  restriction of $M(\tilde G)$ (\cref{lem:matroid-restrict-subgraph}),
  its rank is at most that of $M(\tilde G)$, and KT eq.~(2.4) turns this
  into $\operatorname{def}(\widetilde{G_v}) \ge \operatorname{def}(\tilde G) - D$.
\end{remark}

The step of the induction uses the two bounds together
(\cref{lem:reduction-step}).

\begin{lemma}[The deficiency after splitting off and after removal; KT Lemmas~4.3(i) and~4.4]
  \label{lem:dof-tracking}
  \lean{Graph.dof_tracking}
  \leanok
  \uses{lem:splitoff-deficiency, lem:removal-deficiency, def:graph-operations}
  Let $D \ge 2$, and let $G$ be a $k$-dof-graph with a vertex $v$ of
  degree two, whose edges $e_a$ and $e_b$ join $v$ to vertices
  $a, b \neq v$. Then
  \[
    k - 1 \;\le\; \operatorname{def}(\widetilde{G_v^{ab}}) \;\le\; k
    \qquad\text{and}\qquad
    \operatorname{def}(\widetilde{G_v}) \;\ge\; k.
  \]
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:splitoff-deficiency, lem:removal-deficiency}
  Combine \cref{lem:splitoff-deficiency}, which holds as $D \ge 1$, with
  \cref{lem:removal-deficiency}.
\end{proof}

\begin{lemma}[Base-fiber criterion for the splitting-off dof; KT Lemma 4.3(ii)]
  \label{lem:splitoff-kdof-criterion}
  \lean{Graph.splitOff_exists_base_inter_fiber_lt,
        Graph.splitOff_isKDof_of_exists_base_inter_fiber_lt}
  \leanok
  \uses{def:k-dof, def:graph-operations, def:matroid-MG, lem:edge-splitting,
        lem:splitoff-deficiency, thm:def-eq-corank}
  Let $G$ be a $k$-dof-graph with a degree-2 vertex $v$ (distinct neighbours
  $a, b$, incident edges exactly $e_a \ne e_b$, $e_0 \notin E(G)$ fresh) and
  $D \ge 2$. The splitting-off $G_v^{ab}$ is itself a $k$-dof-graph if and
  only if $M(\widetilde{G_v^{ab}})$ has a base $B'$ with
  $|\widetilde{ab} \cap B'| < D - 1$. This is the refinement of the
  two-sided bound of \cref{lem:dof-tracking} that reads off \emph{which}
  alternative ($k$-dof or $(k-1)$-dof) the splitting-off lands in from the
  short-circuit count of a base.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:forest-surgery-count, lem:balanced-forest-packing}
  Forward (a $k$-dof splitting-off has such a base): reroute a balanced
  $D$-forest packing of a base of $M(\tilde G)$ across $v$
  (\cref{lem:forest-surgery-count}); the rerouted set $I'$ is
  $M(\widetilde{G_v^{ab}})$-independent with $|I'| = |\text{base}| - D$ and
  $|\widetilde{ab} \cap I'| < D - 1$, and since both graphs are $k$-dof the
  def\,$=$\,corank identity (\cref{thm:def-eq-corank}) makes $|I'|$ the full
  rank, so $I'$ is the desired base. Reverse: the edge-splitting extension
  (\cref{lem:edge-splitting}, partial-fiber branch) lifts a base $B'$ with
  $|\widetilde{ab} \cap B'| < D - 1$ to an $M(\tilde G)$-independent set of
  size $|B'| + D$; bounding it by $\operatorname{rank} M(\tilde G) =
  D(|V|-1) - k$ gives $\operatorname{def}(\widetilde{G_v^{ab}}) \ge k$, and
  the splitting-off deficiency bound (\cref{lem:splitoff-deficiency}) gives
  the matching $\le k$.
\end{proof}

\begin{lemma}[Edge-count bound with no proper rigid subgraph; KT Lemma 4.5(i)]
  \label{lem:no-rigid-edge-count}
  \lean{Graph.no_rigid_edge_count}
  \leanok
  \uses{lem:circuit-induces-rigid, def:k-dof, def:rigid-subgraph, def:matroid-MG}
  Let $G$ be a minimal $k$-dof-graph with no proper rigid subgraph and
  $D = \binom{d+1}{2} \ge 2$. Then
  \[
    (D-1)\,|E(G)| \;<\; D\,(|V(G)| - 1) - k + (D-1).
  \]
  This is the edge bound Katoh--Tanigawa use to force a low-degree vertex
  (\cref{lem:reducible-vertex}). At $k = 0$ it recovers the original KT form.
\end{lemma}
\begin{proof}
  \leanok
  \uses{thm:def-eq-corank}
  Read the graph-level bound through the matroid: $|E(\tilde G)| = (D-1)\,|E(G)|$
  (each edge of $G$ contributes a fiber $\tilde e$ of size $D-1$, the
  \texttt{mulTilde}-level count \texttt{Graph.mulTilde\_edgeSet\_ncard}), and a
  base $B^\ast$ has $|B^\ast| = D(|V(G)| - 1) - k$ since $G$ is $k$-dof
  (\cref{thm:def-eq-corank}); so the claim is $|E(\tilde G)| < |B^\ast| + (D-1)$.
  Fix an edge $e$ and let $h^\ast = \min_B |\tilde e \cap
  B|$ over bases $B$ of $M(\tilde G)$, attained at $B^\ast$; minimality of $G$
  gives $h^\ast \ge 1$. We show $E(\tilde G) \setminus \tilde e \subseteq B^\ast$
  (KT eq.~4.3). If some $f \in E(\tilde G) \setminus \tilde e$ were outside
  $B^\ast$, its fundamental circuit $X = \operatorname{fundCircuit}(f, B^\ast)$
  induces, by \cref{lem:circuit-induces-rigid}, a rigid subgraph $G[V(X)]$
  spanning at least two vertices ($G$ is loopless; \texttt{Graph.loopless\_of\_isMinimalKDof});
  the no-proper-rigid hypothesis forces $V(X) = V(G)$ (the spanning step
  \texttt{Graph.fundCircuit\_inducedSpan\_vertexSet\_eq}). Then $X \cap \tilde e
  \neq \emptyset$: otherwise $X - e_j$ is independent of size $D(|V(G)| - 1)$ for any
  $e_j \in X$; any base $B'$ containing $X - e_j$ has $|B'| = D(|V(G)| - 1) - k$, so
  $D(|V(G)| - 1) \le D(|V(G)| - 1) - k$, forcing $k \le 0$, and
  $\operatorname{def}(G) \ge 0$ then pins $k = 0$.
  At $k = 0$, $X - e_j$ is itself a base avoiding $\tilde e$, contradicting minimality.
  Picking $e_j \in X \cap \tilde e$, the base
  exchange $B = B^\ast + f - e_j$ has $|B \cap \tilde
  e| = h^\ast - 1 < h^\ast$, contradicting the choice of $B^\ast$. So
  $E(\tilde G) \setminus \tilde e \subseteq B^\ast$, whence $|E(\tilde G)| =
  |B^\ast| + (|\tilde e| - h^\ast) = D(|V(G)| - 1) - k + (D - 1 - h^\ast) \le
  D(|V(G)| - 1) - k + (D - 2)$.
\end{proof}

\begin{lemma}[Independence of $E(\tilde G)$ at $k > 0$; KT Lemma 4.5(ii)]
  \label{lem:edge-set-indep-pos}
  \lean{Graph.indep_edgeSet_mulTilde_of_noRigid_of_pos,
        Graph.isBase_eq_edgeSet_mulTilde_of_noRigid_of_pos}
  \leanok
  \uses{lem:circuit-induces-rigid, def:k-dof, def:rigid-subgraph, def:matroid-MG,
        thm:def-eq-corank}
  Let $G$ be a minimal $k$-dof-graph with $k > 0$, no proper rigid subgraph, and
  $D \ge 2$. Then $E(\tilde G)$ is independent in $M(\tilde G)$ and is its unique base.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:circuit-induces-rigid, def:k-dof, def:rigid-subgraph, def:matroid-MG,
        thm:def-eq-corank}
  If $E(\tilde G)$ were dependent, it would contain a circuit $C$; by
  \cref{lem:circuit-induces-rigid} the induced span $H = G[V(C)]$ is a rigid subgraph
  of $G$; looplessness gives $|V(H)| \ge 2$; the no-proper-rigid hypothesis forces
  $V(H) = V(G)$. Then $H$ is rigid and spans all of $V(G)$, so
  $\operatorname{rank}M(\tilde G) \ge \operatorname{rank}M(\tilde H) = D(|V(G)|-1)$;
  combined with $\operatorname{rank}M(\tilde G) + \operatorname{def}(G) = D(|V(G)|-1)$
  and $\operatorname{def}(G) = k > 0$, this is a contradiction.
  Uniqueness: $E(\tilde G)$ independent and any base $B \subseteq E(\tilde G)$ gives
  $B = E(\tilde G)$ by \texttt{Matroid.IsBase.eq\_of\_subset\_indep}.
\end{proof}

\begin{lemma}[Strict removal gap; KT Lemma 4.4 (equality clause) and Lemma 4.7]
  \label{lem:removal-deficiency-strict}
  \lean{Graph.exists_isBase_vb_fiber_eq_one_of_removeVertex_isKDof,
        Graph.removeVertex_deficiency_gt_of_noRigid}
  \leanok
  \uses{lem:removal-deficiency, lem:edge-splitting, lem:edge-set-indep-pos,
        def:k-dof, def:rigid-subgraph, thm:def-eq-corank}
  Let $G$ be a $k$-dof-graph with a degree-2 vertex $v$ (distinct neighbours
  $a, b$, incident edges exactly $e_a \ne e_b$, $e_0 \notin E(G)$ fresh). Two
  sharpenings of the removal bound $\operatorname{def}(\widetilde{G_v}) \ge k$
  of \cref{lem:removal-deficiency}: (i) \emph{(KT 4.4, equality clause)} if
  $D \ge 2$ and equality $\operatorname{def}(\widetilde{G_v}) = k$ holds, then
  $M(\tilde G)$ has a base $B$ with $|\widetilde{vb} \cap B| = 1$;
  (ii) \emph{(KT 4.7)} if moreover $G$ is a \emph{minimal} $k$-dof-graph with
  no proper rigid subgraph, $|V(G)| \ge 3$, and $D \ge 3$, the bound is
  strict: $\operatorname{def}(\widetilde{G_v}) > k$.
\end{lemma}
\begin{proof}
  \leanok
  (i) A base $B'$ of $M(\widetilde{G_v})$ is independent in
  $M(\widetilde{G_v^{ab}})$ (matroid restriction along $\widetilde{G_v} \le
  \widetilde{G_v^{ab}}$) and avoids $\widetilde{ab}$ entirely, so the
  edge-splitting extension (\cref{lem:edge-splitting}, partial-fiber branch at
  $h' = 0$) lifts it to an $M(\tilde G)$-independent $I$ of size $|B'| + D =
  (D(|V|-2) - k) + D = D(|V|-1) - k = \operatorname{rank} M(\tilde G)$ (the
  def\,$=$\,corank identity on both sides, \cref{thm:def-eq-corank}) --- a
  base with $|\widetilde{vb} \cap I| = 0 + 1 = 1$. (ii) Suppose
  $\operatorname{def}(\widetilde{G_v}) = k$. At $k = 0$ the removal $G_v$
  would be a rigid subgraph of $G$, proper on $|V| - 1 \ge 2$ vertices ---
  contradicting the no-proper-rigid hypothesis. At $k > 0$, clause (i) gives
  a base meeting $\widetilde{vb}$ in exactly one copy, but
  \cref{lem:edge-set-indep-pos} forces every base to equal $E(\tilde G)$,
  which meets $\widetilde{vb}$ in all $D - 1 \ge 2$ copies ($D \ge 3$ is
  sharp: at $D = 2$ the contradiction vanishes) --- contradiction.
\end{proof}

\begin{lemma}[A low-degree vertex by the average-degree count; KT Lemma 4.6, core]
  \label{lem:low-degree-vertex}
  \lean{Graph.exists_degree_le_two}
  \leanok
  \uses{lem:no-rigid-edge-count, def:k-dof}
  Let $G$ be a minimal $k$-dof-graph with no proper rigid subgraph,
  $V(G) \neq \emptyset$, and $D = \binom{d+1}{2} \ge 3$ (the molecular
  regime $d \ge 2$). Then some vertex $v \in V(G)$ has multigraph degree
  $\deg_G(v) \le 2$.
\end{lemma}
\begin{proof}
  \leanok
  Average-degree count. The all-$k$ edge bound (\cref{lem:no-rigid-edge-count})
  gives $(D-1)\,|E| < D\,(|V|-1) - k + (D-1)$; using $k \ge 0$
  (\texttt{Graph.deficiency\_nonneg}) this implies $(D-1)\,|E| < D|V| - 1$,
  i.e.\ $2|E| < 3|V|$ (after $D \ge 3$ and $|V| \ge 1$). The multigraph handshake
  $\sum_v \deg_G(v) = 2|E|$ then forces some vertex to have degree $\le 2$
  by a finite pigeonhole. The degree function and the handshake are the
  vendored \texttt{Graph.degree} / \texttt{Graph.handshake\_degree\_subtype}
  API (Peter Nelson's \texttt{apnelson1/Matroid} \texttt{Graph.incFun}
  development), reused here rather than re-derived.
\end{proof}

\begin{lemma}[Existence of a degree-2 vertex; KT Lemma 4.6]
  \label{lem:reducible-vertex}
  \lean{Graph.exists_degree_eq_two}
  \leanok
  \uses{lem:low-degree-vertex, lem:two-edge-conn, def:k-dof}
  A minimal $k$-dof-graph $G$ that is $2$-edge-connected, has no proper rigid
  subgraph, $|V| \ge 2$, and $D = \binom{d+1}{2} \ge 3$ (the molecular regime
  $d \ge 2$) contains a vertex $v$ of multigraph degree exactly $2$. The
  $2$-edge-connectivity is supplied at the call site for $k = 0$ by
  \cref{lem:two-edge-conn}. (This degree-2 vertex is the one whose splitting-off
  the reduction step (\cref{lem:reduction-step}) shows preserves minimality and
  the degree of freedom.)
\end{lemma}
\begin{proof}
  \leanok
  The average-degree count (\cref{lem:low-degree-vertex}) supplies a
  vertex of degree $\le 2$; the $2$-edge-connectivity hypothesis forces
  $\deg_G(v) \ge 2$ (\texttt{Graph.two\_le\_degree\_of\_twoEdgeConnected}),
  pinning it to degree exactly $2$. This is the lemma that guarantees the
  induction can always take a step.
\end{proof}

\begin{lemma}[Each reduction strictly shrinks the vertex set; KT Theorem 4.9, measure]
  \label{lem:reduction-measure}
  \lean{Graph.splitOff_vertexSet_ncard_lt, Graph.rigidContract_vertexSet_ncard_lt}
  \leanok
  \uses{def:graph-operations, def:rigid-contraction}
  Both reduction operations strictly decrease the number of vertices, the
  well-founded measure on which Theorem~4.9 inducts. Splitting-off deletes
  the degree-2 vertex, $V(G_v^{ab}) = V(G) \setminus \{v\}$, so
  $|V(G_v^{ab})| < |V(G)|$ for $v \in V(G)$. Contraction collapses $V(H)$ to a
  single representative, so $|V(G / E(H))| \le |V(G)| - |V(H)| + 1 < |V(G)|$
  whenever $|V(H)| \ge 2$ --- the genuine requirement, since contracting a
  single-vertex subgraph is a vertex-set no-op; Case~I always contracts a
  proper rigid subgraph spanning at least two vertices.
\end{lemma}
\begin{proof}
  \leanok
  Splitting-off: immediate from $V(G_v^{ab}) = V(G) \setminus \{v\}$
  (\texttt{Graph.vertexSet\_splitOff}) and $v \in V(G)$. Contraction: the
  image $\operatorname{collapseTo}\,r\,V(H)\,''\,V(G)$ is contained in
  $(V(G) \setminus V(H)) \cup \{r\}$, of cardinality at most $|V(G)| - |V(H)|
  + 1$, which is $< |V(G)|$ exactly when $|V(H)| \ge 2$.
\end{proof}

\begin{lemma}[Reduction step preserves minimality; KT Lemmas 4.7--4.8]
  \label{lem:reduction-step}
  \lean{Graph.splitOff_isMinimalKDof, Graph.contraction_isMinimalKDof}
  \leanok
  \uses{lem:reducible-vertex, lem:contraction-minimality, lem:dof-tracking, lem:reduction-measure, lem:circuit-induces-rigid}
  Each reduction --- splitting-off at the degree-2 vertex of
  \cref{lem:reducible-vertex}, or contraction of a proper rigid
  subgraph (\cref{lem:contraction-minimality}) --- carries a minimal
  $0$-dof-graph (with no proper rigid subgraph and $|V| \ge 3$, in the
  splitting-off case) to a smaller minimal $0$-dof-graph. This is the
  inductive step of KT Theorem~4.9. ($|V| \ge 3$ is the splitting
  branch's standing regime and is genuinely needed: the two-vertex
  double edge splits to a one-vertex loop graph, which is not
  minimal.)
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:circuit-induces-rigid}
  The contraction branch is \cref{lem:contraction-minimality}. For the
  splitting-off branch (KT 4.8(i)): $\operatorname{def}(\tilde
  G_v^{ab}) = 0$ is \cref{lem:dof-tracking} squeezed against
  $\operatorname{def}(\tilde G) = 0$. For minimality, Katoh--Tanigawa's
  claim (4.10) --- every circuit of $M(\tilde G_v^{ab})$ meets the fresh
  short-circuit fiber $\tilde{ab}$ --- is exactly the statement that the
  surviving ground set $E(\tilde G_v) = E(\tilde G_v^{ab}) \setminus
  \tilde{ab}$ is independent in $M(\tilde G_v^{ab})$: otherwise a circuit
  inside it would be a circuit of $M(\tilde G)$ avoiding $v$, inducing a
  proper rigid subgraph (\cref{lem:circuit-induces-rigid}; it spans at
  least two vertices since $G$ is loopless --- a loop copy would be a
  matroid loop inside a base, contradicting minimality;
  \texttt{Graph.loopless\_of\_isMinimalKDof}). So $E(\tilde
  G_v)$ is a base of $M(\tilde G_v)$, and KT 4.7 gives
  $\operatorname{def}(\tilde G_v) > 0$ ($G_v$ is a proper subgraph on
  $|V(G)| - 1 \ge 2$ vertices, so not rigid). A \emph{rank count} replaces KT's iterated fundamental-circuit
  swap: any base $B'$ of $M(\tilde G_v^{ab})$ avoiding a fiber $\tilde e$
  has $|B'| \le |E(\tilde G_v)|$ (split $B'$ across $\tilde{ab}$ and
  $E(\tilde G_v)$), which through
  $|B| + \operatorname{def} = D(|V|-1)$ forces $\operatorname{def}(\tilde
  G_v) \le 0$ --- a contradiction. Hence every base meets every fiber.
\end{proof}

\begin{lemma}[Reduction step preserves minimality, positive dof; KT Lemma 4.8(ii)]
  \label{lem:reduction-step-pos}
  \lean{Graph.splitOff_isMinimalKDof_of_pos}
  \leanok
  \uses{lem:dof-tracking, lem:reduction-measure, lem:splitoff-kdof-criterion, lem:edge-splitting, lem:edge-set-indep-pos, lem:removal-deficiency-strict, lem:circuit-induces-rigid}
  Let $G$ be a minimal $k$-dof-graph with $k > 0$, $D \ge 3$, $|V| \ge 3$, and no proper
  rigid subgraph. Then splitting off at a degree-2 vertex $v$ (with neighbours $a, b$) yields
  a minimal $(k-1)$-dof-graph $G_v^{ab}$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:dof-tracking, lem:splitoff-kdof-criterion, lem:edge-splitting, lem:edge-set-indep-pos, lem:removal-deficiency-strict, lem:circuit-induces-rigid}
  \textbf{Step 1 (dof tracking).} \Cref{lem:dof-tracking} gives
  $\operatorname{def}(G_v^{ab}) \in \{k-1, k\}$. \textbf{Step 2 (ruling out $k$).}
  If $\operatorname{def}(G_v^{ab}) = k$ then by \cref{lem:splitoff-kdof-criterion} some
  base $B$ of $M(\tilde{G}_v^{ab})$ meets the fiber $\widetilde{ab}$ in fewer than $D-1$
  elements. \Cref{lem:edge-set-indep-pos} provides the unique base $E(\tilde{G}_v)$ of
  $M(\tilde{G}_v^{ab})$; a fundamental-circuit argument then applies
  \cref{lem:edge-splitting} (KT 4.2(i)) to produce a proper rigid subgraph, contradicting
  the hypothesis. Hence $\operatorname{def}(G_v^{ab}) = k-1$.
  \textbf{Step 3 (minimality at $k-1$).} For any proper subgraph $H \le G_v^{ab}$: if
  $e_0 \notin E(H)$ then \cref{lem:removal-deficiency-strict} and \cref{lem:edge-splitting}
  (KT 4.2(ii)) show $\operatorname{def}(H) < k-1$; otherwise $H$ lifts to a proper
  subgraph of $G$ via the commuting square
  $\mathtt{induce\_insert\_splitOff}$, and \cref{lem:circuit-induces-rigid} forces the
  deficiency below $k-1$.
\end{proof}

\subsection{Theorem 4.9}
\label{sec:molecular-induction-thm49}

\begin{theorem}[Reduction of minimal $0$-dof-graphs; KT Theorem 4.9]
  \label{thm:minimal-kdof-reduction}
  \lean{Graph.minimal_kdof_reduction}
  \leanok
  \uses{lem:reduction-step, lem:reducible-vertex, lem:contraction-minimality, def:graph-operations, lem:reduction-measure}
  Every minimal $0$-dof-graph $G$ with $|V| \ge 2$ reduces to the
  two-vertex double edge by a sequence of splitting-off and
  proper-rigid-subgraph contraction operations. Equivalently, $G$ is
  generated from the two-vertex double edge by the inverse operations
  (edge-splitting and rigid-subgraph expansion). The formalization
  states this as the well-founded induction principle the reduction
  dichotomy and the vertex-count measure (\cref{lem:reduction-measure})
  drive: a property $P$ closed under the two-vertex base case, under
  splitting off a reducible degree-2 vertex, and under contracting a
  proper rigid subgraph (with the induction hypothesis on every
  strictly-smaller minimal $0$-dof-graph) holds of every minimal
  $0$-dof-graph with $|V| \ge 2$.
\end{theorem}
\begin{proof}
  \leanok
  Strong induction on $|V|$. The base case $|V| = 2$ is the two-vertex
  double edge. For $|V| \ge 3$: if $G$ has a proper rigid subgraph, the
  contraction step applies (\cref{lem:contraction-minimality}); otherwise
  \cref{lem:reducible-vertex} supplies a reducible degree-2 vertex, whose
  two incident edges are recovered as the distinct nonloop edges at $v$
  (the $0$-dof two-edge-connectivity rules out a single loop), and
  splitting it off yields a strictly smaller minimal $0$-dof-graph
  (\cref{lem:reduction-step} and \cref{lem:reduction-measure}) on which the induction
  hypothesis applies. The contraction branch is handed the induction
  hypothesis rather than recursing internally: Case~I of the algebraic
  induction genuinely consumes the induction hypothesis at \emph{two}
  objects (the rigid block and the contraction), so the principle exposes
  the full hypothesis rather than a single recursive call; the
  splitting-off branch carries an unused-edge-label premise for the
  fresh short-circuit edge, quantified over exactly the minimal
  $0$-dof-graphs the induction visits (the unconditioned form over every
  graph on the edge-label type is unsatisfiable --- an
  all-loops-at-one-vertex graph exhausts it). This is
  Katoh--Tanigawa~\cite[Theorem 4.9]{katohTanigawa2011}; it is the
  combinatorial skeleton the algebraic induction
  (\cref{sec:molecular-algebraic-induction}) realizes at the
  rigidity-matrix rank. A companion form of the induction principle
  hands the no-rigid-subgraph branch the same full hypothesis as the
  contraction branch, instead of a supplied degree-2 vertex, for use
  where the degree-2 vertex is more naturally re-extracted after the
  case split than received from it.
\end{proof}

\begin{theorem}[All-$k$ induction skeleton for Theorem~5.5]
  \label{thm:minimal-kdof-reduction-all-k}
  \lean{Graph.minimal_kdof_reduction_all_k}
  \leanok
  \uses{def:k-dof, def:cut-edges-2ec, def:rigid-subgraph}
  A property $P$ of graphs that is closed under four cases holds of every
  nonempty minimal $k$-dof-graph.  The four cases are: \textup{(i)} $|V| \le 2$
  (base region); \textup{(ii)} $|V| \ge 3$ and $G$ is not $2$-edge-connected
  (cut-edge decomposition, \cref{def:cut-edges-2ec}); \textup{(iii)} $|V| \ge 3$
  and $G$ has a proper rigid subgraph (\cref{def:rigid-subgraph}); \textup{(iv)}
  $|V| \ge 3$, $G$ is $2$-edge-connected, has no proper rigid subgraph, with
  the split $k > 0$ vs $k = 0$.  Each case is handed the full conditioned
  induction hypothesis over every strictly-smaller nonempty minimal $k'$-dof-graph
  (IH (6.1) of Katoh--Tanigawa~\cite[Sect.~6]{katohTanigawa2011}).
\end{theorem}
\begin{proof}
  \leanok
  Strong induction on $|V|$, generalizing over $k$.  At $|V| \le 2$ the base
  case applies.  At $|V| \ge 3$: $0 \le k$ from \texttt{deficiency\_nonneg};
  three nested case splits on $2$-edge-connectivity, proper-rigid-subgraph
  existence, and $k = 0$ vs $k > 0$ dispatch to the four cases.
\end{proof}

\begin{lemma}[Nondegenerate-hinge realization at a cut edge; KT Lemma 6.1]
  \label{lem:case-cut-edge-realization}
  \lean{CombinatorialRigidity.Molecular.case_cut_edge_realization_gen}
  \leanok
  \uses{def:genuine-hinge-realization, lem:cut-edge-decomposition,
        thm:minimal-kdof-reduction-all-k, lem:block-rank-cut,
        lem:relative-deficiency-rank-bound}
  Let $G$ be a minimal $k$-dof-graph with $|V(G)| \ge 3$ that is not
  $2$-edge-connected, and suppose every strictly-smaller nonempty minimal
  $k'$-dof-graph has a nondegenerate-hinge panel realization
  \textup{(}\cref{def:genuine-hinge-realization}\textup{)}.  Then $G$ has a
  nondegenerate-hinge panel realization.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:cut-edge-decomposition, lem:block-rank-cut,
        lem:relative-deficiency-rank-bound}
  Apply \cref{lem:cut-edge-decomposition} to obtain $V_1$, $V_2 = V(G)\setminus V_1$
  with $|C| \le 1$ and $k = k_1 + k_2 + D - (D-1)|C|$.  Apply the induction
  hypothesis on each side to get frameworks $F_1, F_2$ at their target ranks.
  Assemble $F$ by inheriting $F_1$'s and $F_2$'s extensors for non-cut edges; for
  the at-most-one cut edge assign a nonzero extensor lying in both endpoint panels
  (from \texttt{exists\_extensor\_in\_two\_panels}).  The rank lower bound is the
  block-rank addition \cref{lem:block-rank-cut}; the upper bound is the
  genericity-free bound of \cref{lem:relative-deficiency-rank-bound}; the deficiency
  arithmetic $k = k_1 + k_2 + D - (D-1)|C|$ and $|V| = |V_1| + |V_2|$ close
  equality.
\end{proof}

\medskip\noindent This produces the nondegenerate-hinge conjunct of the
cut-edge case of the all-$k$ induction (\cref{thm:minimal-kdof-reduction-all-k},
case~(ii)); \cref{lem:case-cut-edge-realization-gp} supplies the companion
generic conjunct for simple graphs.

\begin{lemma}[Generic full-rank realization at a cut edge; KT Lemma 6.1]
  \label{lem:case-cut-edge-realization-gp}
  \lean{CombinatorialRigidity.Molecular.case_cut_edge_realization_gp_gen}

  Let $G$ be a minimal $k$-dof simple graph with $|V(G)| \ge 3$ that is not

\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:cut-edge-decomposition, lem:block-rank-cut,

        lem:genericity-device}
  Cut decomposition as in \cref{lem:case-cut-edge-realization}, yielding sides

\subsection{Simplicity and the adjacent-degree-2-pair lemma ($d=3$)}
\label{sec:simplicity-and-adjacent-pair}

\begin{lemma}[Two-vertex double edge is $0$-dof]
  \label{lem:two-vertex-zero-dof}
  \lean{Graph.isKDof_zero_of_parallel_pair}
  \leanok
  \uses{def:k-dof}
  Let $H$ be a two-vertex multigraph with vertex set $\{x, y\}$ ($x \ne y$) and
  exactly two edges $e_1, e_2$ (with $e_1 \ne e_2$), both linking $x$ to $y$,
  and $D = \binom{d+1}{2} \ge 2$. Then $\operatorname{def}(\tilde H) = 0$, i.e.\ $H$
  is a $0$-dof-graph.
\end{lemma}
\begin{proof}
  \leanok
  For the single-part partition ($f(x) = f(y)$) the deficiency contribution is $0$.
  For the two-part partition ($f(x) \ne f(y)$) both edges cross, giving $D \cdot 1 -
  (D-1) \cdot 2 = 2 - D \le 0$. Since the deficiency is also $\ge 0$
  (\texttt{Graph.deficiency\_nonneg}), the supremum equals $0$.
\end{proof}

\begin{lemma}[Simplicity from minimality and no proper rigid subgraph; KT p.~682]
  \label{lem:simple-minimal-noRigid}
  \lean{Graph.simple_of_isMinimalKDof_of_noRigid}
  \leanok
  \uses{def:k-dof, def:rigid-subgraph, lem:two-vertex-zero-dof}
  Let $G$ be a minimal $k$-dof-graph with no proper rigid subgraph, $|V(G)| \ge 3$,
  and $D = \binom{d+1}{2} \ge 2$. Then $G$ is simple (loopless and no parallel
  edges). The proof is $k$-independent.
\end{lemma}
\begin{proof}
  \leanok
  Looplessness is \texttt{Graph.loopless\_of\_isMinimalKDof}. A parallel pair
  $e \ne f$ between vertices $x, y$ makes the two-vertex subgraph
  $H = (G[\{x,y\}])|_{\{e,f\}}$ a $0$-dof-graph
  (\cref{lem:two-vertex-zero-dof}), with $|V(H)| = 2$ and
  $V(H) \subsetneq V(G)$ (since $|V(G)| \ge 3$), contradicting the
  no-proper-rigid-subgraph hypothesis.
\end{proof}

\begin{lemma}[Adjacent degree-2 pair; KT Lemma 4.6 at $d = 3$]
  \label{lem:adjacent-degree-two-pair}
  \lean{Graph.exists_adjacent_degree_two_pair}
  \leanok
  \uses{lem:no-rigid-edge-count, lem:two-edge-conn, def:k-dof, def:rigid-subgraph}
  Let $G$ be a minimal $0$-dof-graph with no proper rigid subgraph, $|V(G)| \ge 3$,
  and $D = \binom{d+1}{2} \ge 6$ (the $d = 3$ regime). Then there exist adjacent
  vertices $v, a \in V(G)$ both of degree exactly $2$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:two-vertex-zero-dof, lem:simple-minimal-noRigid}
  By contradiction: assume no two adjacent degree-$2$ vertices. Partition $V(G)$ as
  $X_2 = \{v \mid \deg v = 2\}$ and $X_{\ge 3} = V(G) \setminus X_2$.
  For each edge~$e$, the number of $X_2$-endpoints of $e$ is at most $1$: if both
  endpoints were in $X_2$ they would form an adjacent degree-$2$ pair, contradicting
  the assumption. Summing over all edges via the incidence function and swapping the
  order of summation gives $\sum_{v \in X_2} \deg v \le \sum_{w \in X_{\ge 3}} \deg w$.
  Combined with the KT 4.5(i) edge bound (\cref{lem:no-rigid-edge-count}), the
  handshake identity, and $D \ge 6$, this forces $4|X_2| \le 2|E| < 2D|V|/(D-1)$,
  which together with $3|X_{\ge 3}| \le \sum_{X_{\ge 3}} \deg$ yields a numeric
  contradiction via $|X_2| + |X_{\ge 3}| = |V| \ge 3$.
\end{proof}

\begin{lemma}[A four-vertex chain from an adjacent degree-2 pair ($d = 3$; superseded)]
  \label{lem:chain-data-of-noRigid}
  \lean{Graph.exists_chain_data_of_noRigid}
  \leanok
  \uses{lem:adjacent-degree-two-pair, lem:simple-minimal-noRigid, def:rigid-subgraph, def:k-dof}
  \emph{Superseded; no live consumer.} \Cref{lem:case-III} now runs the
  general-$d$ extractor \cref{lem:chain-data-extract} directly, uniformly
  at every grade, so this $d = 3$-only extraction is no longer on the
  live argument; the statement below is retained for the record.

  Let $G$ be a minimal $0$-dof-graph with no proper rigid subgraph,
  $D = \binom{d+1}{2} \ge 6$ (the $d = 3$ regime), and $|V(G)| \ge 4$.
  Then there exist distinct vertices $v, a, b, c$ and edges $e_a, e_b, e_c$
  forming the chain $b - v - a - c$: $G$ links $e_a$ from $v$ to $a$,
  $e_b$ from $v$ to $b$, and $e_c$ from $a$ to $c$; every $v$-incident
  edge is $e_a$ or $e_b$; every $a$-incident edge is $e_a$ or $e_c$;
  $e_a \ne e_b$, $e_a \ne e_c$; and $b \ne c$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:adjacent-degree-two-pair, lem:simple-minimal-noRigid, def:rigid-subgraph}
  \cref{lem:adjacent-degree-two-pair} supplies adjacent degree-$2$ vertices
  $v, a$ with shared edge $e_a$.  Simplicity
  (\cref{lem:simple-minimal-noRigid}) ensures the degree-$2$ closure at $v$
  identifies two distinct edges $e_a, e_b$ (linking $v$ to $a$ and $b$), and
  the closure at $a$ identifies two distinct edges $e_a, e_c$ (linking $a$ to
  $v$ and $c$).  The shared edge $e_a$ pins the companion at each vertex
  (the other endpoint of $e_a$ at $v$ is $a$, and at $a$ is $v$).
  If $b = c$ then $G[\{v, a, b\}]$ with the three edges $e_a, e_b, e_c$ is a
  triangle, hence a proper rigid subgraph of $G$ (since $|V(G)| \ge 4$),
  contradicting the hypothesis.
\end{proof}

\subsection{The general-$d$ chain extraction (KT Lemma 4.6)}

The $d = 3$ pair above (\cref{lem:adjacent-degree-two-pair} and
\cref{lem:chain-data-of-noRigid}) is superseded by the general form
developed here. At general $d$, KT
Lemma~4.6 \cite[pp.~664--665]{katohTanigawa2011} is stronger: a minimal
$0$-dof-graph with no proper rigid subgraph either contains a
\emph{length-$d$ chain} (interior vertices all of degree $2$) or
\emph{is} a short cycle. Both branches matter for the general-$d$
Case~III: the chain supplies the splitting-off recursion, and the cycle
supplies the Lemma-5.4 realization (\cref{lem:cycle-realization});
the extraction is accordingly packaged as the two records and the
dichotomy below.

\begin{definition}[Chain data; KT eq.~6.46]
  \label{def:chain-data}
  \lean{Graph.ChainData}
  \leanok
  \uses{def:graph-operations}
  Length-$n$ chain data on a multigraph $G$: pairwise-distinct vertices
  $v_0, \dots, v_n$ (KT eq.~6.46), pairwise-distinct edges $e_i$ genuinely
  linking $v_i$ to $v_{i+1}$, a fresh short-circuit label $e_0 \notin E(G)$
  (for the splitting-off), and the \emph{interior degree-2 closure}: every
  $G$-edge at an interior vertex $v_i$ ($1 \le i \le n-1$) is one of the two
  chain edges there. The record carries the chain length \emph{exactly}
  equal to $n$ (KT Lemma~4.6 builds the chain to length exactly $d$; the $d$
  Case-III candidate frameworks are the $d$ panels of the chain, and the
  panel-discriminator index alignment $d = k+1$ is recovered from the
  ambient dimension identity at the use sites).
\end{definition}

\begin{definition}[Cycle data; KT Lemma 4.6, cycle branch]
  \label{def:cycle-data}
  \lean{Graph.CycleData, Graph.CycleData.range_vtx, Graph.CycleData.vertexSet_ncard}
  \leanok
  \uses{def:graph-operations}
  Cycle-graph data on a multigraph $G$, packaging the cycle branch of the
  dichotomy: an integer $m \ge 3$, injective
  cyclically-indexed families $\mathrm{vtx} \colon \Z/m \to V(G)$ and
  $\mathrm{edge} \colon \Z/m \to E(G)$ with $\mathrm{edge}\,i$ a genuine
  $G$-link from $\mathrm{vtx}\,i$ to the cyclic successor
  $\mathrm{vtx}\,(i+1)$, covering all of $V(G)$ and $E(G)$ --- so $G$
  \emph{is} the cycle $\mathrm{vtx}\,0 - \mathrm{vtx}\,1 - \dots -
  \mathrm{vtx}\,(m-1) - \mathrm{vtx}\,0$. In particular the cycle vertices
  are exactly $V(G)$ (\texttt{range\_vtx}) and $|V(G)| = m$
  (\texttt{vertexSet\_ncard}).
\end{definition}

\begin{lemma}[Chain-or-cycle dichotomy; KT Lemma 4.6]
  \label{lem:chain-cycle-dichotomy}
  \lean{Graph.chainData_or_cycleData_of_noRigid,
        Graph.cycle_isProperRigidSubgraph,
        Graph.isKDof_zero_of_cycle}
  \leanok
  \uses{def:chain-data, def:cycle-data, def:k-dof, def:rigid-subgraph,
        lem:two-edge-conn, lem:no-rigid-edge-count, lem:simple-minimal-noRigid}
  Let $G$ be a minimal $0$-dof-graph with no proper rigid subgraph,
  $D = \binom{n+1}{2} \ge 3$, $|V(G)| \ge 3$, and some edge name unused by
  $G$ (the fresh short-circuit label). Then $G$ contains a length-$n$ chain
  (\cref{def:chain-data}) or $G$ \emph{is} a cycle on $m \le n$ vertices
  (\cref{def:cycle-data}). Both branches occur: at $n \ge 4$ there are
  minimal $0$-dof cycle graphs on $4 \le m \le n$ vertices admitting no
  length-$n$ chain, so the cycle disjunct cannot be strengthened away.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:two-edge-conn, lem:no-rigid-edge-count, lem:simple-minimal-noRigid,
        def:rigid-subgraph}
  KT pp.~664--665, by contradiction: assume neither branch. Minimality
  forces min-degree $\ge 2$ and connectivity (both from the crossing-edge
  count of \cref{lem:two-edge-conn} at singleton and component cuts). From
  every vertex-edge incidence, extend a walk whose interior vertices have
  degree exactly $2$, capped at length $n$. Reaching the cap would yield the
  chain; closing up into a cycle would make $G$ itself a cycle on $\le n$
  vertices (an all-degree-$2$ closed walk spans a connected graph); and a
  \emph{lollipop} --- the walk revisiting an earlier vertex whose degree is
  $\ge 3$ --- is absurd, since its loop is a cycle on $\le n \le D$ vertices
  hanging strictly inside $G$, and a cycle on at most $D$ vertices is a
  $0$-dof subgraph (the deficiency count: every partition class of the cyclic
  index set has a boundary index, injecting parts into crossing edges), i.e.\
  a proper rigid subgraph, contradicting the hypothesis. So under the
  contradiction assumption every such walk terminates at both ends in
  degree-$\ge 3$ vertices after at most $n-1$ edges. Walks are deterministic
  (two walks sharing their first vertex and edge are prefix-comparable), so
  charging each degree-$2$ vertex to the terminated walks through it double
  counts as $2|X_2| \le (n-2) \sum_{\deg \ge 3} \deg v$ --- KT's
  (4.6)+(4.7). Combined with the KT~4.5(i) edge bound
  (\cref{lem:no-rigid-edge-count}), the handshake identity, and the linking
  identity $i(n-2) + 2 \le (D-1)(i-2)$ for $i \ge 3$ (an arithmetic
  consequence of $D = \binom{n+1}{2}$, KT's display above (4.9)), this
  yields the numeric contradiction of KT (4.8)/(4.9).
\end{proof}

\begin{lemma}[The general-$d$ chain extractor; KT Lemmas 4.6 + 4.8]
  \label{lem:chain-data-extract}
  \lean{Graph.chainData_extract,
        Graph.splitOff_simple_of_noRigid_of_card}
  \leanok
  \uses{def:chain-data, def:cycle-data, lem:chain-cycle-dichotomy,
        lem:reduction-step, lem:reduction-measure, lem:simple-minimal-noRigid}
  Let $G$ be a simple minimal $0$-dof-graph with no proper rigid subgraph,
  $D \ge 3$, $|V(G)| \ge 4$, with a fresh edge name available for $G$ (an
  unused label outside $E(G)$). Then either
  $G$ carries a length-$n$ chain (\cref{def:chain-data}) whose
  $v_1$-split $G_{v_1}^{v_0 v_2}$ is again a \emph{simple} minimal
  $0$-dof-graph on at least two and strictly fewer vertices --- the data the
  general-$d$ Case~III argument recurses on --- or $G$ is a cycle on
  $m \le n$ vertices (\cref{def:cycle-data}). This is the general-$d$ form
  of the $d = 3$ extraction pair (\cref{lem:adjacent-degree-two-pair},
  \cref{lem:chain-data-of-noRigid}, which need $D \ge 6$), with the cycle
  branch genuinely reachable and the floor $D \ge 3$ including the planar case
  $n = 2$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:chain-cycle-dichotomy, lem:reduction-step, lem:reduction-measure,
        lem:simple-minimal-noRigid}
  The dichotomy (\cref{lem:chain-cycle-dichotomy}) yields a chain or a
  cycle; the cycle disjunct forwards unchanged. On the chain disjunct, the
  interior vertex $v_1$ has degree $2$ with its two chain edges oriented
  toward $v_0$ and $v_2$, so the KT Lemma-4.8 splitting-off stack applies at
  $v_1$: the split is minimal $0$-dof (\cref{lem:reduction-step}), simple at
  $|V(G)| \ge 4$ (\texttt{splitOff\_simple\_of\_noRigid\_of\_card}, KT
  Lemma~6.7(ii)), and strictly smaller with at least two vertices
  (\cref{lem:reduction-measure}).
\end{proof}
