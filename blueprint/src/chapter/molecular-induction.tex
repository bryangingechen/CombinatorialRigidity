\section{The combinatorial induction: graph operations and KT Theorem~4.9}
\label{sec:molecular-induction}

This chapter proves Katoh and Tanigawa's Lemmas~3.4 and~3.5 and the
results of their \S4~\cite{katohTanigawa2011} on minimal
$k$-dof-graphs, with the induction and some lemmas of their \S6. Two
operations reduce a minimal $k$-dof-graph $G$ with at least three
vertices to a smaller one. If $G$ has a proper rigid subgraph,
contracting it gives a minimal $k$-dof-graph (KT Lemma~3.5). If $G$ has
none, splitting off at a vertex of degree two gives a minimal
$0$-dof-graph when $k = 0$, and a minimal $(k-1)$-dof-graph when $k > 0$
(KT Lemma~4.8). Such a vertex exists when $G$ is $2$-edge-connected
(KT Lemma~4.6), and every $0$-dof-graph is $2$-edge-connected. So every
minimal $0$-dof-graph with at least two vertices reduces to the
two-vertex double edge, which is KT Theorem~4.9. Katoh and Tanigawa
prove their Theorem~5.5 by an induction on the number of vertices over
minimal $k$-dof-graphs of every deficiency $k$. The algebraic induction
(\cref{sec:molecular-algebraic-induction}) constructs the panel-hinge
realizations that this induction needs.

Throughout, $D = \binom{d+1}{2}$, and $\tilde G = (D-1)\cdot G$ is the
multigraph of \cref{sec:body-hinge}, in which each edge of $G$ is
repeated $D - 1$ times. The matroid $M(\tilde G)$, the deficiency,
$k$-dof-graphs and rigid subgraphs are those of
\cref{sec:molecular-deficiency}.

\Cref{sec:molecular-induction-inherited} proves KT Lemmas~3.4 and~3.5.
By the first, the ends of the edges of a circuit of $M(\tilde G)$ induce
a rigid subgraph. We prove the second for the corresponding contraction
of the matroid $M(\tilde G)$. \Cref{sec:molecular-induction-operations}
defines removal, splitting off and edge-splitting at a vertex of degree
two. \Cref{sec:molecular-induction-forest} proves KT Lemmas~4.1
and~4.2, which compare the independent sets of $M(\tilde G)$ before and
after splitting off and edge-splitting. We prove KT Lemma~4.1 for
bases, and show that a base has the partition into forests that its
proof assumes (\cref{rem:kt-lemma-41}).
\Cref{sec:molecular-induction-dof} proves KT Lemmas~4.3 to~4.8. We bound
the deficiency after splitting off and after removal by comparing
partitions of $V(G)$, and find a vertex of degree two when $G$ has no
proper rigid subgraph. \Cref{sec:molecular-induction-thm49} proves KT
Theorem~4.9, the induction over all deficiencies, and KT Lemma~6.1,
which realizes a graph that is not $2$-edge-connected.

The last two subsections concern a minimal $0$-dof-graph $G$ with at
least three vertices and no proper rigid subgraph.
\Cref{sec:simplicity-and-adjacent-pair} shows that $G$ is simple, and
that it has two adjacent vertices of degree two when $d \ge 3$. The last
subsection proves KT Lemma~4.6 for $G$
(\cref{lem:chain-cycle-dichotomy}): $G$ is a cycle on at most $d$
vertices, or it has a chain of length $d$ whose interior vertices have
degree two. Case~III of the algebraic induction starts from this
dichotomy.

\subsection{Circuits and the contraction of a rigid subgraph}
\label{sec:molecular-induction-inherited}

This subsection proves Katoh and Tanigawa's Lemmas~3.4
and~3.5~\cite[\S3.2]{katohTanigawa2011}. By KT Lemma~3.4, the ends of
the edges of a circuit of $M(\tilde G)$ induce a rigid subgraph.
\Cref{lem:circuit-rigid} is its first step, and we complete it with KT
eq.~(2.4). By KT Lemma~3.5, contracting a rigid subgraph of a minimal
$k$-dof-graph gives a minimal $k$-dof-graph. We define the contraction,
and prove the deficiency and minimality statements of KT Lemma~3.5 for
the corresponding contraction of the matroid $M(\tilde G)$, then for the
graph, the form that the algebraic induction uses.

\begin{definition}[The subgraph induced by the ends of a set of edges]
  \label{def:induced-span}
  \lean{Graph.fiberSpan, Graph.inducedSpan}
  \leanok
  \uses{def:edge-multiply, def:body-bar-framework}
  Let $G$ be a multigraph, let $X$ be a set of edges of $\tilde G$, and
  let $V(X)$ be the set of vertices incident with an edge of $X$. Then
  $G[V(X)]$ is the subgraph of $G$ induced by $V(X)$. Its vertices are
  those of $V(X)$, and its edges are the edges of $G$ with both ends in
  $V(X)$.
\end{definition}

\begin{lemma}[A circuit induces a rigid subgraph; KT Lemma~3.4]
  \label{lem:circuit-induces-rigid}
  \lean{Graph.circuit_induces_isTight, Graph.circuit_induces_isRigidSubgraph}
  \leanok
  \uses{lem:circuit-rigid, def:induced-span, def:rigid-subgraph, thm:def-eq-corank}
  Let $D \ge 1$, let $G$ be a multigraph, let $X$ be a circuit of
  $M(\tilde G)$, and let $e \in X$. Then $|X - e| = D(|V(X)| - 1)$, and
  $X - e$ is the union of $D$ edge-disjoint spanning trees on $V(X)$.
  Moreover, $G[V(X)]$ is a rigid subgraph of $G$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{def:matroid-MG, lem:rank-matroidMG-le, lem:matroid-restrict-subgraph, thm:def-eq-corank}
  For the count we follow Katoh and
  Tanigawa~\cite[p.~658]{katohTanigawa2011}. As $X$ is a circuit, $X - e$
  is independent, so $\tilde G \restriction (X - e)$ is $(D, D)$-sparse
  (\cref{lem:circuit-rigid}). If $X - e$ is not empty, sparsity applied
  to $X - e$ itself gives
  \[
    |X - e| \;\le\; D\,(|V(X - e)| - 1) \;\le\; D\,(|V(X)| - 1).
  \]
  If $X - e$ is empty, the bound holds as $V(X)$ is not empty.
  Conversely, $X$ is dependent, so some nonempty $F \subseteq X$ has
  $|F| > D(|V(F)| - 1)$ (\cref{def:matroid-MG}). If $F$ were a proper
  subset of $X$, it would miss some $f \in X$ and lie in the independent
  set $X - f$, whose sparsity gives $|F| \le D(|V(F)| - 1)$. Hence
  $F = X$, and $|X| > D(|V(X)| - 1)$, that is,
  $|X - e| \ge D(|V(X)| - 1)$. So $|X - e| = D(|V(X)| - 1)$.

  If $X - e$ is empty, then $|V(X)| = 1$, and the $D$ spanning trees are
  empty. Otherwise both inequalities above are equalities, so
  $V(X - e) = V(X)$, and the multigraph on $V(X)$ with the edges $X - e$
  is $(D, D)$-tight. It is connected, by the count in the proof of
  \cref{lem:deficiency-zero-iff-tree-packing}. By
  \cref{cor:k-spanning-trees}, $X - e$ is the union of $D$ edge-disjoint
  spanning trees on $V(X)$.

  Let $H = G[V(X)]$. The ends of each edge of $X$ lie in $V(X)$, so
  $X \subseteq E(\tilde H)$. Hence $X - e$ is independent in
  $M(\tilde G)\,|\,E(\tilde H)$, which is $M(\tilde H)$
  (\cref{lem:matroid-restrict-subgraph}). So
  $\operatorname{rank} M(\tilde H) \ge |X - e| = D(|V(H)| - 1)$, and the
  reverse inequality is \cref{lem:rank-matroidMG-le}. By KT eq.~(2.4)
  (\cref{thm:def-eq-corank}), $\operatorname{def}(\tilde H) = 0$. As $H$
  is a subgraph of $G$, it is a rigid subgraph of $G$.
\end{proof}
\begin{fmlnote}
  The first pin is the count, and the second says that $G[V(X)]$ is a
  rigid subgraph. The partition of $X - e$ into spanning trees is not
  formalized.
\end{fmlnote}

If $G$ has no proper rigid subgraph (\cref{def:rigid-subgraph}), the
lemma shows that every circuit $X$ of $M(\tilde G)$ with
$|V(X)| \ge 2$ has $V(X) = V(G)$. Katoh and Tanigawa use it in this
form in the proofs of their Lemmas~4.5 and~4.8~\cite[pp.~663
and~666]{katohTanigawa2011}.

Katoh and Tanigawa's induction also contracts proper rigid
subgraphs~\cite[p.~659]{katohTanigawa2011}. We define the contraction
for any subgraph, which we contract to one of its vertices.

\begin{definition}[Contracting a subgraph]
  \label{def:rigid-contraction}
  \lean{Graph.rigidContract}
  \leanok
  \uses{def:rigid-subgraph}
  Let $H$ be a subgraph of a multigraph $G$, and let $r$ be a vertex of
  $H$. The \emph{contraction} $G/E(H)$ of $H$ to $r$ is obtained from
  $G$ by deleting the edges of $H$ and identifying every vertex of $H$
  with $r$. Its vertices are $r$ and the vertices of $G$ not in $H$. Its
  edges are the edges of $G$ not in $H$, and each end of such an edge
  that lies in $V(H)$ is replaced by $r$.
\end{definition}

So $G/E(H)$ has $|V(G)| - |V(H)| + 1$ vertices, and
\[
  D\,\bigl(|V(G/E(H))| - 1\bigr) \;=\; D\,\bigl(|V(G)| - |V(H)|\bigr).
\]

Katoh and Tanigawa~\cite[pp.~658--659]{katohTanigawa2011} prove their
Lemma~3.5 with partitions of bases into forests. We work instead with
the contraction of $M(\tilde G)$ by $E(\tilde H)$. For a matroid $M$ on
a set $E$ and a set $T \subseteq E$, the contraction $M/T$ is the
matroid on $E \setminus T$ whose rank function is
$X \mapsto r_M(X \cup T) - r_M(T)$~\cite[\S3.1 and
Proposition~3.1.6]{oxley2011}. The ground set of
$M(\tilde G)/E(\tilde H)$ consists of the copies of the edges of $G$
not in $H$, which are the edges of $\widetilde{G/E(H)}$. We first
compute its rank.

\begin{lemma}[The rank of a rigid subgraph; KT Lemma~3.5, first step]
  \label{lem:rigid-full-rank}
  \lean{Graph.rank_matroidMG_of_isKDof_zero}
  \leanok
  \uses{def:rigid-subgraph, thm:def-eq-corank, def:matroid-MG}
  Let $D \ge 1$, and let $H$ be a rigid subgraph of a multigraph $G$,
  with at least one vertex. Then
  $\operatorname{rank} M(\tilde H) = D(|V(H)| - 1)$.
\end{lemma}
\begin{proof}
  \leanok
  By KT eq.~(2.4) (\cref{thm:def-eq-corank}),
  $\operatorname{rank} M(\tilde H) + \operatorname{def}(\tilde H) =
  D(|V(H)| - 1)$, and $\operatorname{def}(\tilde H) = 0$ as $H$ is
  rigid.
\end{proof}

\begin{lemma}[The rank of $M(\tilde G)/E(\tilde H)$]
  \label{lem:contract-rank-bridge}
  \lean{Graph.contract_matroidMG_rank}
  \leanok
  \uses{def:rigid-contraction, def:matroid-MG, lem:matroid-restrict-subgraph}
  Let $H$ be a subgraph of a multigraph $G$. Then
  \[
    \operatorname{rank}\bigl(M(\tilde G) / E(\tilde H)\bigr)
      + \operatorname{rank} M(\tilde H)
      = \operatorname{rank} M(\tilde G).
  \]
\end{lemma}
\begin{proof}
  \leanok
  A matroid $M$ on $E$ and a set $T \subseteq E$ satisfy
  $\operatorname{rank}(M/T) = \operatorname{rank} M - r_M(T)$, by
  Proposition~3.1.6 of Oxley~\cite{oxley2011} at $X = E \setminus T$.
  Take $M = M(\tilde G)$ and $T = E(\tilde H)$. Then $r_M(T)$ is the
  rank of $M(\tilde G)\,|\,E(\tilde H)$, which is $M(\tilde H)$
  (\cref{lem:matroid-restrict-subgraph}).
\end{proof}

\begin{lemma}[Contracting a rigid subgraph keeps the deficiency; KT Lemma~3.5]
  \label{lem:contract-deficiency-conservation}
  \lean{Graph.contract_matroidMG_deficiency_eq}
  \leanok
  \uses{lem:rigid-full-rank, lem:contract-rank-bridge, thm:def-eq-corank}
  Let $D \ge 1$, and let $H$ be a rigid subgraph of a multigraph $G$,
  with at least one vertex. Then
  \[
    D\bigl(|V(G)| - |V(H)|\bigr) -
    \operatorname{rank}\bigl(M(\tilde G) / E(\tilde H)\bigr)
      = \operatorname{def}(\tilde G).
  \]
\end{lemma}
\begin{proof}
  \leanok
  By \cref{lem:contract-rank-bridge} and \cref{lem:rigid-full-rank},
  \[
    \operatorname{rank}\bigl(M(\tilde G) / E(\tilde H)\bigr) \;=\;
    \operatorname{rank} M(\tilde G) - D\,(|V(H)| - 1).
  \]
  The vertex set of $G$ contains that of $H$, so it is not empty, and
  KT eq.~(2.4) (\cref{thm:def-eq-corank}) gives
  $\operatorname{rank} M(\tilde G) = D(|V(G)| - 1) -
  \operatorname{def}(\tilde G)$. Substituting this into the first
  identity gives the lemma. This is Katoh and Tanigawa's
  count~\cite[p.~658]{katohTanigawa2011}.
\end{proof}

The next lemma is the analogue of \cref{lem:subgraph-minimality} for
the contraction.

\begin{lemma}[Minimality passes to the contraction; KT Lemma~3.5]
  \label{lem:contract-minimality-transport}
  \lean{Graph.contract_minimality_transport}
  \leanok
  \uses{lem:subgraph-minimality, def:k-dof, def:rigid-contraction}
  Let $G$ be a minimal $k$-dof-graph, and let $H$ be a subgraph of $G$.
  Then every base of $M(\tilde G)/E(\tilde H)$ meets $\tilde e$ for every
  edge $e$ of $G$ that is not an edge of $H$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{def:matroid-MG}
  Let $B'$ be a base of $M(\tilde G)/E(\tilde H)$, and let $J$ be a
  maximal subset of $E(\tilde H)$ that is independent in $M(\tilde G)$.
  By Corollary~3.1.8 of Oxley~\cite{oxley2011}, $B' \cup J$ is a base of
  $M(\tilde G)$. Let $e$ be an edge of $G$ that is not an edge of $H$.
  As $G$ is minimal, $B' \cup J$ meets $\tilde e$. The copies of $e$ are
  not in $E(\tilde H)$, which contains $J$, so $B'$ meets $\tilde e$.
  This is Katoh and Tanigawa's argument~\cite[p.~659]{katohTanigawa2011}.
\end{proof}

\begin{lemma}[The contraction of a proper rigid subgraph; KT Lemma~3.5]
  \label{lem:contraction-minimality}
  \lean{Graph.contraction_isMinimalKDof}
  \leanok
  \uses{def:rigid-subgraph, lem:subgraph-minimality, def:k-dof, lem:rigid-full-rank, lem:contract-rank-bridge, lem:contract-deficiency-conservation, lem:contract-minimality-transport}
  Let $D \ge 1$, let $G$ be a minimal $k$-dof-graph, and let $H$ be a
  proper rigid subgraph of $G$. Then
  \[
    D\bigl(|V(G)| - |V(H)|\bigr) -
    \operatorname{rank}\bigl(M(\tilde G) / E(\tilde H)\bigr) = k,
  \]
  and every base of $M(\tilde G)/E(\tilde H)$ meets $\tilde e$ for every
  edge $e$ of $G$ that is not an edge of $H$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:contract-deficiency-conservation, lem:contract-minimality-transport}
  As $H$ is proper, it has at least two vertices, and
  $\operatorname{def}(\tilde G) = k$ as $G$ is a $k$-dof-graph. So the
  identity is \cref{lem:contract-deficiency-conservation}, and the
  second claim is \cref{lem:contract-minimality-transport}.
\end{proof}

Katoh and Tanigawa state their Lemma~3.5 for the graph $G/E(H)$. It
follows from \cref{lem:contraction-minimality} and KT eq.~(2.4) for
$G/E(H)$ if the matroid $M(\widetilde{G/E(H)})$ is
$M(\tilde G)/E(\tilde H)$. The two matroids have the same ground set,
and they are equal because $H$ is rigid. The reason is Katoh and
Tanigawa's (3.1)~\cite[p.~658]{katohTanigawa2011}. Let a base of
$M(\tilde G)$ contain a base of $M(\tilde H)$, and let it be the union
of $D$ disjoint forests. Then each of these forests meets $E(\tilde H)$
in a spanning tree on $V(H)$. The algebraic induction applies its
induction hypothesis to $G/E(H)$. For $D \ge 2$ we prove the identity,
and with it Katoh and Tanigawa's form of their Lemma~3.5.

\begin{lemma}[The contraction is a minimal $k$-dof-graph; KT Lemma~3.5]
  \label{lem:rigidContract-isMinimalKDof}
  \lean{Graph.rigidContract_isMinimalKDof}
  \leanok
  \uses{lem:contraction-minimality, def:rigid-contraction, lem:reduction-measure}
  Let $D \ge 2$, let $G$ be a minimal $k$-dof-graph, let $H$ be a proper
  rigid subgraph of $G$, and let $G/E(H)$ be the contraction of $H$ to a
  vertex of $H$ (\cref{def:rigid-contraction}). Then $G/E(H)$ is a
  minimal $k$-dof-graph with fewer vertices than $G$
  (\cref{lem:reduction-measure}).
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:contraction-minimality, thm:def-eq-corank, lem:two-edge-conn,
        def:matroid-MG, lem:rigid-full-rank, thm:tutte-nash-williams}
  By \cref{lem:contraction-minimality},
  \[
    D\bigl(|V(G)| - |V(H)|\bigr)
      - \operatorname{rank}\bigl(M(\tilde G)/E(\tilde H)\bigr) \;=\; k,
  \]
  and every base of $M(\tilde G)/E(\tilde H)$ meets $\tilde e$ for every
  edge $e$ of $G$ not in $H$. These edges are the edges of $G/E(H)$, and
  their copies form the common ground set of $M(\tilde G)/E(\tilde H)$
  and $M(\widetilde{G/E(H)})$. We show below that
  \[
    M\bigl(\widetilde{G/E(H)}\bigr) \;=\; M(\tilde G)/E(\tilde H).
  \]
  Then every base of $M(\widetilde{G/E(H)})$ meets the copies of every
  edge of $G/E(H)$. The graph $G/E(H)$ has $|V(G)| - |V(H)| + 1$
  vertices (\cref{lem:reduction-measure}), so KT eq.~(2.4)
  (\cref{thm:def-eq-corank}) gives
  \[
    \operatorname{def}\bigl(\widetilde{G/E(H)}\bigr)
    \;=\; D\bigl(|V(G)| - |V(H)|\bigr)
      - \operatorname{rank} M\bigl(\widetilde{G/E(H)}\bigr) \;=\; k.
  \]
  Hence $G/E(H)$ is a minimal $k$-dof-graph.

  The identity is not a renaming of edges. Let $M_c$ be the cycle matroid
  of $\tilde G$, so that $M(\tilde G)$ is the union of $D$ copies of
  $M_c$ (\cref{def:matroid-MG}). As $H$ is rigid, it is connected
  (\cref{lem:two-edge-conn}), and so is $\tilde H$, since $D \ge 2$ gives
  every edge of $H$ a copy in $\tilde H$. So contracting the edges of
  $\tilde H$ in $\tilde G$ identifies all the vertices of $H$, and gives
  $\widetilde{G/E(H)}$. The cycle matroid of $\widetilde{G/E(H)}$ is
  therefore $M_c/E(\tilde H)$~\cite[Proposition~3.2.1]{oxley2011}. So
  $M(\widetilde{G/E(H)})$ is the union of $D$ copies of $M_c/E(\tilde H)$,
  while $M(\tilde G)/E(\tilde H)$ is the contraction of the union of $D$
  copies of $M_c$. In general a union of matroids does not commute with
  contraction. Let $M$ be the matroid of rank one on two elements $a$ and
  $c$, neither of which is a loop. Then $a$ is independent in the
  contraction by $c$ of the union of two copies of $M$, but $a$ is a loop
  of $M/c$, and so of the union of two copies of $M/c$.

  The two matroids coincide here because the union attains $D$ times the
  cycle rank on $E(\tilde H)$. Let $r_c$ be the rank function of $M_c$.
  The rank of $E(\tilde H)$ in $M(\tilde G)$ is the rank of $M(\tilde H)$
  (\cref{lem:matroid-restrict-subgraph}), which is $D(|V(H)| - 1)$ as $H$
  is rigid (\cref{lem:rigid-full-rank}). The connected graph $\tilde H$
  has $r_c(E(\tilde H)) = |V(H)| - 1$. So
  \[
    \operatorname{rank}_{M(\tilde G)} E(\tilde H) \;=\;
    D\, r_c\bigl(E(\tilde H)\bigr).
  \]
  Hence a base of $E(\tilde H)$ in $M(\tilde G)$ has $D(|V(H)| - 1)$
  elements. It is the union of $D$ disjoint forests in $\tilde H$
  (\cref{thm:tutte-nash-williams}), each with at most $|V(H)| - 1$
  edges, and so each of them is a spanning tree on $V(H)$. This is Katoh and Tanigawa's (3.1)~\cite[p.~658]{katohTanigawa2011}, on
  which their proof of Lemma~3.5 rests.

  We derive the identity from this equality. A set $X$ is independent in
  the union of $D$ copies of a matroid with rank function $\rho$ if and
  only if $|Y| \le D\,\rho(Y)$ for every $Y \subseteq X$
  (\cref{thm:matroid-partition}). Let
  $C = E(\tilde H)$, let $J$ be a base of $C$ in $M(\tilde G)$, so that
  $|J| = D\, r_c(C)$, and let $I$ be a set of copies of edges not in
  $H$. Then $I$ is independent in $M(\tilde G)/C$ if and only if
  $I \cup J$ is independent in
  $M(\tilde G)$~\cite[Proposition~3.1.7]{oxley2011}, that is, if and only
  if $|Z| \le D\, r_c(Z)$ for every $Z \subseteq I \cup J$. The rank
  function of $M_c/C$ is $Y \mapsto r_c(Y \cup C) - r_c(C)$~\cite[Proposition~3.1.6]{oxley2011}.
  So $I$ is independent in the union of $D$ copies of $M_c/C$ if and only
  if $|Y| \le D\,\bigl(r_c(Y \cup C) - r_c(C)\bigr)$ for every
  $Y \subseteq I$.

  Suppose this second condition, and let $Z \subseteq I \cup J$. Write
  $Z = Y \cup W$ with $Y \subseteq I$ and $W \subseteq J$. Since
  $W \subseteq C$, submodularity gives
  $r_c(Y \cup C) + r_c(W) \le r_c(Y \cup W) + r_c(C)$. Also
  $|W| \le D\, r_c(W)$, as $J$ is independent in $M(\tilde G)$. Adding
  these two bounds to the bound for $Y$ gives $|Z| \le D\, r_c(Z)$.
  Conversely, suppose the first condition, and let $Y \subseteq I$. Then
  \[
    |Y| + D\, r_c(C) \;=\; |Y \cup J| \;\le\; D\, r_c(Y \cup J)
    \;\le\; D\, r_c(Y \cup C).
  \]
  So the two matroids have the same independent sets. The converse used
  $|J| = D\, r_c(C)$, and it fails without it, as in the example above.
\end{proof}

\subsection{Removal, splitting off and edge-splitting}
\label{sec:molecular-induction-operations}

The other operation of the induction acts at a vertex of degree two.
Katoh and Tanigawa define splitting off at a vertex $v$ as the removal
of $v$ followed by the insertion of new edges between neighbours of
$v$, and they use it only when $v$ has degree two~\cite[\S4.1,
p.~660]{katohTanigawa2011}. They also use its inverse, edge-splitting
(p.~660), and the removal of $v$ alone (p.~662).

\begin{definition}[Removal, splitting off and edge-splitting]
  \label{def:graph-operations}
  \lean{Graph.removeVertex, Graph.splitOff, Graph.edgeSplit}
  \leanok
  \uses{def:k-dof}
  Let $G$ be a multigraph, and let $v$ be a vertex of $G$ of degree two,
  whose two edges join $v$ to vertices $a$ and $b$.
  \begin{itemize}
    \item The \emph{removal} of $v$ gives $G_v = G - v$, obtained from
      $G$ by deleting $v$ and its two edges.
    \item The \emph{splitting off} at $v$ gives $G_v^{ab}$, obtained from
      $G$ by deleting $v$ and its two edges and adding a new edge $ab$.
  \end{itemize}
  Let $H$ be a multigraph with an edge $ab$. The \emph{edge-splitting}
  along $ab$ gives $H_{ab}^v$, obtained from $H$ by deleting $ab$ and
  adding a new vertex $v$ and two new edges $va$ and $vb$.
\end{definition}
\begin{fmlnote}
  \label{fmlnote:graph-operations}
  In the formalization the edges of a multigraph are elements of a fixed
  set of labels. An operation that adds an edge takes the label of the
  new edge as an argument: the label $e_0$ of the edge $ab$ of
  $G_v^{ab}$, and the labels of $va$ and $vb$ in $H_{ab}^v$. The formal
  splitting off deletes every edge with the label $e_0$ before it adds
  $ab$. So it gives $G_v^{ab}$ when no edge of $G - v$ has the label
  $e_0$, for instance when $e_0 \notin E(G)$. A label outside $E(G)$
  exists when there are enough labels (\cref{rem:fresh-edge-supply}).
  The formalization also fixes, for each edge of $G$, an ordered pair of
  its ends (\texttt{endsOf}), the \emph{endpoint choice} of $G$. The
  realizations of Cases~I and~II of the algebraic induction take the
  ends of the edges of $G$ from it
  (\cref{fmlnote:panel-hinge-nondegenerate}).
\end{fmlnote}

Splitting off the new vertex $v$ of $H_{ab}^v$ gives back $H$.

\subsection{Forest surgery: KT Lemmas~4.1 and~4.2}
\label{sec:molecular-induction-forest}

Let $v$ be a vertex of $G$ of degree two, whose edges $e_a$ and $e_b$
join it to vertices $a, b \neq v$. For an edge $e$ of $G$, let
$\tilde e$ be the set of its $D - 1$ copies in $\tilde G$
(\cref{def:k-dof}). The edges of $\tilde G$ incident with $v$ are the
$2(D - 1)$ edges of $\tilde e_a \cup \tilde e_b$. We write
$\widetilde{ab}$ for the set of the $D - 1$ copies of the new edge $ab$
of $G_v^{ab}$. Katoh and Tanigawa~\cite[p.~660]{katohTanigawa2011}
compare the independent sets of $M(\tilde G)$ and of
$M(\widetilde{G_v^{ab}})$ in their Lemma~4.1. They prove it by a
\emph{forest surgery}: they partition an independent set of
$M(\tilde G)$ into $D$ forests of $\tilde G$, and change each forest at
$v$. Their proof assumes that every forest of the partition has an edge
incident with $v$.

We discuss the lemma and this assumption in \cref{rem:kt-lemma-41}, and
then prove the lemma for bases in three steps. The independent sets of
$M(\tilde G)$ are the unions of $D$ forests
(\cref{lem:forest-packing-decomp}). A base has a partition into $D$
forests each of which has an edge incident with $v$
(\cref{lem:balanced-forest-packing}). When $a \neq b$, the surgery on
such a partition gives the two conclusions of the lemma
(\cref{lem:forest-surgery-count}). The second step uses KT Lemma~4.4
(\cref{lem:removal-deficiency}). We prove that lemma in
\cref{sec:molecular-induction-dof} by comparing the values of partitions
of $V(G)$, without forests. KT Lemma~4.2 (\cref{lem:edge-splitting}),
the converse for edge-splitting, ends the subsection.

\begin{remark}[On KT Lemma~4.1]
  \label{rem:kt-lemma-41}
  KT Lemma~4.1~\cite[p.~660]{katohTanigawa2011} states that for any
  independent set $I$ of $M(\tilde G)$ there is an independent set $I'$
  of $M(\widetilde{G_v^{ab}})$ with $|I'| = |I| - D$ and
  $|\widetilde{ab} \cap I'| < D - 1$. The arXiv version of the paper
  (arXiv:0902.0236, 2009, Lemma~5.1, p.~11) has essentially the same
  statement and proof. We separate three points.
  \begin{enumerate}
    \item \emph{The statement holds only for bases.} If $|I| < D$, for
      instance if $I = \emptyset$, then no set $I'$ has
      $|I'| = |I| - D$ (\cref{ex:kt-41-overquantified}). Katoh and
      Tanigawa apply the lemma to a base, in the proof of their
      Lemma~4.3, and we read it for bases.
    \item \emph{The proof assumes a balanced partition.} Let
      $F_1, \dots, F_D$ be a partition of $I$ into forests of $\tilde G$
      (\cref{lem:forest-packing-decomp}). A forest has at most two edges
      incident with $v$, at most one in $\tilde e_a$ and one in
      $\tilde e_b$, since two copies of an edge form a cycle. Let
      $h = |(\tilde e_a \cup \tilde e_b) \cap I|$, and let $h'$ be the
      number of forests with two edges incident with $v$. Katoh and
      Tanigawa note that $2h' + (D - h') = h$, and they turn each $F_i$
      into a forest $F'_i$ of $\widetilde{G_v^{ab}}$ with
      $|F'_i| = |F_i| - 1$. Both steps assume that every $F_i$ has an
      edge incident with $v$. We call such a partition \emph{balanced at
      $v$}. The proof does not show that $I$ has a balanced partition,
      and if $h < D$, it has none.

      We show that a base $B$ has one. First, $B$ contains at least $D$
      edges incident with $v$ (\cref{lem:base-vfiber-count}). Now let
      $F_j$ be a forest of a partition of $B$ with no edge incident with
      $v$. The other $D - 1$ forests contain at least $D$ such edges, so
      one of them contains two. Moving one of these two to $F_j$ gives a
      new partition of $B$ into forests, as the moved edge is a
      pendant edge at $v$ of the new $F_j$
      (\cref{lem:acyclic-insert-vfiber}).
      Each move lowers the number of forests with no edge incident with
      $v$, so repeated moves give a balanced partition
      (\cref{lem:balanced-forest-packing}). So, for bases, this step is a
      gap in the proof and not an error.

      On a balanced partition the surgery goes as Katoh and Tanigawa
      describe it. A forest with two edges $p_a \in \tilde e_a$ and
      $p_b \in \tilde e_b$ replaces them by one copy of $ab$
      (\cref{lem:reroute-cycle-lift}). A forest with one edge incident
      with $v$ loses it and gains nothing
      (\cref{lem:reroute-vfree-transport}). Each forest loses one edge,
      so $|I'| = |I| - D$: the $h$ edges incident with $v$ leave, and
      $h'$ copies of $ab$ enter. This count is the equation
      $2h' + (D - h') = h$. As $h \le 2(D - 1)$, it gives
      $h' \le D - 2$. So the $D - 1$ copies of $ab$ suffice for distinct
      forests to take distinct copies, and
      $|\widetilde{ab} \cap I'| = h' < D - 1$
      (\cref{lem:forest-surgery-count}). The count and KT eq.~(2.4)
      (\cref{thm:def-eq-corank}) give
      $\operatorname{def}(\widetilde{G_v^{ab}}) \le
      \operatorname{def}(\tilde G)$, which is
      \cref{lem:forest-surgery-split}.
    \item \emph{What the induction uses.} The proof of KT Theorem~4.9
      needs KT Lemma~4.1 only through its consequence
      $\operatorname{def}(\widetilde{G_v^{ab}}) \le
      \operatorname{def}(\tilde G)$, the upper bound of KT
      Lemma~4.3(i). We prove the deficiency bounds of KT Lemmas~4.3(i)
      and~4.4 in \cref{sec:molecular-induction-dof}, by comparing the
      values of partitions of $V(G)$, without forests. These bounds
      (\cref{lem:dof-tracking}) are what our proof of KT Theorem~4.9
      uses. The surgery also gives the second conclusion of KT Lemma~4.1,
      $|\widetilde{ab} \cap I'| < D - 1$. From it we derive one half of
      KT Lemma~4.3(ii) (\cref{lem:splitoff-kdof-criterion}), which the
      algebraic induction uses in Cases~II and~III. The other half uses
      KT Lemma~4.2, the converse for edge-splitting, which adds edges at
      the new vertex and needs no balanced partition
      (\cref{lem:edge-splitting}).
  \end{enumerate}
\end{remark}

\begin{example}[KT Lemma~4.1 fails for $I = \emptyset$]
  \label{ex:kt-41-overquantified}
  \lean{Graph.kt_lemma_41_overquantified}
  \leanok
  \uses{def:matroid-MG, rem:kt-lemma-41}
  Let $D \ge 1$. The empty set is independent in $M(\tilde G)$, and no
  set $I'$ has $|I'| + D = |\emptyset| = 0$. So no independent set $I'$
  of $M(\widetilde{G_v^{ab}})$ has $|I'| = |\emptyset| - D$, and KT
  Lemma~4.1, stated for every independent set, fails at
  $I = \emptyset$.
\end{example}

\begin{lemma}[The independent sets of $M(\tilde G)$ are the unions of $D$ forests]
  \label{lem:forest-packing-decomp}
  \lean{Graph.matroidMG_indep_iff_exists_forest_packing}
  \leanok
  \uses{def:matroid-MG}
  Let $G$ be a multigraph. A set $I$ of edges of $\tilde G$ is
  independent in $M(\tilde G)$ if and only if
  $I = F_1 \cup \dots \cup F_D$ for some forests $F_1, \dots, F_D$ of
  $\tilde G$.
\end{lemma}
\begin{proof}
  \leanok
  By \cref{def:matroid-MG}, $M(\tilde G)$ is the union of $D$ copies of
  the cycle matroid of $\tilde G$. The independent sets of the cycle
  matroid are the forests (\cref{sec:body-bar-matroid-union}). The
  independent sets of a union of matroids are the unions of independent
  sets of the matroids (\cref{lem:union-indep-iff}).
\end{proof}

A subset of a forest is a forest. So removing from each $F_i$ the edges
of $F_1, \dots, F_{i-1}$ gives pairwise disjoint forests with the same
union. Hence every independent set of $M(\tilde G)$ has a partition
into $D$ forests. Katoh and Tanigawa begin their proofs of Lemmas~4.1
and~4.2 with such a partition.

\begin{lemma}[A base contains at least $D$ edges at a vertex of degree two]
  \label{lem:base-vfiber-count}
  \lean{Graph.isBase_vfiber_ncard_ge}
  \leanok
  \uses{def:graph-operations, def:matroid-MG, thm:def-eq-corank, lem:removal-deficiency, rem:kt-lemma-41}
  Let $D \ge 2$, and let $G$ be a multigraph with a vertex $v$ of degree
  two, whose edges $e_a$ and $e_b$ join $v$ to vertices $a, b \neq v$.
  Then every base $B$ of $M(\tilde G)$ contains at least $D$ of the
  $2(D - 1)$ edges of $\tilde G$ incident with $v$:
  \[
    \bigl|\, B \cap (\tilde e_a \cup \tilde e_b)\,\bigr| \;\ge\; D,
  \]
  where $\tilde e$ is the set of the $D - 1$ copies of $e$ in
  $\tilde G$.
\end{lemma}
\begin{proof}
  \leanok
  Let $G_v = G - v$ (\cref{def:graph-operations}), and let
  $B_0 = B \setminus (\tilde e_a \cup \tilde e_b)$. An edge of $B_0$ is
  a copy of an edge of $G$ other than $e_a$ and $e_b$, so it is not
  incident with $v$. Hence $B_0$ is a set of edges of
  $\widetilde{G_v}$. It is independent in $M(\tilde G)$, and hence in
  the restriction $M(\widetilde{G_v})$ of $M(\tilde G)$
  (\cref{lem:matroid-restrict-subgraph}). So
  $|B_0| \le \operatorname{rank} M(\widetilde{G_v})$.

  The multigraph $G_v$ has the vertex $a$, so KT eq.~(2.4)
  (\cref{thm:def-eq-corank}) applies to $G$ and to $G_v$:
  \[
    |B| = D\,(|V(G)| - 1) - \operatorname{def}(\tilde G), \qquad
    \operatorname{rank} M(\widetilde{G_v}) = D\,(|V(G)| - 2) -
    \operatorname{def}(\widetilde{G_v}).
  \]
  Hence
  \[
    \bigl|\, B \cap (\tilde e_a \cup \tilde e_b)\,\bigr| \;=\;
    |B| - |B_0| \;\ge\;
    D + \operatorname{def}(\widetilde{G_v}) - \operatorname{def}(\tilde G)
    \;\ge\; D.
  \]
  The last step is KT Lemma~4.4 (\cref{lem:removal-deficiency}), which
  uses $D \ge 2$: it gives
  $\operatorname{def}(\tilde G) \le \operatorname{def}(\widetilde{G_v})$.
\end{proof}

\begin{lemma}[A forest with no edge at $v$ accepts an edge at $v$]
  \label{lem:acyclic-insert-vfiber}
  \lean{Graph.acyclicSet_insert_vfiber_of_not_inc}
  \leanok
  \uses{def:matroid-MG, rem:kt-lemma-41, lem:base-vfiber-count}
  Let $G$ be a multigraph with a vertex $v$ of degree two. Let $F$ be a
  forest of $\tilde G$ none of whose edges is incident with $v$, and let
  $x$ be an edge of $\tilde G$ that joins $v$ to a vertex $w \neq v$.
  Then $F \cup \{x\}$ is a forest of $\tilde G$.
\end{lemma}
\begin{proof}
  \leanok
  In $\tilde G \restriction (F \cup \{x\})$ the only edge incident with
  $v$ is $x$, and $x$ is not a loop. So deleting $x$ leaves $v$
  isolated, and in particular separates $v$ from $w$: the edge $x$ is a
  bridge. A bridge lies on no cycle, and $F$ contains no cycle, so
  $F \cup \{x\}$ contains none. The argument uses only that no edge of
  $F$ is incident with $v$, not the degree of $v$.
\end{proof}

\begin{lemma}[Moving an edge at $v$ into a forest with none]
  \label{lem:packing-move}
  \lean{Graph.exists_packing_move_of_not_inc}
  \leanok
  \uses{def:matroid-MG, lem:acyclic-insert-vfiber, lem:base-vfiber-count}
  Let $G$ be a multigraph with a vertex $v$ of degree two, and let
  $F_1, \dots, F_D$ be forests of $\tilde G$ with union $I$. Suppose
  that no edge of $F_j$ is incident with $v$, and let $x \in I$ be an
  edge of $\tilde G$ that joins $v$ to a vertex $w \neq v$. Then there
  are forests $F'_1, \dots, F'_D$ of $\tilde G$ with union $I$ such that
  $x \in F'_j$.
\end{lemma}
\begin{proof}
  \leanok
  Let $F'_j = F_j \cup \{x\}$, and let $F'_i = F_i \setminus \{x\}$ for
  $i \neq j$. Then $F'_j$ is a forest by
  \cref{lem:acyclic-insert-vfiber}, and each $F'_i$ with $i \neq j$ is
  a forest as a subset of $F_i$. The union is still $I$, since $x \in I$
  moves from the other forests to $F'_j$.
\end{proof}

\begin{lemma}[A base has a balanced partition into forests]
  \label{lem:balanced-forest-packing}
  \lean{Graph.exists_balanced_forest_packing}
  \leanok
  \uses{def:matroid-MG, lem:forest-packing-decomp, lem:base-vfiber-count, lem:acyclic-insert-vfiber, lem:packing-move, rem:kt-lemma-41}
  Let $D \ge 2$, and let $G$ be a multigraph with a vertex $v$ of degree
  two, whose edges join $v$ to vertices $a, b \neq v$. Then every base
  $B$ of $M(\tilde G)$ is the union of $D$ pairwise disjoint forests of
  $\tilde G$, each of which has an edge incident with $v$.
\end{lemma}
\begin{proof}
  \leanok
  By \cref{lem:forest-packing-decomp}, $B$ is the union of $D$ forests,
  which we make pairwise disjoint as above. We induct on the number of
  these forests with no edge incident with $v$. If there is none, we are
  done. Otherwise let $F_j$ be one. By \cref{lem:base-vfiber-count}, $B$
  contains at least $D$ edges incident with $v$, and they lie in the
  $D - 1$ forests other than $F_j$. So one of these, $F_i$, contains two
  of them. Let $x$ be one of the two. It joins $v$ to $a$ or to $b$.

  Move $x$ from $F_i$ to $F_j$, as in the proof of
  \cref{lem:packing-move}. The new $F_j$ is a forest
  (\cref{lem:acyclic-insert-vfiber}), and the forests are still pairwise
  disjoint with union $B$, since $x$ lay in $F_i$ only. Now $F_j$ has an
  edge incident with $v$, and $F_i$ keeps its other one. So the number
  of forests with no edge incident with $v$ drops by one, and the
  induction hypothesis gives the partition.
\end{proof}

We now perform the surgery on a balanced partition of a base.

\begin{lemma}[Removing the edges at $v$ from a forest]
  \label{lem:reroute-vfree-transport}
  \lean{Graph.isAcyclicSet_splitOff_of_diff_fiberAtVertex}
  \leanok
  \uses{def:graph-operations, def:matroid-MG}
  Let $G$ be a multigraph with a vertex $v$ of degree two, whose edges
  join $v$ to vertices $a, b \neq v$, and let $F$ be a forest of
  $\tilde G$. Then the edges of $F$ that are not incident with $v$ form a
  forest of $\widetilde{G_v^{ab}}$.
\end{lemma}
\begin{proof}
  \leanok
  The edges of $\tilde G$ not incident with $v$ are the edges of
  $\widetilde{G_v}$, where $G_v = G - v$ (\cref{def:graph-operations}).
  As $G_v$ is a subgraph of $G_v^{ab}$, they are edges of
  $\widetilde{G_v^{ab}}$ with the same ends. Those of them in $F$
  contain no cycle, since $F$ contains none.
\end{proof}

\begin{lemma}[Replacing the two edges at $v$ of a forest by a copy of $ab$]
  \label{lem:reroute-cycle-lift}
  \lean{Graph.isAcyclicSet_splitOff_reroute}
  \leanok
  \uses{def:graph-operations, def:matroid-MG, lem:reroute-vfree-transport}
  Let $G$ be a multigraph with a vertex $v$ of degree two, whose edges
  $e_a$ and $e_b$ join $v$ to vertices $a, b \neq v$. Let $F$ be a forest
  of $\tilde G$ whose edges incident with $v$ are $p_a \in \tilde e_a$ and
  $p_b \in \tilde e_b$, and let $r \in \widetilde{ab}$. Then
  $(F \setminus \{p_a, p_b\}) \cup \{r\}$ is a forest of
  $\widetilde{G_v^{ab}}$.
\end{lemma}
\begin{proof}
  \leanok
  Let $F_0 = F \setminus \{p_a, p_b\}$, the set of edges of $F$ not
  incident with $v$. By \cref{lem:reroute-vfree-transport}, $F_0$ is a
  forest of $\widetilde{G_v^{ab}}$, so a cycle $C$ of $F_0 \cup \{r\}$
  passes through $r$. Then $C - r$ is a path from $a$ to $b$ in $F_0$. Its
  edges have the same ends in $\tilde G$, and it does not pass through
  $v$. So together with $p_a$ and $p_b$ it forms a cycle of $\tilde G$ in
  $F$, which is impossible. Hence $F_0 \cup \{r\}$ contains no cycle.
\end{proof}

\begin{lemma}[The surgery on a balanced partition]
  \label{lem:forest-surgery-count}
  \lean{Graph.forest_surgery_count}
  \leanok
  \uses{def:graph-operations, def:matroid-MG, lem:balanced-forest-packing, lem:reroute-cycle-lift, lem:reroute-vfree-transport}
  Let $D \ge 2$, and let $G$ be a multigraph with a vertex $v$ of degree
  two, whose edges $e_a$ and $e_b$ join $v$ to distinct vertices
  $a, b \neq v$. Let $F_1, \dots, F_D$ be pairwise disjoint forests of
  $\tilde G$, each of which has an edge incident with $v$, and let $I$ be
  their union. Then there are pairwise disjoint forests
  $F'_1, \dots, F'_D$ of $\widetilde{G_v^{ab}}$ whose union $I'$ is
  independent in $M(\widetilde{G_v^{ab}})$ and satisfies
  \[
    |I'| = |I| - D \qquad\text{and}\qquad
    |\widetilde{ab} \cap I'| < D - 1.
  \]
\end{lemma}
\begin{proof}
  \leanok
  Two copies of an edge form a cycle, so each $F_i$ has at most one edge
  in $\tilde e_a$ and at most one in $\tilde e_b$. These are its only
  edges incident with $v$, and it has at least one, so it has one or two.
  Let $h'$ be the number of forests with two, and let
  $h = |(\tilde e_a \cup \tilde e_b) \cap I|$. As the $F_i$ are disjoint,
  $h = 2h' + (D - h') = D + h'$. As
  $|\tilde e_a \cup \tilde e_b| = 2(D - 1)$, also $h \le 2(D - 1)$, so
  $h' \le D - 2$.

  Since $\widetilde{ab}$ has $D - 1 > h'$ elements, we can choose a
  different copy $r_i \in \widetilde{ab}$ for each forest $F_i$ with two
  edges incident with $v$. For such an $i$, let $F'_i$ be obtained from
  $F_i$ by replacing these two edges by $r_i$. It is a forest of
  $\widetilde{G_v^{ab}}$ by \cref{lem:reroute-cycle-lift}. For every
  other $i$, let $F'_i$ be $F_i$ without its edge incident with $v$, a
  forest of $\widetilde{G_v^{ab}}$ by \cref{lem:reroute-vfree-transport}.
  In both cases $|F'_i| = |F_i| - 1$.

  The edges of the $F'_i$ other than the $r_i$ lie in the disjoint sets
  $F_i$, and none of them is a copy of $ab$. So the $F'_i$ are pairwise
  disjoint, and their union $I'$ has
  $|I'| = \sum_i |F_i| - D = |I| - D$. Its copies of $ab$ are the $h'$
  copies $r_i$, so $|\widetilde{ab} \cap I'| = h' < D - 1$. By
  \cref{lem:forest-packing-decomp}, $I'$ is independent in
  $M(\widetilde{G_v^{ab}})$, as the union of $D$ forests.
\end{proof}

\begin{lemma}[Splitting off does not increase the deficiency; KT Lemma~4.1]
  \label{lem:forest-surgery-split}
  \lean{Graph.forest_surgery_split}
  \leanok
  \uses{def:graph-operations, def:matroid-MG}
  Let $D \ge 2$, and let $G$ be a multigraph with a vertex $v$ of degree
  two, whose edges join $v$ to distinct vertices $a, b \neq v$. Then
  \[
    \operatorname{def}(\widetilde{G_v^{ab}}) \;\le\;
    \operatorname{def}(\tilde G).
  \]
\end{lemma}
\begin{proof}
  \leanok
  \uses{thm:def-eq-corank, lem:balanced-forest-packing, lem:reroute-cycle-lift, lem:reroute-vfree-transport, lem:forest-surgery-count}
  Let $B$ be a base of $M(\tilde G)$. By
  \cref{lem:balanced-forest-packing}, $B$ is the union of $D$ pairwise
  disjoint forests of $\tilde G$, each of which has an edge incident with
  $v$. By \cref{lem:forest-surgery-count}, $M(\widetilde{G_v^{ab}})$ has
  an independent set $I'$ with $|I'| = |B| - D$. Hence
  \[
    \operatorname{rank} M(\widetilde{G_v^{ab}}) \;\ge\; |B| - D.
  \]

  The vertices of $G_v^{ab}$ are the $|V(G)| - 1$ vertices of $G$ other
  than $v$, among them $a$. So KT eq.~(2.4) (\cref{thm:def-eq-corank})
  applies to $G_v^{ab}$ and to $G$, and gives
  \[
    \operatorname{def}(\widetilde{G_v^{ab}}) \;=\;
    D\,(|V(G)| - 2) - \operatorname{rank} M(\widetilde{G_v^{ab}})
    \;\le\; D\,(|V(G)| - 1) - |B| \;=\; \operatorname{def}(\tilde G).
  \]
  Katoh and Tanigawa~\cite[p.~661]{katohTanigawa2011} derive their
  eq.~(4.1), $\operatorname{def}(\widetilde{G_v^{ab}}) \le k$ for a
  minimal $k$-dof-graph $G$, from their Lemma~4.1 in the same way.
\end{proof}

The bound of \cref{lem:forest-surgery-split} is the upper bound of KT
Lemma~4.3(i). We prove the same bound by comparing the values of
partitions (\cref{lem:splitoff-deficiency}), for every $D \ge 1$ and
without assuming $a \neq b$. Here we restate it as a consequence of the
forest surgery.

\begin{corollary}[The upper bound of KT Lemma~4.3(i) by forest surgery]
  \label{cor:forest-surgery-deficiency}
  \lean{Graph.splitOff_deficiency_le_of_forest_surgery}
  \leanok
  \uses{lem:forest-surgery-split, lem:splitoff-deficiency}
  Let $D \ge 2$, and let $G$ be a multigraph with a vertex $v$ of degree
  two, whose edges join $v$ to distinct vertices $a, b \neq v$. Then
  \[
    \operatorname{def}(\widetilde{G_v^{ab}}) \;\le\;
    \operatorname{def}(\tilde G).
  \]
\end{corollary}
\begin{proof}
  \leanok
  This is \cref{lem:forest-surgery-split}.
\end{proof}

We turn to KT Lemma~4.2. Let $H$ be a multigraph with an edge $ab$, and
let $H_{ab}^v$ be obtained from $H$ by edge-splitting along $ab$
(\cref{def:graph-operations}). Katoh and
Tanigawa~\cite[pp.~660--661]{katohTanigawa2011} partition an independent
set $I'$ of $M(\tilde H)$ into $D$ forests and change each forest at the
new vertex $v$. When no forest contains a copy of $ab$, each forest
gains one edge at $v$.

\begin{lemma}[Edge-splitting and forests avoiding $ab$; KT Lemma~4.2(i)]
  \label{lem:forest-surgery-unsplit}
  \uses{def:graph-operations, lem:forest-packing-decomp}
  Let $H$ be a multigraph with an edge $ab$, and let $H_{ab}^v$ be
  obtained from $H$ by edge-splitting along $ab$. Let
  $F'_1, \dots, F'_D$ be pairwise disjoint forests of $\tilde H$, none of
  which contains a copy of $ab$, and let $I'$ be their union. Then there
  are pairwise disjoint forests $F_1, \dots, F_D$ of
  $\widetilde{H_{ab}^v}$ with $F'_i \subseteq F_i$ and
  $|F_i| = |F'_i| + 1$ for each $i$. In particular, their union $I$ has
  $|I| = |I'| + D$.
\end{lemma}
\begin{proof}
  The forests $F'_i$ contain no copy of $ab$, so they are sets of edges of
  $\widetilde{H_{ab}^v}$ with the same ends as in $\tilde H$, and none of
  their edges is incident with $v$. Let $F_i = F'_i \cup \{(va, i)\}$
  for $i \le D - 1$, and $F_D = F'_D \cup \{(vb, 1)\}$, where $(va, i)$
  is the $i$th copy of the new edge $va$. Each added edge joins $v$ to a
  vertex of $H$ and is the only edge of $F_i$ at $v$, so it lies on no
  cycle of $F_i$. Hence $F_i$ is a forest. The added edges are distinct
  and are not edges of $\tilde H$, so the $F_i$ are pairwise disjoint.
  This is Katoh and Tanigawa's construction when no forest contains a
  copy of $ab$.
\end{proof}

Katoh and Tanigawa~\cite[p.~662]{katohTanigawa2011} use this case to
prove their Lemma~4.4, which we prove without forests
(\cref{lem:removal-deficiency}). In the form of independent sets, the
lemma above is the case $h' = 0$ of KT Lemma~4.2(i) below
(\cref{lem:edge-splitting}), with $G = H_{ab}^v$, for $D \ge 2$ and
$a \neq b$. So we do not formalize it separately.

We state KT Lemma~4.2 for a vertex $v$ of $G$ of degree two, whose edges
$e_a$ and $e_b$ join $v$ to distinct vertices $a, b \neq v$. Then $G$ is
obtained from $G_v^{ab}$ by edge-splitting along $ab$, so the lemma
applies to $H = G_v^{ab}$. In its proof, a forest of
$\widetilde{G_v^{ab}}$ that contains a copy of $ab$ loses it and gains a
copy of $e_a$ and a copy of $e_b$, and some of the other forests gain a
copy of $e_a$ or of $e_b$. The next two lemmas show that each change
gives a forest of $\tilde G$.

\begin{lemma}[Replacing a copy of $ab$ in a forest by the two edges at $v$]
  \label{lem:reverse-reroute-cycle-lift}
  \lean{Graph.isAcyclicSet_mulTilde_of_splitOff_reroute}
  \leanok
  \uses{def:graph-operations, def:matroid-MG, lem:reroute-cycle-lift}
  Let $G$ be a multigraph with a vertex $v$ of degree two, whose edges
  $e_a$ and $e_b$ join $v$ to distinct vertices $a, b \neq v$. Let $F'$
  be a forest of $\widetilde{G_v^{ab}}$ that contains a copy $r$ of $ab$,
  and let $p_a \in \tilde e_a$ and $p_b \in \tilde e_b$. Then
  $(F' \setminus \{r\}) \cup \{p_a, p_b\}$ is a forest of $\tilde G$.
\end{lemma}
\begin{proof}
  \leanok
  We reverse the argument of \cref{lem:reroute-cycle-lift}. Two copies
  of $ab$ form a cycle, so $r$ is the only copy of $ab$ in $F'$. Hence the
  edges of $F_0 = F' \setminus \{r\}$ are edges of $\widetilde{G_v}$.
  They have the same ends in $\tilde G$ as in $\widetilde{G_v^{ab}}$,
  and none of them is incident with $v$. Let $C$ be a cycle of $\tilde G$
  in $F_0 \cup \{p_a, p_b\}$. If $C$ does not pass through $v$, then $C$
  lies in $F_0$, and it is a cycle of $\widetilde{G_v^{ab}}$ in $F'$.
  Otherwise $C$ uses both $p_a$ and $p_b$, the only edges of the set at
  $v$. Then the rest of $C$ is a path from $b$ to $a$ in $F_0$, and with
  $r$ it forms a cycle of $\widetilde{G_v^{ab}}$ in $F'$. Both cases are
  impossible, as $F'$ is a forest.
\end{proof}

\begin{lemma}[Adding an edge at $v$ to a forest with no copy of $ab$]
  \label{lem:reverse-pendant-insert}
  \lean{Graph.isAcyclicSet_mulTilde_insert_vfiber_of_splitOff}
  \leanok
  \uses{def:graph-operations, def:matroid-MG, lem:reroute-vfree-transport, lem:acyclic-insert-vfiber}
  Let $G$ be a multigraph with a vertex $v$ of degree two, whose edges
  join $v$ to vertices $a, b \neq v$. Let $F$ be a forest of
  $\widetilde{G_v^{ab}}$ that contains no copy of $ab$, and let $x$ be an
  edge of $\tilde G$ that joins $v$ to a vertex $w \neq v$. Then
  $F \cup \{x\}$ is a forest of $\tilde G$.
\end{lemma}
\begin{proof}
  \leanok
  As $F$ contains no copy of $ab$, its edges are edges of
  $\widetilde{G_v}$. As in the proof of
  \cref{lem:reroute-vfree-transport}, they have the same ends in
  $\tilde G$ as in $\widetilde{G_v^{ab}}$. So $F$ is a forest of
  $\tilde G$, and none of its edges is incident with $v$. By
  \cref{lem:acyclic-insert-vfiber}, $F \cup \{x\}$ is a forest of
  $\tilde G$.
\end{proof}

We add to Katoh and Tanigawa's statement the equation
$I \setminus (\tilde e_a \cup \tilde e_b) = I' \setminus \widetilde{ab}$,
which their proof gives. Their proof of Lemma~4.3 relies on it: a
base of $M(\widetilde{G_v^{ab}})$ with no copy of an edge $e \neq ab$
extends to a base of $M(\tilde G)$ with no copy of
$e$~\cite[p.~661]{katohTanigawa2011}. As $|\widetilde{ab}| = D - 1$, the
two cases of the lemma cover every $I'$.

\begin{lemma}[Independent sets under edge-splitting; KT Lemma~4.2]
  \label{lem:edge-splitting}
  \lean{Graph.splitOff_indep_extend_of_fiber_subset,
    Graph.splitOff_indep_extend_of_fiber_lt}
  \leanok
  \uses{def:graph-operations, def:matroid-MG, lem:forest-packing-decomp,
    lem:reverse-reroute-cycle-lift, lem:reverse-pendant-insert}
  Let $D \ge 2$, and let $G$ be a multigraph with a vertex $v$ of degree
  two, whose edges $e_a$ and $e_b$ join $v$ to distinct vertices
  $a, b \neq v$. Let $I'$ be an independent set of
  $M(\widetilde{G_v^{ab}})$, and let $h' = |\widetilde{ab} \cap I'|$.
  Then $M(\tilde G)$ has an independent set $I$ with
  \[
    I \setminus (\tilde e_a \cup \tilde e_b) \;=\;
    I' \setminus \widetilde{ab}
  \]
  such that (i) if $h' < D - 1$, then $|I| = |I'| + D$ and
  $|\tilde e_b \cap I| = h' + 1$, and (ii) if $h' = D - 1$, then
  $|I| = |I'| + D - 1$.
\end{lemma}
\begin{proof}
  \leanok
  We follow Katoh and Tanigawa's proof. By
  \cref{lem:forest-packing-decomp}, $I'$ is the union of $D$ forests of
  $\widetilde{G_v^{ab}}$, which we make pairwise disjoint as after that
  lemma. Two copies of $ab$ form a cycle, so exactly $h'$ of the forests
  contain a copy of $ab$, one each. Renumbering the forests and the
  copies of $ab$, we call them $F'_1, \dots, F'_D$, where $F'_i$ contains
  $(ab, i)$ for $i \le h'$ and no copy of $ab$ for $i > h'$.

  For $i \le h'$, let $F_i$ be $F'_i$ with $(ab, i)$ replaced by
  $(e_a, i)$ and $(e_b, i)$, a forest of $\tilde G$ by
  \cref{lem:reverse-reroute-cycle-lift}. In case~(i), let
  $F_i = F'_i \cup \{(e_a, i)\}$ for $h' < i \le D - 1$, and
  $F_D = F'_D \cup \{(e_b, h' + 1)\}$. These are forests of $\tilde G$ by
  \cref{lem:reverse-pendant-insert}. In case~(ii), let $F_D = F'_D$. It
  contains no copy of $ab$, so it is a forest of $\tilde G$, as in the
  proof of \cref{lem:reverse-pendant-insert}.

  The added edges are different copies of $e_a$ and $e_b$, and they are
  not edges of $\widetilde{G_v^{ab}}$. So $F_1, \dots, F_D$ are pairwise
  disjoint forests of $\tilde G$, and their union $I$ is independent in
  $M(\tilde G)$ (\cref{lem:forest-packing-decomp}). Deleting the copies
  of $e_a$ and $e_b$ from $I$ leaves the union of the $F'_i$ without
  their copies of $ab$, which is $I' \setminus \widetilde{ab}$. In
  case~(i) every forest gains one edge, so $|I| = |I'| + D$, and the
  copies of $e_b$ in $I$ are $(e_b, 1), \dots, (e_b, h' + 1)$. In
  case~(ii) the forests $F_1, \dots, F_{D-1}$ gain one edge each and
  $F_D$ none, so $|I| = |I'| + D - 1$.
\end{proof}
\begin{fmlnote}
  The first pin is case~(ii), with $h' = D - 1$ stated as
  $\widetilde{ab} \subseteq I'$, and the second is case~(i). The formal
  proof does not renumber the copies of $ab$. A forest that contains the
  copy $(ab, j)$ receives $(e_a, j)$ and $(e_b, j)$, and the copies added
  to the other forests take the indices $j$ for which $(ab, j)$ is not in
  $I'$.
\end{fmlnote}

\subsection{Splitting off and removal: KT Lemmas~4.3--4.8}
\label{sec:molecular-induction-dof}

This subsection proves KT Lemmas~4.3 to~4.8~\cite[pp.~661--666]{katohTanigawa2011}.
Let $v$ be a vertex of $G$ of degree two, whose edges $e_a$ and $e_b$
join it to vertices $a, b \neq v$. KT Lemmas~4.3(i) and~4.4 compare the
deficiency of $G$ with the deficiencies of the splitting off $G_v^{ab}$
and of the removal $G_v$ (\cref{def:graph-operations}). Splitting off
does not raise the deficiency and lowers it by at most one. Removal does
not lower it when $D \ge 2$. Katoh and Tanigawa prove these bounds with
forests and KT eq.~(2.4). We compare the values of partitions of $V(G)$
instead (\cref{def:D-deficiency}), and we compare the two proofs of KT
Lemma~4.4 in \cref{rem:kt-lemma-44}. Two variants of the upper bound for
splitting off follow the first lemma; the pencil chapter uses them.

When $G$ is a $k$-dof-graph, KT Lemma~4.3(ii) tells whether $G_v^{ab}$
is a $k$-dof-graph or a $(k-1)$-dof-graph, in terms of the bases of
$M(\widetilde{G_v^{ab}})$. Its proof uses the forest surgery and
edge-splitting (\cref{sec:molecular-induction-forest}). KT Lemmas~4.5
to~4.7 concern a minimal $k$-dof-graph with no proper rigid subgraph.
They bound its number of edges, find a vertex of degree two in it when
it is $2$-edge-connected, and show that removing such a vertex raises
its deficiency. KT Lemma~4.8 ends the subsection: splitting off at such
a vertex gives a minimal $0$-dof-graph if $k = 0$, and a minimal
$(k-1)$-dof-graph if $k > 0$. This is the step of the induction in KT
Theorem~4.9, together with the contraction of a proper rigid subgraph
(\cref{lem:contraction-minimality}).

\begin{lemma}[Splitting off lowers the deficiency by zero or one; KT Lemma~4.3(i)]
  \label{lem:splitoff-deficiency}
  \lean{Graph.splitOff_deficiency_le, Graph.splitOff_deficiency_ge}
  \leanok
  \uses{def:graph-operations, def:matroid-MG, thm:def-eq-corank}
  Let $D \ge 1$, and let $G$ be a multigraph with a vertex $v$ of degree
  two, whose edges $e_a$ and $e_b$ join $v$ to vertices $a, b \neq v$.
  Then
  \[
    \operatorname{def}(\tilde G) - 1
      \;\le\; \operatorname{def}(\widetilde{G_v^{ab}})
      \;\le\; \operatorname{def}(\tilde G).
  \]
  In particular, if $G$ is a $k$-dof-graph, then $G_v^{ab}$ is a
  $k$-dof-graph or a $(k-1)$-dof-graph.
\end{lemma}
\begin{proof}
  \leanok
  We compare the values $\operatorname{def}_{\tilde G}(P)$ of partitions
  (\cref{def:D-deficiency}). The vertices of $G_v^{ab}$ are those of $G$
  other than $v$. As $D \ge 1$, the value
  $D\,(|P| - 1) - (D - 1)\,d_G(P)$ does not increase when $d_G(P)$
  increases.

  For the upper bound, let $P'$ be a partition of $V(G) - v$, and let
  $P$ be the partition of $V(G)$ obtained by adding $v$ to the part of
  $P'$ that contains $a$. Then $|P| = |P'|$. The edge $e_a$ joins two
  vertices of one part of $P$. The edge $e_b$ joins different parts of
  $P$ exactly when $a$ and $b$ lie in different parts of $P'$, that is,
  when the new edge $ab$ of $G_v^{ab}$ joins different parts of $P'$.
  Every other edge has the same ends in $G$ and in $G_v^{ab}$. Hence
  $d_G(P) = d_{G_v^{ab}}(P')$, and the value of $P'$ for
  $\widetilde{G_v^{ab}}$ is the value of $P$ for $\tilde G$, which is at
  most $\operatorname{def}(\tilde G)$.

  For the lower bound, let $P$ be a partition of $V(G)$ of value
  $\operatorname{def}(\tilde G)$, and let $P'$ be the partition of
  $V(G) - v$ into the nonempty sets $X - v$ with $X \in P$. The edges of
  $G_v^{ab}$ are the edges of $G$ other than $e_a$ and $e_b$, with the
  same ends, and the new edge $ab$.

  Suppose first that $v$ is not alone in its part of $P$. Then
  $|P'| = |P|$. If $ab$ joins different parts of $P'$, then $v$ lies in a
  different part from $a$ or from $b$, so $e_a$ or $e_b$ joins different
  parts of $P$. Hence $d_{G_v^{ab}}(P') \le d_G(P)$, and the value of
  $P'$ is at least $\operatorname{def}(\tilde G)$.

  Suppose now that $\{v\}$ is a part of $P$. Then $|P'| = |P| - 1$, and
  both $e_a$ and $e_b$ join different parts of $P$, so
  $d_{G_v^{ab}}(P') \le d_G(P) - 1$. The value of $P'$ is then at least
  \[
    D\,(|P| - 2) - (D - 1)\,(d_G(P) - 1) \;=\;
    \operatorname{def}(\tilde G) - 1.
  \]
  In both cases
  $\operatorname{def}(\widetilde{G_v^{ab}}) \ge \operatorname{def}(\tilde G) - 1$.
\end{proof}

KT Lemma~4.3 assumes that $G$ is a minimal $k$-dof-graph, and its
part~(i) adds that $G_v^{ab}$ is minimal when it is a
$(k-1)$-dof-graph. Katoh and Tanigawa use this clause only in the proof
of their Lemma~4.8(ii). We prove the minimality there
(\cref{lem:reduction-step-pos}).

The ear steps of the pencil chapter (\cref{sec:main-component-short})
suppress a vertex of degree two, and keep one of its two edges as the
new edge that joins its neighbours.

\begin{lemma}[Suppressing a vertex of degree two]
  \label{lem:splitoff-deficiency-reuse}
  \lean{Graph.splitOff_deficiency_le_of_eq_left}
  \leanok
  \uses{def:graph-operations, def:D-deficiency}
  Let $D \ge 1$, and let $G$ be a multigraph with a vertex $v$ of degree
  two, whose edges $e_a$ and $e_b$ join $v$ to vertices $a, b \neq v$.
  Let $G''$ be obtained from $G$ by replacing the end $v$ of $e_a$ by
  $b$, and deleting $v$ and $e_b$. Then
  \[
    \operatorname{def}(\tilde G'') \;\le\; \operatorname{def}(\tilde G).
  \]
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:splitoff-deficiency}
  The graph $G''$ is $G_v^{ab}$ with $e_a$ as its new edge $ab$, and we
  argue as for the upper bound of \cref{lem:splitoff-deficiency}. Let
  $P'$ be a partition of $V(G) - v$, and let $P$ be the partition of
  $V(G)$ obtained by adding $v$ to the part of $a$. Then $|P| = |P'|$. In
  $G$ the edge $e_a$ joins two vertices of one part of $P$, and $e_b$
  joins different parts of $P$ exactly when $e_a$ joins different parts
  of $P'$ in $G''$. Every other edge has the same ends in $G$ and in
  $G''$. So $P'$ has the same value for $\tilde G''$ as $P$ for
  $\tilde G$.
\end{proof}
\begin{fmlnote}
  In the formalization $G''$ is the splitting off at $v$ with $e_0 = e_a$
  as the label of the new edge (\cref{fmlnote:graph-operations}), while
  the upper bound of \cref{lem:splitoff-deficiency} assumes
  $e_0 \notin E(G)$. The two bounds have one proof, which uses only that
  $e_0$ does not join different parts of $P$ in $G$.
\end{fmlnote}

The ear steps of the pencil chapter whose ends are not adjacent
(\cref{sec:main-component-orbit}) compare a graph with the graph
obtained from it by adding an edge.

\begin{lemma}[Splitting off with a gap in the merged deficiency]
  \label{lem:splitoff-deficiency-merged}
  \lean{Graph.splitOff_deficiency_add_le_of_deficiencyMerged}
  \leanok
  \uses{def:graph-operations, def:D-deficiency, def:deficiency-merged}
  Let $D \ge 1$. Let $v$, $a$ and $b$ be vertices of a multigraph $G$
  with $a, b \neq v$. Let $G'$ be obtained from $G$ by deleting $v$, and
  let $H$ be obtained from $G'$ by adding a new edge $ab$. Suppose that
  \[
    \operatorname{def}(\tilde G'; a, b) \;\le\;
    \operatorname{def}(\tilde G') - (D - 1)
  \]
  (\cref{def:deficiency-merged}). Then
  $\operatorname{def}(\tilde H) \le \operatorname{def}(\tilde G') - (D - 1)$.
\end{lemma}
\begin{proof}
  \leanok
  The graphs $H$ and $G'$ have the same vertices, and $H$ has one more
  edge, $ab$. Let $P$ be a partition of $V(G')$. If $P$ separates $a$
  and $b$, then $d_H(P) = d_{G'}(P) + 1$. So the value of $P$ for
  $\tilde H$ is its value for $\tilde G'$ minus $D - 1$, which is at most
  $\operatorname{def}(\tilde G') - (D - 1)$. If $a$ and $b$ lie in one
  part of $P$, then $d_H(P) = d_{G'}(P)$. So the value of $P$ for
  $\tilde H$ is at most $\operatorname{def}(\tilde G'; a, b)$, which the
  hypothesis bounds.
\end{proof}
\begin{fmlnote}
  The pin builds $H$ with the formal splitting off at $v$
  (\cref{fmlnote:graph-operations}), which is defined at a vertex of any
  degree, and a label $e_0$ for the new edge. It asks only that every
  edge of $G$ with the label $e_0$ have $v$ as an end. So $e_0$ is not
  the label of an edge of $G'$, and it may be the label of an edge at
  $v$, as in \cref{lem:splitoff-deficiency-reuse}.
\end{fmlnote}

\begin{lemma}[Removal does not lower the deficiency; KT Lemma~4.4]
  \label{lem:removal-deficiency}
  \lean{Graph.removeVertex_deficiency_ge}
  \leanok
  \uses{def:graph-operations, lem:splitoff-deficiency}
  Let $D \ge 2$, and let $G$ be a multigraph with a vertex $v$ of degree
  two, whose edges $e_a$ and $e_b$ join $v$ to vertices $a, b \neq v$.
  Then
  \[
    \operatorname{def}(\tilde G) \;\le\;
    \operatorname{def}(\widetilde{G_v}).
  \]
  In particular, if $G$ is a $k$-dof-graph, then
  $\operatorname{def}(\widetilde{G_v}) \ge k$.
\end{lemma}
\begin{proof}
  \leanok
  We compare the values of partitions, as for the lower bound of
  \cref{lem:splitoff-deficiency}. Let $P$ be a partition of $V(G)$ of
  value $\operatorname{def}(\tilde G)$, and let $P'$ be the partition of
  $V(G) - v$ into the nonempty sets $X - v$ with $X \in P$. The edges of
  $G_v$ are the edges of $G$ other than $e_a$ and $e_b$, the only edges
  at $v$, with the same ends. So
  \[
    d_{G_v}(P') \;=\; d_G(P) - c,
  \]
  where $c \in \{0, 1, 2\}$ is the number of the edges $e_a$ and $e_b$
  that join different parts of $P$.

  Suppose first that $v$ is not alone in its part of $P$. Then
  $|P'| = |P|$, and the value of $P'$ for $\widetilde{G_v}$ is
  $\operatorname{def}(\tilde G) + (D - 1)\,c \ge \operatorname{def}(\tilde G)$.

  Suppose now that $\{v\}$ is a part of $P$. Then $|P'| = |P| - 1$. The
  neighbours $a$ and $b$ of $v$ lie in other parts, so both $e_a$ and
  $e_b$ join different parts of $P$, and $c = 2$. The value of $P'$ is
  \[
    D\,(|P| - 2) - (D - 1)\,(d_G(P) - 2) \;=\;
    \operatorname{def}(\tilde G) + D - 2,
  \]
  which is at least $\operatorname{def}(\tilde G)$ as $D \ge 2$. In both
  cases $\operatorname{def}(\widetilde{G_v}) \ge \operatorname{def}(\tilde G)$.
\end{proof}

\begin{remark}[On KT Lemma~4.4]
  \label{rem:kt-lemma-44}
  \uses{lem:splitoff-deficiency, lem:removal-deficiency}
  Katoh and Tanigawa~\cite[p.~662]{katohTanigawa2011} prove KT
  Lemma~4.4 with forests. Let $B'$ be a base of $M(\widetilde{G_v})$. As
  $G_v$ is a subgraph of $G_v^{ab}$, the set $B'$ is independent in
  $M(\widetilde{G_v^{ab}})$, and it contains no copy of $ab$. The graph
  $G$ is obtained from $G_v^{ab}$ by edge-splitting along $ab$, so KT
  Lemma~4.2(i) with $h' = 0$ gives an independent set $I$ of
  $M(\tilde G)$ with $|I| = |B'| + D$ and $|\tilde e_b \cap I| = 1$. In
  the form of forests, each of the $D$ forests whose union is $B'$ gains
  one edge at $v$ (\cref{lem:forest-surgery-unsplit}).

  By KT eq.~(2.4) (\cref{thm:def-eq-corank}),
  $|B'| = D\,(|V(G)| - 2) - \operatorname{def}(\widetilde{G_v})$ and
  $|I| \le D\,(|V(G)| - 1) - \operatorname{def}(\tilde G)$. Hence
  $\operatorname{def}(\tilde G) \le \operatorname{def}(\widetilde{G_v})$.
  If equality holds, then $I$ is a base of $M(\tilde G)$ with
  $|\tilde e_b \cap I| = 1$. This is the second assertion of KT
  Lemma~4.4, and we prove it in this way
  (\cref{lem:removal-deficiency-equality}).

  Katoh and Tanigawa's argument is correct. It also needs $D \ge 2$,
  since KT Lemma~4.2(i) at $h' = 0$ asks that $0 < D - 1$. Our proof of
  \cref{lem:removal-deficiency} compares the values of partitions
  instead, and needs neither forests nor KT eq.~(2.4). The comparison
  works because of the case where $\{v\}$ is a part of $P$. There $P'$
  has one part fewer than $P$, which lowers the value by $D$. But the
  neighbours $a$ and $b$ of $v$ lie in other parts, so both $e_a$ and
  $e_b$ join different parts of $P$. Hence
  $d_{G_v}(P') = d_G(P) - 2$, which raises the value by $2(D - 1)$. The
  value changes by $2(D - 1) - D = D - 2 \ge 0$. When $v$ is not alone in
  its part, the number of parts stays the same, and fewer edges joining
  different parts can only raise the value.

  Without the forced crossing of $e_a$ and $e_b$ the count fails. If we
  used only that the number of edges joining different parts does not
  increase, as in the other case, we would get only
  $\operatorname{def}(\widetilde{G_v}) \ge \operatorname{def}(\tilde G) - D$.
  Rank monotonicity gives the same bound. As $M(\widetilde{G_v})$ is a
  restriction of $M(\tilde G)$ (\cref{lem:matroid-restrict-subgraph}),
  its rank is at most that of $M(\tilde G)$, and KT eq.~(2.4) turns this
  into $\operatorname{def}(\widetilde{G_v}) \ge \operatorname{def}(\tilde G) - D$.
\end{remark}

The step of the induction uses the two bounds together
(\cref{lem:reduction-step}).

\begin{lemma}[The deficiency after splitting off and after removal; KT Lemmas~4.3(i) and~4.4]
  \label{lem:dof-tracking}
  \lean{Graph.dof_tracking}
  \leanok
  \uses{lem:splitoff-deficiency, lem:removal-deficiency, def:graph-operations}
  Let $D \ge 2$, and let $G$ be a $k$-dof-graph with a vertex $v$ of
  degree two, whose edges $e_a$ and $e_b$ join $v$ to vertices
  $a, b \neq v$. Then
  \[
    k - 1 \;\le\; \operatorname{def}(\widetilde{G_v^{ab}}) \;\le\; k
    \qquad\text{and}\qquad
    \operatorname{def}(\widetilde{G_v}) \;\ge\; k.
  \]
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:splitoff-deficiency, lem:removal-deficiency}
  Combine \cref{lem:splitoff-deficiency}, which holds as $D \ge 1$, with
  \cref{lem:removal-deficiency}.
\end{proof}

\begin{lemma}[When splitting off keeps the deficiency; KT Lemma~4.3(ii)]
  \label{lem:splitoff-kdof-criterion}
  \lean{Graph.splitOff_exists_base_inter_fiber_lt,
        Graph.splitOff_isKDof_of_exists_base_inter_fiber_lt}
  \leanok
  \uses{def:k-dof, def:graph-operations, def:matroid-MG, lem:edge-splitting,
        lem:splitoff-deficiency, thm:def-eq-corank}
  Let $D \ge 2$, and let $G$ be a $k$-dof-graph with a vertex $v$ of
  degree two, whose edges $e_a$ and $e_b$ join $v$ to distinct vertices
  $a, b \neq v$. Then $G_v^{ab}$ is a $k$-dof-graph if and only if
  $M(\widetilde{G_v^{ab}})$ has a base $B'$ with
  $|\widetilde{ab} \cap B'| < D - 1$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:forest-surgery-count, lem:balanced-forest-packing}
  The vertices of $G_v^{ab}$ are the $|V(G)| - 1$ vertices of $G$ other
  than $v$. So KT eq.~(2.4) (\cref{thm:def-eq-corank}) gives
  \[
    \operatorname{rank} M(\tilde G) \;=\; D\,(|V(G)| - 1) - k,
    \qquad
    \operatorname{rank} M(\widetilde{G_v^{ab}}) \;=\;
    D\,(|V(G)| - 2) - \operatorname{def}(\widetilde{G_v^{ab}}).
  \]

  Suppose that $G_v^{ab}$ is a $k$-dof-graph, and let $B$ be a base of
  $M(\tilde G)$. By \cref{lem:balanced-forest-packing}, $B$ is the union
  of $D$ pairwise disjoint forests of $\tilde G$, each of which has an
  edge incident with $v$. The surgery on these forests
  (\cref{lem:forest-surgery-count}) gives an independent set $I'$ of
  $M(\widetilde{G_v^{ab}})$ with $|I'| = |B| - D$ and
  $|\widetilde{ab} \cap I'| < D - 1$. Then
  $|I'| = D(|V(G)| - 2) - k$, which is the rank of
  $M(\widetilde{G_v^{ab}})$, so $I'$ is a base. This is Katoh and
  Tanigawa's argument~\cite[p.~661]{katohTanigawa2011}, with the balanced
  partition that they assume (\cref{rem:kt-lemma-41}).

  Conversely, let $B'$ be a base of $M(\widetilde{G_v^{ab}})$ with
  $|\widetilde{ab} \cap B'| < D - 1$. By case~(i) of KT Lemma~4.2
  (\cref{lem:edge-splitting}), $M(\tilde G)$ has an independent set $I$
  with $|I| = |B'| + D$. As $|I| \le \operatorname{rank} M(\tilde G)$, we
  get $|B'| \le D(|V(G)| - 2) - k$, that is,
  $\operatorname{def}(\widetilde{G_v^{ab}}) \ge k$. The upper bound of KT
  Lemma~4.3(i) (\cref{lem:splitoff-deficiency}) gives
  $\operatorname{def}(\widetilde{G_v^{ab}}) \le k$.
\end{proof}

The rest of the subsection follows Katoh and Tanigawa's
\S4.2~\cite[pp.~663--666]{katohTanigawa2011}, on minimal
$k$-dof-graphs with no proper rigid subgraph
(\cref{def:rigid-subgraph}). Katoh and Tanigawa's multigraphs have no
loops~\cite[\S2.5]{katohTanigawa2011}, and ours may have them. A
minimal $k$-dof-graph $G$ has none. Indeed, the copies of a loop $e$ of
$G$ are loops of $\tilde G$. A loop is a cycle, so it lies in no forest,
and hence in no independent set of $M(\tilde G)$
(\cref{lem:forest-packing-decomp}). So no base of $M(\tilde G)$ meets
$\tilde e$, against minimality (\cref{def:k-dof}). Hence every circuit
$X$ of $M(\tilde G)$ has $|V(X)| \ge 2$. If $G$ also has no proper rigid
subgraph, then $V(X) = V(G)$, as we noted after
\cref{lem:circuit-induces-rigid}.

\begin{lemma}[An edge bound with no proper rigid subgraph; KT Lemma~4.5(i)]
  \label{lem:no-rigid-edge-count}
  \lean{Graph.no_rigid_edge_count}
  \leanok
  \uses{lem:circuit-induces-rigid, def:k-dof, def:rigid-subgraph, def:matroid-MG}
  Let $D \ge 2$, and let $G$ be a minimal $k$-dof-graph with at least one
  vertex and no proper rigid subgraph. Then
  \[
    (D-1)\,|E(G)| \;<\; D\,(|V(G)| - 1) - k + (D-1).
  \]
\end{lemma}
\begin{proof}
  \leanok
  \uses{thm:def-eq-corank}
  We follow Katoh and Tanigawa's proof~\cite[p.~663]{katohTanigawa2011},
  which they give for $k = 0$. If $G$ has no edge, then the partition of
  $V(G)$ into single vertices has the largest value, so
  $k = D(|V(G)| - 1)$, and the bound reads $0 < D - 1$. Otherwise, fix an
  edge $e$ of $G$. Let $h^*$ be the least value of $|\tilde e \cap B|$
  over the bases $B$ of $M(\tilde G)$, and let $B^*$ be a base with
  $|\tilde e \cap B^*| = h^*$. As $G$ is minimal, $h^* \ge 1$. By KT
  eq.~(2.4) (\cref{thm:def-eq-corank}), every base of $M(\tilde G)$ has
  $D(|V(G)| - 1) - k$ elements.

  We show that every edge of $\tilde G$ outside $\tilde e$ lies in $B^*$
  (KT eq.~(4.3)). Suppose that an edge $f \notin \tilde e$ of $\tilde G$
  is not in $B^*$, and let $X$ be the fundamental circuit of $f$ with
  respect to $B^*$, the only circuit contained in $B^* + f$. Then
  $V(X) = V(G)$, as $G$ has no loops and no proper rigid subgraph.
  Suppose first that $X$ does not meet $\tilde e$, and let $x \in X$. By
  \cref{lem:circuit-induces-rigid}, $X - x$ is an independent set with
  $D(|V(G)| - 1)$ elements. It lies in a base, which has
  $D(|V(G)| - 1) - k$ elements, so $k \le 0$. As $V(G)$ is not empty,
  $k \ge 0$. So $k = 0$, and $X - x$ is a base of $M(\tilde G)$ that does
  not meet $\tilde e$, against the minimality of $G$. Hence $X$ contains
  a copy $e_j \in \tilde e$. Then $B^* + f - e_j$ is a base of
  $M(\tilde G)$ that meets $\tilde e$ in $h^* - 1$ elements, against the
  choice of $B^*$.

  So the edges of $\tilde G$ outside $B^*$ are the $D - 1 - h^*$ copies
  of $e$ outside $B^*$. As $\tilde G$ has $(D-1)\,|E(G)|$ edges,
  \[
    (D-1)\,|E(G)| \;=\; |B^*| + (D - 1 - h^*) \;\le\;
    D\,(|V(G)| - 1) - k + (D - 2).
  \]
\end{proof}

\begin{lemma}[The edges of $\tilde G$ are independent when $k > 0$; KT Lemma~4.5(ii)]
  \label{lem:edge-set-indep-pos}
  \lean{Graph.indep_edgeSet_mulTilde_of_noRigid_of_pos,
        Graph.isBase_eq_edgeSet_mulTilde_of_noRigid_of_pos}
  \leanok
  \uses{lem:circuit-induces-rigid, def:k-dof, def:rigid-subgraph, def:matroid-MG,
        thm:def-eq-corank}
  Let $D \ge 2$, and let $G$ be a minimal $k$-dof-graph with $k > 0$ and
  no proper rigid subgraph. Then $E(\tilde G)$ is independent in
  $M(\tilde G)$. In particular, $E(\tilde G)$ is the only base of
  $M(\tilde G)$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:circuit-induces-rigid, def:k-dof, def:rigid-subgraph, def:matroid-MG,
        thm:def-eq-corank}
  Suppose that $E(\tilde G)$ contains a circuit $X$ of $M(\tilde G)$. As
  $G$ has no loops and no proper rigid subgraph, $V(X) = V(G)$. So
  $H = G[V(X)]$ is a rigid subgraph of $G$ with the vertices of $G$
  (\cref{lem:circuit-induces-rigid}). By \cref{lem:rigid-full-rank},
  $\operatorname{rank} M(\tilde H) = D(|V(G)| - 1)$, and $M(\tilde H)$ is
  a restriction of $M(\tilde G)$ (\cref{lem:matroid-restrict-subgraph}).
  So $\operatorname{rank} M(\tilde G) \ge D(|V(G)| - 1)$, and KT
  eq.~(2.4) (\cref{thm:def-eq-corank}) gives
  $k = \operatorname{def}(\tilde G) \le 0$, a contradiction. Hence
  $E(\tilde G)$ is independent. A base of $M(\tilde G)$ is a maximal
  independent subset of $E(\tilde G)$, so it is $E(\tilde G)$.
\end{proof}

Katoh and Tanigawa state part~(ii) of their Lemma~4.5 for every
$k$-dof-graph with $k > 0$ and no proper rigid subgraph. They use that
a circuit $X$ has $|V(X)| \ge 2$, which holds as their multigraphs have
no loops. We assume that $G$ is minimal, which excludes loops. They add that $(D-1)\,|E(G)| = D(|V(G)| - 1) - k$,
which follows by KT eq.~(2.4).

For $k > 0$, Katoh and Tanigawa derive KT Lemma~4.7 from KT
Lemma~4.5(ii) and the second assertion of KT Lemma~4.4, which they prove
with KT Lemma~4.2 (\cref{rem:kt-lemma-44}). The next two lemmas prove
that assertion and KT Lemma~4.7.

\begin{lemma}[When removal keeps the deficiency; the second assertion of KT Lemma~4.4]
  \label{lem:removal-deficiency-equality}
  \lean{Graph.exists_isBase_vb_fiber_eq_one_of_removeVertex_isKDof}
  \leanok
  \uses{def:k-dof, def:graph-operations, def:matroid-MG}
  Let $D \ge 2$, and let $G$ be a $k$-dof-graph with a vertex $v$ of
  degree two, whose edges $e_a$ and $e_b$ join $v$ to distinct vertices
  $a, b \neq v$. If $\operatorname{def}(\widetilde{G_v}) = k$, then
  $M(\tilde G)$ has a base $B$ with $|\tilde e_b \cap B| = 1$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:edge-splitting, thm:def-eq-corank, lem:matroid-restrict-subgraph,
        lem:splitoff-edge-substitution, def:edge-multiply}
  We argue as Katoh and Tanigawa do (\cref{rem:kt-lemma-44}). Let $B'$ be
  a base of $M(\widetilde{G_v})$. As $G_v$ is a subgraph of $G_v^{ab}$,
  the set $B'$ is independent in $M(\widetilde{G_v^{ab}})$
  (\cref{lem:matroid-restrict-subgraph}), and it contains no copy of
  $ab$. By case~(i) of KT Lemma~4.2 with $h' = 0$
  (\cref{lem:edge-splitting}), $M(\tilde G)$ has an independent set $I$
  with $|I| = |B'| + D$ and $|\tilde e_b \cap I| = 1$. The vertices of
  $G_v$ are the $|V(G)| - 1$ vertices of $G$ other than $v$. So KT
  eq.~(2.4) (\cref{thm:def-eq-corank}), for $G_v$ and for $G$, gives
  \[
    |I| \;=\; D\,(|V(G)| - 2) - k + D \;=\; D\,(|V(G)| - 1) - k
    \;=\; \operatorname{rank} M(\tilde G),
  \]
  and $I$ is a base.
\end{proof}
\begin{fmlnote}
  The pin also takes a label $e_0 \notin E(G)$ for the new edge of
  $G_v^{ab}$, which the proof uses (\cref{fmlnote:graph-operations}).
\end{fmlnote}

\begin{lemma}[Removal raises the deficiency; KT Lemma~4.7]
  \label{lem:removal-deficiency-strict}
  \lean{Graph.removeVertex_deficiency_gt_of_noRigid}
  \leanok
  \uses{lem:removal-deficiency, lem:edge-set-indep-pos, def:k-dof,
        def:rigid-subgraph}
  Let $D \ge 3$, and let $G$ be a minimal $k$-dof-graph with at least
  three vertices and no proper rigid subgraph. Let $v$ be a vertex of $G$
  of degree two, whose edges $e_a$ and $e_b$ join $v$ to distinct
  vertices $a, b \neq v$. Then $\operatorname{def}(\widetilde{G_v}) > k$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:removal-deficiency-equality}
  KT Lemma~4.4 (\cref{lem:removal-deficiency}) gives
  $\operatorname{def}(\widetilde{G_v}) \ge k$, and we show that
  equality fails. Suppose that $\operatorname{def}(\widetilde{G_v}) = k$.
  If $k = 0$, then $G_v$ is a rigid subgraph of $G$ with
  $|V(G)| - 1 \ge 2$ vertices, not all of those of $G$, which is
  impossible. If $k > 0$, then by \cref{lem:removal-deficiency-equality}
  some base $B$ of $M(\tilde G)$ has $|\tilde e_b \cap B| = 1$. But
  $B = E(\tilde G)$ (\cref{lem:edge-set-indep-pos}), which contains all
  $D - 1 \ge 2$ copies of $e_b$. This is Katoh and Tanigawa's
  proof~\cite[p.~665]{katohTanigawa2011}.
\end{proof}
\begin{fmlnote}
  The pin also takes a label $e_0 \notin E(G)$ for the new edge of
  $G_v^{ab}$, which the case $k > 0$ of the proof uses
  (\cref{fmlnote:graph-operations}).
\end{fmlnote}

KT Lemma~4.6 shows that a $2$-edge-connected minimal $k$-dof-graph with
no proper rigid subgraph is a cycle on at most $d$ vertices, or contains
a chain $v_0 v_1 \dots v_d$ of length $d$ whose interior vertices have
degree two. We prove this for $k = 0$ (\cref{lem:chain-cycle-dichotomy}).
Katoh and Tanigawa's proof begins with a vertex of degree two, which the
next two lemmas give.

\begin{lemma}[A vertex of degree at most two; from the proof of KT Lemma~4.6]
  \label{lem:low-degree-vertex}
  \lean{Graph.exists_degree_le_two}
  \leanok
  \uses{lem:no-rigid-edge-count, def:k-dof}
  Let $D \ge 3$, and let $G$ be a minimal $k$-dof-graph with at least one
  vertex and no proper rigid subgraph. Then $G$ has a vertex of degree at
  most two.
\end{lemma}
\begin{proof}
  \leanok
  As $V(G)$ is not empty, $k \ge 0$, so \cref{lem:no-rigid-edge-count}
  gives
  \[
    (D-1)\,|E(G)| \;<\; D\,(|V(G)| - 1) + D - 1 \;<\; D\,|V(G)|.
  \]
  The degrees of the vertices of $G$ sum to $2|E(G)|$, so their average
  is less than $2D/(D-1) = 2 + 2/(D-1)$, which is at most $3$ as
  $D \ge 3$. Hence some vertex has degree at most two. This is KT
  eq.~(4.5)~\cite[p.~664]{katohTanigawa2011}.
\end{proof}

\begin{lemma}[A vertex of degree two; from the proof of KT Lemma~4.6]
  \label{lem:reducible-vertex}
  \lean{Graph.exists_degree_eq_two}
  \leanok
  \uses{lem:low-degree-vertex, lem:two-edge-conn, def:k-dof}
  Let $D \ge 3$, and let $G$ be a $2$-edge-connected minimal
  $k$-dof-graph with at least two vertices and no proper rigid subgraph.
  Then $G$ has a vertex of degree two.
\end{lemma}
\begin{proof}
  \leanok
  By \cref{lem:low-degree-vertex}, $G$ has a vertex of degree at most
  two. As $G$ is $2$-edge-connected and has at least two vertices, every
  vertex of $G$ has degree at least two
  (\cref{lem:two-edge-connected-degree}).
\end{proof}

A $0$-dof-graph is $2$-edge-connected (KT Lemma~3.1,
\cref{lem:two-edge-conn}). So for $k = 0$ the hypothesis of
$2$-edge-connectivity holds automatically.

Katoh and Tanigawa prove their Theorem~4.9 by induction on the number of
vertices~\cite[pp.~666--667]{katohTanigawa2011}.

\begin{lemma}[The reductions lower the number of vertices; from the proof of KT Theorem~4.9]
  \label{lem:reduction-measure}
  \lean{Graph.splitOff_vertexSet_ncard_lt, Graph.rigidContract_vertexSet_ncard,
        Graph.rigidContract_vertexSet_ncard_lt}
  \leanok
  \uses{def:graph-operations, def:rigid-contraction}
  Let $G$ be a multigraph. If $v$ is a vertex of $G$ of degree two, whose
  edges join $v$ to vertices $a$ and $b$, then
  $|V(G_v^{ab})| < |V(G)|$. If $H$ is a subgraph of $G$, and $G/E(H)$ is
  the contraction of $H$ to one of its vertices, then
  $|V(G/E(H))| = |V(G)| - |V(H)| + 1$, which is less than $|V(G)|$ if
  $H$ has at least two vertices.
\end{lemma}
\begin{proof}
  \leanok
  The vertices of $G_v^{ab}$ are those of $G$ other than $v$
  (\cref{def:graph-operations}). The vertices of $G/E(H)$ are the chosen
  vertex of $H$ and the vertices of $G$ not in $H$
  (\cref{def:rigid-contraction}), which gives the count.
\end{proof}

\begin{lemma}[Splitting off and contraction keep a minimal $0$-dof-graph minimal; KT Lemmas~4.8(i) and~3.5]
  \label{lem:reduction-step}
  \lean{Graph.splitOff_isMinimalKDof, Graph.contraction_isMinimalKDof}
  \leanok
  \uses{lem:reducible-vertex, lem:contraction-minimality, lem:dof-tracking, lem:reduction-measure, lem:circuit-induces-rigid}
  Let $D \ge 2$, and let $G$ be a minimal $0$-dof-graph.
  \begin{enumerate}
    \item Suppose that $G$ has at least three vertices and no proper
      rigid subgraph, and let $v$ be a vertex of $G$ of degree two, whose
      edges join $v$ to vertices $a, b \neq v$. Then $G_v^{ab}$ is a
      minimal $0$-dof-graph.
    \item Let $H$ be a proper rigid subgraph of $G$. Then
      \[
        \operatorname{rank}\bigl(M(\tilde G)/E(\tilde H)\bigr) \;=\;
        D\,(|V(G)| - |V(H)|),
      \]
      and every base of $M(\tilde G)/E(\tilde H)$ meets $\tilde e$ for
      every edge $e$ of $G$ that is not an edge of $H$.
  \end{enumerate}
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:circuit-induces-rigid}
  Part~(ii) is the case $k = 0$ of \cref{lem:contraction-minimality}.

  For (i) we follow Katoh and Tanigawa's
  proof~\cite[pp.~665--666]{katohTanigawa2011} up to its last step. Let
  $G' = G_v^{ab}$, and let $G_v = G - v$. Both have the $|V(G)| - 1$
  vertices of $G$ other than $v$, and the edges of $\widetilde{G'}$ are
  those of $\widetilde{G_v}$ and the $D - 1$ copies of $ab$. By
  \cref{lem:dof-tracking},
  $\operatorname{def}(\widetilde{G'}) \le \operatorname{def}(\tilde G) = 0$,
  and $\operatorname{def}(\widetilde{G'}) \ge 0$ as $G'$ has a vertex. So
  $G'$ is a $0$-dof-graph. The subgraph $G_v$ of $G$ has
  $|V(G)| - 1 \ge 2$ vertices, but not all of those of $G$. As $G$ has
  no proper rigid subgraph, $G_v$ is not rigid, and so
  $\operatorname{def}(\widetilde{G_v}) > 0$. This is the case $k = 0$ of
  KT Lemma~4.7.

  Every circuit $X$ of $M(\widetilde{G'})$ meets $\widetilde{ab}$ (KT
  eq.~(4.10)). Otherwise $X$ is a set of edges of $\widetilde{G_v}$. As
  $G_v$ is a subgraph of both $G'$ and $G$, $X$ is then a circuit of
  $M(\widetilde{G_v})$ and of $M(\tilde G)$
  (\cref{lem:matroid-restrict-subgraph}). As $G$ has no loops,
  $|V(X)| \ge 2$, and no edge of $X$ is incident with $v$. So $G[V(X)]$
  is a proper rigid subgraph of $G$ (\cref{lem:circuit-induces-rigid}),
  which is impossible. Hence $E(\widetilde{G_v})$ contains no circuit of
  $M(\widetilde{G'})$, and it is independent in $M(\widetilde{G'})$ and
  in its restriction $M(\widetilde{G_v})$, whose ground set it is. Thus
  $E(\widetilde{G_v})$ is the only base of $M(\widetilde{G_v})$, and KT
  eq.~(2.4) (\cref{thm:def-eq-corank}) gives
  \[
    |E(\widetilde{G_v})| \;=\; D\,(|V(G)| - 2) -
    \operatorname{def}(\widetilde{G_v}) \;<\; D\,(|V(G)| - 2).
  \]

  Now let $B'$ be a base of $M(\widetilde{G'})$, and let $e$ be an edge
  of $G'$. By KT eq.~(2.4), $|B'| = D(|V(G)| - 2)$. Suppose that $B'$
  does not meet $\tilde e$. If $e$ is the new edge $ab$, then $B'$ is a
  set of edges of $\widetilde{G_v}$. Otherwise $e$ is an edge of $G_v$,
  and $B'$ consists of at most $D - 1$ copies of $ab$ and of edges of
  $\widetilde{G_v}$ outside $\tilde e$, of which there are
  $|E(\widetilde{G_v})| - (D - 1)$. In both cases
  $|B'| \le |E(\widetilde{G_v})| < D(|V(G)| - 2)$, a contradiction. So
  every base of $M(\widetilde{G'})$ meets $\tilde e$ for every edge $e$
  of $G'$, and $G'$ is minimal.

  Katoh and Tanigawa end differently. They start from a base of
  $M(\widetilde{G'})$ that misses $\tilde e$ for an edge $e \neq ab$, and
  use KT eq.~(4.10) to exchange its copies of $ab$, one at a time, for
  copies of $e$. This gives a base of $M(\widetilde{G'})$ with no copy of
  $ab$, which is a base of $M(\widetilde{G_v})$ with $D(|V(G)| - 2)$
  elements, against KT Lemma~4.7. We replace this exchange by the count
  above.
\end{proof}

In part~(i) of \cref{lem:reduction-step}, the bound $|V(G)| \ge 3$ is
needed. The two-vertex double edge is a minimal $0$-dof-graph with no
proper rigid subgraph, and both of its vertices have degree two.
Splitting off either of them leaves one vertex with a loop. This graph
is a $0$-dof-graph, as it has one vertex, but it is not minimal, as the
only base of its matroid is empty.

By \cref{lem:reduction-measure}, the graphs $G_v^{ab}$ and $G/E(H)$ of
the lemma have fewer vertices than $G$. For $D \ge 2$, its part~(ii)
gives Katoh and Tanigawa's form of their Lemma~3.5, as we explained
after \cref{lem:contraction-minimality}: the graph $G/E(H)$ is again a
minimal $0$-dof-graph (\cref{lem:rigidContract-isMinimalKDof}).

\begin{lemma}[Splitting off when $k > 0$; KT Lemma~4.8(ii)]
  \label{lem:reduction-step-pos}
  \lean{Graph.splitOff_isMinimalKDof_of_pos}
  \leanok
  \uses{lem:dof-tracking, lem:reduction-measure, lem:splitoff-kdof-criterion, lem:edge-splitting, lem:edge-set-indep-pos, lem:removal-deficiency-strict, lem:circuit-induces-rigid}
  Let $D \ge 3$, and let $G$ be a minimal $k$-dof-graph with $k > 0$, at
  least three vertices and no proper rigid subgraph. Let $v$ be a vertex
  of $G$ of degree two, whose edges $e_a$ and $e_b$ join $v$ to distinct
  vertices $a, b \neq v$. Then $G_v^{ab}$ is a minimal
  $(k-1)$-dof-graph.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:dof-tracking, lem:splitoff-kdof-criterion, lem:edge-splitting, lem:edge-set-indep-pos, lem:removal-deficiency-strict, lem:circuit-induces-rigid}
  Let $G' = G_v^{ab}$. Its vertices are the $|V(G)| - 1$ vertices of $G$
  other than $v$, and by \cref{lem:dof-tracking},
  $\operatorname{def}(\widetilde{G'})$ is $k - 1$ or $k$.

  We show, as Katoh and Tanigawa do~\cite[p.~666]{katohTanigawa2011},
  that $\operatorname{def}(\widetilde{G'}) \neq k$. Suppose that
  $\operatorname{def}(\widetilde{G'}) = k$. By KT Lemma~4.3(ii)
  (\cref{lem:splitoff-kdof-criterion}), $M(\widetilde{G'})$ has a base
  $B'$ that misses a copy $p$ of $ab$. Let $X$ be the fundamental circuit
  of $p$ with respect to $B'$, and let $H = G'[V(X)]$, a rigid subgraph
  of $G'$ (\cref{lem:circuit-induces-rigid}). If $V(H) = V(G')$, then
  $\operatorname{rank} M(\widetilde{G'}) \ge \operatorname{rank}
  M(\tilde H) = D(|V(G')| - 1)$ (\cref{lem:rigid-full-rank}), and KT
  eq.~(2.4) gives $\operatorname{def}(\widetilde{G'}) \le 0 < k$. So
  $V(H)$ is a proper subset of $V(G')$. It contains the ends $a$ and $b$
  of $p$.

  Let $K = G[V(H) \cup \{v\}]$. Then $v$ has degree two in $K$, and
  $K_v^{ab} = H$. The set $I = X - p$ is independent in $M(\tilde H)$,
  with $|I| = D(|V(H)| - 1)$ (\cref{lem:circuit-induces-rigid}). As
  $I \subseteq B'$ and $p \notin B'$, also
  $|\widetilde{ab} \cap I| < D - 1$. By case~(i) of KT Lemma~4.2
  (\cref{lem:edge-splitting}) for $K$, $M(\tilde K)$ has an independent
  set with $|I| + D = D(|V(K)| - 1)$ elements. So
  $\operatorname{def}(\tilde K) = 0$ by KT eq.~(2.4), and $K$ is a rigid
  subgraph of $G$. Its vertex set $V(H) \cup \{v\}$ has at least two
  elements, and it is a proper subset of $V(G)$. This contradicts the
  hypothesis, and so $\operatorname{def}(\widetilde{G'}) = k - 1$.

  It remains to show that $G'$ is minimal: every base $B'$ of
  $M(\widetilde{G'})$ meets $\tilde e$ for every edge $e$ of $G'$. This
  is the minimality clause of KT Lemma~4.3(i). By KT eq.~(2.4),
  $|B'| = D(|V(G)| - 2) - (k - 1)$. Suppose that $B'$ misses $\tilde e$.
  If $e = ab$, then $B'$ is an independent set of $M(\widetilde{G_v})$,
  where $G_v = G - v$. So KT eq.~(2.4) gives
  $\operatorname{def}(\widetilde{G_v}) \le k - 1$, against KT Lemma~4.7
  (\cref{lem:removal-deficiency-strict}). If $e \neq ab$, then $e$ is an
  edge of $G$ other than $e_a$ and $e_b$, and we follow Katoh and
  Tanigawa's proof~\cite[p.~661]{katohTanigawa2011}. As $G'$ is not a
  $k$-dof-graph, $B'$ contains all $D - 1$ copies of $ab$
  (\cref{lem:splitoff-kdof-criterion}). By case~(ii) of KT Lemma~4.2
  (\cref{lem:edge-splitting}), $M(\tilde G)$ has an independent set $J$
  with $|J| = |B'| + D - 1 = D(|V(G)| - 1) - k$ and
  $J \setminus (\tilde e_a \cup \tilde e_b) = B' \setminus \widetilde{ab}$.
  By KT eq.~(2.4), $J$ is a base of $M(\tilde G)$, and it does not meet
  $\tilde e$, against the minimality of $G$.
\end{proof}

\subsection{KT Theorem~4.9 and the induction for KT Theorem~5.5}
\label{sec:molecular-induction-thm49}

Katoh and Tanigawa combine their Lemmas~3.5, 4.6 and~4.8 into their
Theorem~4.9: a minimal $0$-dof-graph with at least two vertices reduces
to the two-vertex double edge by splitting off at vertices of degree two
and contracting proper rigid
subgraphs~\cite[pp.~666--667]{katohTanigawa2011}. We state it with an
induction principle on the number of vertices, from which it follows
(\cref{thm:minimal-kdof-reduction}). Katoh and Tanigawa prove their
Theorem~5.5 by a second induction on the number of vertices, over
minimal $k$-dof-graphs of every deficiency $k$, with four
cases~\cite[p.~671]{katohTanigawa2011}. This induction is
\cref{thm:minimal-kdof-reduction-all-k}. In its first case the graph is
not $2$-edge-connected. Katoh and Tanigawa realize such a graph in their
Lemma~6.1, which ends the subsection in two forms.

\begin{theorem}[Reduction of minimal $0$-dof-graphs; KT Theorem~4.9]
  \label{thm:minimal-kdof-reduction}
  \lean{Graph.minimal_kdof_reduction}
  \leanok
  \uses{lem:reduction-step, lem:reducible-vertex, lem:contraction-minimality, def:graph-operations, lem:reduction-measure}
  Let $D \ge 3$.

  \textup{(i)} Let $G$ be a minimal $0$-dof-graph with at least two
  vertices. Then there are minimal $0$-dof-graphs
  $G = G_1, G_2, \dots, G_m$ such that $G_m$ has two vertices and two
  edges, each joining them, and for $1 \le i < m$, $G_{i+1}$ is obtained
  from $G_i$ by splitting off at a vertex of degree two or by contracting
  a proper rigid subgraph.

  \textup{(ii)} Let $P$ be a property of multigraphs such that for every
  minimal $0$-dof-graph $G$: (a) $P(G)$ holds if $G$ has two vertices;
  (b) if $G$ has no proper rigid subgraph, and $v$ is a vertex of $G$ of
  degree two whose edges join it to vertices $a, b \neq v$, then
  $P(G_v^{ab})$ implies $P(G)$; (c) if $G$ has at least three vertices
  and a proper rigid subgraph, and $P$ holds for every minimal
  $0$-dof-graph with at least two and fewer than $|V(G)|$ vertices, then
  $P(G)$ holds. Then $P$ holds for every minimal $0$-dof-graph with at
  least two vertices.
\end{theorem}
\begin{proof}
  \leanok
  We prove (ii) by induction on $|V(G)|$. If $G$ has two vertices, then
  $P(G)$ holds by (a). Let $G$ have at least three vertices, and suppose
  that $P$ holds for every minimal $0$-dof-graph with at least two and
  fewer than $|V(G)|$ vertices. If $G$ has a proper rigid subgraph, then
  $P(G)$ holds by (c).

  Otherwise, $G$ is $2$-edge-connected, as it is a $0$-dof-graph
  (\cref{lem:two-edge-connected-degree}). So $G$ has a vertex $v$ of
  degree two (\cref{lem:reducible-vertex}). As we showed before
  \cref{lem:no-rigid-edge-count}, $G$ has no loops. So the two edges at
  $v$ join it to vertices $a, b \neq v$. By part~(i) of
  \cref{lem:reduction-step}, $G_v^{ab}$ is a minimal $0$-dof-graph. Its
  vertices are the $|V(G)| - 1 \ge 2$ vertices of $G$ other than $v$
  (\cref{lem:reduction-measure}). So $P(G_v^{ab})$ holds by the induction
  hypothesis, and $P(G)$ holds by (b).

  Part~(i) is Katoh and Tanigawa's statement. As in their proof, it
  follows from (ii), with $P(G)$ the existence of such a sequence for
  $G$. A minimal $0$-dof-graph with two vertices has two edges, each
  joining them (\cref{lem:two-vertex-trichotomy}), which gives (a). A
  sequence for $G_v^{ab}$ extends to one for $G$, which gives (b).

  For (c), let $H$ be a proper rigid subgraph of $G$, and let $G/E(H)$ be its
  contraction to a vertex of $H$. It has $|V(G)| - |V(H)| + 1$ vertices,
  at least two and fewer than $|V(G)|$ (\cref{lem:reduction-measure}).
  By KT Lemma~3.5, $G/E(H)$ is a minimal $0$-dof-graph. We prove this
  lemma for the matroid $M(\tilde G)/E(\tilde H)$
  (\cref{lem:contraction-minimality}), and for the graph $G/E(H)$
  (\cref{lem:rigidContract-isMinimalKDof}). So $G/E(H)$ has a sequence,
  which extends to one for $G$.
\end{proof}
\begin{fmlnote}
  The formalization proves part~(ii), and no declaration states
  part~(i). The declaration also assumes that every minimal
  $0$-dof-graph has an edge label outside its edge set, and in (b) it
  forms $G_v^{ab}$ with such a label for the new edge
  (\cref{fmlnote:graph-operations}).
\end{fmlnote}

In~(c), $P$ is assumed for every smaller graph, and not only for
$G/E(H)$. In Katoh and Tanigawa's proof of their Theorem~5.5, the case of
a proper rigid subgraph $H$ needs it for both $H$ and
$G/E(H)$~\cite[p.~673]{katohTanigawa2011}. That proof is the following
induction over all deficiencies. In each case it assumes the induction
hypothesis KT eq.~(6.1), which we state for a property $P$.

\begin{theorem}[Induction over all deficiencies; KT p.~671]
  \label{thm:minimal-kdof-reduction-all-k}
  \lean{Graph.minimal_kdof_reduction_all_k}
  \leanok
  \uses{def:k-dof, def:cut-edges-2ec, def:rigid-subgraph}
  Let $P$ be a property of multigraphs. Suppose that $P(G)$ holds for
  every minimal $k$-dof-graph $G$ with one or two vertices, for every
  $k$. Suppose also that $P(G)$ holds for every minimal $k$-dof-graph
  $G$ with at least three vertices, in each of the following cases,
  provided that $P$ holds for every minimal $k'$-dof-graph with at least
  one and fewer than $|V(G)|$ vertices, for every $k'$: \textup{(i)} $G$
  is not $2$-edge-connected; \textup{(ii)} $G$ has a proper rigid
  subgraph; \textup{(iii)} $G$ is $2$-edge-connected, has no proper rigid
  subgraph, and $k > 0$; \textup{(iv)} $G$ is $2$-edge-connected, has no
  proper rigid subgraph, and $k = 0$. Then $P$ holds for every minimal
  $k$-dof-graph with at least one vertex, for every $k$.
\end{theorem}
\begin{proof}
  \leanok
  We argue by induction on $|V(G)|$, for all $k$ at once. Let $G$ be a
  minimal $k$-dof-graph with at least one vertex, and suppose that $P$
  holds for every minimal $k'$-dof-graph with at least one and fewer than
  $|V(G)|$ vertices. If $G$ has one or two vertices, then $P(G)$ holds by
  hypothesis. Otherwise one of the four cases applies. Indeed, if $G$ is
  $2$-edge-connected and has no proper rigid subgraph, then $k \ge 0$, as
  the partition of $V(G)$ into one part has value $0$
  (\cref{def:D-deficiency}). In each case $P(G)$ holds by hypothesis.
\end{proof}

Katoh and Tanigawa's Lemma~6.1~\cite[p.~672]{katohTanigawa2011} gives
the realization in case~(i). Its hypothesis is KT eq.~(6.1): every
minimal $k'$-dof-graph $H$ with fewer vertices than $G$ has a
panel-hinge realization $(H, p_H)$ with
$\operatorname{rank} R(H, p_H) = D(|V(H)| - 1) - k'$, nonparallel if $H$
is simple. The algebraic induction states KT eq.~(6.1) for two kinds of
realization, and we state KT Lemma~6.1 for each. A
nondegenerate-hinge panel realization of $G$
(\cref{def:genuine-hinge-realization}) is a body-hinge framework $(G,p)$
with a nonzero normal $n_v \in K^{d+1}$ at each body $v$, such that
$\operatorname{rank} R(G,p) = D(|V(G)| - 1) - \operatorname{def}(\tilde G)$.
Its hinge $C(p(e))$ at each edge $e$ with ends $u$ and $v$ is nonzero,
and it is $x_1 \wedge \dots \wedge x_{d-1}$ for some vectors $x_i$ in
both panels $n_u^\perp$ and $n_v^\perp$. A multigraph is \emph{simple}
if it has no loops and no two of its edges join the same pair of
vertices~\cite[p.~668]{katohTanigawa2011}. A generic full-rank
realization of $G$ (\cref{def:rank-hypothesis}) is a panel-hinge
framework on $G$ (\cref{def:panel-hinge-framework}) of the same rank
whose normals are pairwise linearly independent
(\cref{def:panel-general-position}). This is the form of Katoh and
Tanigawa's nonparallel realizations that the algebraic induction uses
(\cref{fmlnote:generic-full-rank}).

\begin{lemma}[Nondegenerate-hinge realization at a cut of at most one edge; KT Lemma~6.1]
  \label{lem:case-cut-edge-realization}
  \lean{CombinatorialRigidity.Molecular.case_cut_edge_realization_gen}
  \leanok
  \uses{def:genuine-hinge-realization, lem:cut-edge-decomposition,
        thm:minimal-kdof-reduction-all-k, lem:block-rank-cut,
        lem:relative-deficiency-rank-bound}
  Let $D \ge 2$, and let $G$ be a minimal $k$-dof-graph with at least
  three vertices that is not $2$-edge-connected. Suppose that for every
  $k'$, every minimal $k'$-dof-graph with at least one and fewer than
  $|V(G)|$ vertices has a nondegenerate-hinge panel realization. Then $G$
  has a nondegenerate-hinge panel realization.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:cut-edge-decomposition, lem:block-rank-cut,
        lem:relative-deficiency-rank-bound}
  Part~(2) of \cref{lem:cut-edge-decomposition} (KT Lemmas~3.3
  and~3.6) applies, as $G$ is not $2$-edge-connected. It gives a set
  $V_1$ of vertices with $\emptyset \neq V_1 \neq V(G)$ and the
  following properties. Let $V_2 = V(G) \setminus V_1$. At most one edge
  joins $V_1$ to $V_2$; let $C$ be the set of such edges. Then $G[V_1]$
  and $G[V_2]$ are minimal $k_1$- and $k_2$-dof-graphs, with
  $k = k_1 + k_2 + D - (D-1)\,|C|$. Both have fewer vertices than $G$.
  So by hypothesis they have nondegenerate-hinge panel realizations
  $(G[V_1], p_1)$ and $(G[V_2], p_2)$, with
  $\operatorname{rank} R(G[V_i], p_i) = D(|V_i| - 1) - k_i$.

  Let $p$ have the normals of $p_i$ at the bodies of $V_i$, and the
  hinges of $p_i$ at the edges of $G[V_i]$. Suppose that $C = \{e\}$,
  where $e$ joins $u \in V_1$ to $v \in V_2$. The subspace
  $n_u^\perp \cap n_v^\perp$ of $K^{d+1}$ has dimension at least
  $d - 1$, so it contains linearly independent vectors
  $x_1, \dots, x_{d-1}$. Let $p$ have the hinge
  $x_1 \wedge \dots \wedge x_{d-1} \neq 0$ (\cref{def:extensor}) at $e$.
  Katoh and Tanigawa take the intersection of the two panels as this
  hinge, after they move $p_2$ so that the panels are not parallel. Here
  any nonzero extensor of vectors in both panels serves. So $(G,p)$ has
  nonzero normals, and each of its hinges is nonzero and lies in the
  panels of the ends of its edge.

  The hinges of $p$ at the edges of $G[V_i]$ are those of $p_i$, so
  $\operatorname{rank} R(G[V_i], p) = D(|V_i| - 1) - k_i$. By
  \cref{lem:block-rank-cut},
  \[
    \operatorname{rank} R(G,p) \;\ge\;
    D(|V_1| - 1) - k_1 + (D-1)\,|C| + D(|V_2| - 1) - k_2
    \;=\; D(|V(G)| - 1) - k,
  \]
  by the formula for $k$ and as $|V(G)| = |V_1| + |V_2|$. As every hinge
  of $p$ is nonzero and $\operatorname{def}(\tilde G) = k$,
  \cref{lem:relative-deficiency-rank-bound} gives
  $\operatorname{rank} R(G,p) \le D(|V(G)| - 1) - k$.
\end{proof}

\begin{lemma}[Generic full-rank realization at a cut of at most one edge; KT Lemma~6.1]
  \label{lem:case-cut-edge-realization-gp}
  \lean{CombinatorialRigidity.Molecular.case_cut_edge_realization_gp_gen}
  \leanok
  \uses{def:genuine-hinge-realization, lem:cut-edge-decomposition,
        thm:minimal-kdof-reduction-all-k, lem:block-rank-cut,
        lem:relative-deficiency-rank-bound, lem:rank-polynomial-of-le-finrank,
        lem:genericity-device}
  Let $D \ge 2$, and let $G$ be a simple minimal $k$-dof-graph with at
  least three vertices that is not $2$-edge-connected. Suppose that for
  every $k'$, every simple minimal $k'$-dof-graph with at least one and
  fewer than $|V(G)|$ vertices has a generic full-rank realization. Then
  $G$ has a generic full-rank realization.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:cut-edge-decomposition, lem:block-rank-cut,
        lem:relative-deficiency-rank-bound, lem:rank-polynomial-of-le-finrank,
        lem:genericity-device}
  Let $V_1$, $V_2$, $C$, $k_1$ and $k_2$ be as in the proof of
  \cref{lem:case-cut-edge-realization}. As $G$ is simple, so are
  $G[V_1]$ and $G[V_2]$. By hypothesis they have generic full-rank
  realizations, with normals $q_1$ and $q_2$. For $i = 1, 2$, there is a
  polynomial $Q_i$ in the normals with $Q_i(q_i) \neq 0$ such that
  $\operatorname{rank} R(G[V_i], q) \ge D(|V_i| - 1) - k_i$ at all
  normals $q$ with $Q_i(q) \neq 0$
  (\cref{lem:rank-polynomial-of-le-finrank}). Let $Q_0$ be the product,
  over the pairs of distinct bodies $u$ and $v$, of the $2 \times 2$
  minor of the first two coordinates of $n_u$ and $n_v$. Where $Q_0$
  does not vanish, the normals are pairwise linearly independent. At
  normals on the moment curve each of these minors is a nonzero
  Vandermonde determinant (\cref{lem:moment-curve-general-position}), so
  $Q_0$ is not the zero polynomial. Hence the product $Q_1 Q_2 Q_0$ of
  three nonzero polynomials is not the zero polynomial. Since $K$ is
  infinite, there are normals $q_0$ at which it does not vanish. This is
  the argument of KT Claim~6.4 (\cref{lem:genericity-device}).

  Let $(G, q_0)$ be the panel-hinge framework on $G$ with the normals
  $q_0$, whose hinge at each edge is the intersection of the panels of
  its ends (\cref{def:panel-hinge-framework}). Its normals are pairwise
  linearly independent, so its hinges are nonzero
  (\cref{lem:general-position-support-nonzero}), and
  $\operatorname{rank} R(G[V_i], q_0) \ge D(|V_i| - 1) - k_i$ for
  $i = 1, 2$. As in the proof of \cref{lem:case-cut-edge-realization},
  \cref{lem:block-rank-cut} and \cref{lem:relative-deficiency-rank-bound}
  give $\operatorname{rank} R(G, q_0) = D(|V(G)| - 1) - k$.
\end{proof}
\begin{fmlnote}
  A generic full-rank realization also records the two ends of each edge,
  in some order (\cref{fmlnote:generic-full-rank}). The framework
  $(G, q_0)$ takes them from the endpoint choice of $G$
  (\cref{fmlnote:graph-operations}). The polynomial $Q_i$ is formed at
  the framework on $G[V_i]$ with the normals $q_i$ and these ends. Its
  rank is that of the given realization, as reversing the ends of an
  edge does not change the space of infinitesimal motions. The
  hypothesis of the formal statement also asks that every minimal
  $k'$-dof-graph with at least one and fewer than $|V(G)|$ vertices have
  a nondegenerate-hinge panel realization
  (\cref{def:genuine-hinge-realization}). With it, the hypothesis is
  that of \cref{thm:minimal-kdof-reduction-all-k} for both kinds of
  realization at once. The proof does not use it.
\end{fmlnote}

\subsection{Simplicity, and two adjacent vertices of degree two}
\label{sec:simplicity-and-adjacent-pair}

Katoh and Tanigawa remark that a minimal $k$-dof-graph with at least
three vertices and no proper rigid subgraph is simple, as two edges that
join the same pair of vertices form a proper rigid
subgraph~\cite[p.~677]{katohTanigawa2011}. We prove this remark
(\cref{lem:simple-minimal-noRigid}) after a computation for the
two-vertex double edge. For $d = 3$, Katoh and Tanigawa derive from KT
Lemma~4.6 that such a graph with $k = 0$ has two adjacent vertices $v$
and $a$ of degree two, and they write $N_G(v) = \{a, b\}$ and
$N_G(a) = \{v, c\}$~\cite[p.~680]{katohTanigawa2011}.
We prove the first fact for $d \ge 3$ by a count of degrees
(\cref{lem:adjacent-degree-two-pair}), and derive the chain $b\,v\,a\,c$
from it (\cref{lem:chain-data-of-noRigid}). Case~III of the algebraic
induction uses neither: it extracts a chain of length $d$ at every $d$
(\cref{lem:chain-data-extract}).

\begin{lemma}[The two-vertex double edge is a $0$-dof-graph]
  \label{lem:two-vertex-zero-dof}
  \lean{Graph.isKDof_zero_of_parallel_pair}
  \leanok
  \uses{def:k-dof}
  Let $D \ge 2$, and let $H$ be a multigraph with two vertices
  $x \neq y$ and two edges $e_1 \neq e_2$, each joining $x$ and $y$.
  Then $H$ is a $0$-dof-graph.
\end{lemma}
\begin{proof}
  \leanok
  The partitions of $V(H)$ are $\{V(H)\}$, of value $0$, and
  $\{\{x\}, \{y\}\}$. Both edges join the two parts of the second, so
  its value is $D - 2(D - 1) = 2 - D \le 0$ (\cref{def:D-deficiency}).
  Hence $\operatorname{def}(\tilde H) = 0$. This is the computation in
  the proof of \cref{lem:two-vertex-trichotomy}.
\end{proof}

\begin{lemma}[A minimal $k$-dof-graph with no proper rigid subgraph is simple; KT p.~677]
  \label{lem:simple-minimal-noRigid}
  \lean{Graph.simple_of_isMinimalKDof_of_noRigid}
  \leanok
  \uses{def:k-dof, def:rigid-subgraph, lem:two-vertex-zero-dof}
  Let $D \ge 2$, and let $G$ be a minimal $k$-dof-graph with at least
  three vertices and no proper rigid subgraph. Then $G$ is simple.
\end{lemma}
\begin{proof}
  \leanok
  As we showed before \cref{lem:no-rigid-edge-count}, $G$ has no loops.
  Suppose that two edges $e \neq f$ join vertices $x$ and $y$, so that
  $x \neq y$. The subgraph $H$ of $G$ with the vertices $x$, $y$ and the
  edges $e$, $f$ is a $0$-dof-graph (\cref{lem:two-vertex-zero-dof}),
  and $1 < |V(H)| = 2 < |V(G)|$. So $H$ is a proper rigid subgraph of
  $G$ (\cref{def:rigid-subgraph}), which is impossible.
\end{proof}

\begin{lemma}[Two adjacent vertices of degree two; from KT Lemma~4.6, for $d \ge 3$ (superseded)]
  \label{lem:adjacent-degree-two-pair}
  \lean{Graph.exists_adjacent_degree_two_pair}
  \leanok
  \uses{lem:no-rigid-edge-count, lem:two-edge-conn, def:k-dof, def:rigid-subgraph}
  Let $D \ge 6$, that is, $d \ge 3$, and let $G$ be a minimal
  $0$-dof-graph with at least three vertices and no proper rigid
  subgraph. Then $G$ has two adjacent vertices of degree two.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:two-vertex-zero-dof, lem:simple-minimal-noRigid}
  Suppose not. As we showed before \cref{lem:no-rigid-edge-count}, $G$
  has no loops. As $G$ is a $0$-dof-graph with at least two vertices,
  every vertex of $G$ has degree at least two
  (\cref{lem:deficiency-zero-connected}).
  Let $X_2$ be the set of vertices of degree two, and let $X_3$ be the
  set of the other vertices, which have degree at least three. No edge
  joins two vertices of $X_2$, so every edge at a vertex of $X_2$ has its
  other end in $X_3$. Hence
  \[
    \sum_{v \in X_2} \deg v \;\le\; \sum_{w \in X_3} \deg w.
  \]
  The degrees of the vertices of $G$ sum to $2|E(G)|$. So
  $2|E(G)| \ge 2\sum_{v \in X_2} \deg v = 4|X_2|$, and
  $2|E(G)| \ge 2|X_2| + 3|X_3|$. Adding the first inequality to four
  times the second gives $10|E(G)| \ge 12(|X_2| + |X_3|)$, that is,
  $5|E(G)| \ge 6|V(G)|$.

  On the other hand, KT Lemma~4.5(i) (\cref{lem:no-rigid-edge-count}) at
  $k = 0$ gives $(D-1)\,|E(G)| < D\,|V(G)| - 1$. So
  \[
    5|E(G)| \;<\; \frac{5D}{D-1}\,|V(G)| \;\le\; 6|V(G)|,
  \]
  as $D \ge 6$. This is a contradiction. Katoh and Tanigawa prove their
  Lemma~4.6 with maximal chains~\cite[pp.~664--665]{katohTanigawa2011};
  for two adjacent vertices of degree two, the count above suffices.
\end{proof}

\begin{lemma}[A chain $b\,v\,a\,c$ with inner vertices of degree two; KT p.~680 (superseded)]
  \label{lem:chain-data-of-noRigid}
  \lean{Graph.exists_chain_data_of_noRigid}
  \leanok
  \uses{lem:adjacent-degree-two-pair, lem:simple-minimal-noRigid, def:rigid-subgraph, def:k-dof}
  Let $D \ge 6$, and let $G$ be a minimal $0$-dof-graph with at least
  four vertices and no proper rigid subgraph. Then $G$ has distinct
  vertices $v$, $a$, $b$, $c$ and edges $e_a = va$, $e_b = vb$ and
  $e_c = ac$, such that $e_a$ and $e_b$ are the only edges at $v$, and
  $e_a$ and $e_c$ are the only edges at $a$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:adjacent-degree-two-pair, lem:simple-minimal-noRigid, def:rigid-subgraph}
  By \cref{lem:adjacent-degree-two-pair}, $G$ has adjacent vertices $v$
  and $a$ of degree two. Let $e_a$ be an edge that joins them. By
  \cref{lem:simple-minimal-noRigid}, $G$ is simple. So the other edge
  $e_b$ at $v$ joins $v$ to a vertex $b \notin \{v, a\}$, and the other
  edge $e_c$ at $a$ joins $a$ to a vertex $c \notin \{v, a\}$.

  Suppose that $b = c$. Then the subgraph $T$ of $G$ with the vertices
  $v$, $a$, $b$ and the edges $e_a$, $e_b$, $e_c$ is a triangle. It is a
  $0$-dof-graph. Indeed, two edges of $T$ join the parts of a partition
  of $V(T)$ into two parts, of value $D - 2(D-1) \le 0$, and three join
  those of the partition into three parts, of value
  $2D - 3(D-1) \le 0$ (\cref{def:D-deficiency}). As
  $1 < |V(T)| = 3 < |V(G)|$, $T$ is a proper rigid subgraph of $G$
  (\cref{def:rigid-subgraph}), which is impossible. Hence $b \neq c$.
\end{proof}

\subsection{A chain or a short cycle: KT Lemma~4.6}

We now prove KT Lemma~4.6 for $k = 0$, which we stated before
\cref{lem:low-degree-vertex}. Katoh and Tanigawa use it in the proof of
their Lemma~6.13~\cite[p.~692]{katohTanigawa2011}, which is Case~III of
the algebraic induction at general $d$. If $G$ is a cycle on at most $d$
vertices, they realize it by their Lemma~5.4
(\cref{lem:cycle-realization}). Otherwise they split off at the interior
vertices of a chain $v_0 v_1 \dots v_d$, beginning with
$G_{v_1}^{v_0 v_2}$, which is a minimal $0$-dof-graph by KT Lemma~4.8.
We define chains and cycles, show that a short cycle is a
$0$-dof-graph, and prove the dichotomy
(\cref{lem:chain-cycle-dichotomy}). Then we show that
$G_{v_1}^{v_0 v_2}$ is also simple and smaller than $G$ when $G$ has at
least four vertices (\cref{lem:chain-data-extract}), as Case~III needs.

\begin{definition}[A chain of length $d$; KT Lemma~4.6]
  \label{def:chain-data}
  \lean{Graph.ChainData}
  \leanok
  \uses{def:graph-operations}
  Let $G$ be a multigraph, and let $d \ge 1$. A \emph{chain of length
  $d$} in $G$, written $v_0 v_1 \dots v_d$, consists of distinct vertices
  $v_0, \dots, v_d$ of $G$ and, for $0 \le i \le d - 1$, an edge $f_i$ of
  $G$ joining $v_i$ and $v_{i+1}$, such that for $1 \le i \le d - 1$ the
  only edges of $G$ at the \emph{interior} vertex $v_i$ are $f_{i-1}$ and
  $f_i$.
\end{definition}
\begin{fmlnote}
  The formal chain also records a label $e_0 \notin E(G)$, which
  \cref{lem:chain-data-extract} uses for the new edge of
  $G_{v_1}^{v_0 v_2}$ (\cref{fmlnote:graph-operations}). So the
  declarations that give a chain, pinned on
  \cref{lem:chain-cycle-dichotomy} and \cref{lem:chain-data-extract},
  assume that a label outside $E(G)$ exists.
\end{fmlnote}

\begin{definition}[Cycle graphs; KT Lemma~4.6]
  \label{def:cycle-data}
  \lean{Graph.CycleData, Graph.CycleData.range_vtx}
  \leanok
  \uses{def:graph-operations}
  Let $G$ be a multigraph, and let $m \ge 3$. Then $G$ is a \emph{cycle}
  on $m$ vertices, or a cycle graph, if it has distinct vertices
  $v_0, \dots, v_{m-1}$ and, for $0 \le i \le m - 1$, an edge $f_i$
  joining $v_i$ and $v_{i+1}$, with indices modulo $m$, such that every
  vertex and every edge of $G$ is among them.
\end{definition}
\begin{fmlnote}
  The second pin says that $V(G) = \{v_0, \dots, v_{m-1}\}$. So
  $|V(G)| = m$, as the $v_i$ are distinct
  (\texttt{CycleData.\allowbreak{}vertexSet\_\allowbreak{}ncard}).
\end{fmlnote}

\begin{lemma}[A short cycle is a $0$-dof-graph]
  \label{lem:cycle-zero-dof}
  \lean{Graph.isKDof_zero_of_cycle}
  \leanok
  \uses{def:cycle-data, def:k-dof}
  Let $C$ be a cycle on $m$ vertices (\cref{def:cycle-data}), with
  $3 \le m \le D$. Then $C$ is a $0$-dof-graph.
\end{lemma}
\begin{proof}
  \leanok
  \uses{def:D-deficiency}
  Let $P$ be a partition of $V(C)$ into at least two parts. Each part is
  a proper subset of $V(C)$, so it has a vertex whose successor along
  $C$ lies in another part. The edge between them joins two parts, and
  different parts give different edges. So at least $|P|$ edges join
  different parts, and the value of $P$ (\cref{def:D-deficiency}) is at
  most $D(|P| - 1) - (D-1)|P| = |P| - D \le 0$, as $|P| \le m \le D$.
  The partition into one part has value $0$, so
  $\operatorname{def}(\tilde C) = 0$.
\end{proof}
\begin{fmlnote}
  The pin takes the vertices and the edges of $C$ as two maps on
  $\{0, \dots, m-1\}$, and it does not ask that the vertices be distinct.
\end{fmlnote}

\begin{lemma}[A short cycle or a chain; KT Lemma~4.6]
  \label{lem:chain-cycle-dichotomy}
  \lean{Graph.chainData_or_cycleData_of_noRigid,
        Graph.cycle_isProperRigidSubgraph}
  \leanok
  \uses{def:chain-data, def:cycle-data, def:k-dof, def:rigid-subgraph,
        lem:two-edge-conn, lem:no-rigid-edge-count, lem:simple-minimal-noRigid}
  Let $D \ge 3$, and let $G$ be a minimal $0$-dof-graph with at least
  three vertices and no proper rigid subgraph. Then $G$ is a cycle on at
  most $d$ vertices (\cref{def:cycle-data}), or $G$ has a chain of length
  $d$ (\cref{def:chain-data}).
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:two-edge-conn, lem:no-rigid-edge-count, lem:simple-minimal-noRigid,
        def:rigid-subgraph, lem:cycle-zero-dof}
  We follow Katoh and Tanigawa's
  proof~\cite[pp.~664--665]{katohTanigawa2011}, and give its count in
  full. As $D \ge 3$, we have $d \ge 2$, and $d \le D$. By KT Lemma~3.1
  (\cref{lem:two-edge-conn}), $G$ is $2$-edge-connected. So it is
  connected, and each of its vertices has degree at least two
  (\cref{lem:deficiency-zero-connected}). By
  \cref{lem:simple-minimal-noRigid}, $G$ is simple.

  Let $u_0$ be a vertex of $G$, and let $g_1$ be an edge at $u_0$,
  joining it to $u_1$. We extend the path $u_0 u_1$ one edge at a time.
  While its last vertex $u_j$ has degree two and $j < d$, let $g_{j+1}$
  be the edge at $u_j$ other than $g_j$, and let $u_{j+1}$ be its other
  end. The vertices $u_1, \dots, u_{j-1}$ have degree two and both of
  their edges are on the path, so $u_{j+1}$ is not one of them. As $G$ is
  simple, $u_{j+1} \neq u_j$, and $u_2 \neq u_0$. So $u_{j+1}$ is a new
  vertex, or $u_{j+1} = u_0$ with $j \ge 2$. We call the path so obtained
  the \emph{walk} from $u_0$ along $g_1$. It ends in one of three ways.

  If the walk reaches length $d$, then $u_0 u_1 \dots u_d$ is a chain of
  length $d$ (\cref{def:chain-data}).

  If the walk closes up, with $u_{j+1} = u_0$, then $u_0, \dots, u_j$ and
  $g_1, \dots, g_{j+1}$ form a cycle $C$ on $m = j + 1$ vertices, where
  $3 \le m \le d$. If $u_0$ has degree two, then no edge of $G$ leaves
  $C$. As $G$ is connected, $G = C$ is a cycle on at most $d$ vertices
  (\cref{def:cycle-data}). If $u_0$ has degree at least three, then $C$
  is a $0$-dof-graph, as $m \le d \le D$ (\cref{lem:cycle-zero-dof}).
  A third edge at $u_0$ has its other end outside $C$, since the other
  vertices of $C$ have degree two. So $C$ is a proper rigid subgraph of
  $G$ (\cref{def:rigid-subgraph}), which is impossible. Katoh and Tanigawa's maximal chains may begin and
  end at the same vertex. A closed one of length $d$ has only $d$
  distinct vertices, and this case excludes it.

  Otherwise the walk stops at a vertex $u_j$ of degree at least three,
  with $1 \le j \le d - 1$.

  Suppose now that $G$ is not a cycle on at most $d$ vertices and has no
  chain of length $d$. Then every walk stops at a vertex of degree at
  least three, after at most $d - 1$ edges. So a walk that starts at a
  vertex of degree at least three has at most $d - 2$ interior vertices,
  all of degree two. For $i \ge 2$, let $X_i$ be the set of vertices of
  $G$ of degree $i$. Let $v \in X_2$, and let $e$ be an edge at $v$. The
  walk from $v$ along $e$ stops at a vertex $u$ of degree at least three.
  Read backwards, it is a path that leaves $u$ along an edge $g$ and
  passes through vertices of degree two up to $v$. At a vertex of degree
  two, a walk leaves by the edge by which it did not arrive. So this path
  is the beginning of the walk from $u$ along $g$, and $v$, of degree
  two, is an interior vertex of that walk. The pair $(u, g)$ and the
  position of $v$ on its walk determine $v$. They also determine $e$, as
  the edge by which that walk arrives at $v$. There are
  $\sum_{i \ge 3} i\,|X_i|$ pairs $(u, g)$ with $u$ of degree at least
  three, and so
  \[
    2\,|X_2| \;\le\; (d-2) \sum_{i \ge 3} i\,|X_i|.
  \]
  Katoh and Tanigawa obtain this inequality from their eqs.~(4.6)
  and~(4.7), by counting maximal chains.

  As every vertex of $G$ has degree at least two,
  $|V(G)| = \sum_{i \ge 2} |X_i|$. Adding $2|X_i|$ for each $i \ge 3$ to
  both sides of the inequality gives KT eq.~(4.8),
  \[
    2\,|V(G)| \;\le\; \sum_{i \ge 3} \bigl(i(d-2) + 2\bigr)|X_i|.
  \]
  For $i \ge 3$ we have $i(d-2) + 2 \le (D-1)(i-2)$, the inequality
  before KT eq.~(4.9). Indeed, at $i = 3$ twice the right side minus
  twice the left side is $(d-2)(d-3) \ge 0$, as $D = d(d+1)/2$. Each step
  from $i$ to $i + 1$ adds $d - 2$ to the left side and $D - 1 \ge d - 2$
  to the right side. The degrees of $G$ sum to $2|E(G)|$, so
  $\sum_{i \ge 3} (i-2)|X_i| = 2|E(G)| - 2|V(G)|$, and KT eq.~(4.9) reads
  \[
    2\,|V(G)| \;\le\; (D-1) \sum_{i \ge 3} (i-2)|X_i|
    \;=\; (D-1)\bigl(2|E(G)| - 2|V(G)|\bigr).
  \]
  Hence $D\,|V(G)| \le (D-1)\,|E(G)|$. But KT Lemma~4.5(i)
  (\cref{lem:no-rigid-edge-count}, at $k = 0$) gives
  $(D-1)\,|E(G)| < D(|V(G)| - 1) + D - 1 < D\,|V(G)|$, a contradiction.
\end{proof}
\begin{fmlnote}
  The first pin is the lemma. The second is the case of a walk that
  closes up at a vertex of degree at least three, in a simple
  multigraph: the cycle is a proper rigid subgraph.
\end{fmlnote}

The cycle alternative cannot be dropped. Let $3 \le m \le d$, and let
$C$ be a cycle on $m$ vertices. It is a $0$-dof-graph
(\cref{lem:cycle-zero-dof}). A subgraph of $C$ that misses an edge or a
vertex has no cycle, so its partition into single vertices has positive
value when it has at least two vertices. So removing an edge of $C$
raises the deficiency, and $C$ is minimal, as we noted after
\cref{def:k-dof}. Also, $C$ has no proper rigid subgraph. A chain of
length $d$ has $d + 1 > m$ vertices, so $C$ has none. Hence the cycle
alternative occurs for every $d \ge 3$. Under the bound $|V(G)| \ge 4$
of the next lemma, it occurs exactly when $d \ge 4$.

\begin{lemma}[The chain and its splitting off; KT Lemmas~4.6, 4.8 and~6.7]
  \label{lem:chain-data-extract}
  \lean{Graph.chainData_extract,
        Graph.splitOff_simple_of_noRigid_of_card}
  \leanok
  \uses{def:chain-data, def:cycle-data, lem:chain-cycle-dichotomy,
        lem:reduction-step, lem:reduction-measure, lem:simple-minimal-noRigid}
  Let $D \ge 3$, and let $G$ be a simple minimal $0$-dof-graph with at
  least four vertices and no proper rigid subgraph. Then $G$ is a cycle
  on at most $d$ vertices, or $G$ has a chain $v_0 v_1 \dots v_d$ of
  length $d$ such that $G_{v_1}^{v_0 v_2}$ is a simple minimal
  $0$-dof-graph with at least two and fewer than $|V(G)|$ vertices.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:chain-cycle-dichotomy, lem:reduction-step, lem:reduction-measure,
        lem:simple-minimal-noRigid, lem:cycle-zero-dof}
  By \cref{lem:chain-cycle-dichotomy}, $G$ is a cycle on at most $d$
  vertices, or it has a chain $v_0 v_1 \dots v_d$ of length $d$. Suppose
  the second. As $D \ge 3$, we have $d \ge 2$, so $v_1$ is an interior
  vertex of the chain. Its only edges are $f_0$ and $f_1$, which join it
  to $v_0$ and to $v_2$, and $v_0, v_1, v_2$ are distinct. As $G$ has at
  least three vertices and no proper rigid subgraph, $G_{v_1}^{v_0 v_2}$
  is a minimal $0$-dof-graph by KT Lemma~4.8(i) (part~(i) of
  \cref{lem:reduction-step}). Its vertices are the $|V(G)| - 1 \ge 3$
  vertices of $G$ other than $v_1$ (\cref{lem:reduction-measure}).

  It remains to show that $G_{v_1}^{v_0 v_2}$ is simple. This is KT
  Lemma~6.7(ii)~\cite[p.~677]{katohTanigawa2011}, and we follow its
  proof. The graph $G$ is simple, by hypothesis and also by
  \cref{lem:simple-minimal-noRigid}, and the new edge $v_0 v_2$ is not a
  loop. So if $G_{v_1}^{v_0 v_2}$ were not simple, $G$ would have an edge
  $g$ joining $v_0$ and $v_2$. Then $v_0, v_1, v_2$ and $f_0, f_1, g$
  form a cycle on three vertices in $G$. As $3 \le D$, this cycle is a
  $0$-dof-graph (\cref{lem:cycle-zero-dof}). As $|V(G)| \ge 4$, it is a
  proper rigid subgraph of $G$, which is impossible.
\end{proof}
\begin{fmlnote}
  The first pin is the lemma. The second pin is KT Lemma~6.7(ii), for
  every simple multigraph with at least four vertices and no proper
  rigid subgraph.
\end{fmlnote}
