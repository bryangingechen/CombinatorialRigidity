\subsection{Case I: a proper rigid subgraph}
\label{sec:molecular-algebraic-induction-caseI}

This subsection proves Katoh--Tanigawa's Case~I: a minimal $k$-dof-graph
with a proper rigid subgraph $H$ realizes its target rank by contracting
$H$ to a single pinned body and gluing the rigid block to the
contraction's own inductive realization (KT eq.~(6.3)). Two pieces of
shared infrastructure come first: the panel-cycle realization (KT
Lemma~5.4), used later in Case~III, and the block pin --- the
framework-side model of contracting a rigid block to one body. The Case~I
accounting itself (\cref{lem:case-I}) is genericity-free; genericity
enters only through KT Claim~6.4 (\cref{lem:claim-6-4}), needed to realize
both legs of the block-triangular splice at a single common placement.
The realization lemmas (\cref{lem:case-I-realization}, \cref{lem:case-I-realization-nonsimple},
\cref{lem:case-I-realization-h65}, \cref{lem:case-I-dispatch}) assemble
this into KT Lemmas~6.2, 6.3, and 6.5.

We first prove KT Lemma~5.4, which Case~III uses. It states that a
cycle on $m$ vertices, for $3 \le m \le D$, has an infinitesimally rigid
nonparallel panel-hinge
realization~\cite[p.~670]{katohTanigawa2011}. Katoh and Tanigawa note
that a cycle can be analyzed directly from the definition of
infinitesimal motions. For the computation in dimension~$3$ they refer
to Crapo and Whiteley~\cite[Proposition~3.4]{crapoWhiteley1982} and to
Whiteley~\cite[Proposition~3]{whiteley1999}, and they note that it
works in every dimension. Crapo and Whiteley show that a cycle of $m$
panels and $m$ hinges in dimension~$3$ is infinitesimally rigid if and
only if the extensors of its hinges are linearly independent. We need
one direction: in every dimension, a cycle whose hinges have linearly
independent extensors is infinitesimally rigid
(\cref{lem:cycle-realization-rigid}). Since $\bigwedge^{d-1} K^{d+1}$
has dimension $D$, the hinges of a cycle on $m$ bodies can have linearly
independent extensors only if $m \le D$
(\cref{lem:cycle-realization-dim-bound}), the bound of KT Lemma~5.4.
For $m \le d + 1$, the normals of \cref{lem:cycle-normals} give a
generic full-rank realization of the cycle
(\cref{lem:cycle-realization}).

The case of two bodies joined by two edges is the double edge of KT
Lemma~5.3, which \cref{lem:theorem-55-base} treats for body-hinge
frameworks. For panel-hinge frameworks it reads as follows.

\begin{lemma}[Two bodies joined by two independent hinges]
  \label{lem:cycle-realization-base}
  \lean{CombinatorialRigidity.Molecular.PanelHingeFramework.toBodyHinge_rankHypothesis_zero}
  \leanok
  \uses{def:panel-hinge-framework, def:rank-hypothesis, lem:theorem-55-base}
  Let $(G,p)$ be a panel-hinge framework, and let $e_1$ and $e_2$ be
  edges of $G$ that both join the bodies $u \neq v$, with $C(p(e_1))$ and
  $C(p(e_2))$ linearly independent. Then every infinitesimal motion $S$
  of $(G,p)$ has $S(u) = S(v)$. In particular, if $u$ and $v$ are the
  only bodies, then $(G,p)$ is infinitesimally rigid, of rank $D$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:theorem-55-base}
  This is \cref{lem:theorem-55-base} for the body-hinge interpretation of
  $(G,p)$ (\cref{def:panel-hinge-framework}). An infinitesimally rigid
  framework on two bodies has rank $D(2 - 1) = D$
  (\cref{lem:trivial-motions-rank-bound}).
\end{proof}

\begin{lemma}[At most $D$ independent hinges]
  \label{lem:cycle-realization-dim-bound}
  \lean{CombinatorialRigidity.Molecular.card_le_screwDim_of_linearIndependent,
        CombinatorialRigidity.Molecular.PanelHingeFramework.card_le_screwDim_of_supportExtensor_linearIndependent}
  \leanok
  \uses{def:panel-hinge-framework, def:rigidity-matrix}
  Let $(G,p)$ be a panel-hinge framework, and let $e_1, \dots, e_m$ be
  edges of $G$ with $C(p(e_1)), \dots, C(p(e_m))$ linearly independent.
  Then $m \le D$.
\end{lemma}
\begin{proof}
  \leanok
  The extensors lie in $\bigwedge^{d-1} K^{d+1}$, which has dimension
  $\binom{d+1}{d-1} = D$ (the remark after
  \cref{lem:extensor-independence}).
\end{proof}

The bound of \cref{lem:cycle-realization-dim-bound} is attained by panel
support extensors (\cref{lem:exists-independent-panel-extensor}). We
first reduce their linear independence to that of joins. The proofs of
\cref{lem:triangle-normals} and \cref{lem:cycle-normals} use this
reduction.

\begin{lemma}[Independence of panel support extensors]
  \label{lem:panel-support-extensor-independence}
  \lean{CombinatorialRigidity.Molecular.panelSupportExtensor_linearIndependent_iff}
  \leanok
  \uses{def:panel-support-extensor, def:meet-complement-iso}
  Let $a_i, b_i \in K^{d+1}$ for $1 \le i \le m$. Then the extensors
  $C(a_i, b_i)$ with $1 \le i \le m$ are linearly independent if and
  only if the joins $a_i \vee b_i$ are linearly independent in
  $\bigwedge^2 K^{d+1}$.
\end{lemma}
\begin{proof}
  \leanok
  By definition, $C(a_i, b_i) = *(a_i \vee b_i)$
  (\cref{def:panel-support-extensor}), and the complement isomorphism $*$
  (\cref{def:meet-complement-iso}) is a linear isomorphism. A linear
  isomorphism preserves and reflects linear independence.
\end{proof}

\begin{lemma}[Up to $D$ independent panel support extensors]
  \label{lem:exists-independent-panel-extensor}
  \lean{CombinatorialRigidity.Molecular.exists_independent_panelSupportExtensor,
        CombinatorialRigidity.Molecular.exists_independent_normalsJoin}
  \leanok
  \uses{def:panel-support-extensor, lem:panel-support-extensor-independence,
        lem:extensor-independence}
  Let $m \le D$. Then there are $a_i, b_i \in K^{d+1}$ for
  $1 \le i \le m$ such that the extensors $C(a_i, b_i)$ are linearly
  independent, and so are the joins $a_i \vee b_i$.
\end{lemma}
\begin{proof}
  \leanok
  Let $e_0, \dots, e_d$ be the standard basis of $K^{d+1}$. The
  $\binom{d+1}{2} = D$ joins $e_s \vee e_t$ with $s < t$ are linearly
  independent (the remark after \cref{def:extensor}). Choose $m$ distinct
  pairs $s_i < t_i$, and let $a_i = e_{s_i}$ and $b_i = e_{t_i}$. Then
  the joins $a_i \vee b_i$ are linearly independent, and so are the
  extensors $C(a_i, b_i)$ (\cref{lem:panel-support-extensor-independence}).
\end{proof}

The pairs $(a_i, b_i)$ of \cref{lem:exists-independent-panel-extensor}
need not be the normals at the ends of the edges of a cycle, whose
consecutive hinges share a normal. For a cycle, the normals of
\cref{lem:cycle-normals} serve instead (\cref{lem:cycle-realization}).

\begin{lemma}[A cycle with independent hinges is rigid]
  \label{lem:cycle-realization-rigid}
  \lean{CombinatorialRigidity.Molecular.PanelHingeFramework.toBodyHinge_rankHypothesis_zero_cycle,
        CombinatorialRigidity.Molecular.BodyHingeFramework.rankHypothesis_zero_of_cycle,
        CombinatorialRigidity.Molecular.BodyHingeFramework.isTrivialMotion_of_isInfinitesimalMotion_cycle,
        CombinatorialRigidity.Molecular.BodyHingeFramework.theorem_55_cycle}
  \leanok
  \uses{def:panel-hinge-framework, def:rank-hypothesis, lem:cycle-realization-base}
  Let $(G,p)$ be a body-hinge framework, let $m \ge 1$, and let
  $v_0, \dots, v_{m-1}$ be bodies, not necessarily distinct. Let
  $f_0, \dots, f_{m-1}$ be edges of $G$ such that $f_i$ joins $v_i$ and
  $v_{i+1}$, with indices modulo $m$, and suppose that
  $C(p(f_0)), \dots, C(p(f_{m-1}))$ are linearly independent. Then every
  infinitesimal motion $S$ of $(G,p)$ has
  $S(v_0) = S(v_1) = \cdots = S(v_{m-1})$. In particular, if
  $v_0, \dots, v_{m-1}$ are all the bodies, then $(G,p)$ is
  infinitesimally rigid.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:rank-parallel-full}
  We argue as for KT Lemma~5.3 (\cref{lem:rank-parallel-full}) and
  \cref{lem:theorem-55-triangle}, which are the cases $m = 2$ and $m = 3$
  with distinct bodies. Let $S$ be an infinitesimal motion. By the hinge
  constraints (\cref{def:hinge-constraint}), there are scalars
  $\lambda_0, \dots, \lambda_{m-1}$ with
  $S(v_i) - S(v_{i+1}) = \lambda_i C(p(f_i))$ for each $i$.

  Since the indices are taken modulo $m$, the $m$ differences sum to
  zero. Hence $\sum_i \lambda_i C(p(f_i)) = 0$, and by the linear
  independence every $\lambda_i$ is zero. So $S(v_i) = S(v_{i+1})$ for
  every $i$. If $v_0, \dots, v_{m-1}$ are all the bodies, then $S$ is
  trivial (\cref{def:dof-generic}).
\end{proof}

\begin{fmlnote}
  The first three pinned theorems are the case in which the bodies are
  $0, 1, \dots, m - 1$ and $v_i = i$: for a panel-hinge framework in the
  form $\dim Z(G,p) = D$, and for a body-hinge framework in that form and
  in the form that every infinitesimal motion is trivial. The fourth is
  the lemma as stated.
\end{fmlnote}

\begin{lemma}[Realizing a cycle; KT Lemma 5.4]
  \label{lem:cycle-realization}
  \lean{CombinatorialRigidity.Molecular.PanelHingeFramework.cycle_realization}
  \leanok
  \uses{def:rigidity-matrix, def:rank-hypothesis, def:cycle-data, def:k-dof,
        lem:cycle-realization-base, lem:cycle-realization-dim-bound,
        lem:cycle-realization-rigid, lem:cycle-normals,
        lem:case-III-claim612-line-in-panel-union, lem:genericity-device}
  Let $K$ be an infinite field, let $3 \le m \le D$, and let $G$ be a
  cycle on $m$ vertices (\cref{def:cycle-data}) that is a minimal
  $0$-dof-graph. Then $G$ has a generic full-rank realization
  (\cref{def:rank-hypothesis}), of rank $D(m - 1)$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:cycle-realization-rigid,
        lem:cycle-normals, lem:case-III-claim612-line-in-panel-union}
  We give the proof for $m \le d + 1$, the range of the formalization;
  the note after the proof compares this range with Katoh and Tanigawa's.
  Let
  $v_0, \dots, v_{m-1}$ and $f_0, \dots, f_{m-1}$ be the vertices and the
  edges of $G$, where $f_i$ joins $v_i$ and $v_{i+1}$, with indices
  modulo $m$ (\cref{def:cycle-data}). Let $e_0, \dots, e_d$ be the
  standard basis of $K^{d+1}$, and give the body $v_i$ the normal $e_i$,
  which is possible as $m \le d + 1$. Up to sign, the hinge at $f_i$ is
  $C(e_i, e_{i+1})$, again with indices modulo $m$
  (\cref{def:panel-hinge-framework}). These $m$ extensors are linearly
  independent, by \cref{lem:cycle-normals} and its proof, and so are the
  hinges.

  So every infinitesimal motion is constant on
  $\{v_0, \dots, v_{m-1}\} = V(G)$ (\cref{lem:cycle-realization-rigid}).
  As $G$ is a $0$-dof-graph, $\operatorname{def}(\tilde G) = 0$, and the
  realization is at the target rank $D(m - 1)$
  (\cref{rem:rank-hypothesis-relative}). Any two distinct
  standard basis vectors are linearly independent, so the normals are in
  general position (\cref{def:panel-general-position}). Hence the
  realization is a generic full-rank realization of $G$.
\end{proof}

\begin{fmlnote}
  The pinned theorem assumes $m \le d + 1$. This is Katoh and Tanigawa's
  range when $d = 2$, where $D = 3$. For $d \ge 3$ we have $d + 1 < D$,
  and the cycles with $d + 1 < m \le D$ are not covered. Case~III applies
  the lemma to cycles on at most $d$ vertices, as Katoh and Tanigawa do
  in their proof of Lemma~6.13 (KT p.~692; \cref{lem:chain-cycle-dichotomy}),
  and to triangles, for which $m = 3 = d + 1$ when $d = 2$
  (\cref{lem:case-III}).

  The formal framework also has bodies outside $V(G)$
  (\cref{fmlnote:generic-full-rank}), and general position concerns their
  normals too. The pinned theorem gives those bodies the zero normal.
  It then chooses new normals for all the bodies, in general position, at
  which the rank of $R(G,p)$ is still $D(m - 1)$. Such normals exist
  because $K$ is infinite: each of the two conditions holds off the zero
  set of a nonzero polynomial in the normals. This is the argument of
  \cref{lem:genericity-device}, later in this chapter
  (\cref{sec:molecular-algebraic-induction-genericity}), for one
  realization. The step is the last theorem pinned on
  \cref{lem:case-III-claim612-line-in-panel-union}
  (\texttt{hasGenericFullRankRealization\_\allowbreak{}of\_\allowbreak{}rigidOn\_\allowbreak{}ofNormals}).
\end{fmlnote}

Katoh and Tanigawa's Case~I contracts a proper rigid subgraph $H$ of $G$
to one vertex (KT \S6.2, p.~673). In terms of infinitesimal motions, the
corresponding operation pins all the bodies of $H$ at once, as
$\operatorname{pinned}(v)$ does for one body in KT Lemma~5.1
(\cref{lem:rank-delete-vertex}).

\begin{definition}[The block pin of a set of bodies]
  \label{def:pinned-motions-on}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.pinnedMotionsOn,
        CombinatorialRigidity.Molecular.BodyHingeFramework.pinnedMotionsOn_singleton,
        CombinatorialRigidity.Molecular.BodyHingeFramework.pinnedMotionsOn_eq_iInf}
  \leanok
  \uses{def:rigidity-matrix, lem:rank-delete-vertex}
  Let $(G,p)$ be a body-hinge framework, and let $s$ be a set of bodies.
  The \emph{block pin} of $s$ is the subspace
  \[
    Z_s(G,p) \;=\; \{S \in Z(G,p) : S(v) = 0 \text{ for every } v \in s\}
  \]
  of the infinitesimal motions that vanish on $s$.
\end{definition}

For a single body, $Z_{\{v\}}(G,p)$ is the subspace
$\operatorname{pinned}(v)$ of \cref{lem:rank-delete-vertex}. For a
nonempty set $s$, $Z_s(G,p)$ is the intersection of the subspaces
$\operatorname{pinned}(v)$ with $v \in s$.

\begin{lemma}[Deleting edges enlarges the block pin]
  \label{lem:pinned-motions-on-mono}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.pinnedMotionsOn_le_withGraph_of_le,
        CombinatorialRigidity.Molecular.BodyHingeFramework.finrank_pinnedMotionsOn_le_of_graph_le}
  \leanok
  \uses{def:pinned-motions-on, def:framework-with-graph, lem:motions-mono-of-graph-le}
  Let $(G,p)$ be a body-hinge framework, let $s$ be a set of bodies, and
  let $(G',p)$ be the restriction of $(G,p)$ to a subgraph $G'$ of $G$
  (\cref{def:framework-with-graph}). Then
  $Z_s(G,p) \subseteq Z_s(G',p)$. In particular,
  $\dim Z_s(G,p) \le \dim Z_s(G',p)$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:motions-mono-of-graph-le}
  The conditions $S(v) = 0$ for $v \in s$ do not involve the graph, and
  $Z(G,p) \subseteq Z(G',p)$ (\cref{lem:motions-mono-of-graph-le}).
\end{proof}

For one body, $\dim \operatorname{pinned}(v) + D = \dim Z(G,p)$
(\cref{lem:rank-delete-vertex}). For a larger set of bodies, only an
inequality holds.

\begin{lemma}[The block pin and the trivial motions]
  \label{lem:pinned-motions-on-rank-bound}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.screwDim_add_finrank_pinnedMotionsOn_le}
  \leanok
  \uses{def:pinned-motions-on, lem:rank-delete-vertex, lem:trivial-motions-rank-bound}
  Let $(G,p)$ be a body-hinge framework, and let $s$ be a nonempty set of
  bodies. Then
  \[
    D + \dim Z_s(G,p) \;\le\; \dim Z(G,p).
  \]
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:rank-delete-vertex}
  Let $v \in s$. Then $Z_s(G,p) \subseteq \operatorname{pinned}(v)$, and
  $\operatorname{pinned}(v)$ meets the space of trivial motions only in
  $0$ (\cref{lem:rank-delete-vertex}). The trivial motions form a
  $D$-dimensional subspace of $Z(G,p)$
  (\cref{lem:trivial-motions-rank-bound}). So $Z(G,p)$ contains the
  direct sum of this subspace and $Z_s(G,p)$.
\end{proof}

A block pin also detects infinitesimal rigidity.

\begin{lemma}[Rigidity and the block pin]
  \label{lem:pinned-motions-on-rigid-iff}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.pinnedMotionsOn_eq_bot_of_isInfinitesimallyRigid,
        CombinatorialRigidity.Molecular.BodyHingeFramework.finrank_pinnedMotionsOn_eq_zero_of_isInfinitesimallyRigid,
        CombinatorialRigidity.Molecular.BodyHingeFramework.isInfinitesimallyRigid_of_block_of_pinnedMotionsOn_eq_bot}
  \leanok
  \uses{def:pinned-motions-on}
  Let $(G,p)$ be a body-hinge framework, and let $s$ be a nonempty set of
  bodies. If $(G,p)$ is infinitesimally rigid, then $Z_s(G,p) = 0$.
  Conversely, if every infinitesimal motion of $(G,p)$ is constant on $s$
  and $Z_s(G,p) = 0$, then $(G,p)$ is infinitesimally rigid.
\end{lemma}
\begin{proof}
  \leanok
  Let $(G,p)$ be infinitesimally rigid, and let $S \in Z_s(G,p)$. Then
  $S$ is trivial, that is, constant (\cref{def:dof-generic}), and it
  vanishes at a body of $s$. So $S = 0$.

  Conversely, let $S$ be an infinitesimal motion, let $v \in s$, and let
  $T$ be the trivial motion with the value $S(v)$ at every body. Then
  $S - T$ is an infinitesimal motion, and it vanishes on $s$ because $S$
  is constant on $s$. So $S - T \in Z_s(G,p) = 0$, and $S = T$ is
  trivial.
\end{proof}

\begin{lemma}[Case I: rigid-subgraph contraction realizes the rank; KT Lemmas 6.2, 6.3, 6.5]
  \label{lem:case-I}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.isInfinitesimallyRigidOn_iff_pinnedMotionsOn_le,
        CombinatorialRigidity.Molecular.PanelHingeFramework.toBodyHinge_rankHypothesis_iff_finrank_pinnedMotionsOn,
        CombinatorialRigidity.Molecular.BodyHingeFramework.rankHypothesis_iff_finrank_pinnedMotionsOn}
  \leanok
  \uses{thm:minimal-kdof-reduction, lem:contraction-minimality,
        lem:rank-delete-vertex, def:pinned-motions-on,
        lem:pinned-motions-on-rank-bound, lem:genericity-device}
  Let $G$ be a minimal $k$-dof-graph with a proper rigid subgraph $H$ on
  the (nonempty) body set $s = V(H)$. Contracting $H$ to a single pinned
  body is the block pin $Z_s(G,p)$ (\cref{def:pinned-motions-on}), the
  framework-side carrier of the contraction $G/E(H)$ (a smaller minimal
  $k$-dof-graph by \cref{lem:contraction-minimality}, supplied the
  induction hypothesis). Then $(G,p)$ realizes the $V(G)$-relative target
  rank at $k'$ (\cref{def:rank-hypothesis}) \emph{iff} the rigid block $H$
  contributes its full $D(|V(H)|-1)$ rows and the residual contraction
  contributes the rest --- the block-triangular rank addition of
  KT~(6.3): $\operatorname{rank} R(G,p) = D(|V(H)|-1) +
  \operatorname{rank} R(G/E(H), p_2)$, glued via the pin-a-body
  Lemma~5.1 (\cref{lem:rank-delete-vertex}).
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:rank-delete-vertex, lem:pinned-motions-on-rank-bound,
        lem:genericity-device}
  Pin the rigid block $H$ and place the contraction $G/E(H)$ at its
  inductive rank. The block pin and the $D$-dimensional trivial motions
  are disjoint, giving the genericity-free lower bound
  (\cref{lem:pinned-motions-on-rank-bound}). The combined rigidity matrix
  is block-triangular --- the $H$-rows occupy only the $V(H)$ columns,
  the contraction rows the rest --- so at a generic point the ranks add
  and the reverse bound holds (the entries are polynomials in the panel
  coordinates, so a common non-root of the relevant minors attains
  the maximal rank, KT's Claim~6.4, \cref{lem:genericity-device}). The
  two bounds together pin the parent rank to $D(|V(G)|-1) - k$. This
  accounting depends only on $s, t \subseteq V(G)$, not on the ambient
  body type, so it composes through the vertex-reducing induction; the
  companion nullity form (relating $\dim Z(G,p)$ to the ambient body
  type) remains available for the deficiency and Proposition~1.1 path.
\end{proof}

\begin{lemma}[Panel-coordinate row coordinatization; KT Claim 6.4]
  \label{lem:rows-polynomial-in-normals}
  \lean{CombinatorialRigidity.Molecular.PanelHingeFramework.exists_good_realization_ofParam}
  \leanok
  \uses{def:panel-support-extensor, def:rigidity-matrix, lem:genericity-device}
  Regard a panel-hinge realization of a fixed graph $G$ as a point of
  panel-coordinate space: an assignment $q \colon V_\alpha \times \{1,
  \dots, k+2\} \to K$ of a normal $n_v = q(v, -) \in K^{k+2}$ to each
  body. Then the rigidity rows of $R(G,q)$ form a finite family of
  functionals on $K^{D|V_\alpha|}$ whose coordinates, in the standard
  basis, are degree-$2$ polynomials in the entries of $q$ --- exactly the
  input shape the genericity device (\cref{lem:genericity-device}) needs
  to be applied to a \emph{varying} panel realization, rather than a
  constant one.
\end{lemma}
\begin{proof}
  \leanok
  A coordinate of the grade-$2$ join $n_u \wedge n_v$ against the
  standard dual basis of $\Lambda^2 K^{k+2}$ is the $2 \times 2$ minor
  $n_u(i)\,n_v(j) - n_u(j)\,n_v(i)$, a determinant of the two normals'
  coordinates and hence bilinear, so it is a degree-$2$ polynomial in the
  panel coordinates $q$. The supporting extensor $C(p(e)) =
  \mathrm{complementIso}(n_u \wedge n_v)$ (\cref{def:panel-support-extensor})
  is a fixed linear image of the join, so its coordinates stay degree-$2$.
  Each rigidity row spans the annihilator of its edge's supporting
  extensor in $(K^{k+2})^*$, and its coordinates are in turn a fixed
  linear combination of the extensor's coordinates at the two endpoints,
  so they too are degree-$2$ in $q$. Applying the genericity device to
  this family, with the witnessed independent subfamily supplied by the
  general-position seed (\cref{lem:exists-independent-panel-extensor}),
  produces a normal assignment attaining the codimension bound
  $\#s + \dim Z(G,q) \le D|V_\alpha|$ at the same corank.
\end{proof}

\begin{lemma}[Case I splice glue; KT eq.\ (6.3)]
  \label{lem:case-I-splice-seed}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.isInfinitesimallyRigidOn_of_splice}
  \leanok
  \uses{def:framework-with-graph, def:rank-hypothesis, lem:case-I}
  Let $H$ (on $s_H = V(H)$) be a proper rigid subgraph and $G/E(H)$ (on
  $s_c$) its contraction, both subgraphs of $G$ sharing the contracted
  body $c \in s_H \cap s_c$ and covering $V(G) \subseteq s_H \cup s_c$.
  If $F$ restricted to $H$ (\cref{def:framework-with-graph}) is
  infinitesimally rigid on $s_H$, and $F$ restricted to $G/E(H)$ is
  infinitesimally rigid on $s_c$, then $F$ is infinitesimally rigid on
  $V(G)$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:motions-mono-of-graph-le}
  Each leg transports to the parent by the monotonicity that re-adding
  edges only shrinks the motion space (\cref{lem:motions-mono-of-graph-le},
  since $H, G/E(H) \le G$), so a parent motion is already constant on
  $s_H$ and on $s_c$. The two constants agree at the shared contracted
  body $c$, so every motion is constant across $s_H \cup s_c \supseteq
  V(G)$.
\end{proof}

This is Katoh--Tanigawa's block-triangular splice (KT eq.~(6.3)), read as
two overlapping rigid pieces glued along a shared body. What the eq.~(6.3)
block form records --- and the reason it keeps the two blocks apart rather
than working on the parent body set $V(G)$ throughout --- is that the two
pieces are rigid on two \emph{different} body sets. Write $V' = V(H)$ for the
rigid subgraph's bodies and $r$ for the single body the contraction
identifies $V'$ to (KT's $v^*$), so that the contraction $G/E(H)$ lives on
$s_c = (V(G) \setminus V(H)) \cup \{r\}$. The rigid block --- KT's
$R(G',p_1)$ with $G' = H$ --- occupies the $V'$ columns at rank $D(|V'|-1)$;
the surviving block is KT's $R(G,p;\, E \setminus E',\, V \setminus V')$, the
parent rigidity matrix cut to the surviving edges $E(G) \setminus E(H)$ and
the surviving-body columns $V(G) \setminus V(H)$. That surviving block is at
the contraction realization's own rank $D(|s_c|-1) - k$: deleting the one
column of the contracted body $r$ loses no rank (the pin-a-body Lemma~5.1,
\cref{lem:rank-delete-vertex}; KT eqs.~(6.5) and (6.9)), so its rank on the
$V \setminus V'$ columns equals the contraction's rank on all of $s_c$.

The contraction leg is rigid on $s_c$ and not on any larger set --- in
particular not on the whole parent body set $V(G)$. Contracting $H$
identifies the whole of $V(H)$ to the single body $r$, so the interior bodies
$V(H) \setminus \{r\}$ are not bodies of $G/E(H)$ at all; its only edges are
the surviving edges $E(G) \setminus E(H)$, each with its $V(H)$ endpoints sent
to $r$, so they run among the surviving bodies $V(G) \setminus V(H)$ and $r$
and touch no interior body. Nothing in the contraction leg constrains
$V(H) \setminus \{r\}$: those bodies are held rigid only by the subgraph's own
edges $E(H)$, which the contraction leg does not include, so an infinitesimal
motion of that leg may move them freely, and its rigidity holds on $s_c$
alone. This is why the splice glue (\cref{lem:case-I-splice-seed}) takes a
body set for each leg, $s_H = V(H)$ for the rigid block and $s_c$ for the
contraction, in place of asserting rigidity of both on $V(G)$. Among all the
pieces the induction assembles the contraction is the first that is rigid on
a proper subset of the ambient body set rather than on the whole of it.

The body-set split is what closes the rank count. After projecting away the
$V'$ columns the rigid block's rows vanish (KT's top-right zero block), so the
two row families are disjoint and their ranks add,
\[
  D(|V'|-1) + \bigl(D(|s_c|-1) - k\bigr) = D(|V(G)|-1) - k,
\]
the target rank for $G$ (KT eq.~(6.3), closing line of Lemma~6.3). The
arithmetic balances because the two body sets overlap in exactly the one body
$r$: with $s_H \cap s_c = \{r\}$ one has $|V'| + |s_c| = |V(G)| + 1$, so the
two ``$-1$'' corrections absorb that single shared body,
$(|V'|-1) + (|s_c|-1) = |V(G)|-1$. Counting $r$ once from each side is what
lets a rigid piece on $V(H)$ and a rigid piece on $s_c$ glue to a rigid parent
on all of $V(G)$.

The splice itself is genericity-free: the genericity device
(\cref{lem:genericity-device}) enters only at the realization lemma
\cref{lem:case-I-realization}, where the two legs must be realized at a single
generic placement --- a point in the intersection of the two legs'
Zariski-open rigid loci. The hypotheses of \cref{lem:case-I-splice-seed} are
the satisfiable inductive facts, relative rigidity of each piece on a common
placement $F$, not the parent rank they conclude.

\begin{lemma}[Graph-level minimality of the rigid-subgraph contraction]
  \label{lem:rigidContract-isMinimalKDof}
  \lean{Graph.rigidContract_isMinimalKDof}
  \leanok
  \uses{lem:contraction-minimality, def:rigid-contraction, lem:reduction-step}
  Contracting a proper rigid subgraph $H$ of a minimal $k$-dof-graph $G$
  again yields a minimal $k$-dof-graph: $(G/E(H))$ is minimal $k$-dof,
  with strictly fewer vertices (\cref{lem:reduction-step}).
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:contraction-minimality, thm:def-eq-corank, lem:two-edge-conn,
        def:matroid-MG, lem:rigid-full-rank, thm:tutte-nash-williams}
  The reduction theorem (\cref{thm:minimal-kdof-reduction}) supplies only
  the existence of a rigid subgraph together with the induction hypothesis
  for the reduction, leaving the contraction's own minimality to be
  established here. On the matroid side it is already settled --- the
  matroid contraction $M(\tilde G) / E(\tilde H)$ is a minimal $k$-dof
  matroid at the reduced ambient $D(|V(G)| - |V(H)|)$
  (\cref{lem:contraction-minimality}). What remains is the passage to the
  \emph{graph}: that the count matroid of the graph contraction is that
  matroid contraction,
  \[
    M\bigl(\widetilde{G / E(H)}\bigr) \;=\; M(\tilde G) / E(\tilde H).
  \]
  Granting this identity, both halves of minimality transport. Its bases
  are the bases of $M(\tilde G) / E(\tilde H)$, and every edge of $G/E(H)$
  is a surviving parent edge $e \in E(G) \setminus E(H)$, so the
  base-meets-every-surviving-fiber condition carries over verbatim; the
  deficiency transports through the def-equals-corank bridge
  (\cref{thm:def-eq-corank}) on $G/E(H)$, whose collapsed vertex-count
  $|V(G/E(H))| = |V(G)| - |V(H)| + 1$ makes its ambient $D(|V(G)| -
  |V(H)|)$, so its corank equals that of $M(\tilde G) / E(\tilde H)$ --- a
  corank \cref{lem:contraction-minimality} has already conserved at
  $\operatorname{def}(\tilde G) = k$.

  The identity reads like a bookkeeping rename --- contract the fibers,
  relabel the collapsed vertices --- but it is not, and it holds only
  because $H$ is rigid. Here $M(\tilde G)$ is the $(D,D)$-count matroid,
  equivalently the $D$-fold union of the cycle matroid of $\tilde G$
  (\cref{def:matroid-MG}; Tutte--Nash-Williams,
  \cref{thm:tutte-nash-williams}), and contraction does not distribute over
  a matroid union: for matroids $M_1, \dots, M_D$ and a set $C$, the
  contraction of the union $\bigl(\bigcup_i M_i\bigr) / C$ and the union of
  the contractions $\bigcup_i (M_i / C)$ share a ground set but not, in
  general, their independent sets --- a base of $C$ in the union need not
  restrict to a base of $C$ in each factor. The graph contraction $G/E(H)$
  produces exactly the union-of-contracted-factors on the surviving fibers
  $E(\tilde G) \setminus E(\tilde H)$, whereas $M(\tilde G) / E(\tilde H)$
  is the contraction-of-the-union; the identity asserts these coincide,
  which for a general subgraph they do not.

  They coincide here because the contracted-out fibers $E(\tilde H)$
  \emph{rank-saturate} the union. Since $H$ is rigid --- $0$-dof, so
  $\operatorname{def}(\tilde H) = 0$ --- its count matroid attains the top
  rank $\operatorname{rank} M(\tilde H) = D(|V(H)| - 1)$
  (\cref{lem:rigid-full-rank}), while the single cycle matroid of the
  connected $\tilde H$ (\cref{lem:two-edge-conn}) ranks $E(\tilde H)$ at
  $|V(H)| - 1$. So the $D$-fold union ranks $E(\tilde H)$ at exactly $D$
  times the single-factor rank, and a base of $E(\tilde H)$ splits into one
  cycle-matroid base per factor: the fibers pack into $D$ edge-disjoint
  spanning trees on $V(H)$. This packing is the pivot of
  Katoh--Tanigawa's own proof of Lemma~3.5 (their equation~(3.1),
  \cite[p.~658]{katohTanigawa2011}): a base of $M(\tilde G)$ containing a
  base of the fibers restricts, within $E(\tilde H)$, to a spanning tree on
  $V(H)$ in each of the $D$ forests packing it. Under this saturation ---
  and only under it --- the per-factor bases of $C = E(\tilde H)$ line up,
  so the two matroids have the same independent sets. For a non-rigid $H$
  the fibers fall short of the packing, and collapsing $V(H)$ to a single
  body over-identifies --- it presumes $V(H)$ fully rigidified, which the
  fibers no longer certify --- so the identity fails.

  What forces the coincidence is a rank count, not a matching of the two
  matroids' decompositions: an arbitrary splitting into $D$ factors of a
  set $I \cup J$ --- surviving fibers $I$ together with a fiber-basis $J$
  of $E(\tilde H)$ --- need not keep $J$'s part factor-aligned, so a naive
  factor-by-factor matching does not go through. Instead both sides are
  read through the $(D,D)$-count condition ($E'$ is independent iff $|Y|
  \le D \cdot \operatorname{rank}_{\mathrm{cyc}}(Y)$ for every $Y \subseteq
  E'$), and equality falls out of the submodularity and monotonicity of the
  cycle rank once the saturation $|J| = D(|V(H)| - 1)$ is in hand.
\end{proof}

\begin{lemma}[KT Claim 6.4: the surviving block attains its rank at a generic placement (eq.\ (6.9))]
  \label{lem:claim-6-4}
  \lean{CombinatorialRigidity.Molecular.PanelHingeFramework.rigidContract_exterior_rank_transport_htransport}
  \leanok
  \uses{def:rigidity-matrix, def:rigid-contraction, lem:rank-delete-vertex,
        lem:contraction-minimality}
  Let $G$ be a simple minimal $0$-dof-graph with a proper rigid subgraph
  $H$ on $s = V(H)$, and write $s_c = (V(G) \setminus V(H)) \cup \{r\}$
  for the surviving body set with representative $r$. Given the
  contraction $G/E(H)$ realized at its inductive rank, the surviving-edge
  block of the rigidity matrix --- restricted to the edges $E(G)
  \setminus E(H)$ and to the columns outside $V(H)$ --- attains rank
  $D(|s_c|-1)$ at a generic placement: at a Zariski-open locus of panel
  coordinates, the surviving rows are linearly independent after
  projecting away the columns of $V(H)$ (KT's bottom-right block,
  eq.~(6.9); the row-count companion of the pin-a-body Lemma~5.1,
  \cref{lem:rank-delete-vertex}).
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:rank-delete-vertex, lem:contraction-minimality}
  The argument transports the contraction's rank across the collapse map
  $f$ sending every body of $V(H)$ to $r$ and fixing the rest, from a
  generic realization of the abstract relabelled graph $G/E(H)$ to the
  surviving-edge block of the parent.

  First, the contraction's generic realization is transported to the
  endpoint selector recording the links of $G/E(H)$ through $f$: since
  both selectors record the same links, they agree up to swap and the two
  realizations have the same rigidity and motion space. This gives a
  free-normal panel framework on $G/E(H)$ in general position, rigid on
  its whole vertex set $s_c$.

  Second --- the genuine content of the claim --- the exterior-column
  projection that drops exactly the $V(H)$-columns loses no rank on this
  framework's rigidity-row span. Because the framework is rigid on $s_c$
  and $s_c \cap V(H) = \{r\}$, the trivial motions and the projection's
  kernel together span everything, so the projected rows stay
  independent, of size at least $D(|s_c|-1)$: KT's bottom-right block,
  the pin-a-body content (\cref{lem:rank-delete-vertex}) at the row
  level.

  Third, the witness placement is KT's degenerate seed (eq.~(6.7)): the
  collapsed normal field $n \circ f$ on $G \setminus E(H)$. A per-edge
  identity --- both framings read the same support extensor, the
  projection reconciling their differing endpoints --- carries the
  projected independence of the relabelled contraction's rows back to
  the projected independence of the parent's surviving-edge rows at this
  seed.

  This discharges what KT \S5.1/Claim~6.4 states under the simplifying
  assumption that the panel coordinates are algebraically independent
  over $\Q$, realizing that genericity as the contraction's inductive
  general-position realization, the transport as the endpoint-selector
  agreement, and the rank preservation as the zero-rank-loss of the
  exterior-column projection on a rigid block.
\end{proof}

The three realization lemmas that follow discharge Case~I in its three
sub-cases \cite[pp.~673--676]{katohTanigawa2011}, and that three-way split
is forced, not cosmetic. Its origin is the contraction $G/E(H)$ itself:
the one operation Case~I recurses through that need not preserve
simplicity. Every other operation the argument performs on a graph ---
deleting the edges $E(H)$, deleting a vertex, restricting to a subset of
edges, passing to an induced subgraph --- is a \emph{subgraph} operation:
it only removes edges and vertices, so it carries a simple graph to a
simple graph automatically. Contraction is different in kind. After
deleting $E(H)$ it \emph{identifies} the whole body set $V(H)$ to a single
representative $r$ --- a relabelling of vertices, not a passage to a
subgraph --- and identifying vertices is exactly what can manufacture the
two configurations simplicity forbids. A \emph{loop} appears when a
surviving edge $e \in E(G) \setminus E(H)$ has both of its ends in $V(H)$:
both are sent to $r$, and $e$ becomes a self-loop at $r$. A \emph{parallel
pair} appears when two distinct surviving edges have their end-pairs
identified to a single collapsed pair --- for instance two edges from a
common body $w \notin V(H)$ into $V(H)$, which after the collapse both
join $w$ to $r$. Simplicity of $G$ constrains $G$'s own edges but says
nothing about how the collapse identifies their endpoints, so neither mode
can be excluded in advance; ``$G/E(H)$ is simple'' is a genuine hypothesis,
not a consequence of $G$ being simple.

This is why Case~I trifurcates rather than recursing uniformly on
contraction. If $G$ is not simple it already carries a parallel pair, and
\cref{lem:case-I-realization-nonsimple} (KT Lemma~6.2) contracts that pair
directly. If $G$ is simple, the argument turns on whether some proper
rigid subgraph $H$ has a \emph{simple} contraction: when one does, $G/E(H)$
is a smaller simple minimal $k$-dof-graph to recurse on and
\cref{lem:case-I-realization} (KT Lemma~6.3) realizes $G$ from its
inductive realization; when \emph{every} proper rigid subgraph has a
non-simple contraction --- a mode the previous paragraph shows cannot be
ruled out --- no simplicity-preserving contraction recursion is available.
There \cref{lem:case-I-realization-h65} (KT Lemma~6.5, through Claim~6.6)
instead removes a degree-two vertex; vertex removal is a subgraph
operation, so it returns a smaller \emph{simple} minimal $0$-dof-graph and
restores the inductive hypothesis. The failure of contraction-simplicity
is thus precisely the signal that routes the argument out of the
contraction case (Lemma~6.3) and into the vertex-removal case
(Lemma~6.5).

\begin{lemma}[Case I realization: simple-contraction case, all deficiencies; KT Lemma~6.3]
  \label{lem:case-I-realization}
  \lean{CombinatorialRigidity.Molecular.PanelHingeFramework.case_I_realization_all_k_gen}
  \leanok
  \uses{def:rank-hypothesis, lem:case-I, lem:rigidContract-isMinimalKDof,
        lem:rank-polynomial-IH-relabel-proj, lem:genericity-device,
        lem:rows-polynomial-in-normals, lem:isInfRigidOn-of-relative-count}
  Let $G$ be a simple minimal $k$-dof-graph with $|V(G)| \ge 3$, a proper
  rigid subgraph $H$ on $s = V(H)$ with representative body $r$, such
  that the contraction $G/E(H)$ is itself simple (KT Lemma~6.3's case
  hypothesis; the non-simple-contraction case is
  \cref{lem:case-I-realization-h65}). If the induction hypothesis
  supplies generic realizations of both $H$ (at deficiency $0$, since $H$
  is rigid) and the contraction $G/E(H)$ (at deficiency $k$, minimal
  $k$-dof by \cref{lem:rigidContract-isMinimalKDof}), then $G$ carries a
  generic realization at rank $D(|V(G)|-1) - k$ (\cref{def:rank-hypothesis}),
  discharging the Case~I contraction premise of \cref{thm:theorem-55} in
  this case.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:rows-polynomial-in-normals, lem:rank-polynomial-IH-relabel-proj,
        lem:genericity-device, lem:case-I, lem:rigidContract-isMinimalKDof,
        lem:isInfRigidOn-of-relative-count}
  Work on a single framework $F$ at a general-position seed $q_0$, KT's
  block-triangular rank addition (eq.~(6.3)). The rigid-block rows
  ($E(H)$) are independent and number $D(|V(H)|-1)$, from the $H$-leg's
  inductive generic realization at deficiency $0$, fed through the row
  coordinatization of \cref{lem:rows-polynomial-in-normals}. The
  surviving-edge rows ($E(G) \setminus E(H)$, on $s_c = (V(G) \setminus
  V(H)) \cup \{r\}$) number $D(|s_c|-1) - k$ and, after the
  exterior-column projection killing the $V(H)$ columns, are independent
  (\cref{lem:rank-polynomial-IH-relabel-proj}, the deficiency-aware
  surviving-block rank polynomial applied to the contraction's inductive
  realization). The rigid-block rows vanish under this projection (KT's
  top-right zero block), so the two row families are disjoint and their
  union gives $D(|V(H)|-1) + D(|s_c|-1) - k = D(|V(G)|-1) - k$
  independent rows (using $|s_H \cap s_c| = 1$). The genericity device
  (\cref{lem:genericity-device}) then reads rigidity on $V(G)$ off this
  row count via the relative-count adapter
  (\cref{lem:isInfRigidOn-of-relative-count}), at $q_0$ itself; since
  $q_0$ is general position, the realization is generic.
\end{proof}

\medskip\noindent\emph{Genericity of the rank versus general position of the
configuration.} The realization above lands at \emph{one} seed $q_0$ on which
both legs attain their ranks and, at the same time, the panel normals are in
general position; the closing step reads rigidity off that count and observes
that, because $q_0$ is general position, the realization is generic. That a
single seed carries both facts at once is where the formalization makes explicit
a separation Katoh--Tanigawa fold into one hypothesis. In \S5.1
\cite[p.~668]{katohTanigawa2011} a nonparallel realization is called
\emph{generic} when the coordinates of its normal field $c \colon V(G) \to \R^d$
are algebraically independent over $\Q$, and that single word bundles two
conditions of different character. The first is a \emph{non-degeneracy} of the
configuration --- KT's \emph{nonparallel} condition, that $\Pi(u) \cap \Pi(v)$
is a $(d-2)$-flat for every distinct pair, equivalently that $c(u)$ and $c(v)$
are linearly independent; this is exactly what the present development calls
general position (\cref{def:panel-general-position}). The second is
\emph{rank-maximality} --- that $R(G,p)$ attains the largest rank of any
nonparallel realization of $G$. Claim~6.4 \cite[p.~675]{katohTanigawa2011} reads
\emph{both} off the one assumption: its proof writes each rigidity-matrix entry
as a polynomial in the panel coefficients and then, ``since $p_1$ and $p_2$ have
been taken so that the coefficients are algebraically independent over $\Q$,''
concludes the surviving block ``takes the maximum rank,'' with no separate
non-degeneracy check and no explicit intersection of loci. The fusion is
deliberate: footnote~4 (\emph{ibid.}) records that the algebraic-independence
definition cuts out a strictly smaller class of frameworks than the conventional
rank-maximality notion of generic, and that KT adopt it only ``to make our proof
simpler.''

Working with concrete non-roots of concrete polynomials rather than with a
posited algebraically-independent point, the formalization cannot read both
conditions off one hypothesis; it carries them as two separate polynomial
factors. Rank-maximality is the genericity device
(\cref{lem:genericity-device}): a maximal non-vanishing minor of $R(G,q_0)$ is
one polynomial in the normal coordinates, and its non-vanishing locus is where
the rank stays at least that large. Non-degeneracy is a \emph{different}
polynomial condition, carried by its own \emph{general-position polynomial}: the
product, over the pairs $(a,b)$ of distinct bodies, of the leading $2 \times 2$
minor $q(a,0)\,q(b,1) - q(a,1)\,q(b,0)$ of the two normals. An assignment is a
non-root of the product exactly when every pair's leading minor is nonzero,
which forces each pair of normals to be linearly independent, i.e.\ general
position; and the product is not the zero polynomial, because the moment-curve
assignment of \cref{lem:moment-curve-general-position} makes each factor a
nonzero Vandermonde difference $\mathrm{param}(b) - \mathrm{param}(a)$.

The two factors are coupled by a shared non-root of their product. Each leg's
inductive rigidity yields a rank polynomial nonzero at that leg's own seed,
hence a nonzero polynomial; the product of the two rank polynomials with the
general-position polynomial is again a nonzero polynomial, and over the infinite
field $K$ a nonzero polynomial has a non-root $q_0$ --- equivalently, the
finitely many Zariski-open loci, each the complement of a proper algebraic
subset, meet in a dense set. At $q_0$ every factor is nonzero at once, so both
legs attain their ranks \emph{and} the normals are in general position: the two
conditions KT's single hypothesis delivers together, here recovered as the
common non-root of an explicit product. In this sense the one ``algebraically
independent over $\Q$'' assumption of \S5.1 is, on the formalization's side, two
loci --- one of rank-maximality, one of general position --- whose shared points
the product exhibits.

\begin{lemma}[Case I realization: non-simple $G$; KT Lemma~6.2]

\end{lemma}

  The parallel pair $\{e,f\}$ spans a proper rigid subgraph

  Contract to $G' = G/H'$ and apply the induction hypothesis to obtain a
  panel-hinge framework $F_c$ on $G'$; install coincident panels at $a$

  choose two linearly independent extensors $C_e, C_f$ in that shared

\begin{lemma}[Case I realization: removing a degree-two vertex; KT Lemma~6.5, Claim~6.6]
  \label{lem:case-I-realization-h65}
  \lean{CombinatorialRigidity.Molecular.PanelHingeFramework.case_I_realization_h65_gen}
  \leanok
  \uses{def:rank-hypothesis, lem:genericity-device}
  Let $G$ be a simple minimal $0$-dof-graph with $|V(G)| \ge 3$ in which
  every proper rigid subgraph has a non-simple contraction. Given the
  induction hypothesis for every smaller minimal $0$-dof-graph, $G$
  carries a generic full-rank realization (\cref{def:rank-hypothesis}).
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:genericity-device}
  By KT Claim~6.6, a vertex-inclusionwise maximal proper rigid subgraph
  $G'$ of $G$ has a non-simple contraction only because some vertex
  $v \notin V(G')$ has degree two, with two neighbours $a, b \in V(G')$
  and $G = G' + v$; then $G - v$ is a smaller simple minimal $0$-dof-graph,
  which the induction hypothesis realizes at full rank. Take a seed $q$
  chosen as a common non-root of three nonzero polynomials in the seed
  coordinates, all fixed before the seed: the rank polynomial of $G - v$,
  a general-position polynomial, and the $3 \times 3$ determinant of the
  normal rows at $v, a, b$; over the infinite field $K$ their product has
  a non-root. Re-attach $v$ at that seed: keep every endpoint selection
  from $G-v$'s realization, give $v$ its own normal $q(v)$, and direct the
  two new edges $va, vb$ from $v$. Off $\{va, vb\}$ the framework agrees
  with the vertex-removal framework at the same seed, which attains full
  rank there (the rank-polynomial factor is nonvanishing), so
  $D(|V(G-v)|-1)$ independent rigidity rows transport
  unchanged, vanishing on $v$'s screw column. The two new rows at $va,
  vb$ span the full $D$-dimensional block through $v$'s column: their
  supporting extensors $C(q(v), q(a))$ and $C(q(v), q(b))$ are linearly
  independent because the triple $(q(v), q(a), q(b))$ is --- the
  determinant factor is nonvanishing at the chosen seed, with no
  perturbation argument needed. The two row blocks are disjoint (one
  vanishes on, the other is pinned through, $v$'s column) and together
  give $D + D(|V(G-v)|-1) = D(|V(G)|-1)$ independent rows, so $G$ is
  infinitesimally rigid on $V(G)$ at this seed; the genericity device
  (\cref{lem:genericity-device}) upgrades this fixed-seed rigidity to a
  generic realization.
\end{proof}

\begin{lemma}[Case I dispatch: KT Lemmas 6.2, 6.3, 6.5]
  \label{lem:case-I-dispatch}
  \lean{CombinatorialRigidity.Molecular.case_I_dispatch_gen}
  \leanok
  \uses{def:genuine-hinge-realization, def:rank-hypothesis,
        lem:case-I-realization-nonsimple, lem:case-I-realization,
        lem:case-I-realization-h65}
  Let $G$ be a minimal $0$-dof-graph with $|V(G)| \ge 3$ carrying a
  proper rigid subgraph. Then $G$ has a nondegenerate-hinge panel
  realization at the target rank (\cref{def:genuine-hinge-realization}),
  and if $G$ is simple, a generic one (\cref{def:rank-hypothesis}).
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:case-I-realization-nonsimple, lem:case-I-realization,
        lem:case-I-realization-h65, lem:generic-yields-genuine-hinge}
  If $G$ is not simple, \cref{lem:case-I-realization-nonsimple} (KT
  Lemma~6.2) applies directly. If $G$ is simple, case on whether some
  proper rigid subgraph $H$ with representative body $r$ has a simple
  contraction $G/E(H)$: if so, \cref{lem:case-I-realization} (KT
  Lemma~6.3) gives the generic realization; otherwise every contraction is
  non-simple and \cref{lem:case-I-realization-h65} (KT Lemma~6.5) gives
  it instead. Either way \cref{lem:generic-yields-genuine-hinge} then
  gives the nondegenerate-hinge form.
\end{proof}
