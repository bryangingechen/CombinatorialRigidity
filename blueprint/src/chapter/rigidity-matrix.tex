\section{The panel-hinge rigidity matrix \texorpdfstring{$R(G,p)$}{R(G,p)}}
\label{sec:molecular-rigidity-matrix}

This chapter defines the rigidity matrix $R(G,p)$ of a body-hinge
framework $(G,p)$, following Katoh and
Tanigawa~\cite[\S\S2.2--2.4]{katohTanigawa2011}, and proves the lemmas
on its rank that their proof of the molecular conjecture uses. In
\cref{sec:body-hinge} a hinge was replaced by $D - 1$ bars, where
$D = \binom{d+1}{2}$, and the rigidity of a body-hinge framework on $G$
was that of the resulting body-bar framework on
$\tilde G = (D-1)\cdot G$. Here each edge $e$ has a hinge $p(e)$,
an affine subspace of dimension $d - 2$, and the hinge constrains the
screw centers of the two bodies of $e$ through its extensor
(\cref{sec:molecular-extensor}). As in \cref{sec:molecular}, we work
over an arbitrary field $K$.

\Cref{sec:molecular-rigidity-matrix-def} defines $R(G,p)$ and the space
$Z(G,p)$ of infinitesimal motions, and shows that
$\operatorname{rank} R(G,p) \le D(|V|-1)$. The next chapter defines the
deficiency $\operatorname{def}(\tilde G)$ and sharpens the bound to
$\operatorname{rank} R(G,p) \le D(|V|-1) - \operatorname{def}(\tilde G)$
at every realization (\cref{sec:molecular-deficiency-rank-bound}).
Equality holds at a generic realization. This rank formula is Jackson and
Jord\'an's~\cite[Theorem~6.1]{jacksonJordan2009}, which Katoh and
Tanigawa state as their Proposition~2.3, and we prove it after the
algebraic induction (\cref{prop:rigidity-matrix-prop11}). By the theorem
of Tutte and Nash-Williams~\cite{tutte1961,nashWilliams1961},
$\operatorname{def}(\tilde G) = 0$ exactly when $\tilde G$ contains $D$
edge-disjoint spanning trees~\cite[p.~655]{katohTanigawa2011}. So the
formula implies Tay and Whiteley's theorem in the form that KT
Proposition~1.1 states (\cref{sec:body-hinge-tay}).

\Cref{sec:molecular-rigidity-matrix-rank} proves KT Lemmas~5.1--5.3 on
the rank of $R(G,p)$. \Cref{sec:molecular-rigidity-matrix-blocks}
proves the rank-addition lemmas that the algebraic induction
(\cref{sec:molecular-algebraic-induction}) and the induction of the
pencil chapters (\cref{sec:main-component}) apply when they split a
graph into smaller ones.

\subsection{Body-hinge frameworks and the rigidity matrix}
\label{sec:molecular-rigidity-matrix-def}

This subsection defines the rigidity matrix of a body-hinge framework,
following Katoh and Tanigawa~\cite[\S\S2.2--2.4]{katohTanigawa2011}.
They identify an infinitesimal motion of a rigid body in $\R^d$ with a
vector of $\R^D$, its screw center, where $D = \binom{d+1}{2}$. A hinge
joining two bodies constrains the difference of their screw centers to
the span of a $(d-1)$-extensor supporting the hinge. We take this
constraint as the definition, over a field $K$. Screw centers lie in
the following space, as do the supporting extensors of
\cref{def:affine-subspace-extensor}.

\begin{definition}[Screw space]
  \label{def:screw-space}
  \lean{CombinatorialRigidity.Molecular.ScrewSpace,
    CombinatorialRigidity.Molecular.screwDim,
    CombinatorialRigidity.Molecular.screwSpace_finrank}
  \leanok
  The \emph{screw space} is $\bigwedge^{d-1}K^{d+1}$. It has dimension
  $\binom{d+1}{d-1} = D$, where $D = \binom{d+1}{2}$.
\end{definition}

\begin{fmlnote}
  The dimension is written $d = k + 1$, so that the screw space is
  $\bigwedge^k K^{k+2}$. The pins are the screw space, the number $D$
  and the dimension of the screw space.
\end{fmlnote}

\begin{definition}[Hinge constraint via the supporting extensor]
  \label{def:hinge-constraint}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework,
    CombinatorialRigidity.Molecular.BodyHingeFramework.hingeConstraint}
  \leanok
  \uses{def:affine-subspace-extensor, def:screw-space}
  A \emph{body-hinge framework} $(G,p)$ in $K^d$ is a multigraph
  $G = (V,E)$ together with a map $p$ that assigns to each edge $e$ a
  $(d-2)$-dimensional affine subspace $p(e)$ of $K^d$, the \emph{hinge}
  of $e$. Let $C(p(e)) \in \bigwedge^{d-1}K^{d+1}$ be a
  $(d-1)$-extensor supporting $p(e)$
  (\cref{def:affine-subspace-extensor}). A \emph{screw assignment} is a
  map $S$ from $V$ to the screw space $\bigwedge^{d-1}K^{d+1}$. It
  satisfies the \emph{hinge constraint} at an edge $e$ with ends $u$
  and $v$ if
  \[
    S(u) - S(v) \in \operatorname{span} C\bigl(p(e)\bigr).
  \]
\end{definition}

The hinge $p(e)$ is spanned by $d - 1$ affinely independent points, so
$C(p(e)) \neq 0$, and $p(e)$ determines $C(p(e))$ up to a nonzero
scalar (the remark after \cref{def:affine-subspace-extensor}). So the
hinge constraint does not depend on the choice of $C(p(e))$. It is
symmetric in $u$ and $v$, since $S(v) - S(u) = -(S(u) - S(v))$.

\begin{fmlnote}
  The formalization records a body-hinge framework by its multigraph
  and, for each edge $e$, an element $C(e)$ of the screw space in place
  of $C(p(e))$. It does not require $C(e)$ to be nonzero or to come from
  points, so the same definitions apply to hinges given as the meet of
  two panels (\cref{def:panel-hinge-framework}). A lemma that needs
  $C(e) \neq 0$ assumes it. Screw assignments are defined on a fixed set
  of bodies, which may contain bodies that lie on no edge of $G$
  (\cref{sec:molecular-rigidity-matrix-blocks}).
\end{fmlnote}

Katoh and Tanigawa write the hinge constraint as linear
equations~\cite[\S2.4]{katohTanigawa2011}. For this we identify
$\bigwedge^{d-1}K^{d+1}$ with $K^D$ by the basis of the products
$e_{i_1} \wedge \cdots \wedge e_{i_{d-1}}$ with
$i_1 < \cdots < i_{d-1}$ (the remark after \cref{def:plucker-coords}),
and write $x \cdot y$ for the dot product of $K^D$.

\begin{definition}[Hinge-row block]
  \label{def:hinge-row-block}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.hingeRowBlock,
    CombinatorialRigidity.Molecular.BodyHingeFramework.hingeConstraint_iff_hingeRowBlock,
    CombinatorialRigidity.Molecular.BodyHingeFramework.finrank_hingeRowBlock,
    CombinatorialRigidity.Molecular.BodyHingeFramework.mem_hingeRowBlock_iff}
  \leanok
  \uses{def:hinge-constraint}
  Let $(G,p)$ be a body-hinge framework and let $e$ be an edge of $G$.
  Let $r_1(p(e)), \dots, r_{D-1}(p(e))$ be a basis of the orthogonal
  complement of $\operatorname{span} C(p(e))$ in $K^D$, and let
  $r(p(e))$ be the $(D-1) \times D$ matrix with these rows. Then a screw
  assignment $S$ satisfies the hinge constraint at $e$, with ends $u$
  and $v$, if and only if
  \[
    r_i\bigl(p(e)\bigr) \cdot \bigl(S(u) - S(v)\bigr) = 0
    \qquad (1 \le i \le D-1).
  \]
\end{definition}

The orthogonal complement has dimension $D - 1$ because
$C(p(e)) \neq 0$. The equivalence holds because a subspace of $K^D$ is
the orthogonal complement of its orthogonal complement, as the dot
product is nondegenerate. Through the dot product, the rows of
$r(p(e))$ form a basis of the linear forms on $K^D$ that vanish on
$\operatorname{span} C(p(e))$. A linear form vanishes on
$\operatorname{span} C(p(e))$ if and only if it vanishes at $C(p(e))$,
as the span consists of the multiples of $C(p(e))$; we then say that it
vanishes on $C(p(e))$.

\begin{fmlnote}
  The first pin is the space of linear forms on the screw space that
  vanish on $\operatorname{span} C(p(e))$, with no basis chosen, and the
  second is the equivalence. The third gives this space the dimension
  $D - 1$ when $C(p(e)) \neq 0$, and the fourth says that a form lies in
  it if and only if it vanishes at $C(p(e))$.
\end{fmlnote}

The rigidity matrix collects these equations for all the edges.

\begin{definition}[Panel-hinge rigidity matrix]
  \label{def:rigidity-matrix}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.IsInfinitesimalMotion,
    CombinatorialRigidity.Molecular.BodyHingeFramework.infinitesimalMotions,
    CombinatorialRigidity.Molecular.BodyHingeFramework.hingeRow,
    CombinatorialRigidity.Molecular.BodyHingeFramework.rigidityRows,
    CombinatorialRigidity.Molecular.BodyHingeFramework.infinitesimalMotions_eq_dualCoannihilator}
  \leanok
  \uses{def:hinge-row-block}
  Let $(G,p)$ be a body-hinge framework. Its \emph{rigidity matrix}
  $R(G,p)$ is the $(D-1)|E| \times D|V|$ matrix with a block of $D - 1$
  rows for each edge and a block of $D$ columns for each body. Orient
  each edge $e$ from one end $u$ to the other end $v$. The rows of $e$
  have $r(p(e))$ in the columns of $u$, $-r(p(e))$ in the columns of
  $v$, and zeros elsewhere (\cref{def:hinge-row-block}). An
  \emph{infinitesimal motion} of $(G,p)$ is a screw assignment that
  satisfies the hinge constraint at every edge, that is, a vector of
  $\ker R(G,p) \subseteq K^{D|V|}$. We write $Z(G,p)$ for the space of
  infinitesimal motions.
\end{definition}

The space $Z(G,p)$ depends neither on the choice of the rows of the
matrices $r(p(e))$ nor on the orientation of the
edges~\cite[p.~654]{katohTanigawa2011}. So neither does the rank of
$R(G,p)$, which is
\[
  \operatorname{rank} R(G,p) \;=\; D|V| - \dim Z(G,p).
\]

\begin{fmlnote}
  The formalization fixes no basis of the screw space. A row of
  $R(G,p)$ is the linear form $S \mapsto r(S(u) - S(v))$ on screw
  assignments, for an edge $e$ with ends $u$ and $v$ and a linear form
  $r$ on the screw space that vanishes on $C(p(e))$. The rank of
  $R(G,p)$ is the dimension of the span of these forms, and $Z(G,p)$ is
  the space on which they all vanish. The first two pins are the
  infinitesimal motions and $Z(G,p)$, the next two the rows of
  $R(G,p)$, and the last the description of $Z(G,p)$ by the rows.
\end{fmlnote}

We record two facts about the rows of $R(G,p)$. First, the rows of an
edge with distinct ends are linearly independent.

\begin{lemma}[The rows of an edge are independent]
  \label{lem:edge-rows-independent}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.linearIndependent_hingeRow,
    CombinatorialRigidity.Molecular.BodyHingeFramework.exists_independent_rigidityRows_of_edge}
  \leanok
  \uses{def:rigidity-matrix}
  Let $(G,p)$ be a body-hinge framework, let $u \neq v$ be bodies, and
  let $r_1, \dots, r_n \in K^D$ be linearly independent. Then the linear
  forms $S \mapsto r_i \cdot (S(u) - S(v))$ on screw assignments are
  linearly independent. In particular, if $e$ is an edge with ends $u$
  and $v$ and $C(p(e)) \neq 0$, then the $D - 1$ rows of $e$ are
  linearly independent.
\end{lemma}
\begin{proof}
  \leanok
  The map $S \mapsto S(u) - S(v)$ takes the screw assignments onto
  $K^D$, as $u \neq v$. So a linear relation among the forms gives the
  same relation among the $r_i$, and its coefficients are zero. If
  $C(p(e)) \neq 0$, the rows of $e$ are the forms for $r_i = r_i(p(e))$,
  $1 \le i \le D - 1$, which are linearly independent
  (\cref{def:hinge-row-block}).
\end{proof}

\begin{fmlnote}
  The first pin is the claim for linear forms, and the second its form
  for rows.
\end{fmlnote}

Second, the rows of several edges are jointly independent when each of
the edges has an end of its own.

\begin{lemma}[Independent rows at edges with ends of their own]
  \label{lem:own-end-rows-independent}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.linearIndependent_hingeRow_starForest,
    CombinatorialRigidity.Molecular.BodyHingeFramework.exists_independent_rigidityRows_of_starForest,
    CombinatorialRigidity.Molecular.BodyHingeFramework.linearIndependent_hingeRow_star}
  \leanok
  \uses{def:rigidity-matrix}
  Let $(G,p)$ be a body-hinge framework. Let $u_1, \dots, u_m$ be
  distinct bodies, and let $w_1, \dots, w_m$ be bodies none of which is
  equal to any $u_j$. For each $j$, let
  $r_{j,1}, \dots, r_{j,n_j} \in K^D$ be linearly independent. Then the
  linear forms $S \mapsto r_{j,i} \cdot (S(u_j) - S(w_j))$ on screw
  assignments are linearly independent. In particular, if $e_j$ is an
  edge with ends $u_j$ and $w_j$ and $C(p(e_j)) \neq 0$ for each $j$,
  then the $(D-1)m$ rows of $e_1, \dots, e_m$ are linearly independent.
\end{lemma}
\begin{proof}
  \leanok
  Take a linear combination of the forms that vanishes, and evaluate it
  at the screw assignment with value $x$ at $u_j$ and $0$ elsewhere. A
  form of index $j' \neq j$ vanishes there, as $u_{j'} \neq u_j$ and
  $w_{j'} \neq u_j$. So the coefficients $c_i$ of the forms of index $j$
  give $\sum_i c_i\, r_{j,i} \cdot x = 0$ for every $x \in K^D$, and
  every $c_i$ is zero, as the $r_{j,i}$ are linearly independent. If
  $C(p(e_j)) \neq 0$, the rows of $e_j$ are the forms for
  $r_{j,i} = r_i(p(e_j))$, $1 \le i \le D - 1$, which are linearly
  independent (\cref{def:hinge-row-block}).
\end{proof}

For example, the edges of a star at a body $w$ satisfy the hypothesis
if their other ends are distinct bodies other than $w$.

\begin{fmlnote}
  The first pin is the claim for linear forms, the second its form for
  rows, and the third the case of a star.
\end{fmlnote}

The next definitions are Katoh and
Tanigawa's~\cite[\S\S2.3--2.4]{katohTanigawa2011}, who take that of a
generic framework from Jackson and
Jord\'an~\cite[\S6]{jacksonJordan2009}. A body-hinge framework on $G$
is also called a \emph{realization} of $G$.

\begin{definition}[Trivial motions; infinitesimal rigidity]
  \label{def:trivial-motions-rigid}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.IsTrivialMotion,
    CombinatorialRigidity.Molecular.BodyHingeFramework.trivialMotions,
    CombinatorialRigidity.Molecular.BodyHingeFramework.IsInfinitesimallyRigid}
  \leanok
  \uses{def:rigidity-matrix}
  Let $(G,p)$ be a body-hinge framework. A screw assignment $S$ is
  \emph{trivial} if $S(u) = S(v)$ for all bodies $u$ and $v$, and the
  trivial screw assignments form the space of \emph{trivial motions}.
  The framework $(G,p)$ is \emph{infinitesimally rigid} if all its
  infinitesimal motions are trivial.
\end{definition}

Katoh and Tanigawa call an infinitesimal motion trivial under the same
condition~\cite[\S2.3]{katohTanigawa2011}. By
\cref{lem:trivial-motions-rank-bound} below, every trivial screw
assignment is an infinitesimal motion.

\begin{definition}[Degree of freedom; generic realization]
  \label{def:dof-generic}
  \uses{def:rigidity-matrix}
  Let $(G,p)$ be a body-hinge framework. The \emph{degree of freedom}
  of $(G,p)$ is $D(|V|-1) - \operatorname{rank} R(G,p)$. The framework
  $(G,p)$ is \emph{generic} if $R(G,p)$ and each of its edge-induced
  submatrices (the rows of the edges of a subset of $E$) have the
  largest rank over all realizations of $G$.
\end{definition}

\begin{fmlnote}
  The formalization defines neither notion. Its statements give the
  rank of $R(G,p)$ directly, and the note after
  \cref{prop:rigidity-matrix-prop11} says how they replace genericity.
\end{fmlnote}

By the next lemma, the trivial motions form a $D$-dimensional subspace
of $Z(G,p)$ when $V$ is nonempty. So the degree of freedom is
$\dim Z(G,p) - D \ge 0$, and it is $0$ if and only if $(G,p)$ is
infinitesimally rigid.

\begin{lemma}[The $D$ trivial motions and the rank bound]
  \label{lem:trivial-motions-rank-bound}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.isInfinitesimalMotion_of_isTrivialMotion,
    CombinatorialRigidity.Molecular.BodyHingeFramework.finrank_trivialMotions,
    CombinatorialRigidity.Molecular.BodyHingeFramework.infinitesimalMotions_eq_trivialMotions_iff,
    CombinatorialRigidity.Molecular.BodyHingeFramework.rankHypothesis_zero_iff}
  \leanok
  \uses{def:rigidity-matrix, def:trivial-motions-rigid}
  Let $(G,p)$ be a body-hinge framework with $V$ nonempty, and for
  $1 \le i \le D$ let $S^*_i$ be the screw assignment whose value at
  every body is the $i$-th basis vector of $K^D$. Then
  $S^*_1, \dots, S^*_D$ are infinitesimal motions, and they form a
  basis of the space of trivial motions. Hence
  $\operatorname{rank} R(G,p) \le D(|V|-1)$, with equality if and only
  if $(G,p)$ is infinitesimally rigid.
\end{lemma}
\begin{proof}
  \leanok
  A constant screw assignment $S$ has $S(u) - S(v) = 0$ at every edge,
  so it is an infinitesimal motion. The trivial motions are the
  constant assignments (\cref{def:trivial-motions-rigid}). Hence they
  form a subspace of $Z(G,p)$
  (\texttt{trivialMotions\_\allowbreak{}le\_\allowbreak{}infinitesimalMotions}).
  As $V$ is nonempty, sending $x \in K^D$ to the constant assignment
  with value $x$ is an isomorphism from $K^D$ onto them
  (\texttt{trivialMotions\_\allowbreak{}eq\_\allowbreak{}range\_\allowbreak{}const}),
  and it sends the standard basis to $S^*_1, \dots, S^*_D$. So the
  trivial motions form a $D$-dimensional subspace of $Z(G,p)$. The
  screw assignments form a space of dimension $D|V|$
  (\texttt{finrank\_\allowbreak{}screwAssignment}), so
  $\operatorname{rank} R(G,p) = D|V| - \dim Z(G,p) \le D(|V|-1)$.
  Equality holds if and only if $Z(G,p)$ is the space of trivial
  motions, that is, if and only if $(G,p)$ is infinitesimally rigid.
\end{proof}

\begin{fmlnote}
  The first pin says that the trivial screw assignments are
  infinitesimal motions, the second that they form a space of dimension
  $D$, and the third that $Z(G,p)$ is this space exactly when $(G,p)$ is
  infinitesimally rigid. The fourth states the equality case for the
  dimension of $Z(G,p)$, which is $D|V| - \operatorname{rank} R(G,p)$:
  $\dim Z(G,p) = D$ exactly when $(G,p)$ is infinitesimally rigid.
\end{fmlnote}

\subsection{Rank lemmas}
\label{sec:molecular-rigidity-matrix-rank}

This subsection proves two rank lemmas of Katoh and
Tanigawa~\cite[\S5]{katohTanigawa2011}, and compares two frameworks on
one graph. Deleting the columns of one
body does not change the rank of $R(G,p)$ (KT Lemma~5.1). Two edges
between the same two bodies, with linearly independent supporting
extensors, force the two bodies to have the same screw center in every
infinitesimal motion (KT Lemma~5.3). The last lemma says that if every
hinge of one framework has a supporting extensor in the span of the
other's, then every infinitesimal motion of the first is one of the
second. Katoh and Tanigawa's remaining lemma of their \S5, KT
Lemma~5.2, rotates a panel without decreasing the rank; the
formalization takes its steps with the realizations of the algebraic
induction (the passage before
\cref{lem:generic-realization-of-rigid}).

\begin{lemma}[Deleting one body's columns preserves rank; KT Lemma 5.1]
  \label{lem:rank-delete-vertex}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.trivialMotions_sup_pinnedMotions,
    CombinatorialRigidity.Molecular.BodyHingeFramework.trivialMotions_inf_pinnedMotions_eq_bot,
    CombinatorialRigidity.Molecular.BodyHingeFramework.finrank_pinnedMotions_add_screwDim}
  \leanok
  \uses{def:rigidity-matrix, lem:trivial-motions-rank-bound}
  Let $(G,p)$ be a body-hinge framework, let $v$ be a body, and let
  $\operatorname{pinned}(v) = \{S \in Z(G,p) : S(v) = 0\}$. Then
  $Z(G,p)$ is the direct sum of the space of trivial motions and
  $\operatorname{pinned}(v)$, so
  $\dim \operatorname{pinned}(v) + D = \dim Z(G,p)$. Equivalently,
  deleting the $D$ columns of $v$ from $R(G,p)$ does not change its
  rank.
\end{lemma}
\begin{proof}
  \leanok
  A trivial motion that vanishes at $v$ vanishes everywhere. An
  infinitesimal motion $S$ is the sum of the trivial motion with value
  $S(v)$ and the screw assignment $u \mapsto S(u) - S(v)$, which
  vanishes at $v$. The latter is again an infinitesimal motion, since it
  has the same differences $S(u) - S(w)$ as $S$. So $Z(G,p)$ is the
  direct sum, and the trivial motions have dimension $D$
  (\cref{lem:trivial-motions-rank-bound}).

  Deleting the columns of $v$ leaves the matrix whose null space
  consists of the restrictions to $V \setminus \{v\}$ of the motions in
  $\operatorname{pinned}(v)$. Its rank is therefore
  $D(|V|-1) - \dim \operatorname{pinned}(v) = D|V| - \dim Z(G,p)$, the
  rank of $R(G,p)$.
\end{proof}

Katoh and Tanigawa prove the lemma by adding to $R(G,p)$ the $D$ rows
that tie down the body $v$~\cite[p.~668]{katohTanigawa2011}. They cite
White and Whiteley~\cite[\S2]{whiteWhiteley1987} for the fact that
fixing one body does not change the space of nontrivial motions. The
direct sum above expresses that fact.

\begin{lemma}[Two hinges of parallel edges give the full block; KT Lemma 5.3]
  \label{lem:rank-parallel-full}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.span_inf_span_eq_bot_of_linearIndependent,
    CombinatorialRigidity.Molecular.BodyHingeFramework.eq_of_hingeConstraint_two_parallel}
  \leanok
  \uses{def:rigidity-matrix}
  Let $(G,p)$ be a body-hinge framework, and let $e$ and $f$ be edges of
  $G$ that both join the bodies $u \neq v$, with $C(p(e))$ and
  $C(p(f))$ linearly independent. Then every screw assignment $S$ that
  satisfies the hinge constraints at $e$ and $f$ has $S(u) = S(v)$.
  Equivalently, the rows of $R(G,p)$ at $e$ and $f$, restricted to the
  $D$ columns of $v$, have rank $D$.
\end{lemma}
\begin{proof}
  \leanok
  The difference $S(u) - S(v)$ lies in
  $\operatorname{span} C(p(e)) \cap \operatorname{span} C(p(f))$
  (\cref{def:hinge-constraint}). Two lines spanned by linearly
  independent vectors meet only in $0$, so $S(u) = S(v)$. In the
  columns of $v$, the rows at $e$ and $f$ are $\pm r(p(e))$ and
  $\pm r(p(f))$. Their common null space in $K^D$ is the same
  intersection (\cref{def:hinge-row-block}), so they have rank $D$.
\end{proof}

Katoh and Tanigawa state the lemma for the graph with the two bodies
$u$ and $v$ and the two edges $e$ and $f$, with distinct hinges
$p(e) \neq p(f)$, and their proof shows that distinct hinges satisfy
the hypothesis above~\cite[p.~670]{katohTanigawa2011}. By the remark
after \cref{def:affine-subspace-extensor}, $C(p(e))$ is nonzero, it is
determined by $p(e)$ up to a nonzero scalar, and it determines $p(e)$.
So $C(p(e))$ and $C(p(f))$ are proportional only if $p(e) = p(f)$, and
two nonzero vectors that are not proportional are linearly
independent. For example, two hinges spanned by $d - 1$ of $d + 1$
affinely independent points, omitting different pairs of them, have
linearly independent extensors by KT Lemma~2.1
(\cref{lem:extensor-independence}). Distinct hinges can lie in a common
hyperplane, so the two-body graph has an infinitesimally rigid
panel-hinge realization. This graph is the base case of the algebraic
induction (\cref{lem:theorem-55-base}).

\begin{lemma}[Smaller hinge spans give fewer infinitesimal motions]
  \label{lem:rank-rotation-semicont}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.infinitesimalMotions_mono_of_span_le,
    CombinatorialRigidity.Molecular.BodyHingeFramework.finrank_infinitesimalMotions_le_of_span_le}
  \leanok
  \uses{def:rigidity-matrix}
  Let $(G,p)$ and $(G,p')$ be body-hinge frameworks on the same
  multigraph $G$, and suppose that
  $\operatorname{span} C(p'(e)) \subseteq \operatorname{span} C(p(e))$
  at every edge $e$. Then $Z(G,p') \subseteq Z(G,p)$. In particular,
  $\dim Z(G,p') \le \dim Z(G,p)$ and
  $\operatorname{rank} R(G,p) \le \operatorname{rank} R(G,p')$.
\end{lemma}
\begin{proof}
  \leanok
  Let $S \in Z(G,p')$, and let $e$ be an edge with ends $u$ and $v$.
  Then $S(u) - S(v)$ lies in $\operatorname{span} C(p'(e))$
  (\cref{def:hinge-constraint}), hence in
  $\operatorname{span} C(p(e))$, so $S$ satisfies the hinge constraint
  of $(G,p)$ at $e$. So $Z(G,p') \subseteq Z(G,p)$, and
  $\dim Z(G,p') \le \dim Z(G,p)$. The rank of $R(G,p)$ is
  $D|V| - \dim Z(G,p)$ (the remark after \cref{def:rigidity-matrix}),
  and likewise for $p'$, which gives the bound on the ranks.
\end{proof}
\begin{fmlnote}
  The first pin is the inclusion, and the second the bound on the
  dimensions, for a finite set of bodies; the bound on the ranks follows
  by the remark after \cref{def:rigidity-matrix}. Both pins assume the
  inclusion of spans at every edge label, also outside $E(G)$. The
  formalization lets $C(p(e))$ be zero (the note after
  \cref{def:hinge-constraint}), so the inclusion can be strict, at an
  edge with $C(p'(e)) = 0 \neq C(p(e))$. When all the extensors are
  nonzero, the hypothesis says that the spans are equal, and then
  $Z(G,p') = Z(G,p)$.
\end{fmlnote}

\subsection{Rank-addition lemmas}
\label{sec:molecular-rigidity-matrix-blocks}

This subsection computes or bounds the rank of $R(G,p)$ from the ranks
of pieces of $G$. In the first lemmas the pieces share at most two
bodies. If two sides are joined by at most one edge, the
rank of the whole is at least the sum of their ranks plus $D - 1$ for
the joining edge; this is the rank step in the proof of KT Lemma~6.1.
If two sides share one body, their ranks add. If they share two bodies
$u$ and $v$, the rank of the whole is the sum of the ranks of the sides
minus a correction, which depends on the motions of $v$ relative to $u$
that each side allows. We compute the correction for a path attached at
its two ends. The next lemmas bound the rank from below when a rigid
subgraph is contracted to one body (KT Lemma~6.2) and when a body of
degree two is restored after splitting off (KT Lemmas~6.8 and~6.10),
from the block-triangular shape of the rigidity matrix. The last one
deletes the columns of the bodies of a contracted subgraph, which Case~I
of the algebraic induction does when it compares the rows outside the
subgraph with a realization of the contraction (KT Claim~6.4,
\cref{sec:molecular-algebraic-induction-caseI}).

Throughout, $(G,p)$ is a body-hinge framework. A subgraph $H$ of $G$
keeps the hinges of $p$ at its edges and the bodies of $(G,p)$: its
screw assignments are those of $(G,p)$, maps
$B \to \bigwedge^{d-1}K^{d+1}$ on a finite set $B$ of bodies that
contains $V(G)$. We write $R(H,p)$ and $Z(H,p)$ for its rigidity matrix
and its space of infinitesimal motions, defined as before. A body of
$B \setminus V(H)$ lies on no edge of $H$, so no hinge constraint of $H$
involves it. Its columns in $R(H,p)$ are zero, and the rank of $R(H,p)$
does not depend on these bodies. As after
\cref{def:rigidity-matrix}, we regard the rows of $R(H,p)$ as linear
forms on screw assignments: the rows of an edge $e$ with ends $u$ and
$v$ are the forms $S \mapsto r_i(p(e)) \cdot (S(u) - S(v))$. The
\emph{row space} of $R(H,p)$ is the span of its rows, and its dimension
is the rank of $R(H,p)$. For a set $V_1$ of bodies, $G[V_1]$ is the
subgraph of $G$ induced on $V_1$.

\begin{lemma}[Rank addition at a cut edge; from the proof of KT Lemma~6.1]
  \label{lem:block-rank-cut}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.le_finrank_span_rigidityRows_of_cut,
    CombinatorialRigidity.Molecular.BodyHingeFramework.add_le_finrank_span_rigidityRows_induce_union_singleton}
  \leanok
  \uses{def:rigidity-matrix, def:hinge-row-block}
  Let $(G,p)$ be a body-hinge framework with $C(p(e)) \neq 0$ at every
  edge $e$. Let $V_1$ be a nonempty proper subset of $V(G)$, let
  $V_2 = V(G) \setminus V_1$, and let $C$ be a set of at most one edge
  of $G$, each with one end in $V_1$ and the other in $V_2$. Then
  \[
    \operatorname{rank} R(G[V_1],p) + (D-1)\,|C|
    + \operatorname{rank} R(G[V_2],p)
    \;\le\;
    \operatorname{rank} R(G,p).
  \]
  In particular, if an edge of $G$ joins $V_1$ to a body $w \in V_2$,
  then
  \[
    \operatorname{rank} R(G[V_1],p) + D - 1 \;\le\;
    \operatorname{rank} R(G[V_1 \cup \{w\}],p).
  \]
\end{lemma}
\begin{proof}
  \leanok
  \uses{def:rigidity-matrix, def:hinge-row-block, lem:edge-rows-independent}
  Let $\mathcal R_1$, $\mathcal R_2$ and $\mathcal R$ be the row spaces of
  $R(G[V_1],p)$, $R(G[V_2],p)$ and $R(G,p)$. The rows of $G[V_1]$ and of
  $G[V_2]$ are rows of $G$, so $\mathcal R_1 + \mathcal R_2 \subseteq
  \mathcal R$. A row of $G[V_1]$ depends only on the screws at the bodies
  of $V_1$, and a row of $G[V_2]$ only on those at the bodies of $V_2$.
  Let $\varphi \in \mathcal R_1 \cap \mathcal R_2$, let $S$ be a screw
  assignment, and let $S'$ agree with $S$ on $V_1$ and vanish elsewhere.
  As $\varphi \in \mathcal R_1$, it takes the same value at $S$ and at
  $S'$. As $\varphi \in \mathcal R_2$ and $S'$ vanishes on $V_2$, that
  value is $0$. Hence $\mathcal R_1 \cap \mathcal R_2 = 0$, which proves
  the bound when $C$ is empty.

  Otherwise $C = \{e\}$, where $e$ has ends $u \in V_1$ and $v \in V_2$.
  By \cref{lem:edge-rows-independent}, the rows of $e$ span a
  subspace $\mathcal R_e$ of $\mathcal R$ of dimension $D - 1$. Its
  members are the forms $S \mapsto r(S(u) - S(v))$, where
  $r$ is a linear form on $K^D$ that vanishes on $C(p(e))$
  (\cref{def:hinge-row-block}). For $x \in K^D$, let $S^x$ be the screw
  assignment equal to $x$ on $V_1$ and to $0$ elsewhere. A row of an edge
  inside $V_1$ or inside $V_2$ vanishes at $S^x$, and the member of
  $\mathcal R_e$ given by $r$ takes the value $r(x)$ there. So a member
  of $\mathcal R_e$ that lies in $\mathcal R_1 + \mathcal R_2$ has
  $r(x) = 0$ for every $x$, and it is zero. Therefore $\mathcal R_1$,
  $\mathcal R_e$ and $\mathcal R_2$ are independent. Their sum lies in
  $\mathcal R$ and has dimension
  $\operatorname{rank} R(G[V_1],p) + (D - 1) + \operatorname{rank}
  R(G[V_2],p)$.

  For the particular case, apply the bound to $G[V_1 \cup \{w\}]$,
  $V_1$ and the given edge. The first side is $G[V_1]$, and the rank of
  the second side, the single body $w$, is at least $0$.
\end{proof}

Katoh and Tanigawa prove this bound for a cut edge $uv$ in the proof of
KT Lemma~6.1~\cite[p.~672]{katohTanigawa2011}. They delete the columns
of $u$ from the rows of $G[V_1]$, which keeps the rank of these rows by
KT Lemma~5.1 (\cref{lem:rank-delete-vertex}). They then read the bound
off the block-triangular shape of the resulting matrix.

When two sides share a body, we compare their row spaces through their
motion spaces, by the following lemma.

\begin{lemma}[The row space is the annihilator of the motions]
  \label{lem:span-rigidityRows-eq-of-motions-eq}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.span_rigidityRows_eq_dualAnnihilator_infinitesimalMotions,
    CombinatorialRigidity.Molecular.BodyHingeFramework.span_rigidityRows_eq_of_infinitesimalMotions_eq}
  \leanok
  \uses{def:rigidity-matrix}
  Let $(G,p)$ be a body-hinge framework with finitely many bodies. Then
  the row space of $R(G,p)$ is the space of linear forms on screw
  assignments that vanish on $Z(G,p)$. In particular, two body-hinge
  frameworks on the same bodies with the same infinitesimal motions have
  the same row space, and so the same rank.
\end{lemma}
\begin{proof}
  \leanok
  \uses{def:rigidity-matrix}
  The space $Z(G,p)$ consists of the screw assignments at which every
  row of $R(G,p)$ vanishes (\cref{def:rigidity-matrix}). The screw
  assignments form a finite-dimensional space. In the dual of a
  finite-dimensional space, every subspace is the space of the forms
  that vanish wherever all its members vanish. Applied to the row space,
  this gives the first claim, and the second follows from it.
\end{proof}

\begin{lemma}[Rank addition at a cut vertex]
  \label{lem:block-rank-cut-vertex}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.finrank_span_rigidityRows_cutVertex_eq}
  \leanok
  \uses{def:rigidity-matrix, def:hinge-constraint}
  Let $(G,p)$ be a body-hinge framework, let $v$ be a body, and let
  $V_1, V_2$ be sets of bodies with $V_1 \cap V_2 \subseteq \{v\}$ such
  that every edge of $G$ lies inside $V_1$ or inside $V_2$. Then
  \[
    \operatorname{rank} R(G,p)
    = \operatorname{rank} R(G[V_1],p) + \operatorname{rank} R(G[V_2],p).
  \]
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:span-rigidityRows-eq-of-motions-eq}
  Every edge of $G$ is an edge of $G[V_1]$ or of $G[V_2]$, so the row
  space of $R(G,p)$ is the sum of the row spaces of $R(G[V_1],p)$ and
  $R(G[V_2],p)$. We show that these two row spaces meet in $0$; then
  the ranks add. By \cref{lem:span-rigidityRows-eq-of-motions-eq}, a
  form in both vanishes on $Z(G[V_1],p) + Z(G[V_2],p)$, so it suffices
  that this sum contains every screw assignment $S$. Let $T$ be the
  assignment equal to $S(v)$ on $V_1$ and to $S$ elsewhere. It is
  constant on $V_1$, so it is a motion of $G[V_1]$. The difference
  $S - T$ vanishes outside $V_1$ and at $v$, hence on $V_2$, so it is a
  motion of $G[V_2]$.
\end{proof}

Two sides that share two bodies $u$ and $v$ interact through the
motions of $v$ relative to $u$.

\begin{definition}[Relative screws and joint rows]
  \label{def:relative-screws}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.relScrews,
    CombinatorialRigidity.Molecular.BodyHingeFramework.jointRows}
  \leanok
  \uses{def:rigidity-matrix}
  Let $(G,p)$ be a body-hinge framework and $u, v$ two bodies. The
  \emph{relative screws} of $u, v$ in $(G,p)$ form the subspace
  \[
    \rho_{uv}(G,p) = \{\, S(v) - S(u) : S \in Z(G,p) \,\} \subseteq K^{D}.
  \]
  For a subspace $U \subseteq K^{D}$, the \emph{joint rows} at $u, v$
  confining the relative screw to $U$ are the linear forms
  $S \mapsto r\bigl(S(u) - S(v)\bigr)$ on screw assignments, one for each
  linear form $r$ on $K^D$ that vanishes on $U$.
\end{definition}

The joint rows at $u, v$ confining the relative screw to $U$ all vanish
at a screw assignment $S$ if and only if $S(u) - S(v) \in U$, as for the
hinge-row block (\cref{def:hinge-row-block}). For an edge $e$ with ends
$u$ and $v$, the joint rows confining the relative screw to
$\operatorname{span} C(p(e))$ span the same space as the rows of $e$ in
$R(G,p)$. At $U = 0$ the joint rows force $S(u) = S(v)$. If $u \neq v$,
the map $S \mapsto S(u) - S(v)$ takes the screw assignments onto $K^D$,
so the joint rows span a space of dimension $D - \dim U$.

The next lemma glues two subgraphs that share two bodies. We state it
for any two subgraphs whose edges together are those of $G$, not only
for induced subgraphs: \cref{lem:block-rank-ear} below glues $G[V_1]$ to
a path, which is not an induced subgraph when its ends are adjacent.

\begin{lemma}[Rank gluing at two shared bodies]
  \label{lem:block-rank-two-cut}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.inf_span_rigidityRows_of_twoCut_of_isLink,
    CombinatorialRigidity.Molecular.BodyHingeFramework.finrank_span_rigidityRows_twoCut_eq_of_isLink}
  \leanok
  \uses{def:relative-screws, def:rigidity-matrix}
  Let $(G,p)$ be a body-hinge framework and let $u \neq v$ be bodies. Let
  $V_1, V_2$ be sets of bodies with $V_1 \cap V_2 \subseteq \{u, v\}$,
  and let $G_1, G_2$ be subgraphs of $G$ such that every edge of $G$ is an
  edge of $G_1$ or of $G_2$, and every edge of $G_i$ lies inside $V_i$.
  Let $\rho_i = \rho_{uv}(G_i, p)$ (\cref{def:relative-screws}). Then the
  row spaces of $R(G_1,p)$ and $R(G_2,p)$ meet in the span of the joint
  rows at $u, v$ confining the relative screw to $\rho_1 + \rho_2$, and
  \[
    \operatorname{rank} R(G,p) = \operatorname{rank} R(G_1,p)
    + \operatorname{rank} R(G_2,p) + \dim(\rho_1 + \rho_2) - D.
  \]
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:span-rigidityRows-eq-of-motions-eq}
  Let $\mathcal R_1$ and $\mathcal R_2$ be the row spaces of $R(G_1,p)$
  and $R(G_2,p)$, and let $\delta(S) = S(u) - S(v)$. By
  \cref{lem:span-rigidityRows-eq-of-motions-eq}, a form lies in
  $\mathcal R_1 \cap \mathcal R_2$ if and only if it vanishes on
  $Z(G_1,p) + Z(G_2,p)$.

  This sum contains every screw assignment $S$ with $S(u) = S(v)$.
  Indeed, let $T$ be the assignment equal to $S(v)$ on $V_2$ and to $S$
  elsewhere. It is constant on $V_2$, so it is a motion of $G_2$. The
  difference $S - T$ vanishes outside $V_2$, and on
  $V_1 \cap V_2 \subseteq \{u, v\}$ because $S(u) = S(v)$. So it vanishes
  on $V_1$, and it is a motion of $G_1$. As $\delta$ maps $Z(G_i,p)$
  onto $-\rho_i = \rho_i$, the sum $Z(G_1,p) + Z(G_2,p)$ is therefore the
  space of screw assignments $S$ with $\delta(S) \in \rho_1 + \rho_2$.

  Since $u \neq v$, $\delta$ is onto $K^D$, so every form that vanishes
  on the kernel of $\delta$ is $r \circ \delta$ for a linear form $r$ on
  $K^D$. Such a form vanishes on $Z(G_1,p) + Z(G_2,p)$ exactly when $r$
  vanishes on $\rho_1 + \rho_2$, that is, when it is a joint row at
  $u, v$ confining the relative screw to $\rho_1 + \rho_2$. This proves
  the first claim. These joint rows span a space of dimension
  $D - \dim(\rho_1 + \rho_2)$, by the remarks after
  \cref{def:relative-screws}.

  Every edge of $G$ is an edge of $G_1$ or of $G_2$, with the same hinge,
  so the row space of $R(G,p)$ is $\mathcal R_1 + \mathcal R_2$. The rank
  formula follows from
  $\dim(\mathcal R_1 + \mathcal R_2) = \dim \mathcal R_1
  + \dim \mathcal R_2 - \dim(\mathcal R_1 \cap \mathcal R_2)$.
\end{proof}

\begin{corollary}[Rank gluing at a vertex $2$-cut]
  \label{cor:block-rank-vertex-two-cut}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.inf_span_rigidityRows_of_vertexTwoCut,
    CombinatorialRigidity.Molecular.BodyHingeFramework.finrank_span_rigidityRows_vertexTwoCut_eq}
  \leanok
  \uses{lem:block-rank-two-cut}
  Let $(G,p)$ be a body-hinge framework, let $u \neq v$ be bodies, and
  let $V_1, V_2$ be sets of bodies with $V_1 \cap V_2 = \{u, v\}$ such
  that every edge of $G$ lies inside $V_1$ or inside $V_2$. Let
  $\rho_i = \rho_{uv}(G[V_i], p)$. Then the row spaces of $R(G[V_1],p)$
  and $R(G[V_2],p)$ meet in the span of the joint rows at $u, v$
  confining the relative screw to $\rho_1 + \rho_2$, and
  \[
    \operatorname{rank} R(G,p) = \operatorname{rank} R(G[V_1],p)
    + \operatorname{rank} R(G[V_2],p) + \dim(\rho_1 + \rho_2) - D.
  \]
\end{corollary}
\begin{proof}
  \leanok
  Apply \cref{lem:block-rank-two-cut} to $G_i = G[V_i]$. The edges of
  $G[V_i]$ lie inside $V_i$, and every edge of $G$ is an edge of
  $G[V_1]$ or of $G[V_2]$.
\end{proof}

We apply \cref{lem:block-rank-two-cut} to a path attached to $G[V_1]$ at
two bodies $a \neq b$; \cref{def:open-ear} calls such a path
an open ear on $V_1$. The rank and the relative screws of a path come
first.

\begin{lemma}[The rank of a path]
  \label{lem:block-rank-path}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.le_finrank_span_rigidityRows_path,
    CombinatorialRigidity.Molecular.BodyHingeFramework.finrank_span_rigidityRows_path_le}
  \leanok
  \uses{def:rigidity-matrix}
  Let $(P,p)$ be a body-hinge framework whose edges are those of a path
  $a = x_0 - x_1 - \cdots - x_k - x_{k+1} = b$ ($k \geq 0$) on distinct
  bodies, and let $C_i = C(p(x_i x_{i+1})) \neq 0$ for
  $0 \le i \le k$. Then $\operatorname{rank} R(P,p) = (D-1)(k+1)$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:block-rank-cut, def:hinge-row-block, def:cut-edges-2ec}
  The matrix $R(P,p)$ has $D - 1$ rows for each of its $k + 1$ edges
  (\cref{def:hinge-row-block}), so its rank is at most $(D-1)(k+1)$.
  Conversely, for $0 \le i \le k$ the edge $x_i x_{i+1}$ is the only edge
  of $P$ that joins $x_{i+1}$ to $\{x_0, \dots, x_i\}$. So adding
  $x_{i+1}$ to $\{x_0, \dots, x_i\}$ raises the rank of the subgraph of
  $P$ induced on these bodies by at least $D - 1$
  (\cref{lem:block-rank-cut}). The subgraph induced on $\{x_0\}$ has no
  edge, and the subgraph induced on all the bodies of the path is $P$.
\end{proof}

\begin{lemma}[The relative screws of a path]
  \label{lem:relative-screws-path}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.span_range_le_relScrews_path,
    CombinatorialRigidity.Molecular.BodyHingeFramework.relScrews_path_le_span_range}
  \leanok
  \uses{def:relative-screws}
  Let $(P,p)$ be a body-hinge framework whose edges are those of a path
  $a = x_0 - x_1 - \cdots - x_k - x_{k+1} = b$ ($k \geq 0$) on distinct
  bodies, and let $C_i = C(p(x_i x_{i+1}))$ for $0 \le i \le k$. Then
  $\rho_{ab}(P,p) = \operatorname{span}(C_0, \dots, C_k)$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{def:hinge-constraint}
  Let $S$ be an infinitesimal motion of $(P,p)$. Each difference
  $S(x_{i+1}) - S(x_i)$ lies in $\operatorname{span} C_i$
  (\cref{def:hinge-constraint}), and these differences add up to
  $S(b) - S(a)$, which therefore lies in
  $\operatorname{span}(C_0, \dots, C_k)$. Conversely, fix $i$ and let $S$
  be the screw assignment equal to $C_i$ at $x_{i+1}, \dots, x_{k+1}$ and
  to $0$ elsewhere. Its difference across the edge $x_i x_{i+1}$ is
  $C_i$, and across every other edge of $P$ it is $0$. So $S$ is an
  infinitesimal motion with $S(b) - S(a) = C_i$, and the subspace
  $\rho_{ab}(P,p)$ contains every $C_i$.
\end{proof}

\begin{lemma}[Rank addition along an open ear]
  \label{lem:block-rank-ear}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.finrank_span_rigidityRows_ear_eq}
  \leanok
  \uses{def:relative-screws, def:rigidity-matrix}
  Let $(G,p)$ be a body-hinge framework and $V_1$ a set of bodies. Let
  $a - x_1 - \cdots - x_k - b$ ($k \geq 0$) be a path in $G$ with ends
  $a \neq b$ in $V_1$ and distinct bodies $x_1, \dots, x_k$ outside
  $V_1$, such that every edge of $G$ off the path lies inside $V_1$.
  Write $x_0 = a$ and $x_{k+1} = b$, suppose that
  $C_i = C(p(x_i x_{i+1})) \neq 0$ for $0 \le i \le k$, and let
  $\Lambda = \operatorname{span}(C_0, \dots, C_k)$ and
  $\rho = \rho_{ab}(G[V_1], p)$. Then
  \[
    \operatorname{rank} R(G,p) = \operatorname{rank} R(G[V_1],p)
    + (D - 1)(k + 1) + \dim(\rho + \Lambda) - D.
  \]
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:block-rank-two-cut, lem:block-rank-path, lem:relative-screws-path}
  Let $P$ be the path, as a subgraph of $G$, and let $V_2$ be the set of
  its bodies. Then $V_1 \cap V_2 = \{a, b\}$, and every edge of $G$ is an
  edge of $P$ or of $G[V_1]$. By \cref{lem:block-rank-two-cut} at
  $u = a$ and $v = b$, with $G_1 = G[V_1]$ and $G_2 = P$,
  \[
    \operatorname{rank} R(G,p) = \operatorname{rank} R(G[V_1],p)
    + \operatorname{rank} R(P,p) + \dim\bigl(\rho + \rho_{ab}(P,p)\bigr) - D.
  \]
  The path has rank $(D-1)(k+1)$ (\cref{lem:block-rank-path}) and
  relative screws $\rho_{ab}(P,p) = \Lambda$
  (\cref{lem:relative-screws-path}).
\end{proof}

We also need notation for deleting columns. For a set $W$ of bodies and
a screw assignment $S$, let $S^W$ agree with $S$ off $W$ and vanish on
$W$. For a linear form $\varphi$ on screw assignments, let
$\pi_W(\varphi)$ be the form $S \mapsto \varphi(S^W)$. If $\varphi$ is a
row of a rigidity matrix, then $\pi_W(\varphi)$ is the same row with
zeros in the columns of $W$, and we say that the columns of $W$ are
deleted from it. The map $\pi_W$ is linear.

In the proof of KT Lemma~6.2~\cite[pp.~673--674]{katohTanigawa2011},
where $H$ is a rigid subgraph of $G$, Katoh and Tanigawa read the rank
of $R(G,p)$ off its block-triangular shape (KT eq.~(6.3)). They compare
the rank of the rows of the edges of $G$ not in $H$, with the columns of
$V(H)$ deleted, with that of a realization of the contraction $G/E(H)$,
in which $H$ becomes one body (\cref{def:rigid-contraction}; KT
eqs.~(6.4) and~(6.5)). The next lemma is the count behind this
comparison, for any linear map $\pi$ in place of $\pi_{V(H)}$.

\begin{lemma}[Block-triangular rank addition; from the proof of KT Lemma~6.2]
  \label{lem:rigidityRows-splice-rank-add}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.le_finrank_span_rigidityRows_of_splice}
  \leanok
  \uses{def:rigidity-matrix, def:hinge-row-block}
  Let $F$, $F_H$ and $F_c$ be body-hinge frameworks on one set of bodies,
  with row spaces $\mathcal R$, $\mathcal R_H$ and $\mathcal R_c$, and let
  $\pi$ be a linear map from the space of linear forms on screw
  assignments to itself. Suppose that
  $\mathcal R_H \subseteq \mathcal R \cap \ker \pi$, that
  $\pi(\mathcal R_c) \subseteq \pi(\mathcal R)$, and that $\pi$ is
  injective on $\mathcal R_c$. Then
  \[
    \operatorname{rank} R(F_H) + \operatorname{rank} R(F_c)
    \;\le\; \operatorname{rank} R(F).
  \]
\end{lemma}
\begin{proof}
  \leanok
  \uses{def:rigidity-matrix, def:hinge-row-block}
  By rank-nullity for $\pi$ on $\mathcal R$,
  $\dim \mathcal R = \dim \pi(\mathcal R) + \dim(\mathcal R \cap \ker \pi)$.
  The first hypothesis gives
  $\dim \mathcal R_H \le \dim(\mathcal R \cap \ker \pi)$. The other two
  give $\dim \mathcal R_c = \dim \pi(\mathcal R_c) \le \dim \pi(\mathcal R)$.
  Adding the two bounds proves the lemma.
\end{proof}

In the proof of KT Lemma~6.2 (\cref{lem:case-I-realization-nonsimple}),
$F$ realizes $G$, $F_H$ is its restriction to $H$, $F_c$ realizes the
contraction $G/E(H)$, and $\pi = \pi_{V(H)}$. The first hypothesis holds
because a row of an edge of $H$ depends only on the screws at the bodies
of $V(H)$, so $\pi_{V(H)}$ maps it to $0$. The second holds
because the edges of $G \setminus E(H)$ have the same hinges in both
realizations (KT eq.~(6.4)), so that their rows agree once the columns
of $V(H)$ are deleted.

In KT Lemmas~6.8 and~6.10 a body $v$ of degree two is restored after
splitting off. Katoh and Tanigawa bring the rigidity matrix to a
block-triangular form by column and row operations (KT eqs.~(6.16)
and~(6.29), pp.~679 and~686). In this form a block of rows has nonzero
entries only in the columns of $v$, and they read the rank off it. The
next lemma gives such a count for two families of linear forms in the
row space of a rigidity matrix, which need not be its rows: one family
is independent on the screws of $v$, and the other vanishes on them.

\begin{lemma}[Rank addition at one body; from the proofs of KT Lemmas~6.8 and~6.10]
  \label{lem:rigidityRows-pinned-placement-rank-add}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.le_finrank_span_rigidityRows_of_pinned_placement,
    CombinatorialRigidity.Molecular.BodyHingeFramework.le_finrank_span_rigidityRows_of_pinned_placement_augment}
  \leanok
  \uses{def:rigidity-matrix, lem:case-II-placement-block-independent, lem:case-III-conditional-block}
  Let $F$ be a body-hinge framework, let $v$ be a body, and let $(\varphi_i)_{i \in I}$ and $(\psi_j)_{j \in J}$ be
  finite families of linear forms in the row space of $R(F)$. Suppose
  that the restrictions of the $\varphi_i$ to the screw assignments that
  vanish off $v$ are linearly independent, and that the $\psi_j$ are
  linearly independent and vanish on these screw assignments. Then
  $|I| + |J| \le \operatorname{rank} R(F)$. In particular, if $\omega$ is
  a further form in the row space such that the restrictions of the
  $\varphi_i$ and of $\omega$ to these screw assignments are linearly
  independent, then $|I| + 1 + |J| \le \operatorname{rank} R(F)$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:case-II-placement-block-independent, lem:case-III-conditional-block}
  Suppose that $\sum_i a_i \varphi_i + \sum_j b_j \psi_j = 0$. Every
  $\psi_j$ vanishes on the screw assignments that vanish off $v$, so
  $\sum_i a_i \varphi_i$ vanishes on them, and every $a_i$ is $0$ by the
  first hypothesis. Then $\sum_j b_j \psi_j = 0$, and every $b_j$ is $0$.
  So the $|I| + |J|$ forms are linearly independent
  (\cref{lem:case-II-placement-block-independent}). They lie in the row
  space of $R(F)$, whose dimension is therefore at least $|I| + |J|$. The
  second claim is the first, applied to the family of the $\varphi_i$
  and $\omega$ (\cref{lem:case-III-conditional-block}).
\end{proof}

The independence of the two families is
\cref{lem:case-II-placement-block-independent} of the algebraic
induction, for KT Lemma~6.8, and with $\omega$ added it is
\cref{lem:case-III-conditional-block}, for KT eq.~(6.29). In Case~II (KT Lemma~6.8) the $\varphi_i$ are the $D - 1$
rows of an edge at $v$, and the $\psi_j$ are rows of the graph obtained
by splitting off $v$, which lie in the row space of $R(F)$ without being
its rows. In KT Lemma~6.10, $\omega$ is the further row that the row
operations of eq.~(6.29) produce.

Case~I of the algebraic induction contracts a proper rigid subgraph $H$
of $G$ (\cref{def:rigid-subgraph}). Let $r$ be a body of
$H$. The contraction $G/E(H)$ has the edges of $G \setminus E(H)$, with
every end in $V(H)$ replaced by $r$, so its bodies meet $V(H)$ only in
$r$. Deleting the columns of $V(H)$ from its rows therefore deletes
those of $r$. By KT Lemma~5.1 this does not change the rank, as Katoh
and Tanigawa note at eq.~(6.5) (p.~674). The next lemma states this in
the form of the third hypothesis of
\cref{lem:rigidityRows-splice-rank-add}, for $\pi = \pi_{V(H)}$.

\begin{lemma}[Deleting the columns of a set that meets $V(G)$ in one body; KT eq.~(6.5)]
  \label{lem:extProj-preserves-rank-of-inter}
  \lean{CombinatorialRigidity.Molecular.infinitesimalMotions_sup_range_extProj_eq_top_of_inter_eq_singleton,
    CombinatorialRigidity.Molecular.BodyHingeFramework.injOn_extProj_dualMap_rigidityRows_of_inter_eq_singleton,
    CombinatorialRigidity.Molecular.BodyHingeFramework.finrank_span_rigidityRows_map_extProj_dualMap_of_inter_eq_singleton}
  \leanok
  \uses{def:rigidity-matrix}
  Let $(G,p)$ be a body-hinge framework, and let $t$ be a set of bodies
  with $V(G) \cap t = \{r\}$. Then every screw
  assignment is the sum of an infinitesimal motion of $(G,p)$ and a screw
  assignment that vanishes on $t$. Hence $\pi_t$ is injective on the row
  space of $R(G,p)$, and deleting the columns of $t$ from $R(G,p)$ does
  not change its rank.
\end{lemma}
\begin{proof}
  \leanok
  \uses{def:rigidity-matrix}
  Let $S$ be a screw assignment, and let $T$ be equal to $S(r)$ on $V(G)$
  and to $S$ elsewhere. Both ends of every edge lie in $V(G)$, where $T$
  is constant, so $T$ is an infinitesimal motion. The difference $S - T$
  vanishes at $r$ and off $V(G)$. As $t \setminus \{r\}$ lies off $V(G)$,
  the difference vanishes on $t$. This proves the first claim.

  Let $\varphi$ be a form in the row space with $\pi_t(\varphi) = 0$. Then
  $\varphi$ vanishes on the screw assignments that vanish on $t$, and,
  like every row of $R(G,p)$, on $Z(G,p)$ (\cref{def:rigidity-matrix}). By
  the first claim it vanishes everywhere, so $\varphi = 0$. Hence $\pi_t$
  is injective on the row space, and the row space has the same dimension
  as its image, the row space of $R(G,p)$ with the columns of $t$
  deleted.
\end{proof}

The motion $T$ plays the part of the trivial motion with value $S(r)$ in
the proof of KT Lemma~5.1 (\cref{lem:rank-delete-vertex}). In Case~I the
rows of $G \setminus E(H)$ are compared with those of $G/E(H)$ after this
deletion (KT Claim~6.4 and eq.~(6.9)), at every deficiency in
\cref{lem:rank-polynomial-IH-relabel-proj} and when $G/E(H)$ is rigid in
\cref{lem:claim-6-4}.
