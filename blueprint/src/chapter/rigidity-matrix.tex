\section{The panel-hinge rigidity matrix \texorpdfstring{$R(G,p)$}{R(G,p)}}
\label{sec:molecular-rigidity-matrix}

This chapter builds the panel-hinge (body-hinge) rigidity matrix of
Katoh--Tanigawa~\cite[\S2.2--2.4]{katohTanigawa2011}, the substrate on
which the conjecture's algebraic induction
(\cref{sec:molecular-algebraic-induction}) runs, using the extensor
algebra of \cref{sec:molecular}. Where \cref{sec:body-hinge}
\emph{defined} rigidity by reduction to body-bar frameworks on the
multiplied graph $(D-1)\cdot G$ --- a standard-basis witness with no
rank computation, the \emph{existence} form --- this chapter assigns
each edge an explicit $(d-2)$-dimensional affine \emph{hinge} subspace,
builds the rigidity matrix $R(G,p)$ of a realization $p$, and computes
its rank directly. Following \cref{sec:molecular}, the screw space and the
rigidity matrix $R(G,p)$ are built over an arbitrary field $K$;
Katoh--Tanigawa's original argument is the $K = \R$ specialization. The
infinitesimal-motion vectors are $D$-dimensional
\emph{screw centers} $S(v) \in K^{D}$ with $D = \binom{d+1}{2}$,
realized as the degree-$(d-1)$ graded piece $\bigwedge^{d-1}K^{d+1}$ of
the extensor algebra of \cref{sec:molecular-homog}.

\Cref{def:hinge-constraint}, \cref{def:hinge-row-block}, and
\cref{def:rigidity-matrix} build the matrix $R(G,p)$ and its null space
$Z(G,p)$; \cref{lem:trivial-motions-rank-bound} exhibits the
$D$-dimensional space
of trivial (rigid-body) motions, the resulting bound
$\operatorname{rank} R(G,p) \le D(|V|-1)$, and the codimension bound
$D + \operatorname{def}(\tilde G) \le \dim Z(G,p)$ that holds at every
realization, generic or not (the lower-bound half of Tay--Whiteley's
Proposition~1.1, \cref{prop:rigidity-matrix-prop11}).
\Cref{sec:molecular-rigidity-matrix-rank} proves Katoh--Tanigawa's three
rank lemmas --- pinning a body preserves rank (Lemma~5.1,
\cref{lem:rank-delete-vertex}), rank is lower-semicontinuous along a
panel rotation (Lemma~5.2, \cref{lem:rank-rotation-semicont}), and two
hinges of parallel edges in general position force full rank (Lemma~5.3,
\cref{lem:rank-parallel-full}, the base case $|V| = 2$ of the algebraic
induction) --- and \cref{sec:molecular-rigidity-matrix-blocks} proves the
block-rank-addition lemmas underlying the induction's structural cases:
a vertex cut (KT Lemma~6.1, \cref{lem:block-rank-cut}), a shared-body
contraction splice (KT Lemma~6.2,
\cref{lem:rigidityRows-splice-rank-add}), and the pin-a-body splitting
geometry shared by KT Lemmas~6.8 and~6.10
(\cref{lem:rigidityRows-pinned-placement-rank-add}). Equating
$\operatorname{rank} R(G,p) = D(|V|-1)$ with $(D,D)$-tightness
of $(D-1)\cdot G$ (\cref{thm:body-hinge-tay}) needs the analytic
generic-rank theorem of Jackson--Jord\'an
(\cref{prop:rigidity-matrix-prop11}) and is proved alongside the
algebraic induction, once the matroidal bridge
$\operatorname{def}(\tilde G) = \operatorname{corank} M(\tilde G)$
(\cref{sec:molecular-deficiency}) is in hand.

\subsection{Body-hinge frameworks and the rigidity matrix}
\label{sec:molecular-rigidity-matrix-def}

This subsection defines the rigidity matrix of a body-hinge framework,
following Katoh and Tanigawa~\cite[\S\S2.2--2.4]{katohTanigawa2011}.
They identify an infinitesimal motion of a rigid body in $\R^d$ with a
vector of $\R^D$, its screw center, where $D = \binom{d+1}{2}$. A hinge
joining two bodies constrains the difference of their screw centers to
the span of a $(d-1)$-extensor supporting the hinge. We take this
constraint as the definition, over a field $K$. Screw centers lie in
$\bigwedge^{d-1}K^{d+1}$, which has dimension
$\binom{d+1}{d-1} = D$, and the supporting extensors are those of
\cref{def:affine-subspace-extensor}.

\begin{definition}[Hinge constraint via the supporting extensor]
  \label{def:hinge-constraint}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework,
    CombinatorialRigidity.Molecular.BodyHingeFramework.hingeConstraint}
  \leanok
  \uses{def:affine-subspace-extensor}
  A \emph{body-hinge framework} $(G,p)$ in $K^d$ is a multigraph
  $G = (V,E)$ together with a map $p$ that assigns to each edge $e$ a
  $(d-2)$-dimensional affine subspace $p(e)$ of $K^d$, the \emph{hinge}
  of $e$. Let $C(p(e)) \in \bigwedge^{d-1}K^{d+1}$ be a
  $(d-1)$-extensor supporting $p(e)$
  (\cref{def:affine-subspace-extensor}). A \emph{screw assignment} is a
  map $S \colon V \to \bigwedge^{d-1}K^{d+1}$. It satisfies the
  \emph{hinge constraint} at an edge $e$ with ends $u$ and $v$ if
  \[
    S(u) - S(v) \in \operatorname{span} C\bigl(p(e)\bigr).
  \]
\end{definition}

The hinge $p(e)$ is spanned by $d - 1$ affinely independent points, so
$C(p(e)) \neq 0$, and $p(e)$ determines $C(p(e))$ up to a nonzero
scalar (the remark after \cref{def:affine-subspace-extensor}). So the
hinge constraint does not depend on the choice of $C(p(e))$. It is
symmetric in $u$ and $v$, since $S(v) - S(u) = -(S(u) - S(v))$.

\begin{fmlnote}
  The formalization records a body-hinge framework by its multigraph
  and, for each edge $e$, an element $C(e)$ of the screw space in place
  of $C(p(e))$. It does not require $C(e)$ to be nonzero or to come from
  points, so the same definitions apply to hinges given as the meet of
  two panels (\cref{def:panel-hinge-framework}). A lemma that needs
  $C(e) \neq 0$ assumes it. Screw assignments are defined on a fixed set
  of bodies, which may contain bodies that lie on no edge of $G$
  (\cref{lem:relative-deficiency-rank-bound}). The dimension is written
  $d = k + 1$, so that the screw space is $\bigwedge^k K^{k+2}$.
\end{fmlnote}

Katoh and Tanigawa write the hinge constraint as linear
equations~\cite[\S2.4]{katohTanigawa2011}. For this we identify
$\bigwedge^{d-1}K^{d+1}$ with $K^D$ by the basis of the products
$e_{i_1} \wedge \cdots \wedge e_{i_{d-1}}$ with
$i_1 < \cdots < i_{d-1}$ (the remark after \cref{def:plucker-coords}),
and write $x \cdot y$ for the dot product of $K^D$.

\begin{definition}[Hinge-row block]
  \label{def:hinge-row-block}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.hingeRowBlock,
    CombinatorialRigidity.Molecular.BodyHingeFramework.hingeConstraint_iff_hingeRowBlock,
    CombinatorialRigidity.Molecular.BodyHingeFramework.finrank_hingeRowBlock}
  \leanok
  \uses{def:hinge-constraint}
  Let $(G,p)$ be a body-hinge framework and let $e$ be an edge of $G$.
  Let $r_1(p(e)), \dots, r_{D-1}(p(e))$ be a basis of the orthogonal
  complement of $\operatorname{span} C(p(e))$ in $K^D$, and let
  $r(p(e))$ be the $(D-1) \times D$ matrix with these rows. Then a screw
  assignment $S$ satisfies the hinge constraint at $e$, with ends $u$
  and $v$, if and only if
  \[
    r_i\bigl(p(e)\bigr) \cdot \bigl(S(u) - S(v)\bigr) = 0
    \qquad (1 \le i \le D-1).
  \]
\end{definition}

The orthogonal complement has dimension $D - 1$ because
$C(p(e)) \neq 0$. The equivalence holds because a subspace of $K^D$ is
the orthogonal complement of its orthogonal complement, as the dot
product is nondegenerate. Through the dot product, the rows of
$r(p(e))$ form a basis of the linear forms on $K^D$ that vanish on
$C(p(e))$.

The rigidity matrix collects these equations for all the edges.

\begin{definition}[Panel-hinge rigidity matrix]
  \label{def:rigidity-matrix}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.IsInfinitesimalMotion,
    CombinatorialRigidity.Molecular.BodyHingeFramework.infinitesimalMotions,
    CombinatorialRigidity.Molecular.BodyHingeFramework.hingeRow,
    CombinatorialRigidity.Molecular.BodyHingeFramework.rigidityRows,
    CombinatorialRigidity.Molecular.BodyHingeFramework.infinitesimalMotions_eq_dualCoannihilator,
    CombinatorialRigidity.Molecular.BodyHingeFramework.exists_finite_spanning_rigidityRows,
    CombinatorialRigidity.Molecular.BodyHingeFramework.linearIndependent_hingeRow,
    CombinatorialRigidity.Molecular.BodyHingeFramework.exists_independent_rigidityRows_of_edge,
    CombinatorialRigidity.Molecular.BodyHingeFramework.linearIndependent_hingeRow_star,
    CombinatorialRigidity.Molecular.BodyHingeFramework.linearIndependent_hingeRow_forest,
    CombinatorialRigidity.Molecular.BodyHingeFramework.exists_independent_rigidityRows_of_forest,
    CombinatorialRigidity.Molecular.ScrewSpace,
    CombinatorialRigidity.Molecular.screwDim,
    CombinatorialRigidity.Molecular.screwSpace_finrank}
  \leanok
  \uses{def:hinge-row-block}
  Let $(G,p)$ be a body-hinge framework. Its \emph{rigidity matrix}
  $R(G,p)$ is the $(D-1)|E| \times D|V|$ matrix with a block of $D - 1$
  rows for each edge and a block of $D$ columns for each body. Orient
  each edge $e$ from one end $u$ to the other end $v$. The rows of $e$
  have $r(p(e))$ in the columns of $u$, $-r(p(e))$ in the columns of
  $v$, and zeros elsewhere (\cref{def:hinge-row-block}). An
  \emph{infinitesimal motion} of $(G,p)$ is a screw assignment that
  satisfies the hinge constraint at every edge, that is, a vector of
  $\ker R(G,p) \subseteq K^{D|V|}$. We write $Z(G,p)$ for the space of
  infinitesimal motions.
\end{definition}

The space $Z(G,p)$ depends neither on the choice of the rows of the
matrices $r(p(e))$ nor on the orientation of the
edges~\cite[p.~654]{katohTanigawa2011}. So neither does the rank of
$R(G,p)$, which is
\[
  \operatorname{rank} R(G,p) \;=\; D|V| - \dim Z(G,p).
\]

\begin{fmlnote}
  The formalization fixes no basis of the screw space. A row of
  $R(G,p)$ is the linear form $S \mapsto r(S(u) - S(v))$ on screw
  assignments, for an edge $e$ with ends $u$ and $v$ and a linear form
  $r$ on the screw space that vanishes on $C(p(e))$. The rank of
  $R(G,p)$ is the dimension of the span of these forms, and $Z(G,p)$ is
  the space on which they all vanish.
\end{fmlnote}

Two facts about the rows of $R(G,p)$ are used later. First, let $e$ be
an edge with ends $u \neq v$. Its $D - 1$ rows are the linear forms
$S \mapsto r_i(p(e)) \cdot (S(u) - S(v))$. The vectors $r_i(p(e))$ are
linearly independent, and $S \mapsto S(u) - S(v)$ maps the screw
assignments onto $K^D$, so these rows are linearly independent.

Second, the rows of several edges are jointly independent when each of
the edges has an end of its own. Let $e_1, \dots, e_m$ be edges such
that $e_j$ has ends $u_j$ and $w_j$, the bodies $u_1, \dots, u_m$ are
distinct, and no $w_j$ is equal to any $u_{j'}$. Then the $(D-1)m$ rows
of $e_1, \dots, e_m$ are linearly independent. Indeed, take a linear
combination of them that vanishes, and evaluate it at the screw
assignment with value $x$ at $u_j$ and $0$ elsewhere. Only the rows of
$e_j$ contribute, and their coefficients $c_i$ give
$\sum_i c_i\, r_i(p(e_j)) \cdot x = 0$ for every $x \in K^D$, so every
$c_i$ is zero. For example, the edges of a star at a body $w$ satisfy
the hypothesis if their other ends are distinct bodies other than $w$.

The next definitions are Katoh and
Tanigawa's~\cite[\S\S2.3--2.4]{katohTanigawa2011}, who take that of a
generic framework from Jackson and
Jord\'an~\cite[\S6]{jacksonJordan2009}. A body-hinge framework on $G$
is also called a \emph{realization} of $G$.

\begin{definition}[Degree of freedom; generic realization]
  \label{def:dof-generic}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.IsInfinitesimallyRigid}
  \uses{lem:trivial-motions-rank-bound}
  Let $(G,p)$ be a body-hinge framework. An infinitesimal motion $S$ of
  $(G,p)$ is \emph{trivial} if $S(u) = S(v)$ for all bodies $u$ and $v$,
  and $(G,p)$ is \emph{infinitesimally rigid} if all its infinitesimal
  motions are trivial. The \emph{degree of freedom} of $(G,p)$ is
  $D(|V|-1) - \operatorname{rank} R(G,p)$. The framework $(G,p)$ is
  \emph{generic} if $R(G,p)$ and each of its edge-induced submatrices
  (the rows of the edges of a subset of $E$) have the largest rank over
  all realizations of $G$.
\end{definition}

By the next lemma, the trivial motions form a $D$-dimensional subspace
of $Z(G,p)$ when $V$ is nonempty. So the degree of freedom is
$\dim Z(G,p) - D \ge 0$, and it is $0$ if and only if $(G,p)$ is
infinitesimally rigid.

The lemma also bounds $\dim Z(G,p)$ from below by the deficiency, which
the next chapter defines (\cref{def:D-deficiency}). We recall it here.
Let $\tilde G = (D-1)\cdot G$ (\cref{sec:body-hinge-reduction}). For a
partition $P$ of $V$ into nonempty parts, let $d_G(P)$ be the number of
edges of $G$ that join different parts. Then
\[
  \operatorname{def}_{\tilde G}(P) \;=\; D\,(|P| - 1) - (D-1)\,d_G(P),
\]
and $\operatorname{def}(\tilde G)$ is the largest value of
$\operatorname{def}_{\tilde G}(P)$ over the partitions $P$ of $V$.

\begin{lemma}[The $D$ trivial motions and the rank bound]
  \label{lem:trivial-motions-rank-bound}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.IsTrivialMotion,
    CombinatorialRigidity.Molecular.BodyHingeFramework.isInfinitesimalMotion_of_isTrivialMotion,
    CombinatorialRigidity.Molecular.BodyHingeFramework.trivialMotions,
    CombinatorialRigidity.Molecular.BodyHingeFramework.trivialMotions_le_infinitesimalMotions,
    CombinatorialRigidity.Molecular.BodyHingeFramework.trivialMotions_eq_range_const,
    CombinatorialRigidity.Molecular.BodyHingeFramework.infinitesimalMotions_eq_trivialMotions_iff,
    CombinatorialRigidity.Molecular.BodyHingeFramework.finrank_trivialMotions,
    CombinatorialRigidity.Molecular.BodyHingeFramework.finrank_screwAssignment,
    CombinatorialRigidity.Molecular.BodyHingeFramework.screwDim_add_deficiency_le_finrank_infinitesimalMotions}
  \leanok
  \uses{def:rigidity-matrix, def:D-deficiency}
  Let $(G,p)$ be a body-hinge framework with $V$ nonempty, and for
  $1 \le i \le D$ let $S^*_i$ be the screw assignment whose value at
  every body is the $i$-th basis vector of $K^D$. Then
  $S^*_1, \dots, S^*_D$ are infinitesimal motions, and they form a
  basis of the space of trivial motions. Hence
  $\operatorname{rank} R(G,p) \le D(|V|-1)$, with equality if and only
  if $(G,p)$ is infinitesimally rigid. Moreover
  \[
    \dim Z(G,p) \;\ge\; D + \operatorname{def}(\tilde G),
  \]
  that is,
  $\operatorname{rank} R(G,p) \le D(|V|-1) - \operatorname{def}(\tilde G)$.
\end{lemma}
\begin{proof}
  \leanok
  A constant screw assignment $S$ has $S(u) - S(v) = 0$ at every edge,
  so it is an infinitesimal motion. The trivial motions are exactly the
  constant assignments. As $V$ is nonempty, sending $x \in K^D$ to the
  constant assignment with value $x$ is an isomorphism from $K^D$ onto
  them, and it sends the standard basis to $S^*_1, \dots, S^*_D$. So
  the trivial motions form a $D$-dimensional subspace of $Z(G,p)$, and
  $\operatorname{rank} R(G,p) = D|V| - \dim Z(G,p) \le D(|V|-1)$.
  Equality holds if and only if $Z(G,p)$ is the space of trivial
  motions, that is, if and only if $(G,p)$ is infinitesimally rigid.

  For the second bound, let $P$ be a partition of $V$, and let $W_P$ be
  the space of screw assignments that are constant on each part of $P$.
  It has dimension $D|P|$. On $W_P$, the hinge constraint at an edge
  inside a part always holds. At an edge $e$ that joins two parts it
  consists of $D - 1$ linear equations, since $C(p(e)) \neq 0$
  (\cref{def:hinge-row-block}). So the infinitesimal motions in $W_P$
  form a subspace of dimension at least
  \[
    D|P| - (D-1)\,d_G(P) \;=\; D + \operatorname{def}_{\tilde G}(P).
  \]
  A partition $P$ with
  $\operatorname{def}_{\tilde G}(P) = \operatorname{def}(\tilde G)$
  gives the bound.
\end{proof}

The second bound holds at every realization, generic or not, and it is
one of the two inequalities of the rank formula in
\cref{prop:rigidity-matrix-prop11}. Jackson and Jord\'an derive its
rank form from a count of independent
rows~\cite[\S4]{jacksonJordan2009}; we count motions instead.

Later chapters consider a framework on a subgraph $G$ of a fixed graph
together with all the bodies of the fixed graph. Its screw assignments
are maps $B \to \bigwedge^{d-1}K^{d+1}$ on a finite set $B$ of bodies
that contains $V(G)$, and $R(G,p)$ and $Z(G,p)$ are defined as before.
A body of $B \setminus V(G)$ lies on no edge of $G$, so no hinge
constraint involves it.

\begin{lemma}[The relative deficiency bound]
  \label{lem:relative-deficiency-rank-bound}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.screwDim_mul_compl_add_deficiency_le_finrank_infinitesimalMotions,
    CombinatorialRigidity.Molecular.BodyHingeFramework.finrank_span_rigidityRows_add_deficiency_le}
  \leanok
  \uses{def:rigidity-matrix, def:D-deficiency}
  Let $(G,p)$ be a body-hinge framework whose screw assignments are
  defined on a finite set $B$ of bodies with
  $\emptyset \neq V(G) \subseteq B$, and suppose that $C(p(e)) \neq 0$
  at every edge $e$ of $G$. Then
  \[
    D\,(|B \setminus V(G)| + 1) + \operatorname{def}(\tilde G) \;\le\;
      \dim Z(G,p),
  \]
  that is,
  $\operatorname{rank} R(G,p) \le D(|V(G)| - 1) - \operatorname{def}(\tilde G)$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:trivial-motions-rank-bound}
  Let $P$ be a partition of $V(G)$ with
  $\operatorname{def}_{\tilde G}(P) = \operatorname{def}(\tilde G)$, and
  let $P'$ be the partition of $B$ whose parts are those of $P$ and the
  singletons $\{b\}$ for $b \in B \setminus V(G)$. No edge meets these
  singletons, so exactly $d_G(P)$ edges join different parts of $P'$.
  The count in the proof of \cref{lem:trivial-motions-rank-bound},
  applied to $P'$, gives
  \[
    \dim Z(G,p) \;\ge\; D|P'| - (D-1)\,d_G(P) \;=\;
    D\,|B \setminus V(G)| + D + \operatorname{def}_{\tilde G}(P).
  \]
  The rank form follows, as
  $\operatorname{rank} R(G,p) = D|B| - \dim Z(G,p)$.
\end{proof}

\subsection{Rank lemmas}
\label{sec:molecular-rigidity-matrix-rank}

This subsection proves three rank lemmas of Katoh and
Tanigawa~\cite[\S5]{katohTanigawa2011}. Deleting the columns of one
body does not change the rank of $R(G,p)$ (KT Lemma~5.1). Two edges
between the same two bodies, with linearly independent supporting
extensors, force the two bodies to have the same screw center in every
infinitesimal motion (KT Lemma~5.3). The last lemma concerns
panel-hinge frameworks, in which the hinges at each body lie in a
common hyperplane, the \emph{panel} of the body
(\cref{sec:body-hinge-tay}): the rank does not drop when a panel
rotates into a generic position (KT Lemma~5.2).

\begin{lemma}[Deleting one body's columns preserves rank; KT Lemma 5.1]
  \label{lem:rank-delete-vertex}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.trivialMotions_sup_pinnedMotions,
    CombinatorialRigidity.Molecular.BodyHingeFramework.trivialMotions_inf_pinnedMotions_eq_bot,
    CombinatorialRigidity.Molecular.BodyHingeFramework.finrank_pinnedMotions_add_screwDim}
  \leanok
  \uses{def:rigidity-matrix, lem:trivial-motions-rank-bound}
  Let $(G,p)$ be a body-hinge framework, let $v$ be a body, and let
  $\operatorname{pinned}(v) = \{S \in Z(G,p) : S(v) = 0\}$. Then
  $Z(G,p)$ is the direct sum of the space of trivial motions and
  $\operatorname{pinned}(v)$, so
  $\dim \operatorname{pinned}(v) + D = \dim Z(G,p)$. Equivalently,
  deleting the $D$ columns of $v$ from $R(G,p)$ does not change its
  rank.
\end{lemma}
\begin{proof}
  \leanok
  A trivial motion that vanishes at $v$ vanishes everywhere. An
  infinitesimal motion $S$ is the sum of the trivial motion with value
  $S(v)$ and the screw assignment $u \mapsto S(u) - S(v)$, which
  vanishes at $v$. The latter is again an infinitesimal motion, since it
  has the same differences $S(u) - S(w)$ as $S$. So $Z(G,p)$ is the
  direct sum, and the trivial motions have dimension $D$
  (\cref{lem:trivial-motions-rank-bound}).

  Deleting the columns of $v$ leaves the matrix whose null space
  consists of the restrictions to $V \setminus \{v\}$ of the motions in
  $\operatorname{pinned}(v)$. Its rank is therefore
  $D(|V|-1) - \dim \operatorname{pinned}(v) = D|V| - \dim Z(G,p)$, the
  rank of $R(G,p)$.
\end{proof}

Katoh and Tanigawa prove the lemma by adding to $R(G,p)$ the $D$ rows
that tie down the body $v$~\cite[p.~668]{katohTanigawa2011}. They cite
White and Whiteley~\cite[\S2]{whiteWhiteley1987} for the fact that
fixing one body does not change the space of nontrivial motions. The
direct sum above expresses that fact.

\begin{lemma}[Two hinges of parallel edges give the full block; KT Lemma 5.3]
  \label{lem:rank-parallel-full}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.span_inf_span_eq_bot_of_linearIndependent,
    CombinatorialRigidity.Molecular.BodyHingeFramework.eq_of_hingeConstraint_two_parallel}
  \leanok
  \uses{def:rigidity-matrix, lem:extensor-independence}
  Let $(G,p)$ be a body-hinge framework, and let $e$ and $f$ be edges of
  $G$ that both join the bodies $u \neq v$, with $C(p(e))$ and
  $C(p(f))$ linearly independent. Then every screw assignment $S$ that
  satisfies the hinge constraints at $e$ and $f$ has $S(u) = S(v)$.
  Equivalently, the rows of $R(G,p)$ at $e$ and $f$, restricted to the
  $D$ columns of $v$, have rank $D$.
\end{lemma}
\begin{proof}
  \leanok
  The difference $S(u) - S(v)$ lies in
  $\operatorname{span} C(p(e)) \cap \operatorname{span} C(p(f))$
  (\cref{def:hinge-constraint}). Two lines spanned by linearly
  independent vectors meet only in $0$, so $S(u) = S(v)$. In the
  columns of $v$, the rows at $e$ and $f$ are $\pm r(p(e))$ and
  $\pm r(p(f))$. Their common null space in $K^D$ is the same
  intersection (\cref{def:hinge-row-block}), so they have rank $D$.
\end{proof}

Katoh and Tanigawa state the lemma for the graph with the two bodies
$u$ and $v$ and the two edges $e$ and $f$, with distinct hinges
$p(e) \neq p(f)$, and their proof shows that distinct hinges satisfy
the hypothesis above~\cite[p.~670]{katohTanigawa2011}. By the remark
after \cref{def:affine-subspace-extensor}, $C(p(e))$ is nonzero, it is
determined by $p(e)$ up to a nonzero scalar, and it determines $p(e)$.
So $C(p(e))$ and $C(p(f))$ are proportional only if $p(e) = p(f)$, and
two nonzero vectors that are not proportional are linearly
independent. For example, two hinges spanned by $d - 1$ of $d + 1$
affinely independent points, omitting different pairs of them, have
linearly independent extensors by KT Lemma~2.1
(\cref{lem:extensor-independence}). Distinct hinges can lie in a common
hyperplane, so the two-body graph has an infinitesimally rigid
panel-hinge realization. This graph is the base case of the algebraic
induction (\cref{lem:theorem-55-base}).

\begin{lemma}[Lower-semicontinuity of rank under a panel rotation; KT Lemma 5.2]
  \label{lem:rank-rotation-semicont}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.infinitesimalMotions_mono_of_span_le,
    CombinatorialRigidity.Molecular.BodyHingeFramework.finrank_infinitesimalMotions_le_of_span_le}
  \leanok
  \uses{def:rigidity-matrix, def:dof-generic}
  Along a one-parameter family of realizations of $G$ in which a single
  panel rotates, the rank of $R(G,p)$ is lower semicontinuous: its
  value at a generic parameter is at least its value at every member of
  the family. In monotone
  form, let $p$ and $p'$ be realizations of $G$ with
  $\operatorname{span} C(p(e)) \subseteq \operatorname{span} C(p'(e))$
  at every edge $e$. Then $Z(G,p') \subseteq Z(G,p)$, so
  $\dim Z(G,p') \le \dim Z(G,p)$ and
  $\operatorname{rank} R(G,p) \le \operatorname{rank} R(G,p')$.
\end{lemma}
\begin{proof}
  \leanok
  The entries of $R(G,p)$ can be written as polynomials in the
  coordinates of the panels, as Katoh and
  Tanigawa~\cite[p.~668]{katohTanigawa2011} note with a reference to
  Jackson and Jord\'an~\cite[Lemma~7.1]{jacksonJordan2009}. Along the
  family they are polynomials in the parameter $t$. A $\rho \times \rho$
  minor that is nonzero at one parameter is then a nonzero polynomial in
  $t$, so it vanishes at only finitely many parameters, and at every
  other parameter the rank is at least $\rho$. Katoh and Tanigawa argue by
  continuity over $\R$ instead~\cite[p.~669]{katohTanigawa2011}; the
  polynomial argument works over every field.

  In the monotone form, $C(p(e))$ and $C(p'(e))$ are nonzero, as the
  hinges are $(d-2)$-dimensional, so the hypothesis says that
  $\operatorname{span} C(p(e)) = \operatorname{span} C(p'(e))$ at every
  edge. The two realizations then have the same hinge constraints, so
  $Z(G,p') = Z(G,p)$, and their ranks are equal.
\end{proof}
\begin{fmlnote}
  The formalization proves only the monotone form, for frameworks in
  which $C(p(e))$ may be zero (\cref{def:hinge-constraint}), and with
  the inclusion of spans reversed. If
  $\operatorname{span} C(p'(e)) \subseteq \operatorname{span} C(p(e))$
  at every edge, then each hinge constraint of $(G,p')$ implies the
  corresponding one of $(G,p)$, so $Z(G,p') \subseteq Z(G,p)$ and
  $\dim Z(G,p') \le \dim Z(G,p)$. When all the extensors are nonzero,
  either inclusion of spans says that the spans are equal.
\end{fmlnote}

\subsection{Block-rank-addition lemmas for the case analyses}
\label{sec:molecular-rigidity-matrix-blocks}

The algebraic induction's structural cases each reduce a rank count on
$G$ to rank counts on smaller pieces plus a block correction.
\Cref{lem:block-rank-cut} is the vertex-cut form (KT Lemma~6.1); the
remaining lemmas cover the two body-sharing geometries a vertex cut
cannot express --- \emph{collapse}, where a rigid subgraph $H$ and its
contraction $G/E(H)$ share one body (KT Lemma~6.2,
\cref{lem:rigidityRows-splice-rank-add}), and \emph{pin-a-body}, where a
new block of rows and an old block share a body's screw column (KT
Lemmas~6.8 and~6.10, \cref{lem:rigidityRows-pinned-placement-rank-add}).
The supporting lemmas discharge the two geometries' respective
hypotheses: injectivity of the collapse's column-deletion map
(\cref{lem:extProj-preserves-rank-of-inter}), rank invariance under equal
motion spaces (\cref{lem:span-rigidityRows-eq-of-motions-eq}), and the
deficiency-aware rank transport a collapsed leg needs when its
contraction is not yet fully rigid
(\cref{lem:rank-transport-relabel-of-le-finrank} and
\cref{lem:rank-polynomial-IH-relabel-proj}). Two sides that share a pair
of bodies $u, v$ rather than one are glued through the relative screws of
the pair (\cref{def:relative-screws}): the ranks add up to a correction
by the joined relative screws, for any two graphs sharing out the edges
(\cref{lem:block-rank-two-cut}) and in particular for the two sides of a
vertex $2$-cut (\cref{cor:block-rank-vertex-two-cut}). A path between
the two bodies has an explicit rank (\cref{lem:block-rank-path}) and
relative screws (\cref{lem:relative-screws-path}), which give the rank
added by an open ear (\cref{lem:block-rank-ear}); these serve the ear
steps on pencil configurations (\cref{sec:main-component-chain}). The
block-triangular
bound at a core (\cref{lem:block-rank-contract}) and the projected rank
polynomial nonzero at its witness (\cref{lem:rank-polynomial-proj-eval})
serve the contraction step (\cref{sec:main-component-contract}).

\begin{lemma}[Block-rank addition for a vertex cut; KT Lemma 6.1]
  \label{lem:block-rank-cut}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.le_finrank_span_rigidityRows_of_cut,
    CombinatorialRigidity.Molecular.BodyHingeFramework.add_le_finrank_span_rigidityRows_induce_union_singleton}
  \leanok
  \uses{def:rigidity-matrix, def:hinge-row-block}
  Let $(G,p)$ be a body-hinge framework with all nonzero supporting extensors,
  and let $V_1 \subsetneq V(G)$ be a nonempty proper vertex subset with cut
  $C = \operatorname{cutEdges}(G,V_1)$ satisfying $|C| \le 1$.  Write
  $V_2 = V(G) \setminus V_1$ and let $F_i = (G[V_i],p)$ be the induced
  subframeworks.  Then
  \[
    \operatorname{rank} R(G[V_1],p) + (D-1)\,|C|
    + \operatorname{rank} R(G[V_2],p)
    \;\le\;
    \operatorname{rank} R(G,p).
  \]
  This is the vertex-disjoint block-rank-addition lower bound for a
  cut-edge decomposition; it replaces the isometry argument of
  Katoh--Tanigawa's Lemma~6.1~\cite{katohTanigawa2011} with an argument
  via three mutually disjoint subspaces of the rigidity-row span at a
  fixed placement. In particular, if exactly one edge joins a vertex set
  $V_1$ to a body $w \notin V_1$, then
  $\operatorname{rank} R(G[V_1],p) + (D-1) \le
  \operatorname{rank} R(G[V_1 \cup \{w\}],p)$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{def:rigidity-matrix, def:hinge-row-block}
  When $|C| = 0$ the two sides are disconnected: $S_1, S_2 \le S$ and
  $S_1 \cap S_2 = \bot$ (a flow-sum argument separates them), giving
  $\operatorname{finrank}(S_1) + \operatorname{finrank}(S_2)
  \le \operatorname{finrank}(S)$.
  When $|C| = 1$ the single cut edge $e_c$ contributes a $(D-1)$-dimensional
  subspace $S_c$ (from the hinge-row block of $e_c$, nonzero by hypothesis);
  the three subspaces $S_1, S_c, S_2$ are pairwise disjoint, and a
  calc chain of disjoint-rank additions closes the inequality.
  For the particular case, apply the bound to $G[V_1 \cup \{w\}]$ at
  $V_1$: its only crossing edge is the one joining $w$, its $V_1$ side is
  $G[V_1]$, and the rank of the single body $w$ is dropped.
\end{proof}

\begin{lemma}[Rank addition at a cut vertex]
  \label{lem:block-rank-cut-vertex}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.finrank_span_rigidityRows_cutVertex_eq}
  \leanok
  \uses{def:rigidity-matrix, def:hinge-constraint}
  Let $(G,p)$ be a body-hinge framework and let $V_1, V_2$ be vertex sets
  with $V_1 \cap V_2 \subseteq \{v\}$ and every edge of $G$ inside $V_1$
  or inside $V_2$. Then
  \[
    \operatorname{rank} R(G,p)
    = \operatorname{rank} R(G[V_1],p) + \operatorname{rank} R(G[V_2],p).
  \]
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:span-rigidityRows-eq-of-motions-eq}
  The row set of $R(G,p)$ is the union of the two sides' row sets. The
  two row spans meet in $0$, because the two motion spaces together span
  every screw assignment $S$: the assignment equal to $S_v$ on $V_1$ and
  to $S$ elsewhere is a motion of the first side, and the rest vanishes
  on $V_2$, so it is a motion of the second. A row span is the
  annihilator of its motions, so the dimensions add.
\end{proof}

\begin{definition}[Relative screws and joint rows]
  \label{def:relative-screws}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.relScrews,
    CombinatorialRigidity.Molecular.BodyHingeFramework.jointRows}
  \leanok
  \uses{def:rigidity-matrix, def:hinge-row-block}
  Let $(G,p)$ be a body-hinge framework and $u, v$ two bodies. The
  \emph{relative screws} of $u, v$ in $(G,p)$ form the subspace
  \[
    \rho_{uv}(G,p) = \{\, S(v) - S(u) : S \in Z(G,p) \,\} \subseteq K^{D}.
  \]
  For a subspace $U \subseteq K^{D}$, the \emph{joint rows} at $u, v$
  confining the relative screw to $U$ are the functionals
  $S \mapsto r\bigl(S(u) - S(v)\bigr)$ on screw assignments, one for each
  $r$ in the annihilator $U^{\circ} \subseteq (K^{D})^{*}$.
\end{definition}

A screw assignment is annihilated by every joint row exactly when
$S(u) - S(v) \in U$, by the double-annihilator identity, as for the
hinge-row block (\cref{def:hinge-row-block}); at $U = 0$ the joint welds
$u$ and $v$ rigidly together. At distinct bodies the map
$S \mapsto S(u) - S(v)$ is onto, so the joint rows span a space of
dimension $D - \dim U$.

\begin{lemma}[Rank gluing at a $2$-cut]
  \label{lem:block-rank-two-cut}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.inf_span_rigidityRows_of_twoCut_of_isLink,
    CombinatorialRigidity.Molecular.BodyHingeFramework.finrank_span_rigidityRows_twoCut_eq_of_isLink}
  \leanok
  \uses{def:relative-screws, def:rigidity-matrix}
  Let $(G,p)$ be a body-hinge framework and $u \neq v$ two bodies. Let
  $G_1, G_2$ be graphs whose edges, with their ends, are together exactly
  the edges of $G$, such that every edge of $G_i$ lies inside a vertex
  set $V_i$, where $V_1 \cap V_2 \subseteq \{u, v\}$; give each $G_i$ the
  hinges of $p$, and write $\rho_i$ for the relative screws of $u, v$ in
  $(G_i, p)$ (\cref{def:relative-screws}). Then the row spaces of
  $R(G_1,p)$ and $R(G_2,p)$ meet in the span of the joint rows at $u, v$
  confining the relative screw to $\rho_1 + \rho_2$, and
  \[
    \operatorname{rank} R(G,p) = \operatorname{rank} R(G_1,p)
    + \operatorname{rank} R(G_2,p) + \dim(\rho_1 + \rho_2) - D.
  \]
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:span-rigidityRows-eq-of-motions-eq}
  A row space is the annihilator of the corresponding motion space
  (\cref{lem:span-rigidityRows-eq-of-motions-eq}), so a functional lies
  in both row spaces exactly when it annihilates
  $Z(G_1,p) + Z(G_2,p)$. That sum contains every screw assignment $S$
  with $S(u) = S(v)$: the assignment equal to $S(v)$ on $V_2$ and to $S$
  elsewhere is a motion of $G_2$, and the difference vanishes on $V_1$
  (on $V_1 \cap V_2 \subseteq \{u, v\}$ because $S(u) = S(v)$), so it is
  a motion of $G_1$. A subspace containing the kernel of
  $S \mapsto S(u) - S(v)$ has as annihilator the functionals
  $S \mapsto r(S(u) - S(v))$ with $r$ annihilating its image, here
  $\rho_1 + \rho_2$: the span of the joint rows, of dimension
  $D - \dim(\rho_1 + \rho_2)$ since $u \neq v$. The row space of
  $R(G,p)$ is the sum of the two row spaces, and the dimension formula
  for a sum and an intersection gives the rank. No non-adjacency of $u$
  and $v$ is needed: an edge $uv$ belongs to $G_1$ or to $G_2$, and its
  rows are the same on either side.
\end{proof}

\begin{corollary}[Rank gluing at a vertex $2$-cut]
  \label{cor:block-rank-vertex-two-cut}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.inf_span_rigidityRows_of_vertexTwoCut,
    CombinatorialRigidity.Molecular.BodyHingeFramework.finrank_span_rigidityRows_vertexTwoCut_eq}
  \leanok
  \uses{lem:block-rank-two-cut}
  Let $(G,p)$ be a body-hinge framework, $u \neq v$ two bodies, and
  $V_1, V_2$ sets of bodies with $V_1 \cap V_2 = \{u, v\}$ such that
  every edge of $G$ lies inside $V_1$ or inside $V_2$. Write $\rho_i$ for
  the relative screws of $u, v$ in $(G[V_i], p)$. Then the row spaces of
  $R(G[V_1],p)$ and $R(G[V_2],p)$ meet in the span of the joint rows at
  $u, v$ confining the relative screw to $\rho_1 + \rho_2$, and
  \[
    \operatorname{rank} R(G,p) = \operatorname{rank} R(G[V_1],p)
    + \operatorname{rank} R(G[V_2],p) + \dim(\rho_1 + \rho_2) - D.
  \]
\end{corollary}
\begin{proof}
  \leanok
  Apply \cref{lem:block-rank-two-cut} to $G_i = G[V_i]$: the edges of
  $G[V_i]$ lie inside $V_i$ by definition, and together they are the
  edges of $G$.
\end{proof}

\begin{lemma}[The rank of a path]
  \label{lem:block-rank-path}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.le_finrank_span_rigidityRows_path,
    CombinatorialRigidity.Molecular.BodyHingeFramework.finrank_span_rigidityRows_path_le}
  \leanok
  \uses{def:rigidity-matrix}
  Let $(P,p)$ be a body-hinge framework whose edges are those of a path
  $a = x_0 - x_1 - \cdots - x_k - x_{k+1} = b$ of distinct bodies
  ($k \geq 0$), and let the hinge $C_i$ of the edge $x_i x_{i+1}$ be
  nonzero for every $i$. Then
  $\operatorname{rank} R(P,p) = (D-1)(k+1)$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:block-rank-cut, def:hinge-row-block}
  Each edge contributes at most $D - 1$ independent rows
  (\cref{def:hinge-row-block}). Conversely, the body $x_{i+1}$ hangs
  from $\{x_0, \dots, x_i\}$ by the single edge $x_i x_{i+1}$, so
  \cref{lem:block-rank-cut} adds $D - 1$ to the rank for each of the
  $k + 1$ edges.
\end{proof}

\begin{lemma}[The relative screws of a path]
  \label{lem:relative-screws-path}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.span_range_le_relScrews_path,
    CombinatorialRigidity.Molecular.BodyHingeFramework.relScrews_path_le_span_range}
  \leanok
  \uses{def:relative-screws}
  Let $(P,p)$ be a body-hinge framework whose edges are those of a path
  $a = x_0 - x_1 - \cdots - x_k - x_{k+1} = b$ of distinct bodies
  ($k \geq 0$), with hinge $C_i$ at the edge $x_i x_{i+1}$. Then the
  relative screws of $a, b$ in $(P,p)$ are
  $\operatorname{span}(C_0, \dots, C_k)$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{def:hinge-constraint}
  At a motion $S$, each $S(x_{i+1}) - S(x_i)$ lies in
  $\operatorname{span} C_i$, and these differences add up to
  $S(b) - S(a)$. Conversely, the assignment equal to $C_i$ at
  $x_{i+1}, \dots, x_{k+1}$ and to $0$ elsewhere is a motion with
  relative screw $C_i$.
\end{proof}

\begin{lemma}[Rank addition along an open ear]
  \label{lem:block-rank-ear}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.finrank_span_rigidityRows_ear_eq}
  \leanok
  \uses{def:relative-screws, def:rigidity-matrix}
  Let $(G,p)$ be a body-hinge framework and $V_1$ a set of bodies, and
  let $a - x_1 - \cdots - x_k - b$ ($k \geq 0$) be a path of $G$ with
  $a \neq b$ in $V_1$ and distinct bodies $x_1, \dots, x_k$ outside
  $V_1$, such that every edge of $G$ not on the path lies inside $V_1$.
  Let the hinges $C_0, \dots, C_k$ of the path be nonzero, let
  $\Lambda = \operatorname{span}(C_0, \dots, C_k)$, and let $\rho$ be the
  relative screws of $a, b$ in $(G[V_1], p)$. Then
  \[
    \operatorname{rank} R(G,p) = \operatorname{rank} R(G[V_1],p)
    + (D - 1)(k + 1) + \dim(\rho + \Lambda) - D.
  \]
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:block-rank-two-cut, lem:block-rank-path, lem:relative-screws-path}
  Apply \cref{lem:block-rank-two-cut} to $G_1 = G[V_1]$ and to the path
  $G_2$, with $V_2$ its bodies: $V_1 \cap V_2 = \{a, b\}$, and every edge
  of $G$ is on the path or inside $V_1$. The path has rank $(D-1)(k+1)$
  (\cref{lem:block-rank-path}) and relative screws $\Lambda$
  (\cref{lem:relative-screws-path}). The second side is the path itself,
  not the subgraph induced on its bodies, which also contains the edge
  $ab$ when $a$ and $b$ are adjacent; so the formula holds whether or not
  they are.
\end{proof}

\begin{lemma}[The block-triangular rank bound at a core; KT eq.~(6.3)]
  \label{lem:block-rank-contract}
  \lean{CombinatorialRigidity.Molecular.PanelHingeFramework.finrank_span_rigidityRows_induce_add_map_extProj_le}
  \leanok
  \uses{def:rigidity-matrix, def:panel-hinge-framework}
  Let $(G, p)$ be a panel-hinge framework, $W$ a set of bodies, and
  $H = G[W]$. Let $R_W(G \setminus E(H), p)$ be the rigidity matrix of
  the edges of $G$ not inside $W$ with the columns of the bodies of $W$
  deleted. Then
  \[
    \operatorname{rank} R(H, p) + \operatorname{rank} R_W(G \setminus E(H), p)
    \;\le\; \operatorname{rank} R(G, p).
  \]
\end{lemma}
\begin{proof}
  \leanok
  The rows of $H$ read only the columns of $W$, so deleting those columns
  kills them. Rank-nullity for the column deletion, on the span of the rows
  of $H$ and of the other edges, which lies in the row span of $G$, gives
  the bound. This is the rank bound Katoh and Tanigawa read off the
  block-triangular shape of $R(G, p)$~\cite[eq.~(6.3), pp.~673--674]{katohTanigawa2011}.
\end{proof}

\begin{lemma}[Block-triangular rank-addition for a shared-body splice; KT Lemma 6.2]
  \label{lem:rigidityRows-splice-rank-add}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.le_finrank_span_rigidityRows_of_splice}
  \leanok
  \uses{def:rigidity-matrix, def:hinge-row-block}
  Let $F$, $F_H$, $F_c$ be body-hinge frameworks on the same bodies and hinges.
  Let $S = \operatorname{span}(R(F))$, $S_H = \operatorname{span}(R(F_H))$,
  $S_c = \operatorname{span}(R(F_c))$ be the corresponding rigidity-row spans.
  Let $D$ be a linear endomorphism of the row dual space (in practice, the
  dual of the projection that collapses the contracted body; see
  \cref{sec:molecular-algebraic-induction-caseI}). Suppose:
  \begin{enumerate}
    \item $S_H \le S$ and $S_H \le \ker D$ (the hinge block lies in $S$ and is killed by $D$);
    \item $D(S_c) \le D(S)$ (the contraction block maps into the image of $S$);
    \item $D$ is injective on $S_c$ (so $\operatorname{rank}(S_c) = \operatorname{rank}(D(S_c))$).
  \end{enumerate}
  Then
  \[
    \operatorname{rank}(S_H) + \operatorname{rank}(S_c) \;\le\; \operatorname{rank}(S).
  \]
\end{lemma}
\begin{proof}
  \leanok
  \uses{def:rigidity-matrix, def:hinge-row-block}
  Rank-nullity applied to $D$ restricted to $S$ gives
  $\operatorname{rank}(D(S)) + \operatorname{rank}(S \cap \ker D) = \operatorname{rank}(S)$.
  By hypothesis (1), $S_H \le S \cap \ker D$, so
  $\operatorname{rank}(S_H) \le \operatorname{rank}(S \cap \ker D)$.
  By hypotheses (2) and (3),
  $\operatorname{rank}(S_c) = \operatorname{rank}(D(S_c)) \le \operatorname{rank}(D(S))$.
  Adding the two inequalities and using rank-nullity gives the bound.
\end{proof}

\begin{lemma}[Span-level pinned-placement block-rank lower bound; KT Lemmas 6.8 and 6.10]
  \label{lem:rigidityRows-pinned-placement-rank-add}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.le_finrank_span_rigidityRows_of_pinned_placement,
    CombinatorialRigidity.Molecular.BodyHingeFramework.le_finrank_span_rigidityRows_of_pinned_placement_augment}
  \leanok
  \uses{def:rigidity-matrix, lem:case-II-placement-block-independent, lem:case-III-conditional-block}
  The span-transport analogue of \cref{lem:rigidityRows-splice-rank-add}
  for the \emph{pin-a-body} (vertex-splitting) geometry, as opposed to
  that lemma's \emph{collapse} (projected-column) geometry --- Katoh--Tanigawa's
  own Lemma~6.2 (contraction) versus Lemma~6.8 (splitting) distinction.
  Let $F$ be a body-hinge framework, $v$ a body, and write $S =
  \operatorname{span}(R(F))$ for its rigidity-row span. Let $\{r_n^i\}_{i \in \iota_n}$
  be a \emph{new block} of row functionals and $\{r_o^j\}_{j \in \iota_o}$ an \emph{old
  block}, with:
  \begin{enumerate}
    \item $r_n$ is independent when restricted to screw assignments
      supported at $v$ alone;
    \item $r_o$ vanishes on every screw assignment supported at $v$
      alone;
    \item $r_o$ is independent;
    \item every $r_n^i \in S$ and every $r_o^j \in S$.
  \end{enumerate}
  Then $|\iota_n| + |\iota_o| \le \operatorname{rank}(S)$. The $+1$ augment variant adds one
  further candidate row $w \in S$ to the new block, independent through $v$'s column alongside
  $r_n$, and concludes $|\iota_n| + 1 + |\iota_o| \le \operatorname{rank}(S)$ --- the extra
  row supplying the lift from $D(|V|-1)-1$ to the full $D(|V|-1)$ that
  Case~III needs (KT eq.~(6.29)).
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:case-II-placement-block-independent, lem:case-III-conditional-block}
  The block-triangular pin-a-body split
  (\cref{lem:case-II-placement-block-independent}; for the augment,
  \cref{lem:case-III-conditional-block}) makes the family obtained by
  concatenating $r_n$ and $r_o$ linearly independent. Its span lies in
  $S$ by hypothesis~(4), so its dimension equals the block-size sum and
  is bounded by $\operatorname{rank}(S)$. This is the rank skeleton
  shared by KT Lemma~6.8 and Lemma~6.10's arguments, abstracted away
  from any specific construction: no new linear algebra is needed once
  hypothesis~(4) is discharged, which the callers do differently (the
  split-off $e_0 = e_a + e_b$ row decomposition for Lemma~6.8, the
  relabelled hinge-row span for Case~III). Unlike a literal placement
  bound, the conclusion keys on \emph{membership in} $S$ rather than
  literal rigidity-row membership, so it applies to every reduction
  move --- collapse, split-off, or relabel --- whose rows land only in
  $S$. See \cite[\S6.3]{katohTanigawa2011}.
\end{proof}

\begin{lemma}[The exterior-column projection preserves the rigidity-row rank at any rank]
  \label{lem:extProj-preserves-rank-of-inter}
  \lean{CombinatorialRigidity.Molecular.infinitesimalMotions_sup_range_extProj_eq_top_of_inter_eq_singleton,
    CombinatorialRigidity.Molecular.BodyHingeFramework.injOn_extProj_dualMap_rigidityRows_of_inter_eq_singleton,
    CombinatorialRigidity.Molecular.BodyHingeFramework.finrank_span_rigidityRows_map_extProj_dualMap_of_inter_eq_singleton}
  \leanok
  \uses{def:rigidity-matrix, lem:rank-delete-vertex, lem:claim-6-4}
  Let $F$ be a body-hinge framework and $t$ a body set meeting the vertex
  set in exactly one body, $V(F.\mathrm{graph}) \cap t = \{r\}$.  Let
  $D = (\operatorname{extProj} t)^{\!*}$ be the dual of the projection that
  zeroes the $t$-columns.  Then $D$ is injective on the rigidity-row span
  $S = \operatorname{span}(R(F))$, so the projection preserves its rank,
  $\operatorname{rank}(S) = \operatorname{rank}(D(S))$.  This is the
  injectivity hypothesis (3) of \cref{lem:rigidityRows-splice-rank-add},
  discharged at a \emph{deficient} contraction leg where the
  full-rigidity-gated form (\cref{lem:claim-6-4}) is unavailable.
\end{lemma}
\begin{proof}
  \leanok
  \uses{def:rigidity-matrix, lem:rank-delete-vertex}
  Injectivity on $S$ is $S \cap \ker D = \bot$, equivalently
  $Z \sqcup W = \top$ for $Z$ the infinitesimal motions and
  $W = \operatorname{range}(\operatorname{extProj} t)$ the screw
  assignments vanishing on $t$ (by the dual-annihilator calculus:
  $\ker D = W^\circ$, $S = Z^\circ$, so $S \cap \ker D = (Z \sqcup W)^\circ$).
  Unlike the rigid form, $Z \sqcup W = \top$ holds at \emph{any} rank by an
  explicit decomposition needing no rank count: any assignment $\sigma$
  splits as $z + (\sigma - z)$ with $z(a) = \sigma(r)$ for $a \in V(G)$ and
  $z(a) = \sigma(a)$ otherwise; $z$ is a motion (every constraint has both
  endpoints in $V(G)$, where $z$ is the constant $\sigma(r)$, so the
  relative screw is $0$), and $\sigma - z$ vanishes on $t = \{r\} \cup
  (t \setminus V(G))$ since $V(G) \cap t = \{r\}$.  This is the algebraic
  content KT~\cite{katohTanigawa2011} invokes through Lemma~5.1
  (\cref{lem:rank-delete-vertex}): deleting the single shared body $r$'s
  columns preserves rank, the only motion lost to the projection being the
  trivial $\sigma(r)$-shift recovered by $z$.  Rank-nullity for $D$
  restricted to $S$ then gives $\operatorname{rank}(D(S)) = \operatorname{rank}(S)$.
\end{proof}

\begin{lemma}[Rigidity-row span is the motions' annihilator; equal motions give equal rank]
  \label{lem:span-rigidityRows-eq-of-motions-eq}
  \lean{CombinatorialRigidity.Molecular.BodyHingeFramework.span_rigidityRows_eq_dualAnnihilator_infinitesimalMotions,
    CombinatorialRigidity.Molecular.BodyHingeFramework.span_rigidityRows_eq_of_infinitesimalMotions_eq}
  \leanok
  \uses{def:rigidity-matrix}
  Over a finite body set, the rigidity-row span of a body-hinge
  framework is the dual annihilator of its infinitesimal-motion space,
  $\operatorname{span}(R(F)) = Z(F)^\circ$.  Consequently two frameworks
  with the same motion space have the same rigidity-row span --- hence
  the same rank --- at any deficiency, with no rigidity hypothesis.
\end{lemma}
\begin{proof}
  \leanok
  \uses{def:rigidity-matrix}
  The motion space is the dual coannihilator of the row span
  (\texttt{infinitesimalMotions\_eq\_dualCoannihilator}); over a finite
  body set the screw-assignment dual is finite-dimensional, so the
  double-annihilator identity closes
  $\operatorname{span}(R(F)) = Z(F)^\circ$.  Equal motion spaces then
  give equal annihilators, hence equal spans and equal finrank.
\end{proof}

\begin{lemma}[Deficiency-aware rank transport across a collapse relabel]
  \label{lem:rank-transport-relabel-of-le-finrank}
  \lean{CombinatorialRigidity.Molecular.PanelHingeFramework.finrank_span_rigidityRows_ofNormals_relabel_eq}
  \leanok
  \uses{def:rigidity-matrix, lem:span-rigidityRows-eq-of-motions-eq, def:genuine-hinge-realization}
  Let $H \le G$ be a subgraph with representative body $r \in V(H)$, let
  $G_c := G \setminus E(H)$ be the surviving-edge subgraph, and let $f
  \colon V(G) \to V(G)$ collapse every body of $V(H)$ onto $r$ and fix
  every other body, so that $f(G_c) = G/E(H)$ is the contracted graph.
  Given the contraction's inductive realization --- a general-position
  panel framework on $f(G_c)$ attaining the rigidity-row rank
  $D(|V(f(G_c))|-1) - \operatorname{def}(f(G_c))$, at any deficiency ---
  and a parent selector recording $G_c$'s links, the free-normal panel
  framework on $f(G_c)$ at the \emph{relabel selector} $e \mapsto
  (f(\text{ends}(e)_1), f(\text{ends}(e)_2))$ is again in general
  position and attains \emph{the same} rigidity-row rank. This is the
  deficiency-tolerant sibling of the rigid contraction-leg transport
  (\cref{lem:claim-6-4}): it transports the contraction's
  possibly-deficient rank ($\operatorname{def} = k > 0$) across the
  relabel, where the full-rigidity argument does not apply.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:span-rigidityRows-eq-of-motions-eq}
  The relabel selector and the inductive realization's own selector
  record the same unordered link of every edge of $f(G_c)$, so they
  agree up to a swap of endpoints; since the hinge constraint is
  membership in the (sign-insensitive) line through the supporting
  extensor, swapping endpoints fixes every constraint, hence the whole
  motion space.  Equal motion spaces give equal rigidity-row rank
  (\cref{lem:span-rigidityRows-eq-of-motions-eq}); general position is a
  property of the normals alone, unchanged by the selector.
\end{proof}

\begin{lemma}[Deficiency-aware rank polynomial for the relabeled contraction leg (superseded)]
  \label{lem:rank-polynomial-IH-relabel}
  \lean{CombinatorialRigidity.Molecular.PanelHingeFramework.exists_rankPolynomial_of_IH_relabel_linking}
  \leanok
  \uses{lem:rank-transport-relabel-of-le-finrank, lem:rank-polynomial-of-le-finrank}
  \emph{Superseded; no live consumer.} This lemma produces the
  contraction framework's \emph{full-span} surviving rank --- the input
  \cref{lem:rigidityRows-splice-rank-add} was designed to consume. The
  realization this was built for cannot use that lemma directly, because
  its surviving-edge support extensors are not in general position for a
  free-normal panel framework; the working argument instead reads the
  \emph{exterior-projected} surviving rows
  (\cref{lem:rank-polynomial-IH-relabel-proj}). The statement below is
  true and its proof's shared step (\cref{lem:rank-transport-relabel-of-le-finrank})
  remains in use; the lemma itself is retained for the record.

  Under the setup of \cref{lem:rank-transport-relabel-of-le-finrank} (with $f(G_c)$ loopless),
  there exists a natural number $N$ with
  $(N : \mathbb{Z}) = D(|V(f(G_c))| - 1) - \operatorname{def}(f(G_c))$
  and a nonzero polynomial $Q$ over the body-normal parameters of the original graph
  such that at every $Q$-non-root seed $q$, the rigidity-row span of
  $\operatorname{ofNormals}(f(G_c), e \mapsto (f(\text{ends}(e)_1), f(\text{ends}(e)_2)), q)$
  has rank at least $N$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:rank-transport-relabel-of-le-finrank, lem:rank-polynomial-of-le-finrank}
  Apply \cref{lem:rank-transport-relabel-of-le-finrank} to obtain a
  witness seed $q_0$ at which the rigidity-row rank equals $N$ exactly
  and the framework is in general position. Then apply the
  deficiency-aware rank polynomial (\cref{lem:rank-polynomial-of-le-finrank})
  at $q_0$ with the trivial bound $N \le N$: the support extensor is
  nonzero at every link of $f(G_c)$ because general position forces it
  and looplessness ensures distinct endpoints. The resulting polynomial
  $Q$ is nonzero (its evaluation at $q_0$ is nonzero), and
  it transfers the rank lower bound $N$ to every non-root seed.
\end{proof}

\begin{lemma}[Deficiency-aware exterior-projected rank polynomial for the surviving block]
  \label{lem:rank-polynomial-IH-relabel-proj}
  \lean{CombinatorialRigidity.Molecular.PanelHingeFramework.exists_rankPolynomial_of_IH_relabel_linking_set_proj}
  \leanok
  \uses{lem:rank-transport-relabel-of-le-finrank, lem:extProj-preserves-rank-of-inter, lem:rank-polynomial-of-le-finrank}
  Let $H \le G$ be a proper rigid subgraph with representative body
  $r \in V(H) \subseteq V(G)$, and let the contraction $G/E(H)$ be a minimal
  $k'$-dof-graph carrying its inductive general-position realization (the
  deficiency $k' > 0$ permitted).  Write $D = (\operatorname{extProj} V(H))^{\!*}$
  for the dual of the projection zeroing the $V(H)$-columns and
  $\mathrm{sc} = (V(G) \setminus V(H)) \cup \{r\}$ for the surviving body set.
  Then there is a nonzero polynomial $Q$ over the body-normal
  parameters of $G$ such that at every $Q$-non-root seed $q$, the
  $D$-projected surviving-edge panel rows of $\operatorname{ofNormals}(G
  \setminus E(H), \text{ends}, q)$ contain a linearly independent subfamily of
  size at least $D(|\mathrm{sc}|-1) - k'$.
  This is the surviving-leg rank input Case~I's block-triangular
  argument needs when the contraction leg is not yet fully rigid. It is
  the deficiency-aware twin of the rigid contraction-leg transport
  (\cref{lem:claim-6-4}): where the rigid form derives the surviving rank
  from rigidity and reads it off the un-relabeled rows, this transports
  the contraction's possibly-deficient rank across the collapse relabel,
  projecting away the rigid block's columns.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:rank-transport-relabel-of-le-finrank, lem:extProj-preserves-rank-of-inter, lem:rank-polynomial-of-le-finrank}
  \Cref{lem:rank-transport-relabel-of-le-finrank} supplies a
  general-position witness placement at which the relabel-leg
  framework's rigidity-row rank equals $N$ exactly, where
  $(N : \mathbb{Z}) = D(|\mathrm{sc}|-1) - k'$ (the contraction's vertex set is
  $\mathrm{sc}$ and its deficiency is $k'$). At that placement, the
  rigidity-free column-deletion injectivity of
  \cref{lem:extProj-preserves-rank-of-inter} extracts, from the rank $N$,
  a subfamily of relabel-leg links whose $D$-projected collapsed rows are
  independent, of size $\ge N$. The per-edge collapse correspondence
  transports that projected independence to the projected rows of the
  uncollapsed framework at the degenerate placement (KT's $p_2$), giving
  a single-placement witness, which \cref{lem:rank-polynomial-of-le-finrank}'s
  projected form then lifts to the $Q$-non-root rank polynomial.
\end{proof}

\begin{lemma}[Projected row independence persists near its witness]
  \label{lem:rank-polynomial-proj-eval}
  \lean{CombinatorialRigidity.Molecular.PanelHingeFramework.exists_rankPolynomial_of_rigidOn_linking_set_proj_eval}
  \leanok
  \uses{lem:rank-polynomial-of-le-finrank, def:rigidity-matrix}
  Let $t$ be a set of bodies and $D = (\operatorname{extProj} t)^{\!*}$ the
  projection deleting the columns of $t$. Suppose that at panel normals
  $q_0$ a family of panel rows of $G$ at linking edges is linearly
  independent after $D$. Then there is a polynomial $Q$ in the panel
  normals with $Q(q_0) \neq 0$ such that at every $q$ with $Q(q) \neq 0$ a
  family of panel rows of $G$ at linking edges, of the same size, is
  linearly independent after $D$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:rank-polynomial-of-le-finrank}
  The coordinates of the projected panel rows are polynomials in the
  normals, and a maximal minor of the projected family that does not vanish
  at $q_0$ is the polynomial: the projected form of
  \cref{lem:rank-polynomial-of-le-finrank}, which keeps its value at the
  witness.
\end{proof}
