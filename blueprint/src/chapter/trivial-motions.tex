\section{Trivial infinitesimal motions}
\label{sec:trivial-motions}

The translations and rotations of $\R^{d}$ give infinitesimal motions
of every framework in $\R^{d}$, whatever the graph
(\cref{sec:frameworks}). In this section we define the space $T(p)$ of
these trivial motions at a framework $p$, and we show that it lies in
the kernel of the rigidity map $R(G, p)$ for every graph $G$. We then
show that $T(p)$ has dimension at least $d(d+1)/2$ when the points
$p_v$ affinely span $\R^{d}$, so that the kernel of $R(G, p)$ has at
least this dimension too. By rank--nullity, the rank of $R(G, p)$ is
then at most $d|V| - d(d+1)/2$. The proof of Laman's theorem uses this
bound in the plane, where $d(d+1)/2 = 3$
(\cref{lem:rigidityMap-finrank-range-le-of-affinelySpanning}).

We define the translations first, and then the infinitesimal
rotations, which apply a skew-symmetric linear map to every point. The
trivial motions are the linear combinations of the two. For the
dimension bound we exhibit $d(d+1)/2$ trivial motions that are
linearly independent: the translations along the $d$ coordinate axes,
and one infinitesimal rotation in each of the $d(d-1)/2$ coordinate
planes. Whiteley~\cite{whiteley1996} (\S 11.1, before Lemma~11.1.2)
generates the trivial motions in the same way, and notes that these
generators are linearly independent once the points affinely span a
hyperplane of $\R^{d}$. Jord\'an~\cite{jordan2016} (p.~36) states the
resulting dimension, $d(d+1)/2$. Our proof uses the stronger
assumption that the points affinely span $\R^{d}$. We prove only the
lower bound, since the rank bound needs no more; the upper bound is
the count in \cref{sec:frameworks}.

Throughout, $G$ is a graph on a set $V$, and $d \in \N$. An element
$x$ of $(\R^{d})^{V}$ gives each vertex $v$ a velocity
$x_v \in \R^{d}$. It is an infinitesimal motion of $G$ at a framework
$p$ when $R(G, p)(x) = 0$ (\cref{def:rigidityMap}). We write
$e_1, \dots, e_d$ for the standard basis of $\R^{d}$, and $y_i$ for
the $i$-th coordinate of a vector $y \in \R^{d}$. We identify a linear
map of $\R^{d}$ with its matrix in this basis, and we write $E_{ij}$
for the matrix with entry $1$ in row $i$ and column $j$ and all other
entries $0$. A linear map $A$ of $\R^{d}$ is \emph{skew-symmetric} if
$\langle y, A y \rangle = 0$ for every $y \in \R^{d}$. Applying this
to $y + z$ shows that it holds exactly when
$\langle y, A z \rangle = -\langle A y, z \rangle$ for all
$y, z \in \R^{d}$, that is, when $A^{\top} = -A$.

\subsection{Translations}

A translation of $\R^{d}$ moves every point with the same velocity.

\begin{definition}[Translation motion]
  \label{def:translationMotion}
  \lean{SimpleGraph.translationMotion}
  \leanok
  \uses{def:framework}
  Let $b \in \R^{d}$. The \emph{translation} by $b$ is the element
  $\tau_b$ of $(\R^{d})^{V}$ with $(\tau_b)_v = b$ for every vertex
  $v$.
\end{definition}

\begin{lemma}[Translations are infinitesimal motions]
  \label{lem:translationMotion-mem-ker}
  \lean{SimpleGraph.translationMotion_mem_ker}
  \leanok
  \uses{def:translationMotion, def:rigidityMap}
  Let $p$ be a framework on $V$ in $\R^{d}$, and let $b \in \R^{d}$.
  Then $\tau_b \in \ker R(G, p)$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:rigidityMap-apply}
  At each edge $uv$ of $G$, the entry of $R(G, p)(\tau_b)$ is
  $\langle p_u - p_v,\, b - b \rangle = 0$
  (\cref{lem:rigidityMap-apply}).
\end{proof}

\subsection{Infinitesimal rotations}

Let $A$ be a skew-symmetric linear map of $\R^{d}$. The maps
$e^{tA}$, for $t \in \R$, are rotations of $\R^{d}$, and the point
$e^{tA} p_v$ has velocity $A p_v$ at $t = 0$. So the velocities
$A p_v$ describe a rotation of the framework to first order.

\begin{definition}[Infinitesimal rotation]
  \label{def:infinitesimalRotation}
  \lean{SimpleGraph.infinitesimalRotation}
  \leanok
  \uses{def:framework}
  Let $A$ be a linear map of $\R^{d}$, and let $p$ be a framework on
  $V$ in $\R^{d}$. We write $A \circ p$ for the element of
  $(\R^{d})^{V}$ with $(A \circ p)_v = A p_v$ for every vertex $v$.
  When $A$ is skew-symmetric, $A \circ p$ is an \emph{infinitesimal
  rotation} at $p$.
\end{definition}

\begin{lemma}[Infinitesimal rotations are infinitesimal motions]
  \label{lem:infinitesimalRotation-mem-ker}
  \lean{SimpleGraph.infinitesimalRotation_mem_ker}
  \leanok
  \uses{def:infinitesimalRotation, def:rigidityMap}
  Let $A$ be a skew-symmetric linear map of $\R^{d}$, and let $p$ be a
  framework on $V$ in $\R^{d}$. Then $A \circ p \in \ker R(G, p)$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:rigidityMap-apply}
  At each edge $uv$ of $G$, the entry of $R(G, p)(A \circ p)$ is
  $\langle p_u - p_v,\, A p_u - A p_v \rangle$
  (\cref{lem:rigidityMap-apply}). By linearity it equals
  $\langle p_u - p_v,\, A (p_u - p_v) \rangle$, which is zero since $A$
  is skew-symmetric.
\end{proof}

\subsection{The space of trivial motions}

In \cref{sec:frameworks} the trivial motions at $p$ are the motions
$v \mapsto A p_v + b$, with $A$ skew-symmetric and $b \in \R^{d}$. We
define them as a span.

\begin{definition}[Trivial motions]
  \label{def:trivialMotions}
  \lean{SimpleGraph.trivialMotions}
  \leanok
  \uses{def:translationMotion, def:infinitesimalRotation}
  Let $p$ be a framework on $V$ in $\R^{d}$. The space $T(p)$ of
  \emph{trivial infinitesimal motions} at $p$ is the linear span in
  $(\R^{d})^{V}$ of the translations $\tau_b$, for $b \in \R^{d}$, and
  the infinitesimal rotations $A \circ p$, for skew-symmetric linear
  maps $A$ of $\R^{d}$.
\end{definition}
The translations form a linear subspace of $(\R^{d})^{V}$, and so do
the infinitesimal rotations at $p$. So $T(p)$ is their sum, which
consists of the motions $v \mapsto A p_v + b$ above.

\begin{lemma}[Trivial motions are infinitesimal motions]
  \label{lem:trivialMotions-le-ker}
  \lean{SimpleGraph.trivialMotions_le_ker}
  \leanok
  \uses{def:trivialMotions, def:rigidityMap,
    lem:translationMotion-mem-ker, lem:infinitesimalRotation-mem-ker}
  Let $p$ be a framework on $V$ in $\R^{d}$. Then
  $T(p) \subseteq \ker R(G, p)$.
\end{lemma}
\begin{proof}
  \leanok
  The translations lie in $\ker R(G, p)$
  (\cref{lem:translationMotion-mem-ker}), and so do the infinitesimal
  rotations at $p$ (\cref{lem:infinitesimalRotation-mem-ker}). The
  kernel is a linear subspace, so it contains their span $T(p)$.
\end{proof}

\subsection{A lower bound on the dimension}
\label{sec:trivial-motions-d-general}

The points $p_v$ affinely span $\R^{d}$ when their affine hull is all
of $\R^{d}$, or equivalently when $V$ is nonempty and the differences
$p_v - p_w$, for $v, w \in V$, span $\R^{d}$. Under this condition we
exhibit $d(d+1)/2$ linearly independent trivial motions. The rotations
among them come from the following linear maps.

\begin{definition}[Elementary skew-symmetric map]
  \label{def:elemSkewMap}
  \lean{SimpleGraph.elemSkewMap}
  \leanok
  Let $1 \le i, j \le d$. The \emph{elementary skew-symmetric map}
  $E_{ij} - E_{ji}$ is the linear map of $\R^{d}$ that sends $y$ to
  $y_j e_i - y_i e_j$.
\end{definition}
For $i \ne j$, the map $E_{ij} - E_{ji}$ sends $e_j$ to $e_i$ and
$e_i$ to $-e_j$, and it vanishes on the other basis vectors. So the
maps $e^{t(E_{ij} - E_{ji})}$ rotate the plane spanned by $e_i$ and
$e_j$, and they fix the other basis vectors. The map is zero when
$i = j$, and exchanging $i$ and $j$ changes its sign.

\begin{lemma}[$E_{ij} - E_{ji}$ is skew-symmetric]
  \label{lem:inner-elemSkewMap-self}
  \lean{SimpleGraph.inner_elemSkewMap_self}
  \leanok
  \uses{def:elemSkewMap}
  Let $1 \le i, j \le d$. Then
  $\langle y, (E_{ij} - E_{ji}) y \rangle = 0$ for every
  $y \in \R^{d}$, that is, $E_{ij} - E_{ji}$ is skew-symmetric.
\end{lemma}
\begin{proof}
  \leanok
  For $y \in \R^{d}$, we have
  $\langle y,\, y_j e_i - y_i e_j \rangle = y_j y_i - y_i y_j = 0$.
\end{proof}

Since exchanging $i$ and $j$ only changes the sign of
$E_{ij} - E_{ji}$, we use the pairs with $j < i$.

\begin{definition}[Elementary trivial motions]
  \label{def:trivialMotionFamily}
  \lean{SimpleGraph.trivialMotionFamily}
  \leanok
  \uses{def:translationMotion, def:infinitesimalRotation, def:elemSkewMap}
  Let $p$ be a framework on $V$ in $\R^{d}$. The \emph{elementary
  trivial motions} at $p$ are the translations $\tau_{e_i}$, for
  $1 \le i \le d$, and the motions $(E_{ij} - E_{ji}) \circ p$, for
  $1 \le j < i \le d$.
\end{definition}

\begin{lemma}[The elementary trivial motions lie in $T(p)$]
  \label{lem:trivialMotionFamily-mem-trivialMotions}
  \lean{SimpleGraph.trivialMotionFamily_mem_trivialMotions}
  \leanok
  \uses{def:trivialMotionFamily, def:trivialMotions,
    lem:inner-elemSkewMap-self}
  Let $p$ be a framework on $V$ in $\R^{d}$. Then every elementary
  trivial motion at $p$ lies in $T(p)$.
\end{lemma}
\begin{proof}
  \leanok
  The translations $\tau_{e_i}$ are among the generators of $T(p)$. So
  are the motions $(E_{ij} - E_{ji}) \circ p$, since
  $E_{ij} - E_{ji}$ is skew-symmetric (\cref{lem:inner-elemSkewMap-self}).
\end{proof}

\begin{lemma}[Linear independence of the elementary trivial motions]
  \label{lem:trivialMotionFamily-linearIndependent}
  \lean{SimpleGraph.trivialMotionFamily_linearIndependent}
  \leanok
  \uses{def:trivialMotionFamily}
  Let $p$ be a framework on $V$ in $\R^{d}$ whose points affinely span
  $\R^{d}$. Then the elementary trivial motions at $p$ are linearly
  independent in $(\R^{d})^{V}$.
\end{lemma}
\begin{proof}
  \leanok
  Suppose that a linear combination of the elementary trivial motions
  vanishes, with coefficient $c_i$ at $\tau_{e_i}$ and coefficient
  $c_{ij}$ at $(E_{ij} - E_{ji}) \circ p$. Let
  $b = \sum_{i} c_i e_i$ and
  $S = \sum_{j < i} c_{ij} (E_{ij} - E_{ji})$. The combination is the
  motion $v \mapsto b + S p_v$, so $b + S p_v = 0$ for every vertex
  $v$.

  Subtracting this equation at two vertices $v$ and $w$ gives
  $S (p_v - p_w) = 0$. Since the points affinely span $\R^{d}$, these
  differences span $\R^{d}$, so $S = 0$. For the same reason $V$ is
  nonempty, and $b = -S p_v = 0$ for any vertex $v$. The coordinates of
  $b$ are the $c_i$, so every $c_i$ is zero.

  Let $j < i$. Among the matrices $E_{kl} - E_{lk}$ with $l < k$, only
  $E_{ij} - E_{ji}$ has a nonzero entry in row $i$ and column $j$. So
  this entry of $S$ is $c_{ij}$, and it is zero since $S = 0$.
\end{proof}

\begin{lemma}[Dimension lower bound for trivial motions]
  \label{lem:trivialMotions-finrank-ge-of-affinelySpanning}
  \lean{SimpleGraph.trivialMotions_finrank_ge_of_affinelySpanning}
  \leanok
  \uses{def:trivialMotions, lem:trivialMotionFamily-mem-trivialMotions,
    lem:trivialMotionFamily-linearIndependent}
  Let $V$ be finite, and let $p$ be a framework on $V$ in $\R^{d}$
  whose points affinely span $\R^{d}$. Then
  \[
    \dim T(p) \;\ge\; \frac{d(d+1)}{2}.
  \]
\end{lemma}
\begin{proof}
  \leanok
  The elementary trivial motions at $p$ lie in $T(p)$
  (\cref{lem:trivialMotionFamily-mem-trivialMotions}). They are
  linearly independent (\cref{lem:trivialMotionFamily-linearIndependent}),
  so $\dim T(p)$ is at least their number. There are $d$ translations,
  and for each $i$ there are $i - 1$ pairs $(i, j)$ with $j < i$. So
  there are $d + \sum_{i=1}^{d} (i - 1) = d + d(d-1)/2 = d(d+1)/2$
  elementary trivial motions.
\end{proof}

\begin{lemma}[Dimension lower bound for the rigidity-map kernel]
  \label{lem:rigidityMap-ker-finrank-ge-of-affinelySpanning}
  \lean{SimpleGraph.rigidityMap_ker_finrank_ge_of_affinelySpanning}
  \leanok
  \uses{lem:trivialMotions-le-ker,
    lem:trivialMotions-finrank-ge-of-affinelySpanning}
  Let $V$ be finite, and let $p$ be a framework on $V$ in $\R^{d}$
  whose points affinely span $\R^{d}$. Then
  \[
    \dim \ker R(G, p) \;\ge\; \frac{d(d+1)}{2}.
  \]
\end{lemma}
\begin{proof}
  \leanok
  The kernel of $R(G, p)$ contains $T(p)$
  (\cref{lem:trivialMotions-le-ker}). Since the points affinely span
  $\R^{d}$, the dimension of $T(p)$ is at least $d(d+1)/2$
  (\cref{lem:trivialMotions-finrank-ge-of-affinelySpanning}).
\end{proof}
